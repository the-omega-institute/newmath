% Underscores inside \leantarget MUST be \_-escaped.
%
% The block is parsed by lean4/scripts/bedc_ci.py audit and consumed by
% lean4/scripts/critical_path.py to decide whether a chapter is retired
% from the horizon work queue. Both axes are mandatory.
\NewDocumentEnvironment{closurestatus}{m}{%
  \par\smallskip\noindent%
  \begingroup%
    \footnotesize\sffamily\color{purple!55!black}%
    \textbf{Closure status for $\NameCert_{#1}$.}\par
    \def\bedcexternalcalibrated{0}%
    \def\bedcexternalboundary{0}%
    \def\constructivestory##1{\ifx\\##1\\\else\par\textbf{Constructive story:} ##1\par\fi}%
    \def\theoryclosure##1{\par\textbf{Theory closure:} $\TheoryGate(\NameCert_{#1},\,##1)$\par}%
    \def\scopeclosed##1{\textbf{Scope closed:} ##1\par}%
    \def\externalcorrespondence##1{%
      \edef\bedcexternalvalue{\detokenize{##1}}%
      \edef\bedcexternaltarget{\detokenize{\calibratedReconstruction}}%
      \ifx\bedcexternalvalue\bedcexternaltarget\def\bedcexternalcalibrated{1}\fi%
      \textbf{External correspondence:} $\ExternalCorrespondenceStatus(\NameCert_{#1},\,##1)$\par}%
    \def\reference##1{\textbf{External reference:} ##1\par}%
    \def\reconstructionscope##1{\textbf{Reconstruction scope:} ##1\par}%
    \def\boundaryexhaustive##1{\def\bedcexternalboundary{1}\textbf{Boundary exhaustion:} \autoref{##1}\par}%
    \def\formalstatus##1{\textbf{Formal verification:} $\FormalStatus(\NameCert_{#1},\,##1)$\par}%
    \def\leantarget##1{\textbf{Lean target:} \texttt{\detokenize{##1}}\par}%
    \def\bridgestatus##1{\textbf{Bridge status:} ##1\par}%
    \def\notclaimed##1{\textbf{Not claimed:} ##1\par}%
    \def\upgradepath##1{\textbf{Upgrade path:} ##1\par}%
    \def\origin##1{\textbf{Origin:} ##1\par}%
    \def\closureclaimkind##1{\textbf{Claim kind:} ##1\par}%
    \def\closureclassifierincrement##1{\textbf{Classifier increment:} ##1\par}%
    \def\closurenamecert##1{\textbf{NameCert evidence:} ##1\par}%
    \def\closureledger##1{\textbf{Ledger evidence:} ##1\par}%
    \def\closuregate##1{\textbf{Gate evidence:} ##1\par}%
    \def\closureweightprofile##1{\textbf{Weight profile:} ##1\par}%
    \def\closureparents##1{\textbf{Parent evidence:} ##1\par}%
    \def\closurelineage##1{\textbf{Lineage evidence:} ##1\par}%
    \def\dnahash##1{\textbf{DNA hash:} \texttt{\detokenize{##1}}\par}%
}{%
  \ifnum\bedcexternalcalibrated=1
    \ifnum\bedcexternalboundary=0
      \PackageError{BEDC}{calibratedReconstruction requires \string\boundaryexhaustive}{Add \string\boundaryexhaustive\{<label>\} inside the closurestatus block.}%
    \fi
  \fi
  \par\endgroup\smallskip
}
