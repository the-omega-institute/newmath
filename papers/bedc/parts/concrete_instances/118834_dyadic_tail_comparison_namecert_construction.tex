\begin{definition}[Dyadic tail-comparison carrier]
\label{def:dyadic-tail-comparison-carrier}

$$

\begin{enumerate}
\item $X$ and $Y$ are the two $\StreamNameUp$ tail windows whose dyadic endpoints are compared;
\item $W$ is the shared late-window selector, recording the finite indices inspected for both tails;
\item $E$ is the $\DyadicRatCoreUp$ endpoint ledger, including the requested dyadic radius, slack, and lower and upper endpoint rows;
\item $T$ is the dyadic tail-comparison row, recording that the two inspected windows satisfy the displayed dyadic comparison at the requested precision;
\item $R$ is the downstream readback row for $\RegSeqRatUp$, $\CauchyTailComparisonUp$, and $\RealUp$ consumers;
\item $H$ records componentwise $\hsame$ transport for the two windows, dyadic ledger, comparison row, downstream readback, replay, provenance, and local naming rows;

\item $P$ is the $\Pkg$ provenance over $\DyadicTailComparisonUp$, $\DyadicRatCoreUp$, $\StreamNameUp$, $\RegSeqRatUp$, $\CauchyTailComparisonUp$, $\RealUp$, $\BHist$, $\hsame$, $\Cont$, and $\NameCert$;

\label{thm:dyadic-tail-comparison-namecert-obligations}

\end{theorem}
\begin{proof}

\begin{theorem}[Dyadic tail-comparison regular Cauchy route]
\label{thm:dyadic-tail-comparison-regular-cauchy-route}

$$

$$

\end{theorem}
\begin{proof}
