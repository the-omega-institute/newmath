\begin{theorem}[Dependent eliminator gallery boundary]\label{thm:dependent-eliminator-coverage-gallery-boundary}

$$
W\longmapsto F\longmapsto B\longmapsto G\longmapsto O\longmapsto H,C,P,N .
$$

\end{theorem}
\begin{proof}

\end{proof}
