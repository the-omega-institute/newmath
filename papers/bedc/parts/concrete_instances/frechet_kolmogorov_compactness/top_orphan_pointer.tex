\begin{theorem}[Frechet--Kolmogorov top sibling pointer]
\label{thm:frechet-kolmogorov-top-sibling-pointer}
Every finite-window criterion exported by $\FrechetKolmogorovCompactnessUp$ must first display the compact metric source of \autoref{ch:concrete-instances-compactmetric-namecert}, the total-bounded finite-net surface of \autoref{ch:concrete-instances-totallybounded-namecert}, and the Arzela--Ascoli finite-modulus route of \autoref{ch:concrete-instances-arzela-ascoli-finite-modulus-namecert}. The top-level route therefore exposes the sibling lattice before any Frechet--Kolmogorov compactness consumer reads the local carrier.
\end{theorem}

\begin{theorem}[Frechet--Kolmogorov tightness dependency route]
\label{thm:frechet-kolmogorov-compactness-tightness-dependency-route}
Every $\FrechetKolmogorovCompactnessUp$ compactness read must expose the compact metric source of \autoref{ch:concrete-instances-compactmetric-namecert}, the total-bounded finite-net surface of \autoref{ch:concrete-instances-totallybounded-namecert}, the Arzela--Ascoli finite-modulus route of \autoref{ch:concrete-instances-arzela-ascoli-finite-modulus-namecert}, and the locally-convex boundedness and tightness rows before the local carrier is consumed. The dependency route therefore records Frechet--Kolmogorov compactness as a finite lattice read over compact metric, total boundedness, equicontinuity-modulus, and tightness data, not as an ambient compactness oracle.
\end{theorem}
