\begin{definition}[Carleson measure carrier]
\label{def:carleson-measure-carrier}

$$

\begin{enumerate}

\item $H$ is the $\HardySpaceUp$ handoff row from \autoref{ch:concrete-instances-hardy-space-namecert}. It names the Hardy-boundary source that may consume the same box window.

\item $D$ is the $\DyadicRatCoreUp$ tolerance row from \autoref{ch:concrete-instances-dyadicratcore-namecert}. It stores the dyadic scale, arc length, radial depth, and comparison tolerance used by the window estimate.

\end{definition}

\begin{proof}
