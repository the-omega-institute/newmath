\begin{theorem}[Regular Cauchy realization NameCert obligations]
\label{thm:regular-cauchy-realization-namecert-obligations}

\item regularity and Real-seal discipline through the witness row $W$ and terminal $\RealUp$ row $E$;

\end{enumerate}

\begin{proof}
