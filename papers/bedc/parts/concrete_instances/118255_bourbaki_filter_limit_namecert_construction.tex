\begin{enumerate}
\item carrier admission for $L=(B,C,V,W,R,D,E,H,T,P,N)$;
\item filter-base coverage through $B$;
\item Cauchy-filter refinement through $C$;
\item convergence-filter neighborhood coverage through $V$;
\item stream and regular-rational readback through $W$ and $R$;
\item dyadic tolerance exactness and real-seal handoff through $D$ and $E$;

\end{enumerate}
