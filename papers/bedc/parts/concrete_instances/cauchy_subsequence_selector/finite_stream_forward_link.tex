\label{thm:cauchy-subsequence-selector-forward-link}

\FiniteStreamUp,\ \StreamNameUp,\ \CauchyDiagonalSelectionUp,\ \RegSeqRatUp,\ \DyadicRatCoreUp \longrightarrow \RealUp

\end{proof}
