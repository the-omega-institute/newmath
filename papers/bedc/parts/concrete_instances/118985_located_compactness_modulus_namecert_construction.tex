\begin{theorem}[Located compactness modulus NameCert obligations]
\label{thm:located-compactness-modulus-namecert-obligations}

\begin{proof}

\begin{theorem}[Located compactness modulus finite-net extraction]
\label{thm:located-compactness-modulus-finite-net-extraction}
Every accepted $\LocatedCompactnessModulusUp$ packet extracts its modulus through a displayed finite net
$$
\CompactMetricUp \longrightarrow F \longrightarrow Q \longrightarrow B
$$
before any readback row is exported. Thus a compact-uniform-continuity consumer may read a positive rational lower bound only after the located cover evidence and finite fold have appeared in the packet.
\end{theorem}
\begin{proof}
Start from \autoref{thm:located-compactness-modulus-namecert-obligations}. The rows are ordered so that $K$ precedes the finite-net row $F$, the finite-net row precedes the rational ledger $Q$, and the ledger precedes the fold $B$.

The extraction is therefore finite at the point where the lower bound is formed. The fold $B$ has access only to the displayed candidates in $Q$ and the located cover evidence in $F$; it has no access to an external cover selector or an infinite search.

Since $S,G,R,H,C,P,N$ occur downstream of $B$, every exported modulus readback factors through the same finite-net extraction route.
\end{proof}

\end{proof}
