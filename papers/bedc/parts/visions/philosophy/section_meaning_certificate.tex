\section{Section Meaning Certificates}
\label{sec:visions-philosophy-section-meaning-certificate}

\autoref{sec:visions-philosophy-object-knowledge-namecert}
identifies objects with licensed names and knowledge with auditable
witnesses. The present layer applies that discipline to interfaces in
which an observation is invariant under a displayed symmetry while the
semantic label changes along a fiber. Its compressed thesis is:
$$
\begin{gathered}
\text{observation gives a package-level quotient readout},\\
\text{meaning requires a certified section or fiber ledger}.
\end{gathered}
$$
Thus a BEDC interface may accept an orbit-level observation without
accepting a bare representative selector, a hidden canonical section,
or a label that descends through the same package token.

\begin{definition}[Orbit observation packet]
\label{def:philosophy-orbit-observation-packet}
An orbit observation packet consists of a source carrier $X$, an
internal classifier $\approx_X$, a licensed action certificate for a
symmetry interface $G$, an orbit package $Q_G$ with projection
$\pi_G:X\to Q_G$, and a readout $O:X\to Y$ satisfying the displayed
invariance row
$$
\begin{gathered}
O(g\cdot x)\approx_Y O(x).
\end{gathered}
$$
The package $Q_G$ is a BEDC package interface with a gap ledger, not a
primitive quotient type.
\end{definition}

\begin{definition}[Anti-invariant label packet]
\label{def:philosophy-anti-invariant-label-packet}
An anti-invariant label packet over the same source consists of a
two-label interface $\mathsf{B}^{\uparrow}=\{C_0,C_1\}$ with
no-confusion, an involution $\nu$ exchanging the two labels, a certified
involution $\tau\in G$, and a label readout
$L:X\to\mathsf{B}^{\uparrow}$ satisfying
$$
\begin{gathered}
L(\tau\cdot x)=\nu(L(x)).
\end{gathered}
$$
\end{definition}

\begin{definition}[Section meaning certificate]
\label{def:philosophy-section-meaning-certificate}
A section meaning certificate is the derived paper-level interface
$$
\begin{gathered}
\mathsf{SectionMeaningCert}(X,G,O,L,Q_G,\pi_G,\sigma)
\end{gathered}
$$
whose rows include an orbit observation packet, an anti-invariant label
packet, an optional section binding
$\sigma:Q_G\to X$ satisfying $\pi_G(\sigma(q))=q$, and ledgers for the
source, orbit package, observation, label, section authorization, and
boundary. The section row is valid only when it records its source,
authorization, witness, stability, and forbidden-claim surface.
\end{definition}

\begin{theorem}[Observation descends while labels need not]
\label{thm:philosophy-observation-descends-labels-need-not}
In a section meaning certificate, the invariant observation descends to
a licensed readout $\bar O:Q_G\to Y$, while an anti-invariant label
cannot be treated as a readout on $Q_G$ unless an additional section or
fiber ledger is displayed.
\end{theorem}
\begin{proof}
If two source rows have the same orbit package token, the action
certificate supplies a symmetry path between them. Observation
invariance transports $O$ along that path, so $O$ factors through the
package projection as $\bar O\circ\pi_G$. For the label row, $x$ and
$\tau\cdot x$ have the same orbit token, but their labels are exchanged
by $\nu$. A package-level label readout would identify those two
labels, contradicting no-confusion. Therefore the label requires
additional section or fiber data rather than descending through the
orbit package alone.
\end{proof}

\begin{corollary}[No hidden section from pure observation]
\label{cor:philosophy-no-hidden-section-from-pure-observation}
Pure orbit-observation material does not license a distinguished
section $Q_G\to X$.
\end{corollary}
\begin{proof}
The visible observation rows are invariant on each orbit. Choosing a
representative inside an orbit therefore introduces data not contained
in the orbit token or invariant readout. BEDC may expose that data as a
section binding, authorization row, action-response row, bridge, or
boundary, but it may not present the choice as a theorem of the pure
observation packet.
\end{proof}

\begin{theorem}[Meaning is section-relative]
\label{thm:philosophy-meaning-section-relative}
When a valid section binding $\sigma$ is supplied, the expression
$$
\begin{gathered}
L_\sigma(q)=L(\sigma(q))
\end{gathered}
$$
is a licensed meaning readout on $Q_G$ relative to that binding.
\end{theorem}
\begin{proof}
The section binding is a licensed map from the package interface back
to the source carrier, and the label packet is a licensed map from the
source carrier to the two-label interface. Their composite is therefore
a licensed readout. Its ledger is the composite ledger of the section
authorization and the label packet, so the resulting meaning is
relative to the displayed binding rather than intrinsic to the orbit
token.
\end{proof}

\begin{corollary}[Matched twins force a blind ceiling]
\label{cor:philosophy-matched-twins-force-blind-ceiling}
In a balanced matched-twin protocol where each pair has the same
observation and opposite labels, every decision rule depending only on
the orbit observation is bounded by the blind ceiling $\frac12$.
\end{corollary}
\begin{proof}
On each matched pair, the rule sees the same observation on both
members and therefore returns the same decision. The label packet
assigns opposite labels to the two members. With balanced presentation,
the rule is correct on exactly one side of each pair, so averaging over
the finite protocol gives $\frac12$.
\end{proof}

\begin{corollary}[Section access, leakage, or protocol mismatch]
\label{cor:philosophy-section-access-leakage-protocol-mismatch}
If a matched-twin interface is reported above the blind ceiling, then
the certificate audit must expose section access, leakage of non-orbit
data, or a mismatch in the stated protocol object.
\end{corollary}
\begin{proof}
The blind ceiling applies to every orbit-observation rule. Exceeding it
therefore means that the rule is not using only the certified orbit
observation, or that the protocol does not satisfy the matched-twin
rows that define the ceiling. In the first case the additional
coordinate must be ledgered as section access or recorded as leakage;
in the second case the protocol object must be corrected before the
reported number can support a certificate claim.
\end{proof}
