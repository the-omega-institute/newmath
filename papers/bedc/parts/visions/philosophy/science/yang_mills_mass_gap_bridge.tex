\subsection{Yang--Mills mass-gap bridge}
\label{subsec:philosophy-yang-mills-mass-gap-bridge}

The Yang--Mills mass-gap route is admissible in this scientific bridge layer
only as a conditional certificate chain. The finite strong-coupling lattice
surface may close inside a compact group integral and transfer-matrix packet.
The continuum claim needs an additional renormalization bridge that preserves
the physical Hilbert interface, Wilson packet readbacks, and every ultraviolet,
infrared, gauge, topological, and soft-mode residual ledger. Without that
bridge, the finite lattice theorem remains a regulated theorem and not a
continuum mass-gap certificate.

\begin{definition}[Yang--Mills bridge packet]
\label{def:philosophy-yang-mills-bridge-packet}
A Yang--Mills bridge packet is a finite scientific bridge record
$$
\mathcal Y=(G,\mathcal H,\Omega,H,\mathcal W_{\mathrm{fin}},L,A)
$$
where $G$ is a displayed compact simple gauge group row, $\mathcal H$ is the
gauge-invariant physical Hilbert carrier, $\Omega$ is the vacuum row,
$H\ge 0$ is the Hamiltonian readback with $H\Omega=0$,
$\mathcal W_{\mathrm{fin}}$ is the finite Wilson-loop or spin-network packet
surface, $L$ is the ledger for gauge quotient, ultraviolet cutoff, infrared
volume, topological sector, zero mode, and soft residual data, and $A$ is the
audit row recording standard-bridge scope and no-host-leak boundaries.
\end{definition}

\begin{definition}[Finite Wilson packet correlation]
\label{def:philosophy-finite-wilson-packet-correlation}
For a finite Wilson packet
$$
O=\sum_{\alpha\in I}c_{\alpha}W_{\rho_{\alpha},C_{\alpha}}
$$
over a finite index set $I$, write
$$
v_O=O\Omega-\langle\Omega,O\Omega\rangle\Omega
$$
and
$$
C_O(t)=\langle v_O,e^{-tH}v_O\rangle .
$$
The packet is public only through the displayed loop, representation,
coefficient, vacuum-subtraction, correlation, ledger, and provenance rows.
\end{definition}

\begin{definition}[Strong-coupling lattice gap certificate]
\label{def:philosophy-strong-coupling-lattice-gap-certificate}
A strong-coupling lattice gap certificate for a finite lattice
$\Lambda$ consists of compact edge variables $U_e\in G$, plaquette holonomy
rows $U_p$, a Wilson action row, a character-expansion row
$$
e^{\beta\operatorname{Re}\chi_{\mathrm{fund}}(g)}
=
c_0(\beta)
\left(
1+
\sum_{\rho\ne\mathbf 1}a_{\rho}(\beta)\chi_{\rho}(g)
\right),
$$
a polymer ledger with
$$
\kappa(\beta)=\sum_{\rho\ne\mathbf 1}d_{\rho}|a_{\rho}(\beta)|<1,
$$
and a transfer-matrix row. Its strong-coupling bound supplies constants
$m_{\Lambda,\beta}>0$ and $K_O$ such that every non-vacuum finite Wilson
packet satisfies
$$
|C_O(t)|\le K_O e^{-m_{\Lambda,\beta}t}.
$$
\end{definition}

\begin{theorem}[Finite strong-coupling packet decay]
\label{thm:philosophy-finite-strong-coupling-packet-decay}
If the character-expansion ledger of
\autoref{def:philosophy-strong-coupling-lattice-gap-certificate} satisfies
the displayed small-parameter bound strongly enough that the connected
polymer expansion is absolutely convergent, then finite Wilson packet
correlations decay exponentially on that regulated lattice surface.
\end{theorem}
\begin{proof}
Every connected non-vacuum polymer contributing to a connected Wilson packet
correlation contains a non-trivial representation row on each plaquette in
its support. The coefficient ledger bounds the contribution of each such row
by the strong-coupling parameter, while finite lattice connectivity supplies
only a finite combinatorial factor per plaquette. Hence the total weight of
a connected polymer with support size $|\Gamma|$ is bounded by
$$
|w(\Gamma)|\le C^{|\Gamma|}\kappa(\beta)^{|\Gamma|}.
$$
When $C\kappa(\beta)<1$, the polymer sum is absolutely convergent and
separation in Euclidean time forces an exponential penalty for every
connected surface linking the two Wilson packet reads. The transfer-matrix
row reads this Euclidean decay as a non-vacuum transfer spectral gap on the
finite regulated surface.
\end{proof}

\begin{definition}[Wilson packet Hilbert gap certificate]
\label{def:philosophy-wilson-packet-hilbert-gap-certificate}
A Wilson packet Hilbert gap certificate for
\autoref{def:philosophy-yang-mills-bridge-packet} consists of a number
$m>0$ and two displayed rows:
\begin{itemize}
\item the span of vacuum-orthogonal finite Wilson packet vectors $v_O$ is
dense in $\Omega^{\perp}$;
\item for every such $v_O$ there is a constant $K_O$ with
$$
\langle v_O,e^{-tH}v_O\rangle\le K_O e^{-mt}
$$
for all displayed Euclidean times $t\ge 0$.
\end{itemize}
\end{definition}

\begin{theorem}[Wilson decay and the physical Hilbert spectral gap]
\label{thm:philosophy-wilson-decay-physical-hilbert-gap}
For a Yang--Mills bridge packet with self-adjoint non-negative Hamiltonian,
unique vacuum row, and the density row of
\autoref{def:philosophy-wilson-packet-hilbert-gap-certificate}, the following
conditions are equivalent for a fixed $m>0$.
\begin{itemize}
\item The restriction of $H$ to $\Omega^{\perp}$ has spectrum contained in
$[m,\infty)$.
\item Every vacuum-orthogonal finite Wilson packet correlation has a
spectral measure supported in $[m,\infty)$, equivalently it satisfies a
uniform exponential decay certificate with exponent $m$.
\end{itemize}
\end{theorem}
\begin{proof}
If the spectrum of $H$ on $\Omega^{\perp}$ is contained in $[m,\infty)$, then
the spectral theorem gives
$$
\langle v,e^{-tH}v\rangle
=
\int_{[m,\infty)}e^{-tE}\,d\mu_v(E)
\le e^{-mt}\|v\|^2
$$
for every $v\in\Omega^{\perp}$, hence in particular for finite Wilson packet
vectors.

Conversely, suppose the displayed exponential certificate holds for each
finite Wilson packet vector. For any $\varepsilon>0$ and such a vector $v$,
let $\mu_v$ be its spectral measure. The bound
$$
\mu_v([0,m-\varepsilon])e^{-t(m-\varepsilon)}
\le
\int e^{-tE}\,d\mu_v(E)
\le
K_v e^{-mt}
$$
holds for all $t\ge0$. Letting $t$ grow gives
$$
\mu_v([0,m-\varepsilon])=0.
$$
Thus every finite Wilson packet vector is annihilated by the spectral
projection of $H$ onto $[0,m-\varepsilon]$. Since those vectors are dense in
$\Omega^{\perp}$ and spectral projections are bounded, the same projection
annihilates all of $\Omega^{\perp}$. This holds for every $\varepsilon>0$,
so no non-vacuum spectral row lies below $m$.
\end{proof}

\begin{definition}[Soft residual ledger]
\label{def:philosophy-yang-mills-soft-residual-ledger}
A soft residual below a proposed mass threshold $m$ is a non-vacuum Hilbert
state or approximating state family whose spectral mass meets $[0,m)$.
It is ledgered when either some finite Wilson packet vector has nonzero
projection against it, or the residual is recorded in the infrared,
ultraviolet, renormalization, packet-complexity, topological, gauge, or
zero-mode tail row of $L$.
\end{definition}

\begin{definition}[Wilson packet complexity filtration]
\label{def:philosophy-wilson-packet-complexity-filtration}
A Wilson packet complexity filtration for
\autoref{def:philosophy-yang-mills-bridge-packet} is an increasing family
$$
\mathcal V_1\subseteq \mathcal V_2\subseteq\cdots\subseteq\Omega^{\perp}
$$
where $\mathcal V_M$ is generated by vacuum-orthogonal finite Wilson or
spin-network packet vectors whose displayed loop support, representation
weight, and coefficient-description ledgers have total complexity at most
$M$. The filtration has a Peter--Weyl completeness row when
$$
\overline{\bigcup_{M\ge1}\mathcal V_M}=\Omega^{\perp}.
$$
Each fixed $\mathcal V_M$ is a fixed packet-complexity surface. No conclusion
about a growing sequence of packet complexities is licensed unless its tail
ledger is also displayed.
\end{definition}

\begin{definition}[Complexity tail ledger]
\label{def:philosophy-yang-mills-complexity-tail-ledger}
Given the filtration of
\autoref{def:philosophy-wilson-packet-complexity-filtration}, a soft
residual family $\psi_n\in\Omega^{\perp}$ is in the packet-complexity tail
when
$$
\|P_M\psi_n\|\to0
$$
for every fixed $M$, where $P_M$ is the projection onto the displayed fixed
complexity surface. The corresponding ledger row records that detecting the
residual requires packet complexity tending beyond every fixed stratum; it is
not absence of a residual.
\end{definition}

\begin{theorem}[Soft residual fixed-packet or complexity-tail dichotomy]
\label{thm:philosophy-soft-residual-fixed-packet-complexity-tail}
Let $\psi_n\in\Omega^{\perp}$ be a norm-one soft residual family for a
Yang--Mills bridge packet equipped with a Wilson packet complexity
filtration. Then either some fixed complexity surface has
$$
\limsup_n\|P_M\psi_n\|>0,
$$
or the family lies in the packet-complexity tail ledger of
\autoref{def:philosophy-yang-mills-complexity-tail-ledger}.
\end{theorem}
\begin{proof}
This is the exhaustion dichotomy for an increasing filtration. If some
fixed $M$ has positive limsup projection, the residual is visible at that
fixed packet-complexity surface. If no such $M$ exists, then by definition
$\|P_M\psi_n\|\to0$ for every fixed $M$, which is exactly the
packet-complexity tail condition.
\end{proof}

\begin{definition}[Filtered uniform Wilson gap certificate]
\label{def:philosophy-filtered-uniform-wilson-gap-certificate}
A filtered uniform Wilson gap certificate is a Wilson packet Hilbert gap
certificate together with a complexity filtration and a number $m_0>0$ such
that every vacuum-orthogonal finite Wilson packet vector $v_O$ has
$$
\langle v_O,e^{-tH}v_O\rangle\le K_Oe^{-m_0t}
$$
for all displayed Euclidean times $t\ge0$. The constant $K_O$ may depend on
the packet, but the exponent $m_0$ is independent of $O$, of the fixed
complexity stratum, of the loop-support ledger, and of the representation
ledger.
\end{definition}

\begin{definition}[Block tube area-law certificate]
\label{def:philosophy-block-tube-area-law-certificate}
A block tube area-law certificate is a fixed physical block-scale bridge row.
It consists of a block scale $\ell_{\ast}>0$, a blocked gauge graph
$\mathcal L_{\ell_{\ast}}$, a class of blocked finite Wilson or spin-network
packets on that graph, and a connected-tube expansion for every correlation
$C_O(t)$. The certificate supplies constants
$$
\sigma>\kappa,\qquad c_{\ast}>0
$$
such that every connected tube $\Sigma$ joining $O(0)$ to $O(t)$ has weight
and area rows
$$
|w(\Sigma)|\le e^{-\sigma|\Sigma|}
$$
and
$$
|\Sigma|\ge c_{\ast}t-b_O,
$$
where $b_O$ may depend on the displayed packet support but $c_{\ast}$ does
not. It also supplies an entropy row
$$
\#\{\Sigma:|\Sigma|=n,\ \Sigma:O(0)\leftrightarrow O(t)\}
\le A_Oe^{\kappa n}.
$$
All constants are read at the fixed block scale. The certificate makes no
claim for changing block scales or growing packet windows unless those
changes are recorded in the tail ledger.
\end{definition}

\begin{theorem}[Block tube area law gives filtered uniform Wilson decay]
\label{thm:philosophy-block-tube-area-law-gives-filtered-wilson-decay}
If a Yang--Mills bridge packet has a block tube area-law certificate, then
its blocked finite Wilson packet correlations satisfy a filtered uniform
Wilson gap certificate with exponent
$$
m_0=(\sigma-\kappa)c_{\ast}>0.
$$
\end{theorem}
\begin{proof}
The connected-tube expansion bounds the connected correlation by the sum of
absolute tube weights:
$$
|C_O(t)|
\le
\sum_{\Sigma:O(0)\leftrightarrow O(t)}|w(\Sigma)|.
$$
Group the tubes by area $n=|\Sigma|$. The tube lower bound and entropy row
give
$$
|C_O(t)|
\le
\sum_{n\ge c_{\ast}t-b_O}A_Oe^{\kappa n}e^{-\sigma n}.
$$
Since $\sigma>\kappa$, this is a convergent geometric tail:
$$
|C_O(t)|
\le
A'_Oe^{-(\sigma-\kappa)(c_{\ast}t-b_O)}.
$$
Absorbing $e^{(\sigma-\kappa)b_O}$ into the packet-dependent constant gives
$$
|C_O(t)|\le K_Oe^{-(\sigma-\kappa)c_{\ast}t}.
$$
Thus the exponent is $m_0=(\sigma-\kappa)c_{\ast}$ and is independent of the
packet, its complexity stratum, its support ledger, and its representation
ledger. This is exactly the filtered uniform Wilson gap certificate.
\end{proof}

\begin{corollary}[Block tube area law gives the fixed-sector gap]
\label{cor:philosophy-block-tube-area-law-gives-fixed-sector-gap}
Assume the physical Hilbert construction, Wilson packet complexity
filtration, Peter--Weyl completeness row, ledger-exhaustiveness row, and
closedness boundary of
\autoref{thm:philosophy-filtered-yang-mills-gap-theorem}. A block tube
area-law certificate then forces the non-vacuum physical Hilbert spectrum in
the declared fixed block sector to be contained in $[m_0,\infty)$, with
$m_0=(\sigma-\kappa)c_{\ast}$.
\end{corollary}
\begin{proof}
By
\autoref{thm:philosophy-block-tube-area-law-gives-filtered-wilson-decay}, the
block tube certificate supplies the filtered uniform Wilson gap certificate
with exponent $m_0$. The filtered gap theorem then excludes every fixed
packet soft residual below $m_0$. Any residual escaping all fixed
packet-complexity surfaces must enter the packet-complexity or physical tail
ledger, and the closedness boundary rejects hidden unfinished tail data.
\end{proof}

\begin{theorem}[Filtered gap excludes fixed-packet soft residuals]
\label{thm:philosophy-filtered-gap-excludes-fixed-packet-soft-residuals}
If a filtered uniform Wilson gap certificate with exponent $m_0>0$ is
present, no soft residual can have nonzero spectral address in a fixed
finite Wilson packet.
\end{theorem}
\begin{proof}
Suppose a fixed packet vector $v_O$ detects a soft residual. Then the
spectral measure $\mu_O$ of $v_O$ for $H$ has positive mass in arbitrarily
small non-vacuum energy windows. In particular, for some
$0<\varepsilon<m_0$ the measure of $[0,\varepsilon]$ is nonzero. The
correlation then satisfies
$$
\langle v_O,e^{-tH}v_O\rangle
\ge
\mu_O([0,\varepsilon])e^{-\varepsilon t}.
$$
Multiplying by $e^{m_0t}$ contradicts the displayed bound
$K_Oe^{-m_0t}$ as $t$ grows. Hence a fixed packet address is impossible
under the filtered uniform gap certificate.
\end{proof}

\begin{theorem}[Gapless spectrum forces visible or ledgered residual]
\label{thm:philosophy-gapless-spectrum-forces-ledgered-residual}
Under the density, complexity-filtration, and ledger-exhaustiveness rows of
a Yang--Mills bridge packet, failure of the Hilbert spectral gap cannot
disappear silently. A gapless non-vacuum spectral residual is either detected
by a fixed finite Wilson packet correlation, appears on a fixed
packet-complexity surface, or belongs to the displayed packet-complexity,
infrared, ultraviolet, renormalization, topological, gauge, or zero-mode tail
ledger.
\end{theorem}
\begin{proof}
Apply \autoref{thm:philosophy-soft-residual-fixed-packet-complexity-tail}.
If a fixed complexity projection has positive limsup, the displayed fixed
stratum contains a finite packet witness after choosing the corresponding
packet row in that stratum; the residual changes the spectral measure of
that packet below every proposed positive threshold. If every fixed
complexity projection vanishes, the residual is not absent; it has escaped
the fixed packet surfaces and must be recorded as packet-complexity tail.
Any further ultraviolet, infrared, renormalization, topological, gauge, or
zero-mode loss is likewise a ledger row. Therefore an unledgered soft
residual is incompatible with the packet assumptions.
\end{proof}

\begin{theorem}[Filtered Yang--Mills gap theorem]
\label{thm:philosophy-filtered-yang-mills-gap-theorem}
Let a Yang--Mills bridge packet have a physical Hilbert construction, compact
gauge Wilson packet rows, Peter--Weyl completeness for the packet-complexity
filtration, a filtered uniform Wilson gap certificate with exponent
$m_0>0$, and a ledger-exhaustiveness row saying that every residual escaping
all fixed packet-complexity surfaces enters the packet-complexity or
physical tail ledger. If the accepted object is closed against unfinished
tail ledgers, then its physical Hilbert spectrum on $\Omega^{\perp}$ is
contained in $[m_0,\infty)$.
\end{theorem}
\begin{proof}
Assume instead that the non-vacuum spectrum meets energies below $m_0$.
Choose a norm-one approximating soft residual family in
$\Omega^{\perp}$ with energy tending below the proposed gap. By
\autoref{thm:philosophy-soft-residual-fixed-packet-complexity-tail}, this
family either has a fixed packet-complexity address or lies in the
packet-complexity tail.

The first case is excluded by
\autoref{thm:philosophy-filtered-gap-excludes-fixed-packet-soft-residuals}.
In the second case, ledger exhaustiveness requires a packet-complexity or
physical tail row. Such a row is unfinished residual data unless it is
resolved by the bridge, and closedness forbids carrying it as hidden
content. Thus both cases contradict the accepted closed packet assumptions.
Therefore the non-vacuum spectrum is contained in $[m_0,\infty)$.
\end{proof}

\begin{definition}[Filtered renormalization gap stability certificate]
\label{def:philosophy-filtered-renormalization-gap-stability-certificate}
A filtered renormalization gap stability certificate is a bridge row from a
lattice tower
$$
\mathcal Y_0\leftarrow\mathcal Y_1\leftarrow\mathcal Y_2\leftarrow\cdots
\longrightarrow \mathcal Y_{\infty}
$$
to a continuum Yang--Mills bridge packet. It contains:
\begin{itemize}
\item scale-time rows for blocking and continuum refinement;
\item contractive or isometric Hilbert transport rows on the physical
vacuum-orthogonal surfaces;
\item convergence of finite Wilson packet correlations from the lattice tower
to the continuum packet;
\item a uniform lower bound $m_0>0$ for the lattice physical gaps in every
declared fixed Wilson packet-complexity sector, with exponent independent of
the sector;
\item a complete ledger for ultraviolet, infrared, renormalization, gauge,
packet-complexity, topological, and zero-mode residuals.
\end{itemize}
\end{definition}

\begin{theorem}[Renormalization bridge preserves the mass gap]
\label{thm:philosophy-renormalization-bridge-preserves-mass-gap}
Assume a lattice tower has strong-coupling finite Wilson packet decay with a
uniform exponent $m_0>0$ independent of packet complexity, and assume the
tower is connected to a continuum Yang--Mills bridge packet by a filtered
renormalization gap stability certificate. Then the continuum packet has a
physical Hilbert spectral gap at least $m_0$ on the declared Wilson packet
sector, unless a packet-complexity or physical tail ledger remains open.
\end{theorem}
\begin{proof}
Choose a continuum finite Wilson packet vector. By the correlation
convergence row of
\autoref{def:philosophy-filtered-renormalization-gap-stability-certificate}, it is
the limit of lattice Wilson packet vectors in the declared sector. The
uniform lattice gap gives the bound
$$
C_{O,n}(t)\le K_{O,n}e^{-m_0t}
$$
with the same exponent $m_0$ along the tower, after the packet normalization
specified by the bridge row. Passing to the continuum correlation yields the
same spectral-support exclusion below $m_0$ for the continuum packet.

The density and filtration rows then invoke
\autoref{thm:philosophy-filtered-yang-mills-gap-theorem}: finite Wilson
packet decay with exponent $m_0$ forces the physical Hilbert spectrum on the
vacuum-orthogonal declared sector to avoid $[0,m_0)$, except for residuals
that escape every fixed packet-complexity surface. If such a residual tried
to appear during the limit,
\autoref{thm:philosophy-gapless-spectrum-forces-ledgered-residual} would
place it in the displayed packet-complexity or physical tail ledger. The
filtered stability certificate therefore blocks an unrecorded loss of the
mass gap and leaves only explicitly ledgered open tail data.
\end{proof}

\begin{theorem}[Yang--Mills mass-gap bridge boundary]
\label{thm:philosophy-yang-mills-mass-gap-bridge-boundary}
The Yang--Mills mass-gap route in this layer has exactly two admissible
positive surfaces: the regulated finite strong-coupling packet theorem, and
the conditional continuum theorem obtained after a renormalization gap
stability certificate is displayed. It does not claim an unconditional
continuum construction, a continuum gap without the stability certificate, a
gap outside the declared Wilson packet sector, or a host-theory theorem not
listed in the bridge preservation row.
\end{theorem}
\begin{proof}
\autoref{thm:philosophy-finite-strong-coupling-packet-decay} closes only the
finite regulated strong-coupling surface. The passage to a continuum Hilbert
mass gap additionally uses the density, correlation-convergence, Hilbert
transport, packet-complexity filtration, uniform-gap, and tail-ledger rows
of \autoref{def:philosophy-filtered-renormalization-gap-stability-certificate}.
Without those rows the bridge has no public path from lattice transfer-matrix
decay to continuum spectral support. In particular, the statement that every
fixed finite packet has some gap does not imply a uniform physical mass gap;
the exponent must be independent of packet complexity, or the escaping modes
must be recorded in the packet-complexity or physical tail ledger. Standard
bridge discipline also restricts the export to the declared
theorem-preservation list and no-host-leak boundary. The stated positive
surfaces are therefore the only ones licensed here.
\end{proof}
