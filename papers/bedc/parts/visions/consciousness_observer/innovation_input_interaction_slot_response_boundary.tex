\section{First Interaction Row}
\label{sec:innovation-input-interaction-first-row}

\begin{theorem}[First interaction through mass renormalization]
\label{thm:innovation-input-first-interaction-mass-renormalization}
For a scalar row with interaction $\lambda\varphi^4$, the entanglement
entropy has the first-order form
$$
S_{\rm EE}(\lambda)
=
S_{\rm free}(m_*^2)+O(\lambda^2),
\qquad
m_*^2
=
m^2+\frac{\lambda}{2}\langle\varphi^2\rangle.
$$
\end{theorem}

\begin{proof}
The entanglement first law gives
$$
\delta S=\operatorname{tr}(\delta\rho\,K)
$$
at first order. For the Gaussian modular Hamiltonian row, this reduces the
first-order entropy response to the first-order shift of the two-point
function. The tadpole term shifts the mass, while the four-particle
transition does not contribute to the two-point matrix element at this
order. Hence the free entropy evaluated at the shifted mass gives the
displayed expression.
\end{proof}

\begin{principle}[Exact-diagonalization witness]
\label{prin:innovation-input-interaction-diagonalization-witness}
In a four-site $\varphi^4$ chain with a truncated bosonic basis, the
first-law check gives ratio $1.00084$, the zero-coupling entropy agrees
with the covariance method to $1.1\times10^{-7}$, and the measured
derivative
$$
\left.\frac{dS}{d\lambda}\right|_{\lambda=0}
=
-0.166117
$$
matches the mass-renormalization prediction $-0.166344$ at the displayed
precision. The convergence floor of the diagonalization row is below
$3\times10^{-6}$.
\end{principle}

\begin{corollary}[Massive slope inherited by the interacting row]
\label{cor:innovation-input-interaction-massive-slope-inherited}
The universal massive area-law slope remains the free-field slope when
expressed in terms of the renormalized mass $m_*$. The arithmetic blindness
of vacuum entanglement also persists under a global Hamiltonian rescaling:
if a refinement factor changes only the overall energy scale, the
interacting ground-state density matrix on a region is unchanged.
\end{corollary}

\begin{principle}[Critical boundary of the first-order row]
\label{prin:innovation-input-interaction-critical-boundary}
The first-order formula contains the factor
$$
\frac{dm_*}{d\lambda}
=
\frac{\langle\varphi^2\rangle}{4m_*}.
$$
It therefore becomes singular near the massless critical row. The
breakdown is not a failure of the first-order certificate; it locates where
the interaction problem leaves the perturbative mass-shift regime.
\end{principle}

\section{Non-Gaussian Remainder}
\label{sec:innovation-input-interaction-nongaussian-remainder}

\begin{definition}[Gaussian assembly comparison]
\label{def:innovation-input-interaction-gaussian-assembly-comparison}
Let $S_{\rm exact}$ be the exact reduced entropy of the interacting
lattice state. Let $S_{\rm gauss}$ be the Gaussian entropy assembled from
the exact two-point correlations, and let $S_{\rm Hartree}$ be the entropy
obtained from a self-consistent mass approximation. These three rows
separate correlation data from genuinely non-Gaussian entropy.
\end{definition}

\begin{principle}[Quadratic-order witness]
\label{prin:innovation-input-interaction-quadratic-order-witness}
The measured second-order row has stable $\lambda^2$ scaling:
$$
\begin{array}{c|c|c|c}
\lambda & \Delta_2/\lambda^2
& S_{\rm exact}-S_{\rm gauss}
& S_{\rm exact}-S_{\rm Hartree}\\
\hline
0.05 & +0.052 & -9.8\times10^{-6} & +2.8\times10^{-5}\\
0.10 & +0.047 & -3.5\times10^{-5} & +9.5\times10^{-5}\\
0.40 & +0.028 & -3.6\times10^{-4} & +7.6\times10^{-4}
\end{array}
$$
The exact entropy stays below the Gaussian assembly entropy in this
window, while the Hartree row underestimates the entropy.
\end{principle}

\begin{principle}[Two-level interaction program]
\label{prin:innovation-input-interaction-two-level-program}
The measured non-Gaussian deficit is negative but accounts for only a small
part of the displayed second-order effect. Most of the interaction response
is captured by the two-point correlation row and a Gaussian assembly; the
remaining tail belongs to a replica or higher-cumulant calculation. Thus
the interaction extension splits into a two-level program: measure the
correlation row accurately, then compute the residual non-Gaussian entropy.
\end{principle}

\begin{principle}[Hartree sign diagnostic]
\label{prin:innovation-input-interaction-hartree-sign-diagnostic}
The Hartree approximation can have the wrong entropy sign even when it
captures a plausible mass shift. In the measured row it underestimates the
entropy because the missing sunset-type self-energy contribution makes the
effective mass too large. This sign diagnostic is part of the interaction
certificate, not a numerical nuisance.
\end{principle}

\section{Slot-Response Audit}
\label{sec:innovation-input-slot-response-audit}

\begin{definition}[Conformal response channels]
\label{def:innovation-input-conformal-response-channels}
A small conformal perturbation of the spatial metric decomposes the
entanglement response into distinct channels: the coordinate mass channel,
the measure-weight channel, and the geometric slot channel. A slot claim is
valid only after these channels have been separated.
\end{definition}

\begin{principle}[Channel decomposition witness]
\label{prin:innovation-input-slot-channel-decomposition-witness}
For a perturbation amplitude $0.06$ at mass $0.4$, the total measured
curvature response is
$$
-0.05766.
$$
The coordinate mass channel contributes $-0.05912$, while the
measure-weight channel contributes $+0.00156$, closing the total to about
$0.2\%$. The geometric slot channel contributes $+0.00803$ but is hidden
under the larger coordinate mass response. The dominant term matches the
estimate from $\kappa'(m)m\varepsilon\psi(y_c)n$.
\end{principle}

\begin{principle}[Response audit]
\label{prin:innovation-input-response-audit}
A perturbation response must be decomposed by channel before it is assigned
to a geometric invariant. If a claimed curvature slot is mostly a
coordinate mass effect, the measurement protocol exposes the frame
confusion. This response audit is the interaction analogue of the
non-tensor operator audit in
\autoref{prin:innovation-input-thermo-non-tensor-audit}.
\end{principle}

\begin{principle}[Measured inverse-mass slot]
\label{prin:innovation-input-measured-inverse-mass-slot}
A matched double-profile difference that fixes the cut value and slope
shows an inverse-mass response. Across a mass range of about $4.6$, the
product $mD(m)$ remains on a platform between $0.0028$ and $0.0036$,
giving
$$
c_{-1}=0.0034(4).
$$
Thus a covariant inverse-mass slot is present, but the coefficient is not
the naive scalar-curvature coefficient.
\end{principle}

\begin{principle}[Projected curvature coefficient]
\label{prin:innovation-input-projected-curvature-coefficient}
The naive coefficient predicted by a bare $R/6$ slot is about $0.0080$ in
the measured normalization and is excluded by the observed inverse-mass
platform. A projected cone coefficient involving transverse curvature data,
such as a combination of $R$ and $R_{nn}$, gives a value near $0.0040$ and
is compatible at the current precision. The exact coefficient belongs to
the conical heat-kernel projection row rather than to the scalar
heat-kernel trace alone.
\end{principle}

\begin{principle}[Slot boundary]
\label{prin:innovation-input-slot-boundary}
The heat-kernel slot identity for operator traces does not transfer
unchanged to entanglement entropy. Entanglement inserts a cut surface and
therefore a projection onto data normal and tangential to that surface.
Distinguishing candidate projected coefficients requires a geometry where
$R$ and the normal Ricci component are not locked together, such as a
higher-dimensional spatial row or a time-dependent setting.
\end{principle}
