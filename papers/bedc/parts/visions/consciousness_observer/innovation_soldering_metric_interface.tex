The differential-geometry and physics vocabulary used throughout this
chapter --- including coframe, signature, one-form, tetrad, vielbein,
connection, curvature, metric, frame, and related terms --- is schematic
BEDC-internal bookkeeping analogy only. None of it asserts a spacetime
metric, a Lorentzian signature, a cotangent frame, tetrad gravity, a
principal connection, gauge curvature, or any physical differential-geometry
structure. Where such a name appears it labels a row-indexed BEDC ledger
pattern, not the geometric object of the same name, unless a corresponding
structure is separately constructed.


\autoref{ch:visions-section-comparison-section-bundle-boundary} separates the
fibre-indexed state face from the base-comparison face of an
observer-centered section bundle. That separation is intentionally weak:
it says which rows may coexist and which bridge rows are still missing.
The present chapter records a stronger BEDC-internal interface target. A
mere product bundle or direct-sum comparison row does not couple the
rows. The first genuine coupling is a coframe-shaped comparison row: the
orthogonal innovation of the ledger determines an innovation one-form,
and that one-form supplies a comparison pairing after the needed rank and
signature rows are displayed. This is not a Cartan solder form, not a
tetrad field, and not a spacetime metric unless the corresponding base
tangent object, associated fibre object, equivariant identification, and
nondegenerate bilinear form are explicitly constructed.

\section{Product Rows and Coframe-Shaped Rows}
\label{sec:innovation-coupling-product-versus-soldering}

\begin{definition}[Product-level coexistence]
\label{def:innovation-coupling-product-level-coexistence}
A section-bundle description is at \emph{product level} when its total
carrier and comparison row have the form
$$
\mathcal K_R\simeq\mathcal H_R\otimes L^2(\Sigma_{S,R})
$$
and
$$
D_\mu
=
D_\mu^{\mathcal H}
+D_\mu^{\Sigma}
+D_\mu^{\mathrm{base}},
$$
so that its displayed curvature is the sum of sector curvatures after
the cross commutators have been declared absent or untyped. Product
level is a coexistence statement. It is not yet a dynamical coupling
between Hilbert-style ledger rows, phase-coordinate rows, and
comparison-cost rows. The displayed curvature language is
connection-like only when typed transport and loop-composition rows are
also supplied.
\end{definition}

\begin{definition}[Coframe-shaped comparison coupling]
\label{def:innovation-coupling-level-coupling}
A section-bundle description is at \emph{coframe-shaped comparison level}
when the comparison pairing is not supplied independently of the ledger
field. Instead, there is an innovation one-form
$$
\Theta\in\Omega^1(M;\mathscr E)
$$
derived from the ledger field, and the pairing is read as
$$
g_{\mu\nu}=G_{AB}\Theta^A_{\mu}\Theta^B_{\nu}.
$$
The coupling is genuine because a change in the ledger field changes
$\Theta$, and a change in $\Theta$ changes the comparison row. The word
coframe marks only the shape of the row; it is not a Cartan or tetrad
soldering form without the standard bundle data.
\end{definition}

\begin{principle}[Product comparison is only a boundary row]
\label{prin:innovation-coupling-product-connection-boundary-row}
The combined comparison schema of
\autoref{def:section-comparison-combined-connection-row} is a useful boundary
syntax, but it is not by itself a unification theorem. If the loop defect
decomposes as a sectorwise sum, then the rows have been placed in one
ambient bookkeeping space without proving that one sector constrains the
other. A unification claim requires a row such as
\autoref{def:innovation-coupling-level-coupling}, where the comparison
pairing depends on the ledger innovation.
\end{principle}

\section{Continuous Innovation Form}
\label{sec:innovation-coupling-continuous-form}

\begin{definition}[Continuous ledger field]
\label{def:innovation-coupling-continuous-ledger-field}
Let $M$ be a continuous reference-context base and let
$$
\mathscr H\to M
$$
be a Hilbert ledger bundle. A continuous ledger field consists of a
section
$$
B\in\Gamma(\mathscr H),
$$
a closed ledger subbundle
$$
\mathscr L\subseteq\mathscr H,
$$
and the orthogonal projection
$$
P_{\mathscr L}:\mathscr H\to\mathscr L.
$$
The innovation projector is
$$
Q_{\mathscr L}:=I-P_{\mathscr L}.
$$
\end{definition}

\begin{definition}[Innovation one-form]
\label{def:innovation-coupling-covector-row}
Let $D_{\Phi}$ be a context-reshaping comparison operator on
$\mathscr H$. The \emph{innovation one-form} of the ledger field is
$$
\Theta:=Q_{\mathscr L}D_{\Phi}B.
$$
Equivalently, for $u\in T_xM$,
$$
\Theta_x(u)
=
Q_{\mathscr L,x}\bigl(D_{\Phi,u}B\bigr)(x).
$$
Its target bundle is the innovation bundle
$$
\mathscr E:=\mathscr L^{\perp}.
$$
The operator is connection-like only when the packet supplies the
identity, composition, and transport laws needed for a genuine
connection.
\end{definition}

\begin{definition}[Innovation comparison pairing]
\label{def:innovation-coupling-pairing-row}
Assume that the innovation bundle $\mathscr E$ carries a displayed
nondegenerate symmetric bilinear form
$$
G_x:\mathscr E_x\times\mathscr E_x\to\mathbb R.
$$
The innovation comparison pairing is
$$
g_x(u,v):=G_x(\Theta_x(u),\Theta_x(v)).
$$
In local notation,
$$
g_{\mu\nu}=G_{AB}\Theta^A_{\mu}\Theta^B_{\nu}.
$$
This row is not a spacetime metric, not Lorentzian by name alone, and
does not supply causal cones, geodesics, or Einstein equations.
\end{definition}

\begin{principle}[Positive Hilbert norm is not a spacetime metric]
\label{prin:innovation-coupling-hilbert-norm-not-lorentz-metric}
The Hilbert inner product controls orthogonality and projection in the
ledger. It cannot by itself be a Lorentzian spacetime metric, because it
is positive definite. A Lorentz-signature comparison row therefore
requires the additional signature data $G$ on the effective innovation
bundle. Without that row, one may define a positive quadratic size of an
innovation, but not a Lorentzian spacetime metric.
\end{principle}

\begin{theorem}[Innovation one-form gives a Lorentz-signature pairing]
\label{thm:innovation-coupling-covector-row-lorentz-metric}
Assume that, for every $x\in M$, the map
$$
\Theta_x:T_xM\to\mathscr E_x
$$
is injective and that $G_x$ restricted to $\Theta_x(T_xM)$ is
nondegenerate with Lorentz signature. Then
$$
g_x(u,v)=G_x(\Theta_x(u),\Theta_x(v))
$$
defines a nondegenerate Lorentz-signature bilinear form on $M$. Reading
it as a spacetime metric requires an additional physical spacetime
interpretation.
\end{theorem}

\begin{proof}
The map $\Theta_x$ is linear and $G_x$ is bilinear, so $g_x$ is bilinear.
Symmetry of $g_x$ follows from symmetry of $G_x$. If
$$
g_x(u,v)=0
$$
for every $v\in T_xM$, then
$$
G_x(\Theta_x(u),\Theta_x(v))=0
$$
for every $v$. Nondegeneracy of $G_x$ on $\Theta_x(T_xM)$ gives
$\Theta_x(u)=0$. Injectivity of $\Theta_x$ then gives $u=0$. Thus $g_x$
is nondegenerate, and its signature is the Lorentz signature displayed on
the image subspace.
\end{proof}

\begin{theorem}[Comparison variation factors through ledger innovation]
\label{thm:innovation-coupling-metric-variation-through-ledger}
For
$$
g_{\mu\nu}=G_{AB}\Theta^A_{\mu}\Theta^B_{\nu},
\qquad
\Theta=Q_{\mathscr L}D_{\Phi}B,
$$
a ledger variation $B\mapsto B+\delta B$ induces
$$
\delta g_{\mu\nu}
=
(\delta G_{AB})\Theta^A_{\mu}\Theta^B_{\nu}
+
G_{AB}(\delta\Theta^A_{\mu})\Theta^B_{\nu}
+
G_{AB}\Theta^A_{\mu}(\delta\Theta^B_{\nu}),
$$
where
$$
\delta\Theta=\delta(Q_{\mathscr L}D_{\Phi}B).
$$
If $G$ is fixed, the variation reduces to the symmetric cross term
between $\delta\Theta$ and $\Theta$.
\end{theorem}

\begin{proof}
Differentiate the defining expression
$$
g_{\mu\nu}=G_{AB}\Theta^A_{\mu}\Theta^B_{\nu}.
$$
The product rule gives the displayed three terms. The final statement is
the same formula with $\delta G=0$.
\end{proof}

\section{Ledger-Step Reflection and Pythagorean Coding Closure}
\label{sec:innovation-coupling-tick-reflection-pcc}

\begin{definition}[Ledger-step covector and transverse innovation image]
\label{def:innovation-coupling-tick-covector-spatial-image}
Let
$$
A:M\to\mathbb R_{>0}
$$
be a displayed multiplicative scale row. Its logarithmic ledger-step
coordinate and ledger-step covector are
$$
T:=\log A,
\qquad
\tau:=dT.
$$
At a point $x$ with $\tau_x\ne0$, set
$$
S_x:=\ker\tau_x\subset T_xM.
$$
The spatial innovation image is
$$
E_x^S:=\Theta_x(S_x)\subset\mathscr E_x.
$$
This row is part of the Pythagorean coding closure reading: the scale
ledger selects a stage co-direction, while the innovation map supplies
the orthogonal readout surface. It is not a physical clock algebra, not
proper time, and not a quantum time observable.
\end{definition}

\begin{definition}[Ledger-step residual line and reflection]
\label{def:innovation-coupling-tick-residual-line-reflection}
Assume $\operatorname{rank}(\Theta_x|_{S_x})=3$, and let $\Pi_S^E$ be
the Hilbert orthogonal projection onto $E_x^S$. For any
$X_x\in T_xM$ satisfying
$$
\tau_x(X_x)=1,
$$
define the ledger-step residual
$$
r_T(x):=(I-\Pi_S^E)\Theta_x(X_x).
$$
When $r_T(x)\ne0$, the ledger-step residual line is
$$
E_x^T:=\mathbb R\,r_T(x),
$$
and the ledger-step reflection on
$$
E_x^4:=E_x^T\oplus E_x^S
$$
is
$$
J_T:=\Pi_T^E-\Pi_S^E,
$$
where $\Pi_T^E$ is the Hilbert orthogonal projection onto $E_x^T$.
\end{definition}

\begin{lemma}[Ledger-step residual line is independent of the lift]
\label{lem:innovation-coupling-tick-residual-lift-independent}
The vector $r_T(x)$ of
\autoref{def:innovation-coupling-tick-residual-line-reflection} does
not depend on the choice of $X_x$ satisfying $\tau_x(X_x)=1$.
\end{lemma}

\begin{proof}
If $X_x$ and $X'_x$ both satisfy
$$
\tau_x(X_x)=\tau_x(X'_x)=1,
$$
then
$$
\tau_x(X_x-X'_x)=0.
$$
Hence $X_x-X'_x\in S_x$, so
$$
\Theta_x(X_x-X'_x)\in E_x^S.
$$
Applying $I-\Pi_S^E$ kills this vector, and therefore
$$
(I-\Pi_S^E)\Theta_x(X_x)
=
(I-\Pi_S^E)\Theta_x(X'_x).
$$
\end{proof}

\begin{definition}[Reflected innovation tensor]
\label{def:innovation-coupling-signed-innovation-tensor}
On the four-dimensional innovation image $E_x^4$, define
$$
\mathfrak Q^T_x(u,v)
:=
\langle
\Theta_x(u),
J_T\Theta_x(v)
\rangle.
$$
Its real and imaginary reads are
$$
g_x^T(u,v):=\operatorname{Re}\mathfrak Q^T_x(u,v),
$$
and
$$
F_x^\eta(u,v):=2\operatorname{Im}\mathfrak Q^T_x(u,v).
$$
The first row is the Pythagorean comparison read; the second row is the
phase-commutator read carried by the same innovation tensor.
\end{definition}

\begin{theorem}[Ledger-step reflected Gram row has Lorentz signature]
\label{thm:innovation-coupling-tick-reflected-gram-lorentz}
Assume $\dim M=4$, $\tau_x\ne0$,
$$
\operatorname{rank}(\Theta_x|_{\ker\tau_x})=3,
$$
and $r_T(x)\ne0$. Then
$$
g_x^T(u,v)
=
\operatorname{Re}
\langle
\Theta_x(u),
J_T\Theta_x(v)
\rangle
$$
is a nondegenerate symmetric bilinear form on $T_xM$ with signature
$(1,3)$.
\end{theorem}

\begin{proof}
Choose any $X_x$ with $\tau_x(X_x)=1$. Since
$\Pi_S^E\Theta_x(X_x)$ lies in
$$
E_x^S=\Theta_x(S_x),
$$
there is $s_X\in S_x$ with
$$
\Theta_x(s_X)=\Pi_S^E\Theta_x(X_x).
$$
Set
$$
\mathcal T_x:=X_x-s_X.
$$
Then $\tau_x(\mathcal T_x)=1$ and
$$
\Theta_x(\mathcal T_x)=r_T(x).
$$
Every $u\in T_xM$ decomposes as
$$
u=\tau_x(u)\mathcal T_x+s,
\qquad
s\in S_x.
$$
Thus
$$
\Theta_x(u)=\tau_x(u)r_T+\Theta_x(s),
$$
with $r_T\perp E_x^S$. Since $J_T$ is $+I$ on $E_x^T$ and $-I$ on
$E_x^S$,
$$
g_x^T(u,u)
=
\tau_x(u)^2\|r_T\|^2
-
\|\Theta_x(s)\|^2.
$$
The first coefficient is positive because $r_T\ne0$. The second term is
negative definite on the three-dimensional spatial image because
$\Theta_x|_{S_x}$ has rank $3$. Symmetry follows from self-adjointness of
$J_T$. Hence the signature is $(1,3)$.
\end{proof}

\begin{theorem}[Real and imaginary rows share one innovation source]
\label{thm:innovation-coupling-real-imaginary-shared-source}
If $J_T$ is self-adjoint, then
$$
\mathfrak Q^T_x(v,u)=\overline{\mathfrak Q^T_x(u,v)}.
$$
Consequently $g_x^T=\operatorname{Re}\mathfrak Q^T_x$ is symmetric and
$F_x^\eta=2\operatorname{Im}\mathfrak Q^T_x$ is antisymmetric.
\end{theorem}

\begin{proof}
By definition,
$$
\mathfrak Q^T_x(u,v)
=
\langle
\Theta_x(u),
J_T\Theta_x(v)
\rangle.
$$
Self-adjointness of $J_T$ gives
$$
\langle
\Theta_x(v),
J_T\Theta_x(u)
\rangle
=
\overline{
\langle
\Theta_x(u),
J_T\Theta_x(v)
\rangle
}.
$$
Taking real and imaginary parts gives the stated symmetry and
antisymmetry rows.
\end{proof}

\label{def:innovation-coupling-discrete-stencil}

\begin{principle}[One ledger step is not a four-dimensional metric]
\label{prin:innovation-coupling-one-tick-not-full-metric}
A single OCLSD residual supplies one innovation direction. A
four-dimensional metric requires a local stencil of independent
