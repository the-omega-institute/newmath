\chapter{Quantum and Relativistic Section-Bundle Boundary}
\label{ch:visions-quantum-gr-section-bundle-boundary}
\origin{ai}

This chapter records a bundle reading of the
observer-centered section dynamics of
\autoref{ch:visions-observer-centered-logarithmic-section-dynamics}. It
does not derive quantum mechanics, general relativity, or quantum gravity.
Its purpose is to state the interface boundary in which those theories
would enter BEDC: quantum grammar occupies the vertical fibre of a chosen
frame, while relativistic grammar occupies the horizontal comparison of
neighbouring complete sections.

\section{Vertical and Horizontal Faces}
\label{sec:quantum-gr-vertical-horizontal-faces}

\begin{definition}[Section-bundle packet]
\label{def:quantum-gr-section-bundle-packet}
A \emph{section-bundle packet} consists of a reference-frame base
$\mathcal R$, a world Hilbert bundle
$$
\{\mathcal H_R\}_{R\in V_{\mathcal R}},
$$
solenoid phase fibres $\Sigma_{S,R}$ over displayed frame rows, frame
reshaping rows
$$
\Phi_{R'\leftarrow R}:\mathcal H_R\to\mathcal H_{R'},
$$
and ledger rows recording projection, residual, visible readout, and
transport data. Its typical fibre over $R$ is read schematically as
$$
\mathcal H_R\otimes\Sigma_{S,R}.
$$
The pair-indexed notation $\Phi_{R'\leftarrow R}$ is well-defined only
when the reshaping data is path-independent: either the comparison base is
thin (at most one comparison row between any two frames) or a canonical
transport selector is fixed. In general there may be several distinct
comparison rows from $R$ to $R'$, so an honest treatment indexes the
reshaping by the comparison morphism, $\Phi_f$, and a pair-indexed
$\Phi_{R'\leftarrow R}$ silently assumes away holonomy. This
well-definedness dichotomy is discharged as a finite structural fact:
\leanstmt{BEDC.Derived.Visions.ObservationSectionCoherence.pairIndexed\_wellDefined\_iff\_thin\_or\_canonical}
\end{definition}

\begin{definition}[Vertical quantum face]
\label{def:quantum-gr-vertical-quantum-face}
The \emph{vertical quantum face} of a section-bundle packet is the
collection of rows internal to one chosen frame fibre:
$$
\psi_R\in\mathcal H_R,\qquad
\widetilde t_R\in\Sigma_{S,R},
$$
orthogonal axis packets, phase rows, projection rows, Born-weight rows
when displayed, and section-style measurement updates. It is the
fibre-internal grammar already constrained by
\autoref{sec:quantum-measurement-generation-interface}.
\end{definition}

\begin{definition}[Horizontal relativistic face]
\label{def:quantum-gr-horizontal-relativistic-face}
The \emph{horizontal relativistic face} of a section-bundle packet is the
collection of rows comparing neighbouring complete sections:
$$
R'\sim R,
\qquad
\Phi_{R'\leftarrow R},
\qquad
B_{R'}-\Phi_{R'\leftarrow R}(B_R),
$$
together with visible stability bounds, inter-history invariants,
maximum-rate rows when displayed, and any continuous connection or
curvature row introduced over the frame base. It is not a physical signal
inside one fibre; it is the comparison discipline between frame-relative
sections.
\end{definition}

\begin{principle}[Quantum vertical, relativity horizontal]
\label{prin:quantum-gr-vertical-horizontal-reading}
In this packet language, quantum mechanics is read as vertical fibre
dynamics: phase, superposition, projection, and orthogonal residual
admission inside a chosen frame. Relativistic geometry is read as
horizontal base geometry: connection, metric, visible causal rate, and
curvature governing how neighbouring complete sections are compared.
\end{principle}

\section{Quantum Fibre Reading}
\label{sec:quantum-gr-quantum-fibre-reading}

\begin{definition}[Frame-relative quantum state]
\label{def:quantum-gr-frame-relative-quantum-state}
A \emph{frame-relative quantum state} is a state row
$$
\psi_R\in\mathcal H_R
$$
together with the phase and ledger rows required by the chosen
Hilbert-style axis packet. If a displayed orthogonal family
$$
E=\{e_a:a\in A\}
$$
is available, the state may be written at that resolution as
$$
\psi_R=\sum_{a\in A}c_a e_a.
$$
The basis elements are admissible readout directions inside the selected
frame fibre, not absolute local facts outside the packet.
\end{definition}

\begin{principle}[Unitary evolution is log-time phase transport]
\label{prin:quantum-gr-unitary-log-time-transport}
When a multiplicative accumulation row
$$
A_{n+1}=\lambda_{n+1}A_n
$$
is read through
$$
T_n=\log A_n,
$$
a frame-fibre Schrödinger row is a log-time phase transport row:
$$
i\hbar\frac{\partial\psi}{\partial T}=H\psi.
$$
Equivalently, on the multiplicative scale,
$$
i\hbar A\frac{\partial\psi}{\partial A}=H\psi.
$$
This is a coordinate reading of the time parameter. It does not supply
the Hamiltonian, Hilbert-space physical laws, or empirical calibration
rows.
\end{principle}

\begin{definition}[Measurement as fibre projection with residual ledger]
\label{def:quantum-gr-measurement-fibre-projection-residual}
For a displayed projection family $\{P_a\}$ in frame $R$, a measurement
read at outcome $a$ has the section-style form
$$
\psi_R
\longmapsto
\frac{P_a\psi_R}{\|P_a\psi_R\|}
\oplus
\eta_a
$$
when the denominator condition, post-outcome carrier, residual row, and
ledger replay are displayed. The residual must satisfy the appropriate
orthogonality row against the prior accepted ledger.
\end{definition}

\begin{principle}[Collapse is not a propagation row]
\label{prin:quantum-gr-collapse-not-propagation}
The section-style update of
\autoref{def:quantum-gr-measurement-fibre-projection-residual} is a
licensed update of the frame-relative description. It is not, by itself,
a controllable signal propagating inside the selected world-section. A
claim of physical signalling requires an internal propagation row of the
kind refused by
\autoref{prin:observer-centered-reshaping-not-propagation} unless a
separate bridge is supplied.
\end{principle}

\begin{principle}[Entanglement as non-product section residue]
\label{prin:quantum-gr-entanglement-nonproduct-section-residue}
An entangled pair is read as a non-product residue of one complete
section, not as two independently complete worlds joined by a hidden
signal. This refines
\autoref{def:quantum-entanglement-nonproduct-residue}: local outcomes may
be displayed, but the residual fibre need not split as a product of
local residual fibres.
\end{principle}

\section{Relativistic Base Reading}
\label{sec:quantum-gr-relativistic-base-reading}

\begin{definition}[Residual metric candidate]
\label{def:quantum-gr-residual-metric-candidate}
For neighbouring frame rows $R'\sim R$ and a section $B_R$, define the
horizontal residual row
$$
\eta_R(R')
:=
B_{R'}-\Phi_{R'\leftarrow R}(B_R)
$$
whenever both sections and the reshaping row are displayed. A
\emph{residual metric candidate} is a visible quadratic read
$$
g_R(\delta R,\delta R)
:=
\langle
\eta_R(\delta R),\eta_R(\delta R)
\rangle_{\mathrm{vis}}
$$
after the packet supplies the visible projection, signature convention,
and locality certificate needed for the read.
\end{definition}

\begin{principle}[Light-cone bounds are internal propagation bounds]
\label{prin:quantum-gr-light-cone-internal-propagation}
Relativistic maximum-rate bounds constrain visible propagation rows
inside or between displayed histories. They do not constrain bare frame
rewriting as a coordinate or ledger comparison. Thus a global rewrite of
a description is not a superluminal signal unless it also supplies a
controllable internal propagation row.
\end{principle}

\begin{definition}[Frame connection and curvature row]
\label{def:quantum-gr-frame-connection-curvature-row}
In a continuous frame-base reading, a frame connection row is a
comparison operator $\nabla_\mu$ transporting fibre data over the base.
Its curvature row is the noncommutativity
$$
[\nabla_\mu,\nabla_\nu]=\mathcal R_{\mu\nu}.
$$
The row says that transport around an infinitesimal loop can return with
nontrivial holonomy. It is the horizontal obstruction to globally
flattening the frame reshaping discipline.
\end{definition}

\begin{principle}[Equivalence principle as local gauge flattening]
\label{prin:quantum-gr-equivalence-local-gauge-flattening}
The local equivalence-principle reading is that near a frame row $R$ one
may choose a local gauge in which the first-order reshaping row is
trivial:
$$
\Phi_{R+dR\leftarrow R}=I+O(dR^2).
$$
Curvature remains visible through loop holonomy; local gauge flattening
does not erase the horizontal curvature row.
\end{principle}

\section{Unified Boundary}
\label{sec:quantum-gr-unified-boundary}

\begin{definition}[Combined connection row]
\label{def:quantum-gr-combined-connection-row}
A speculative combined connection row over a section-bundle packet has
the schematic form
$$
D_\mu
:=
\partial_\mu+\Gamma_\mu+\frac{i}{\hbar}A_\mu+\mathcal A_\mu^{\mathrm{sol}},
$$
where $\Gamma_\mu$ is a frame-reshaping connection, $A_\mu$ is a visible
phase gauge row, and $\mathcal A_\mu^{\mathrm{sol}}$ is a solenoid
prime-register fibre connection. Its curvature decomposes schematically
as
$$
[D_\mu,D_\nu]
=
\mathcal R_{\mu\nu}
+
\frac{i}{\hbar}F_{\mu\nu}
+
\mathcal F_{\mu\nu}^{\mathrm{sol}}.
$$
This is an interface target, not a closed physical theory.
\end{definition}

\begin{principle}[Conflict of times becomes a bundle-coordinate problem]
\label{prin:quantum-gr-time-conflict-bundle-coordinate-problem}
The usual tension between a fixed quantum time parameter and dynamical
relativistic time is read here as a coordinate mismatch. The quantum time
row is a log-time parameter along a selected fibre evolution; the
relativistic time row is part of the visible geometry of the frame base.
Neither is an external absolute time object in the packet.
\end{principle}

\begin{theorem}[Section-bundle separation of quantum and relativistic roles]
\label{thm:quantum-gr-section-bundle-role-separation}
In any section-bundle packet, vertical rows can state phase, projection,
Born-weight, and non-product residue obligations without specifying a
base curvature row. Horizontal rows can state frame reshaping, visible
stability, maximum-rate, metric-candidate, and curvature obligations
without choosing a Born-weight layer. A physical unification claim
therefore requires bridge rows relating the two faces; it is not
obtained merely by naming both in one packet.
\end{theorem}

\begin{proof}
The vertical face of
\autoref{def:quantum-gr-vertical-quantum-face} is defined using rows
inside one fibre over a frame. Those rows do not include the horizontal
connection or curvature data of
\autoref{def:quantum-gr-horizontal-relativistic-face}. Conversely, the
horizontal face is defined from neighbouring frame comparisons and visible
stability data; those rows do not include a Born-weight layer or a
measurement-update section. Therefore each face is well typed without the
other. A unification claim needs additional compatibility rows, such as
the combined connection target of
\autoref{def:quantum-gr-combined-connection-row}.
\end{proof}

\begin{principle}[Boundary status]
\label{prin:quantum-gr-section-bundle-boundary-status}
The chapter licenses one compressed interpretation: quantum mechanics
uses the vertical grammar of phase, superposition, projection, and
orthogonal residual ledger inside a selected frame fibre; general
relativity uses the horizontal grammar of frame comparison, connection,
metric readout, and curvature over the base. Quantum gravity would be a
certificate showing that these grammars share one ledgered connection
discipline with empirical prediction rows.
\end{principle}
