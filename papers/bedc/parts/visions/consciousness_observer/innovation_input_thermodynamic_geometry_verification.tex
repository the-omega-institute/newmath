\section{Cone Slot and the Two Twelfths}
\label{sec:innovation-input-thermo-verification-cone-slot}

\begin{theorem}[Cone coefficient and curvature slot]
\label{thm:innovation-input-thermo-cone-slot}
Let
$$
s(\alpha)
=
\frac{1}{12}
\left(
\frac{2\pi}{\alpha}
-
\frac{\alpha}{2\pi}
\right)
$$
be the Sommerfeld cone coefficient. In the entanglement limit
$\alpha=2\pi-\delta$, its linear term is $\delta/(12\pi)$. The
distributional cone curvature satisfies $\int R=2\delta$, and the smooth
DeWitt slot $a_1=R/6$ contributes
$$
\frac{1}{4\pi}\frac{2\delta}{6}
=
\frac{\delta}{12\pi}.
$$
Thus the massive area-law coefficient $-1/12$ and the local
heat-geometry correction $+R/12$ occupy the same linear heat-kernel slot.
\end{theorem}

\begin{proof}
The displayed expansion gives
$$
s(2\pi-\delta)
=
\frac{\delta}{12\pi}+O(\delta^2).
$$
The DeWitt contribution of a localized cone curvature is the second
displayed expression. The two linear coefficients are identical.
\end{proof}

\begin{corollary}[Curvature-mass slot target]
\label{cor:innovation-input-thermo-curvature-mass-slot-target}
For minimal coupling, scalar curvature and mass-squared enter the same
heat-kernel slot with opposite signs. The entanglement row therefore has
the numerical target
$$
\frac{\partial S_{\rm EE}}{\partial R}
=
-
\frac16
\frac{\partial S_{\rm EE}}{\partial m^2}
$$
in a regulator comparison where both derivatives are measured in the same
units.
\end{corollary}

\begin{remark}[Linearity scope]
\label{rem:innovation-input-thermo-cone-linearity-scope}
The identification above is a linear-slot statement. Entanglement entropy
uses this linear cone term, while higher Renyi rows probe nonlinear cone
dependence and need not share the same coefficient identity.
\end{remark}

\begin{theorem}[Homogeneous trace expansion]
\label{thm:innovation-input-thermo-homogeneous-trace-expansion}
For a homogeneous smooth geometry,
$$
\operatorname{tr}h
=
\frac{d}{2T}
-
\frac{R}{6}
-
T
\left[
\frac{|\operatorname{Riem}|^2-|\operatorname{Ric}|^2}{90}
+
\frac{\square R}{15}
\right]
+
O(T^2).
$$
The $R^2$ term cancels between the logarithm and the derivative of the
heat-kernel diagonal.
\end{theorem}

\begin{proof}
Use
$$
\operatorname{tr}h=-\partial_T\log K_T(z,z)
$$
and insert the standard $a_2$ coefficient of the heat-kernel diagonal. The
logarithm subtracts the square of the $a_1$ term, eliminating the pure
$R^2$ contribution and leaving the displayed combination.
\end{proof}

\begin{theorem}[Spherical-family consistency constraints]
\label{thm:innovation-input-thermo-spherical-family-consistency}
In the full DeWitt next-order tensor ledger, comparison with the
homogeneous trace identity on the family of standard spheres $S^d$ imposes
four dimension-independent coefficient relations. In the coefficient
coordinates used for the tensor ledger these include
$$
c_5=0,
\qquad

$$
The same relations force the two-dimensional scalar contraction
$$
\frac14(c_6+c_8)+\frac12c_9+c_{10}
=
-\frac1{360}.
$$
Thus the two-dimensional coefficient $\gamma$ in
\autoref{def:innovation-input-thermo-two-dimensional-next-order-basis} is
the spherical-family contraction of the full tensor ledger.
\end{theorem}

\begin{proof}
On a standard sphere, $\nabla R=0$ and all derivative curvature terms vanish.
The homogeneous trace expansion of
\autoref{thm:innovation-input-thermo-homogeneous-trace-expansion} is then
a polynomial identity in the dimension $d$, because the contractions
$|\operatorname{Riem}|^2$ and $|\operatorname{Ric}|^2$ have explicit
polynomial forms in $d$. Matching powers of $d$ against the full tensor
basis gives the displayed relations and the stated two-dimensional
contraction.
\end{proof}

\begin{theorem}[Three-sphere zero trace channel]
\label{thm:innovation-input-thermo-three-sphere-zero-channel}
For the unit standard three-sphere,
$$
|\operatorname{Riem}|^2=|\operatorname{Ric}|^2,
\qquad
\square R=0.
$$
Hence the homogeneous next-order trace coefficient in
\autoref{thm:innovation-input-thermo-homogeneous-trace-expansion}
vanishes exactly.
\end{theorem}

\begin{proof}
For a unit standard sphere of dimension $d$,
$$
|\operatorname{Riem}|^2=2d(d-1),
\qquad
|\operatorname{Ric}|^2=d(d-1)^2.
$$
These are equal at $d=3$, and $R$ is constant, so the $\square R$ term
also vanishes.
\end{proof}
