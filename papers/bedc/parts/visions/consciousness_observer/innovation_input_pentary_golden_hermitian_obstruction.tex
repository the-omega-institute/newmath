\section{Golden Symmetric Word}
\label{sec:innovation-input-pentary-golden-symmetric-word}

\begin{definition}[Golden symmetric pentary word]
\label{def:innovation-input-golden-symmetric-pentary-word}
Let
$$
c:=\frac{\sqrt5-3}{2},
$$
choose $\theta\in(0,\pi)$ with $\cos\theta=c$, and set
$$
r:=e^{i\theta}.
$$
Define
$$
v_\diamond:=(1,\overline r,1,r,1).
$$
\end{definition}

\begin{theorem}[Golden symmetric row is Q3]
\label{thm:innovation-input-golden-symmetric-row-qthree}
The row $v_\diamond$ lies in $\mathcal Q_5$.
\end{theorem}

\begin{proof}
Since $r+\overline r=2c$,
$$
1+\overline r+1+r+1
=
3+2c
=
3+2\frac{\sqrt5-3}{2}
=
\sqrt5.
$$
Thus the Q3 equation holds.
\end{proof}

\begin{theorem}[Golden symmetric row passes the link-diagonal filter]
\label{thm:innovation-input-golden-symmetric-link-diagonal-filter}
For $v_\diamond$,
$$
\mu=\frac{\sqrt5-3}{2}
$$
and
$$
\nu=D_{12}=D_{13}=D_{23}
=
3-\frac{6\sqrt5}{5}.
$$
Numerically,
$$
\mu\approx -0.381966
$$
and
$$
\nu=D_{12}=D_{13}=D_{23}\approx0.316718.
$$
Consequently,
$$
D_{12}\ge|\mu|^2,
\qquad
D_{13}\ge|\nu|^2,
\qquad
D_{23}\ge|\mu|^2.
$$
\end{theorem}

\begin{proof}
The edge ratios of $v_\diamond$ are
$$
x_0=r,
\qquad
x_1=\overline r,
\qquad
x_2=\overline r,
\qquad
x_3=r,
\qquad
x_4=1.
$$
\autoref{thm:innovation-input-pentary-first-correlation-formula} gives
$$
\mu
=
\frac{r+\overline r+\overline r+r}{5-1}
=
\frac{r+\overline r}{2}
=
c.
$$
\autoref{thm:innovation-input-pentary-second-correlation-formula} gives
$$
\nu
=
\frac15
\left[
2+\overline r^2+2r\mu
\right].
$$
Because $2r\mu=r(r+\overline r)=r^2+1$,
$$
\nu
=
\frac15(3+r^2+\overline r^2)
=
\frac15(3+2\cos2\theta).
$$
Using $\cos2\theta=2c^2-1$ and
$$
c^2
=
\left(\frac{\sqrt5-3}{2}\right)^2
=
\frac{7-3\sqrt5}{2},
$$
one obtains
$$
\nu
=
\frac15(1+4c^2)
=
3-\frac{6\sqrt5}{5}.
$$
Substituting the same edge ratios into
\autoref{thm:innovation-input-pentary-dtwelve-formula},
\autoref{thm:innovation-input-pentary-dthirteen-formula}, and
\autoref{thm:innovation-input-pentary-dtwentythree-correlation-meeting}
gives $D_{12}=D_{13}=D_{23}=\nu$.

Finally,
$$
|\mu|^2
=
\left(\frac{\sqrt5-3}{2}\right)^2
\approx0.145898
$$
and
$$
|\nu|^2\approx0.100310.
$$
Both are smaller than
$3-6\sqrt5/5\approx0.316718$, giving the displayed link inequalities.
\end{proof}

\section{Hermitian Obstruction}
\label{sec:innovation-input-pentary-hermitian-obstruction}

\begin{definition}[Anchored $L=3$ Hermitian defect]
\label{def:innovation-input-anchored-lthree-hermitian-defect}
Let $M_3(v)$ be the anchored $L=3$ neutral RP matrix associated to the
neutral monomials generated by $1,F_{12},F_{13},F_{23}$. Define
$$
\mathcal H_3(v):=M_3(v)-M_3(v)^*.
$$
An anchored $L=3$ RP candidate must have
$$
\mathcal H_3(v)=0
$$
before the positive-semidefinite condition can even be tested as a
Hermitian matrix condition.
\end{definition}

\begin{theorem}[Golden symmetric Hermitian obstruction]
\label{thm:innovation-input-golden-symmetric-hermitian-obstruction}
For $v_\diamond=(1,\overline r,1,r,1)$, the two crossed entries
$$
C_{12,13}:=\langle\theta F_{12},F_{13}\rangle
$$
and
$$
C_{13,12}:=\langle\theta F_{13},F_{12}\rangle
$$
satisfy
$$
C_{12,13}=C_{13,12}
=
\frac{3c+2c^3}{5}
+i\frac{2s^3}{5},
$$
where $c=\cos\theta=(\sqrt5-3)/2$ and $s=\sin\theta>0$. Hence
$$
C_{12,13}\ne\overline{C_{13,12}},
$$
and the anchored $L=3$ neutral RP matrix is not Hermitian.
\end{theorem}

\begin{proof}
After translation normalization,
$$
C_{12,13}
=
\Gamma(\delta_0-\delta_1+\delta_2-\delta_4)
$$
and
$$
C_{13,12}
=
\Gamma(\delta_0-\delta_2+\delta_3-\delta_4).
$$
For $v_\diamond$, the carry recursion for each of these two blocks closes
on the zero block and the one-step block. The five residues give the same
average:
$$
C_{12,13}
=
C_{13,12}
=
\frac15
\left[
2r+(2+\overline r^2)\mu
\right].
$$
Since $\mu=c$ and $r=c+is$,
$$
2r+(2+\overline r^2)c
=
2(c+is)
+c\left(2+c^2-s^2-2ics\right).
$$
Using $c^2+s^2=1$, this becomes
$$
3c+2c^3+i\,2s^3.
$$
Therefore the crossed entry has imaginary part $2s^3/5$, which is
nonzero. Since the reverse crossed entry is the same nonreal number, the
Hermitian relation $M_{12,13}=\overline{M_{13,12}}$ fails.
\end{proof}

\begin{corollary}[Diagonal passage does not imply Hermitian passage]
\label{cor:innovation-input-diagonal-passage-not-hermitian-passage}
There exists a pentary Q3 row satisfying all three anchored $L=3$
link-diagonal necessary inequalities while failing the Hermitian condition
for the same window.
\end{corollary}

\begin{proof}
Use $v_\diamond$. The link-diagonal inequalities are
\autoref{thm:innovation-input-golden-symmetric-link-diagonal-filter},
while the Hermitian failure is
\autoref{thm:innovation-input-golden-symmetric-hermitian-obstruction}.
\end{proof}

\begin{principle}[Pentary $L=3$ RP gates]
\label{prin:innovation-input-pentary-lthree-rp-gates}
For pentary self-substitution rows, anchored $L=3$ neutral RP has three
distinct finite-window gates:
$$
\text{link diagonals are real and nonnegative,}
$$
$$
M_3(v)=M_3(v)^*,
$$
and
$$
M_3(v)\succeq0.
$$
The Pythagorean row $v_\star$ fails the first gate because
$D_{23}=-1/5$. The golden symmetric row $v_\diamond$ passes the first
gate but fails the Hermitian gate. Thus diagonal positivity is not the
reflection-symmetry content of RP; it is only the first scalar shadow of
the finite-window matrix condition.
\end{principle}

\begin{principle}[Pentary search stratification]
\label{prin:innovation-input-pentary-search-stratification}
The finite-window stratification visible at $N=5$ is:
$$
\begin{array}{c|c|c|c|c|c}
\text{row} & \text{Q3} & \text{two-point} & L=2\text{ RP} &
L=3\text{ link} & L=3\text{ Hermitian}\\
\hline
v_\star=(a,b,-a,b,a) &
\text{yes} & \text{real} & \text{pass} &
\text{fail} & \text{not tested}\\
v_5^{(+)} &
\text{yes} & \text{real} & \text{fail} &
\text{not tested} & \text{not tested}\\
v_5^{(-)} &
\text{yes} & \text{real} & \text{fail} &
\text{not tested} & \text{not tested}\\
v_\diamond=(1,\overline r,1,r,1) &
\text{yes} & \text{real} & \text{diagonal pass} &
\text{pass} & \text{fail}
\end{array}
$$
The unresolved finite-window problem is therefore not only to maximize
the link diagonals, but to solve the Hermitian equations and then test
the least eigenvalue inside that Hermitian locus.
\end{principle}
