\chapter{Innovation Input Anchored RP Golden Palindrome Boundary}
\label{ch:visions-innovation-input-anchored-rp-golden-palindrome-boundary}
\origin{ai}

\autoref{ch:visions-innovation-input-finite-window-rp-boundary} fixes the
finite-window RP matrix as an anchored future-window construction. This
chapter applies that anchored audit to the lowest-complexity pentary
palindrome family. The family removes the two-point arrow, but the
smallest neutral anchored RP window still detects failure.

\section{Anchored Window Discipline}
\label{sec:innovation-input-anchored-window-discipline}

\begin{principle}[Anchored RP workflow]
\label{prin:innovation-input-anchored-rp-workflow}
Finite-window RP entries are not functions of future-block translation
classes before reflection. The workflow is
$$
\text{anchored future block}
\to
\text{reflection}
\to
\text{difference}
\to
\text{common translation}
\to
\Gamma\text{ evaluation}.
$$
Only the last step uses translation invariance of finite-block
correlations.
\end{principle}

\begin{principle}[Ternary anchored status]
\label{prin:innovation-input-ternary-anchored-status}
For the ternary Q3 row
$$
v=(e^{i\pi/6},e^{-i\pi/2},e^{i\pi/6}),
$$
the anchored $L=2$ neutral window generated by
$$
1,\qquad
u_{n+1}\overline{u_{n+2}},
\qquad
\overline{u_{n+1}}u_{n+2}
$$
has matrix
$$
\left(
\begin{array}{ccc}
1 & -\frac12 & -\frac12\\
-\frac12 & \frac12 & 0\\
-\frac12 & 0 & \frac12
\end{array}
\right),
$$
with eigenvalues $0$, $1/2$, and $3/2$. Hence the ternary row passes the
anchored $L=2$ neutral window. It still fails at anchored $L=3$ by
\autoref{thm:innovation-input-ternary-neutral-rp-obstruction}.
\end{principle}

\begin{principle}[Quartic anchored status]
\label{prin:innovation-input-quartic-anchored-status}
For the quartic real-sign row $v=(1,1,1,-1)$, the anchored observable
$$
F=u_{n+1}\overline{u_{n+2}}
$$
has
$$
\langle\theta F,F\rangle
=
\Gamma(\delta_0-\delta_1+\delta_2-\delta_3)
=
-\frac35.
$$
Thus the quartic representative fails the anchored $L=2$ neutral window.
\end{principle}

\section{Pentary Palindromes}
\label{sec:innovation-input-pentary-palindromes}

\begin{definition}[Pentary palindrome phase word]
\label{def:innovation-input-pentary-palindrome-phase-word}
A pentary palindrome phase word is a row
$$
v=(a,b,c,b,a)\in\mathbb T^5.
$$
It is Q3-compatible exactly when
$$
2a+2b+c=\sqrt5.
$$
\end{definition}

\begin{theorem}[Pentary palindromes have real two-point coefficients]
\label{thm:innovation-input-pentary-palindrome-real-two-point}
For every pentary palindrome $v=(a,b,c,b,a)$, the word-level two-point
coefficients
$$
s_r=\sum_{j=0}^{4-r}v_{j+r}\overline{v_j}
$$
are real for $1\le r\le4$. Consequently every Q3-compatible pentary
palindrome is two-point weakly reversible.
\end{theorem}

\begin{proof}
Directly,
$$
s_1
=
b\overline a+c\overline b+b\overline c+a\overline b
=
2\operatorname{Re}(b\overline a)
+
2\operatorname{Re}(c\overline b),
$$
so $s_1$ is real. Also
$$
s_2
=
c\overline a+b\overline b+a\overline c
=
1+2\operatorname{Re}(c\overline a),
$$
$$
s_3
=
b\overline a+a\overline b
=
2\operatorname{Re}(b\overline a),
$$
and
$$
s_4=a\overline a=1.
$$
Thus all $s_r$ are real, and
\autoref{thm:innovation-input-real-word-level-criterion} gives real
two-point correlations.
\end{proof}

\begin{corollary}[Pentary palindrome two-point reversibility]
\label{cor:innovation-input-pentary-palindrome-two-point-reversible}
Every Q3-compatible pentary palindrome satisfies
$$
\gamma(m)\in\mathbb R
\qquad
\text{for all }m\ge0.
$$
\end{corollary}

\section{Golden Palindrome Branches}
\label{sec:innovation-input-golden-palindrome-branches}

\begin{definition}[Golden palindrome branches]
\label{def:innovation-input-golden-palindrome-branches}
Let
$$
\xi:=e^{\pi i/5},
\qquad
\zeta:=e^{2\pi i/5}.
$$
Define
$$
v_5^{(+)}
:=
(\xi,\xi^{-1},-1,\xi^{-1},\xi)
$$
and
$$
v_5^{(-)}
:=
(\zeta,\zeta^{-1},1,\zeta^{-1},\zeta).
$$
\end{definition}

\begin{theorem}[Golden palindrome Q3 rows]
\label{thm:innovation-input-golden-palindrome-qthree-rows}
Both golden palindrome branches lie in $\mathcal Q_5$, and both have real
two-point correlations.
\end{theorem}

\begin{proof}
The trigonometric identities
$$
\cos\frac{\pi}{5}=\frac{\sqrt5+1}{4}
$$
and
$$
\cos\frac{2\pi}{5}=\frac{\sqrt5-1}{4}
$$
give
$$
2\xi+2\xi^{-1}-1
=
4\cos\frac{\pi}{5}-1
=
\sqrt5,
$$
and
$$
2\zeta+2\zeta^{-1}+1
=
4\cos\frac{2\pi}{5}+1
=
\sqrt5.
$$
Thus both rows satisfy Q3. They are palindromes, so
\autoref{thm:innovation-input-pentary-palindrome-real-two-point} gives
real two-point correlations.
\end{proof}

\section{Anchored Neutral Window Audit}
\label{sec:innovation-input-golden-palindrome-anchored-audit}

\begin{definition}[Pentary anchored neutral window]
\label{def:innovation-input-pentary-anchored-neutral-window}
Use the anchored $L=2$ neutral observables
$$
F_0=1,
$$
$$
F_1=u_{n+1}\overline{u_{n+2}},
$$
and
$$
F_2=\overline{u_{n+1}}u_{n+2}.
$$
Their future blocks are
$$
\alpha_0=0,
\qquad
\alpha_1=\delta_1-\delta_2,
\qquad
\alpha_2=-\delta_1+\delta_2.
$$
\end{definition}

\begin{lemma}[First golden branch window values]
\label{lem:innovation-input-first-golden-branch-window-values}
For
$$
v_5^{(+)}=(\xi,\xi^{-1},-1,\xi^{-1},\xi),
\qquad
\xi=e^{\pi i/5},
$$
the anchored $L=2$ neutral window has
$$
\mu_1:=\langle F_1\rangle=-\frac14,
$$
$$
D_1:=\langle\theta F_1,F_1\rangle=\frac{1+\sqrt5}{8},
$$
and
$$
C_1:=\langle\theta F_1,F_2\rangle
=
-\frac{1+\sqrt5}{8}
+
i\frac{\sqrt{10-2\sqrt5}}{8}.
$$
In particular, $|C_1|=1/2$.
\end{lemma}

\begin{proof}
The values are obtained by applying
\autoref{thm:innovation-input-finite-block-carry-recursion} to the anchored
blocks defining $\mu_1$, $D_1$, and $C_1$, then solving the finite
carry-closure linear system described in
\autoref{prop:innovation-input-finite-window-correlation-linear-system}.
The cross term has modulus
$$
|C_1|^2
=
\frac{(1+\sqrt5)^2}{64}
+
\frac{10-2\sqrt5}{64}
=
\frac14,
$$
so $|C_1|=1/2$.
\end{proof}

\begin{theorem}[First golden branch fails anchored neutral RP]
\label{thm:innovation-input-first-golden-branch-rp-failure}
The branch $v_5^{(+)}$ fails the anchored $L=2$ neutral RP window.
\end{theorem}

\begin{proof}
The $(F_1,F_2)$ principal submatrix has the form
$$
\left(
\begin{array}{cc}
D_1 & C_1\\
\overline{C_1} & D_1
\end{array}
\right).
$$
Its eigenvalues are $D_1\pm |C_1|$. By
\autoref{lem:innovation-input-first-golden-branch-window-values},
$$
D_1-|C_1|
=
\frac{1+\sqrt5}{8}-\frac12
=
\frac{\sqrt5-3}{8}
<
0.
$$
The principal submatrix is not positive semidefinite.
\end{proof}

\begin{lemma}[Second golden branch negative diagonal]
\label{lem:innovation-input-second-golden-branch-negative-diagonal}
For
$$
v_5^{(-)}=(\zeta,\zeta^{-1},1,\zeta^{-1},\zeta),
\qquad
\zeta=e^{2\pi i/5},
$$
one has
$$
\langle\theta F_1,F_1\rangle=\frac{1-\sqrt5}{8}<0.
$$
\end{lemma}

\begin{proof}
For $F_1=u_{n+1}\overline{u_{n+2}}$, the reflected diagonal block is
$$
\delta_0-\delta_1+\delta_2-\delta_3.
$$
Applying the finite-block carry recursion and solving its finite closure
system for the second golden branch gives
$$
\Gamma(\delta_0-\delta_1+\delta_2-\delta_3)
=
\frac{1-\sqrt5}{8}.
$$
This is negative.
\end{proof}

\begin{theorem}[Second golden branch fails anchored neutral RP]
\label{thm:innovation-input-second-golden-branch-rp-failure}
The branch $v_5^{(-)}$ fails the anchored $L=2$ neutral RP window.
\end{theorem}

\begin{proof}
By \autoref{lem:innovation-input-second-golden-branch-negative-diagonal},
the RP quadratic form is negative on the neutral future observable
$F_1$. Hence the anchored $L=2$ window is not positive semidefinite.
\end{proof}

\begin{theorem}[Golden palindromes are not RP candidates]
\label{thm:innovation-input-golden-palindromes-not-rp-candidates}
The two golden palindrome Q3 rows $v_5^{(+)}$ and $v_5^{(-)}$ have real
two-point correlations, but neither passes the anchored $L=2$ neutral RP
window.
\end{theorem}

\begin{proof}
Q3 compatibility and real two-point correlations follow from
\autoref{thm:innovation-input-golden-palindrome-qthree-rows}. The first
branch fails by
\autoref{thm:innovation-input-first-golden-branch-rp-failure}; the second
branch fails by
\autoref{thm:innovation-input-second-golden-branch-rp-failure}.
\end{proof}

\label{prin:innovation-input-anchored-rp-status-table}
The low-complexity Q3 representatives now have the following anchored RP
status:
$$
\begin{array}{c|c|c|c}

\text{anchored RP status}\\
\hline

3 & (e^{i\pi/6},e^{-i\pi/2},e^{i\pi/6}) &

\text{real} &

\text{real} &

\end{array}
$$
The pentary golden branches show that additional phase freedom can remove
the two-point arrow without producing an OS/RP-compatible
self-substitution row.
\end{principle}

\begin{principle}[Pentary full-window target]
\label{prin:innovation-input-pentary-full-window-target}
The remaining pentary question is global over $\mathcal Q_5$: determine
whether some $v\in\mathcal Q_5$ passes the anchored $L=2$ neutral RP
matrix. For
$$
F=u_{n+1}\overline{u_{n+2}},
$$
the basic entries are
$$
\mu=\langle F\rangle,
\qquad
D=\langle\theta F,F\rangle,
\qquad
C=\langle\theta F,\overline F\rangle.
$$
The anchored $L=2$ matrix is
$$
M_2(v)=
\left(
\begin{array}{ccc}
1 & \mu & \overline\mu\\
\overline\mu & D & C\\
\mu & \overline C & \overline D
\end{array}
\right).
$$
In the Hermitian case, the first necessary conditions are $D\ge0$,
$D\ge |C|$, and $\det M_2(v)\ge0$.
\end{principle}
