\section{Coherent Family and Meeting Law}
\label{sec:innovation-input-thermo-meeting-law}

\begin{lemma}[Two-dimensional conformal diffusion exactness]
\label{lem:innovation-input-thermo-two-dimensional-conformal-exactness}
Let a two-dimensional position-dependent jump row have generator
$-\frac12g(x)\Delta$. For the conformal metric $G=g^{-1}\delta$, this is
exactly $-\frac12\Delta_G$, and the reversible volume density is
proportional to $1/g$.
\end{lemma}

\begin{proof}
In two dimensions,
$$
\Delta_G
=
\frac{1}{\sqrt{|G|}}
\partial_i\left(\sqrt{|G|}G^{ij}\partial_j\right)
=
g\Delta.
$$
The detailed-balance density is the inverse jump rate, so the chain is
reversible on the same volume row used by the Laplace--Beltrami operator.
\end{proof}

\begin{definition}[Machine coherent family]
\label{def:innovation-input-thermo-machine-coherent-family}
Let $H$ be the reconstructed Hamiltonian of a reflection-positive
spatialized input, and let $T>0$ be the logarithmic age. The machine
coherent family based at a position $z$ is the normalized state
$$
\psi_z^{(T)}
\propto
e^{-TH}\delta_z.
$$
Its innovation information metric is the QGT real part for the parameter
$z$, written $h_{ij}(z,T)$.
\end{definition}

\begin{theorem}[Meeting law]
\label{thm:innovation-input-thermo-meeting-law}
For a translation-invariant finite-variance input with
$$
H(k)=\frac12g^{ij}k_ik_j+O(|k|^4),
$$
the coherent-family QGT metric satisfies, in the low-momentum regime,
$$
h_{ij}(T)=\frac{1}{2T}(g^{-1})_{ij}.
$$
Thus the kinematic QGT metric and the dynamic diffusion metric meet as
inverse matrices, with conformal factor $1/(2T)$.
\end{theorem}

\begin{proof}
The Fourier weight of the coherent family is proportional to
$$
e^{-2TH(k)}.
$$
At low momentum this is a centered Gaussian with exponent
$-Tg^{ij}k_ik_j$. Since the constant-modulus phase packet reads the QGT
metric as the covariance of the covectors $k_i$, the covariance is
$(2T)^{-1}g^{-1}$.
\end{proof}

\begin{principle}[Anisotropic closure of the metric comparison]
\label{prin:innovation-input-thermo-anisotropic-closure}
The meeting law is matrix-valued. It includes anisotropic and off-diagonal
diffusion tensors; it is not merely a scalar proportionality. The correction
of the diffusion tensor as an inverse metric is therefore a prerequisite for
the comparison row.
\end{principle}

\begin{principle}[Flat entropy production]
\label{prin:innovation-input-thermo-flat-entropy-production}
For the Gaussian coherent family in $d$ dimensions,
$$
\operatorname{tr}(h\,g)=\frac{d}{2T}.
$$
The Shannon entropy of the corresponding position density has derivative
$d/(2T)$. Hence the flat meeting row reads
$$
\operatorname{tr}(h\,g)=\dot S.
$$
This is the entropy-production form of the meeting law.
\end{principle}

\begin{principle}[Finite-mass and massless boundary]
\label{prin:innovation-input-thermo-finite-mass-massless-boundary}
For the massive relativistic multiplier
$H(k)=\sqrt{|k|^2+m^2}-m$, the quadratic row near $k=0$ gives the same
meeting law with the effective diffusion scale set by $m^{-1}$. At
$m=0$, the finite-variance metric row no longer exists, but the coherent
family still carries the ballistic scaling
$$
h=\frac{1}{2T^2}
$$
in the one-dimensional Cauchy row. The covariance geometry and the
massless light-cone scaling therefore separate cleanly.
\end{principle}

\begin{principle}[Numerical meeting witnesses]

\section{Heat Kernel Identity and Curvature Read}
\label{sec:innovation-input-thermo-curvature-read}

\begin{theorem}[Heat-kernel QGT identity]
\label{thm:innovation-input-thermo-heat-kernel-qgt-identity}
Let $K_T(z,z')$ be the heat kernel of the reconstructed geometric
Hamiltonian on the physical volume measure. For the coherent family of
\autoref{def:innovation-input-thermo-machine-coherent-family},
$$
h_{ij}(z,T)
=
\partial_{z^i}\partial_{z'^j}\log K_T(z,z')\big|_{z'=z}.
$$
\end{theorem}

\begin{proof}
The overlap of the unnormalized coherent states is the heat kernel
$K_T(z,z')$ up to factors separated in $z$ and $z'$. In the QGT expression
for normalized states, those separated factors cancel under the mixed
second derivative, leaving the displayed identity.
\end{proof}

\begin{corollary}[Spectral-dimension merger]
\label{cor:innovation-input-thermo-spectral-dimension-merger}
In a homogeneous setting,
$$
\operatorname{tr}h
=
-\partial_T\log K_T(z,z).
$$
With the spectral-dimension convention
$K_T(z,z)\asymp T^{-d_s(T)/2}$, the coherent-family trace reads
$$
\operatorname{tr}h(T)=\frac{d_s(T)}{2T},
$$
up to the derivative correction when $d_s$ runs with $T$. For a smooth
metric, the small-time expansion gives
$$
d_s(T)=d-\frac{T}{3}R+O(T^2).
$$
Thus the meeting row and the spectral-dimension row use the same heat
kernel instrument.
\end{corollary}

\begin{theorem}[Curvature defect of meeting]
\label{thm:innovation-input-thermo-curvature-defect}
For a smooth Riemannian metric $G$ and small $T$,
$$
2T\,h_{ij}(z,T)
=
G_{ij}(z)-\frac{T}{3}\operatorname{Ric}_{ij}(z)+O(T^2).
$$
In two dimensions, for a conformal scalar diffusion read,
$$
2Tg\,h=1-\frac{R}{6}T+O(T^2).
$$
\end{theorem}

\begin{proof}
Insert the Minakshisundaram--Pleijel heat-kernel expansion into
\autoref{thm:innovation-input-thermo-heat-kernel-qgt-identity}. The mixed
second derivative of the world function gives the metric term, and the
mixed derivative of the Van Vleck--Morette factor gives the Ricci
coefficient.
\end{proof}

\begin{principle}[Ricci as a measurable machine output]
\label{prin:innovation-input-thermo-ricci-output}
Ricci curvature is read as the first-order failure of exact meeting:
the information metric and diffusion metric meet at leading order, and the

output of the coherent-family row, not a primitive field-equation input.
\end{principle}

\section{Local Heat-Geometry Identity}
\label{sec:innovation-input-thermo-local-heat-geometry}

\begin{theorem}[Local thermodynamic geometry identity]
\label{thm:innovation-input-thermo-local-heat-geometry}
For the coherent-family position entropy and the diffusion metric trace,
the small-time local relation is
$$
\dot S
=
\operatorname{tr}(h\,g)+\frac{R}{12}+O(T).
$$
The coefficient $1/12$ is dimension-independent in the stated heat-kernel
normalization.
\end{theorem}

\begin{proof}
In normal coordinates, the squared Van Vleck factor and the Riemannian
volume density cancel to the required order. The entropy receives the
volume-density correction, while $\operatorname{tr}(h\,g)$ receives the
Ricci correction from
\autoref{thm:innovation-input-thermo-curvature-defect}. Subtracting the
two expansions gives the displayed $R/12$ term.
\end{proof}

\begin{corollary}[Thermodynamic curvature read]
\label{cor:innovation-input-thermo-thermodynamic-curvature-read}
The scalar curvature can be read by
$$
R=12\bigl(\dot S-\operatorname{tr}(h\,g)\bigr)+O(T).
$$
Thus the geometric curvature read of
\autoref{thm:innovation-input-thermo-curvature-defect} has a second,
thermodynamic measurement protocol.
\end{corollary}

\begin{principle}[Internal consistency test]
\label{prin:innovation-input-thermo-internal-consistency-test}
The same scalar curvature must be obtained from the meeting-defect read
and from the entropy-production read. This gives an internal consistency
test for any proposed input-flow geometry bridge.
\end{principle}

\begin{principle}[Thermodynamic numerical witnesses]
\label{prin:innovation-input-thermo-thermodynamic-numerical-witnesses}
On the sphere, the measured differences
$$
\dot S-\operatorname{tr}(hg)=0.16856,\ 0.16735,\ 0.16789
$$
at $T=0.02,0.05,0.10$ match the prediction $R/12=1/6$ to the observed

\end{principle}

\section{Modular Hamiltonian and Unruh Row}
\label{sec:innovation-input-thermo-modular-unruh}

\begin{definition}[Gaussian modular Hamiltonian row]
\label{def:innovation-input-thermo-gaussian-modular-hamiltonian-row}
For a Gaussian state restricted to a region $A$, let
$$
X=\langle\varphi\varphi\rangle_A,
\qquad
P=\langle\pi\pi\rangle_A.
$$
The modular Hamiltonian is the quadratic operator $K$ such that
$\rho_A\propto e^{-K}$.
\end{definition}

\begin{theorem}[Gaussian modular Hamiltonian normal form]
\label{thm:innovation-input-thermo-gaussian-modular-normal-form}
Let $X_s=X^{1/2}$ and diagonalize
$$
X_sPX_s=O\nu^2O^{\mathsf T},
$$
with symplectic eigenvalues $\nu_a\ge1/2$. Set
$$
\epsilon_a=\log\frac{\nu_a+1/2}{\nu_a-1/2}.
$$
Then
$$
K=\frac12\pi^{\mathsf T}G_K\pi+\frac12\varphi^{\mathsf T}F_K\varphi,
$$
where
$$
G_K=X_sO\,\operatorname{diag}\left(\frac{\epsilon_a}{\nu_a}\right)
O^{\mathsf T}X_s
$$
and
$$
F_K=X_s^{-1}O\,\operatorname{diag}(\epsilon_a\nu_a)
O^{\mathsf T}X_s^{-1}.
$$
\end{theorem}

\begin{proof}
The canonical transformation
$\Phi=O^{\mathsf T}X_s^{-1}\varphi$ and
$\Pi=O^{\mathsf T}X_s\pi$ diagonalizes the covariance data. Each normal
mode is a thermal oscillator with
$\coth(\epsilon_a/2)=2\nu_a$. Pulling the oscillator kernels back gives
the displayed matrices.
\end{proof}

\begin{principle}[Bisognano--Wichmann read]
\label{prin:innovation-input-thermo-bw-read}
For a continuum half-space, the modular Hamiltonian is
$$
K=2\pi\int x\,T_{00}(x)\,dx.
$$
The lattice calculation reads this weakly: smoothed row sums of the
modular kernels approach the boost weight $2\pi x$ near the cut, while
pointwise entries display nonlocal regulator spread.
\end{principle}

\begin{principle}[Weak modular numerical witness]
\label{prin:innovation-input-thermo-weak-modular-numerical-witness}
For a massive lattice scalar with $m=0.15$ and interval length $40$, high
precision covariance diagonalization gives the following ratios to the
Bisognano--Wichmann prediction. At sites $i=0,2,4,6,10$, the
$G_K$ diagonal ratios are
$$
0.9986,\ 0.9944,\ 0.9708,\ 0.9373,\ 0.8567,
$$
the $G_K$ row-sum ratios are
$$
0.9986,\ 0.9978,\ 0.9879,\ 0.9985,\ 1.062,
$$
and the $F_K$ row-sum ratios against $2\pi \tilde x\,m^2$ are
$$
0.996,\ 1.018,\ 0.928,\ 0.926,\ 1.113.
$$
Near the cut, the row-sum tests recover $2\pi\tilde x$ at percent scale,
while the diagonal entries display the expected nonlocal broadening of the
modular kernel. A larger interval in double precision gives ratios near
$1$ at the first two sites, but high precision is needed once the
symplectic eigenvalues approach $1/2$.
\end{principle}

\begin{principle}[Unruh temperature boundary]
\label{prin:innovation-input-thermo-unruh-temperature-boundary}
The weak half-space modular row gives the Unruh temperature
$1/(2\pi)$ in machine units. The finite lattice row has two controlled
deviations: nonlocal broadening near the cut and exponential sensitivity
to a second boundary when the region is a finite interval rather than a
half-space.
\end{principle}
