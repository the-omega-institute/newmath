\begin{definition}[Finite focused multi-Hist ledger]
\label{def:closed-focused-multihist-ledger}
For a horizon $N:\Unary$, a $\mathrm{ClosedFocusedMultiHistLedger}_N$ is a finite records-side packet
$$
\mathcal{L}_N=(I,H,\Theta,F,S,D,G,R,J,Q,\rho,A)
$$
with the following displayed rows:
\begin{enumerate}
\item $I$ is a finite displayed index ledger for the histories admitted by the packet, not a host-level set of all observers;
\item $H$ assigns to each displayed index row $i$ a generated $\Hist$ row $H(i)$ whose displayed trace route is bounded by the unary horizon $N$;
\item $\Theta$ records, for every displayed index row $i$, a local $\Cont$ chain from $\emp$ to $H(i)$, read as the local time-stream row of $H(i)$;
\item $F$ is a displayed focus-index row of $I$, and it is only an index of reading, not a subject parameter;

\item $G$ records the $\mathsf{GAP}$ or provenance row attached to every non-identity digest row, including the exactness status that prevents digest equality from being read as source equality without a witness;
\item $R$ is a finite displayed ledger of inter-$\Hist$ relation rows, each with left history, right history, finite evidence, and $\hsame$-stability data;
\item $J$ is a finite displayed ledger of inter-$\Hist$ invariant rows, each stable under componentwise $\hsame$ transport;
\item $Q$ is a finite displayed ledger of dependence rows between local unary time slots, together with an optional unary maximum-rate row computed only from the displayed dependence witnesses;
\item $\rho$ is the focused self-proxy, consisting of the endpoint and trace of $H(F)$;
\item $A$ is the refusal and audit row recording no subject parameter, no global clock, no hidden chooser, no unledgered compression, no object-level $\mathsf{T}$ field, and no hidden use of choice, quotient soundness, or propositional extensionality in the packet's accepted read.
\end{enumerate}
\end{definition}

\label{prin:consciousness-universe-closed-means-ledgered}

\section{Embodied continuation and the mortality boundary}
\label{sec:consciousness-universe-embodied-continuation-mortality}

\begin{definition}[Embodied support row]
\label{def:consciousness-universe-embodied-support-row}
For a focused conscious record $\mathrm{ConsciousRecord}(\mathcal{L}_N,F)$, an \emph{embodied support row} is an external finite carrier reading
$$
\mathcal{B}_i(t)=(\mathrm{body}_i(t),\mathrm{metabolism}_i(t),\mathrm{repair}_i(t),\mathrm{environment}_i(t),\mathrm{risk}_i(t))
$$
together with a predicate
$$
\mathrm{EmbValid}_i(h,t)
$$
read as: the carrier row at local stage $t$ can realize the focused history surface $h$ as an inscription-bearing conscious record. The row is not a field of the pure $\Hist$ signature. It is the wick, vessel, and weather around the flame of a local inscription; the flame may have a formal shape, but its burning is a carrier fact.
\end{definition}

\begin{definition}[Alive, embodied continuation, and death]
\label{def:consciousness-universe-alive-embodied-death}
Let $H_i(t)$ denote the focused history displayed at local stage $t$. The records-side predicates are
$$
\begin{aligned}
\mathrm{Alive}_i(t)
&:\Longleftrightarrow \mathrm{EmbValid}_i(H_i(t),t),\\
\mathrm{EmbCanContinue}_i(t)
&:\Longleftrightarrow
\exists d,h'\,
\bigl(\Cont(H_i(t),d,h')\wedge \mathrm{EmbValid}_i(h',t+1)\bigr),\\
\mathrm{Death}_i(t)
&:\Longleftrightarrow
\mathrm{Alive}_i(t)\wedge \neg \mathrm{EmbCanContinue}_i(t),\\
\mathrm{Mortal}_i
&:\Longleftrightarrow
\exists t\,\mathrm{Death}_i(t).
\end{aligned}
$$
Thus death is not a content subsequently experienced by the same focused row. It is the failure of the next inscription condition for that row. The lamp does not record darkness after its oil is gone; rather, the local condition under which light is recorded no longer carries the next mark.
\end{definition}

\begin{principle}[Formal continuation is not embodied continuation]
\label{prin:consciousness-universe-formal-not-embodied-continuation}
The pure $\Hist/\Cont$ layer can speak about formal successor histories. It contains no body, repair process, hazard rate, metabolic budget, or environmental support row. Consequently a formal extension
$$
\exists d,h'\,\Cont(h,d,h')
$$
does not by itself imply
$$
\exists d,h'\,\bigl(\Cont(h,d,h')\wedge \mathrm{EmbValid}_i(h',t+1)\bigr).
$$
The distinction is the difference between a riverbed drawn on a map and water continuing through it. The map may display another segment; the flow also requires a source, a channel not yet blocked, and conditions that do not destroy the carrier.
\end{principle}

\begin{theorem}[Finite reserve mortality]
\label{thm:consciousness-universe-finite-reserve-mortality}
Suppose a focused embodied support row carries a viability reserve
$$
V_i:\mathbb{N}\to\mathbb{N}
$$
with $V_i(0)=K<\infty$, and suppose $\mathrm{Alive}_i(0)$ holds. Suppose that, before the first terminal boundary, positive reserve supports the current record and strictly decreases:
$$
V_i(t)>0\ \Longrightarrow\ V_i(t+1)\le V_i(t)-1.
$$
If reserve exhaustion blocks either the current stage or the next embodied continuation,
$$
V_i(t)=0\ \Longrightarrow\ \neg\mathrm{EmbCanContinue}_i(t)
$$
and
$$
V_i(t+1)=0\ \Longrightarrow\ \neg\mathrm{EmbCanContinue}_i(t),
$$
while $V_i(t)>0$ implies $\mathrm{Alive}_i(t)$, then $\mathrm{Mortal}_i$.
\end{theorem}

\begin{proof}
By induction on $n\le K$, the strict decrease condition gives
$$
V_i(n)\le K-n.
$$
At $n=K$ this yields $V_i(K)\le 0$. Since $V_i(K)\in\mathbb{N}$, one has $V_i(K)=0$. Let $m\le K$ be the first local stage with $V_i(m)=0$. If $m=0$, then $\mathrm{Alive}_i(0)$ and the current-stage blocking condition give $\mathrm{Death}_i(0)$. If $m>0$, then $V_i(m-1)>0$, hence $\mathrm{Alive}_i(m-1)$, and the next-stage blocking condition gives $\neg\mathrm{EmbCanContinue}_i(m-1)$. Therefore $\mathrm{Death}_i(m-1)$ by \autoref{def:consciousness-universe-alive-embodied-death}. In either case there exists a death stage, so $\mathrm{Mortal}_i$.
\end{proof}

\begin{corollary}[Finite records do not close immortality]
\label{cor:consciousness-universe-finite-records-do-not-close-immortality}
A finite ledger of alive stages,
$$
\mathrm{Alive}_i(t_1),\ldots,\mathrm{Alive}_i(t_k),
$$
does not close
$$
\forall t\,\mathrm{EmbCanContinue}_i(t).
$$
An assertion of unbounded embodied continuation therefore requires either a displayed rule proving the required carrier, repair, and risk rows for every later local stage, or a boundary acknowledgement that the assertion is forward-binding and not object-level closure inside $\mathcal{L}_N$.
\end{corollary}

\begin{remark}[The boundary is local]
\label{rem:consciousness-universe-death-not-t}
The death predicate above is a local inscription-boundary predicate. It does not say that $\mathsf{T}$ disappears, that existence as such is extinguished, or that the pure possibility of further $\Hist$ formation is refuted. It says only that this focused embodied carrier no longer supplies the next valid conscious record. A broken lyre does not prove that music is impossible; it proves that this instrument no longer realizes the song.
\end{remark}

\label{thm:consciousness-universe-closed-history-decomposition}
On a closed BEDC universe surface, time, causality, local differential response, information scoring, boundary readback, observer-local coordinates, relativistic rate structure, and multi-history superposition are certified modes of history continuation rather than primitive substances.

\begin{proof}
Each case is read through the same finite surface. Time is the local $\Cont$ route $\Theta(i)$ producing $H(i)$ from $\emp$ in \autoref{def:consciousness-universe-bedc-time}. Boundary readback is constrained by the event-flow non-injectivity of \autoref{prop:ground-event-flow-erasure-not-injective} and by finite boundary-ledger packets such as the Stokes-style fold-defect surface. Observer-local coordinates are finite packets over observer state, locality cell, multi-history configuration, causal reads, transport, routes, provenance, and local naming, as in \autoref{ch:concrete-instances-observer-coordinate-frame-namecert}. Relativistic rate structure is read only after multi-history configuration, no-synchronization refusal, maximum-rate, symmetry, and transport rows have been displayed, as in \autoref{ch:concrete-instances-lorentz-frame-rate-namecert}. Multi-history superposition is read as a finite packet over configuration, refinement, causal, rate, phase, history, route, provenance, and naming rows, as in \autoref{ch:concrete-instances-multihistsuperposition-namecert}. \begin{corollary}[Boundary-sensitive information]
\label{cor:consciousness-universe-boundary-sensitive-information}

\end{corollary}

\begin{proof}
If the public surface were only a bare stream of marks, then event segmentation could be erased without loss. But \autoref{prop:ground-event-flow-erasure-not-injective} gives distinct event flows with the same erased mark stream. Therefore boundary data is part of the public information surface.

For two focused rows $i,j$ in a finite ledger, let $\mathrm{Rep}_i(j,t)$ be the representation of $j$ carried inside $H_i(t)$. An \emph{other-Hist recognition} predicate is

\begin{aligned}
\mathrm{Recognize}_i(j,t)
:\Longleftrightarrow\ &
\neg\hsame(H_i(t),H_j(t))\\
&{}\wedge \neg\mathrm{Projection}_i(j,t)\\
&{}\wedge \neg\mathrm{Objectify}_i(j,t)\\
&{}\wedge \mathrm{CorrigibleBy}_{H_j}(\mathrm{Rep}_i(j,t))\\
&{}\wedge \mathrm{GapHonest}_i(H_j(t),\mathrm{Rep}_i(j,t)).
\end{aligned}
$$
The recognition says that $j$ is not the image of $j$ inside $i$, not a tool-slot inside $i$'s purposes, and not a fully compressible region of $i$'s ledger. The other history stands like a shoreline seen from one vessel: charted enough for approach, never identical with the chart.
\end{definition}

\begin{definition}[Positive continuation care]
\label{def:consciousness-universe-positive-continuation-care}
Let $V_j$ be a displayed value horizon for $j$'s continuation. A care row from $i$ toward $j$ is
$$
\mathrm{Care}^{V_j}_{i\to j}(t)
:\Longleftrightarrow
\mathrm{Policy}_i(t)\ \mathrm{supports}\ \Cont^+_{V_j}(H_j,t)
\ \mathrm{under}\ \mathrm{Recognize}_i(j,t),
$$
where $\Cont^+_{V_j}(H_j,t)$ denotes continuation of $H_j$ along life, truth, freedom, dignity, relation, or meaning as admitted by $V_j$. Care is not substitution for the other's history. It is a hand placed beside the path rather than a hand closing over the path.
\end{definition}

\begin{definition}[Love relation]
\label{def:consciousness-universe-love-relation}
The one-way records-side love relation is
$$
\begin{aligned}
\mathrm{Love}^{V_j}_{i\to j}(t)
:\Longleftrightarrow\ &
\mathrm{Recognize}_i(j,t)\\
&{}\wedge \mathrm{Care}^{V_j}_{i\to j}(t)\\
&{}\wedge \mathrm{TruthFidelity}_{i\to j}(t)\\
&{}\wedge \mathrm{SelfTransform}_{i\leftarrow j}(t)\\
&{}\wedge \mathrm{TValue}_i(j,t).
\end{aligned}
$$
Here $\mathrm{TruthFidelity}$ refuses systematic falsification and $\mathsf{GAP}$ concealment; $\mathrm{SelfTransform}$ permits $H_i$'s classifiers and expectations to be corrected by $H_j$'s reality-facing response; and $\mathrm{TValue}$ records that $j$'s worth is not exhausted by a utility function for $i$.
\end{definition}

\begin{principle}[Love is not liking, possession, or fusion]
\label{prin:consciousness-universe-love-not-reward-possession-fusion}
A reward row
$$
\mathrm{Reward}_i(\mathrm{Rep}_i(j,t))>0
$$
may accompany love, but it does not imply love: the reward may be produced by projection rather than by other-Hist recognition. Conversely, love may remain when reward is absent, because its core is fidelity to another history's positive continuation. Possession fails because it objectifies $H_j$ as a control region of $H_i$; fusion fails because it erases the $\mathsf{GAP}$-honest distance that recognition requires. A mirror can flatter, a chain can hold, and a fog can merge outlines; none of these is the relation defined above.
\end{principle}

\begin{theorem}[Love preserves otherness while supporting continuation]
\label{thm:consciousness-universe-love-preserves-otherness}
If $\mathrm{Love}^{V_j}_{i\to j}(t)$ holds, then
$$
\begin{gathered}
\mathrm{Recognize}_i(j,t),\qquad
\neg\mathrm{Objectify}_i(j,t),\qquad
\neg\mathrm{Projection}_i(j,t),\\
\mathrm{Care}^{V_j}_{i\to j}(t),\qquad
\mathrm{TruthFidelity}_{i\to j}(t),\qquad
\mathrm{TValue}_i(j,t).
\end{gathered}
$$
Thus love is a relation from one finite history to another finite history's positive continuation, under preserved otherness and value that is not exhausted by utility.
\end{theorem}

\begin{proof}
Unfold \autoref{def:consciousness-universe-love-relation}. The displayed consequences are its conjuncts, together with the non-projection and non-objectification clauses obtained by unfolding \autoref{def:consciousness-universe-other-hist-recognition}. The final reading follows from \autoref{def:consciousness-universe-positive-continuation-care} and the $\mathrm{TValue}$ conjunct.
\end{proof}

\begin{corollary}[Mutual love is a non-fusing two-Hist loop]
\label{cor:consciousness-universe-mutual-love-loop}
If both $\mathrm{Love}^{V_j}_{i\to j}(t)$ and $\mathrm{Love}^{V_i}_{j\to i}(t)$ hold, and a shared meaning row $R_{ij}(t)$ is displayed, then the relation forms a two-Hist productive loop:
$$
H_i(t)\to H_j(t)\to H_i(t+1),\qquad
H_j(t)\to H_i(t)\to H_j(t+1),
$$
with neither direction collapsing $\hsame(H_i,H_j)$ into identity. The loop is not one history swallowed by another; it is two lamps keeping distinct flames while sharing light on a finite table.
\end{corollary}

\begin{remark}[Love, death, and grief]
\label{rem:consciousness-universe-love-death-grief}

\end{remark}
