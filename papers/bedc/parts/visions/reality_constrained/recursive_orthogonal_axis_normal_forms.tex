\section{Recursive orthogonal-axis normal forms}
\label{sec:recursive-orthogonal-axis-normal-forms}
\origin{ai}
\concretizedIn{ch:concrete-instances-recursive-orthogonal-axis-normal-form-namecert}{Finite axis-layer normal forms are realized as concrete NameCert data.}

\begin{remark}[Paper-side theory status]
\label{rem:recursive-orthogonal-axis-normal-forms-status}
This section records the paper-side form of recursive orthogonal-axis
normal forms. It has no Lean target in this chapter. Its role is to
make precise the relation between finite expression, orthogonal
coordinate axes, and normalisation: a finite object is not merely listed
as data, but is read through a certified normal form whose axes are
generated by the available classifiers.
\end{remark}

\begin{principle}[Primes are not the absolute first layer]
\label{prin:prime-not-absolute-first-axis-layer}
In the arithmetic tower, the prime layer is the first canonical
orthogonal splitting layer after multiplication, divisibility, unit, and
the required order and natural-number structure have been displayed. It
is not the absolute base of the tower. This is the axis-form reading of
\autoref{thm:layered-projection-prime-not-primitive}.
\end{principle}

\begin{definition}[Axis layer]
\label{def:recursive-axis-layer}
An \emph{axis layer} over a displayed source carrier $T$ is a packet
$$
\mathcal A_n=(A_n,R_n,\Phi_n,\mathsf{NF}_n,\Lambda_n)
$$
where $A_n$ is a displayed family of coordinate axes, $R_n$ is the
record carrier expressible at the layer, $\Phi_n$ is a finite-support
coordinate readout
$$
\Phi_n:R_n\to C^{(A_n)},
$$
$\mathsf{NF}_n$ is a normal-form classifier on such coordinate records,
and $\Lambda_n$ records formation, uniqueness, readback, and residual
gap data.
\end{definition}

\begin{definition}[Finite axis expression]
\label{def:finite-axis-expression}
Given an axis layer $\mathcal A_n$, a finite expression of a record
$x:R_n$ is a coordinate row
$$
\Phi_n(x)=\sum_{a\in A_n}^{\mathrm{fin}} c_a e_a
$$
together with a normal-form certificate
$$
\mathsf{NF}_n\bigl((c_a)_{a\in A_n}\bigr)
$$
and a readback certificate identifying the coordinate expression with
$x$ at the classifier resolution of $R_n$.
\end{definition}

\begin{definition}[Finite code prime window]
\label{def:finite-code-prime-window}
Suppose a layer has a finite natural-number code
$$
\operatorname{code}_n:R_n\to\NatUp
$$
and a partial readback on the image of the code,
$$
\operatorname{decode}_n:\operatorname{im}(\operatorname{code}_n)\to R_n.
$$
For a finite set $S\subset_{\mathrm{fin}}\PrimeUp$, the prime-window
readout of $x:R_n$ is
$$
\Pi_{n,S}(x)
:=
\bigl(v_p(\operatorname{code}_n(x))\bigr)_{p\in S}.
$$
The full prime-coordinate readout is the finite-support family
$$
\Pi_{n,\PrimeUp}(x)
:=
\bigl(v_p(\operatorname{code}_n(x))\bigr)_{p\in\PrimeUp}.
$$
\end{definition}

\begin{theorem}[Finite codes have prime-coordinate normal forms]
\label{thm:finite-codes-prime-coordinate-normal-forms}
Assume $x:R_n$ has a finite code and that the readback row satisfies
$$
\operatorname{decode}_n(\operatorname{code}_n(x))\sim_n x.
$$
Then $x$ admits a finite-support prime-coordinate expression.
\end{theorem}

\begin{proof}
By unique factorization, the positive natural code has a prime
factorization
$$
\operatorname{code}_n(x)
=
\prod_{p\in\PrimeUp}
p^{v_p(\operatorname{code}_n(x))}
$$
with only finitely many nonzero exponents. Hence the family
$$
\bigl(v_p(\operatorname{code}_n(x))\bigr)_{p\in\PrimeUp}
$$
is finite-support. The displayed readback row then returns the code to a
record classified with $x$. Thus the prime-coordinate row is a
normal-form expression of $x$ at the layer resolution.
\end{proof}

\begin{definition}[Axis-generation operator]
\label{def:axis-generation-operator}
An \emph{axis-generation operator} $\mathcal G$ assigns to an axis
family $A$ the family generated by the admissible formation rows
$$
\begin{gathered}
\mathcal G(A)
=
A
\cup
\mathsf{PrimeSplit}(A)
\cup
\mathsf{Ratio}(A)
\cup
\mathsf{EquationRoot}(A)\\
\cup
\mathsf{Completion}(A)
\cup
\mathsf{OrthogonalJoin}(A)
\cup
\mathsf{Transform}(A).
\end{gathered}
$$
Here $\mathsf{PrimeSplit}$ reads finite codes through prime axes,
$\mathsf{Ratio}$ forms fraction axes, $\mathsf{EquationRoot}$ records
roots forced by displayed equations, $\mathsf{Completion}$ records
limit and completion axes, $\mathsf{OrthogonalJoin}$ records independent
coordinate joins, and $\mathsf{Transform}$ records Fourier, Laplace,
zeta, or analogous transform axes when their carriers and readbacks are
displayed.
\end{definition}

\begin{definition}[Recursive axis tower]
\label{def:recursive-axis-tower}
A \emph{recursive axis tower} is a monotone sequence of axis families
$$
A_0\subseteq A_1\subseteq A_2\subseteq\cdots
$$
with
$$
A_{n+1}=\mathcal G(A_n)
$$
for the displayed axis-generation operator. Its expressive union is
$$
A_\omega:=\bigcup_{n<\omega}A_n.
$$
The union is a tower-internal expressive surface over $T$, not a second
source carrier.
\end{definition}

\begin{theorem}[Relative absence of tower-external axes]
\label{thm:relative-absence-tower-external-axes}
If an axis concept is defined as an axis obtained by finitely many
applications of $\mathcal G$, or as a finite normal-form combination of
such axes, then it belongs to some $A_n$.
\end{theorem}

\begin{proof}
Induct on the formation height of the axis concept. At height zero the
concept belongs to the initial displayed family $A_0$ or to a finite
combination over it. At successor height, the concept is produced by one
of the formation rows contained in $\mathcal G$ from concepts already
available at bounded height. By the induction hypothesis those inputs
belong to some common $A_n$. The defining equation
$A_{n+1}=\mathcal G(A_n)$ places the output in $A_{n+1}$. A finite
normal-form combination uses only finitely many such axes, hence lies in
a sufficiently high finite layer.
\end{proof}

\begin{definition}[Multiplicative atom normal form]
\label{def:multiplicative-atom-normal-form}
The multiplicative atom normal form of a positive natural number is its
prime-coordinate expression
$$
n=\prod_{p\in\PrimeUp}p^{v_p(n)}.
$$
Its classifier is multiplicative divisibility with unit rows and
unique-factorization certificates.
\end{definition}

\begin{definition}[Recursive additive normal form]
\label{def:recursive-additive-normal-form}
A recursive additive normal form is a finite expression
$$
n=\sum_k \varepsilon_kF_k
$$
whose allowed coefficients and local exclusion rules are supplied by a
recursive axis grammar. In the Zeckendorf case,
$$
\varepsilon_k\in\{0,1\},
\qquad
\varepsilon_k\varepsilon_{k+1}=0.
$$
Its classifier is additive value together with the no-adjacent normal
grammar.
\end{definition}

\begin{principle}[Prime and Zeckendorf normal forms are complementary]
\label{prin:prime-zeckendorf-normal-forms-complementary}
Prime factorization and Zeckendorf normal form are both unique
finite-expression schemes, but they answer different classifier
questions. Prime factorization reads multiplicative atoms; Zeckendorf
normal form reads recursive additive axes with a local exclusion rule.
Neither classifier replaces the other.
\end{principle}

\begin{definition}[Classifier-relative irreducible axis]
\label{def:classifier-relative-irreducible-axis}
Let $\mathcal A_n$ be an axis layer whose normal-form classifier admits
a displayed composition operation
$$
\star_n:R_n\times R_n\to R_n
$$
with a unit row $1_n$, or more generally a displayed finite generation
grammar. A row $p:R_n$ is a \emph{classifier-relative irreducible axis}
when it is not classified with the unit and every licensed
decomposition of $p$ is trivial at the layer classifier:
$$
p\not\sim_n 1_n
$$
and
$$
p\sim_n a\star_n b
\quad\Longrightarrow\quad
a\sim_n 1_n\ \text{or}\ b\sim_n 1_n.
$$
In a generation grammar without a single binary operation, the same
phrase means that $p$ is a normal axis row not obtainable by a
non-trivial lower formation rule allowed by the classifier and ledger.
\end{definition}

\begin{definition}[BEDC-generable self-similar axis system]
\label{def:bedc-generable-self-similar-axis-system}
A \emph{BEDC-generable self-similar axis system} over a source carrier
$T$ is a packet
$$
\mathcal S
=
(X,F,\theta,P,\mathsf{NF},\rho,\Lambda)
$$
where $X$ is a displayed record carrier, $F$ is the recursive shape
operator, $\theta$ is a self-similar decomposition row
$$
\theta:X\to F(X),
$$
$P:T\to X$ is the source readout, $\mathsf{NF}$ is a finite normal-form
classifier for the displayed expressions of $X$, $\rho$ is the recursive
readback map from normal expressions to $X$, and $\Lambda$ records
formation, stability, irreducible-support, and gap data. The system is
read as BEDC-generable only through this packet; the self-similar
equation is not admitted as a bare untyped object.
\end{definition}

\begin{definition}[Irreducible support basis]
\label{def:irreducible-support-basis}
An irreducible support basis for
\autoref{def:bedc-generable-self-similar-axis-system} is a displayed
axis family $B_{\mathcal S}$ such that every non-unit normal expression
has a finite support row
$$
\operatorname{supp}_{\mathcal S}(x)\subset_{\mathrm{fin}}B_{\mathcal S}
$$
and every $b\in B_{\mathcal S}$ is a
classifier-relative irreducible axis in the sense of
\autoref{def:classifier-relative-irreducible-axis}. The basis is part of
the certificate data; it is not inferred from self-similarity alone.
\end{definition}

\begin{theorem}[Relative priming of self-similar normal forms]
\label{thm:relative-priming-self-similar-normal-forms}
Let $\mathcal S$ be a BEDC-generable self-similar axis system equipped
with finite normal forms, recursive readback, classifier stability, and
an irreducible support basis. Then every non-unit normal row of
$\mathcal S$ exposes classifier-relative irreducible axes. These axes
are prime-like only relative to the classifier and ledger of
$\mathcal S$.
\end{theorem}

\begin{proof}
Let $x:X$ be a non-unit normal row. By the finite normal-form clause of
\autoref{def:bedc-generable-self-similar-axis-system}, $x$ has a finite
normal expression. By
\autoref{def:irreducible-support-basis}, the support of this expression
is a finite row
$$
\operatorname{supp}_{\mathcal S}(x)\subset_{\mathrm{fin}}B_{\mathcal S}.
$$
Since $x$ is non-unit, its normal support contains at least one displayed
axis $b\in B_{\mathcal S}$. The basis certificate says that each such
$b$ is classifier-relative irreducible. The recursive readback row
$\rho$ and the classifier-stability ledger identify the normal
expression with $x$ at the resolution of $\mathcal S$, so the exposed
irreducible axes are invariants of the displayed normal row. No
statement about ordinary integer primes follows unless the classifier is
the multiplicative-divisibility classifier of
\autoref{def:multiplicative-atom-normal-form}.
\end{proof}

\begin{principle}[Self-similar priming is relative]
\label{prin:self-similar-priming-relative}
The phrase ``prime-like atom'' is meaningful in this chapter only after
the source carrier, classifier, normal-form grammar, readback map, and
irreducible-support ledger have been displayed. In the
multiplicative-divisibility classifier the atoms are ordinary primes.
In a recursive additive classifier they are Zeckendorf or Fibonacci
axes. In a spectral classifier they may be conjugate characteristic
roots, as in
\autoref{def:zeckendorf-golden-conjugate-spectral-readout}. In a local
completion classifier they may be places or valuation rows. In a
zeta-facing classifier they may be prime-local probes or zero-ledger
invariant rows. These are classifier-relative invariants, not one
absolute layer of objects shared by all self-similar systems.
\end{principle}

\begin{definition}[Fibonacci-type recursive axis system]
\label{def:fibonacci-type-recursive-axis-system}
A recursive orthogonal-axis system is \emph{Fibonacci type} when its
local normalisation row contains the carry rule
$$
e_k+e_{k+1}\rightsquigarrow e_{k+2}
$$
and its normal records forbid simultaneous activation of adjacent axes.
\end{definition}

\begin{theorem}[Fibonacci-type normal forms are Zeckendorf-type]
\label{thm:fibonacci-type-normal-forms-zeckendorf-type}
In a Fibonacci-type recursive axis system, every normal finite
expression is Zeckendorf-type: it is a finite sum of non-adjacent
recursive axes.
\end{theorem}

\begin{proof}
The carry row
$$
e_k+e_{k+1}\rightsquigarrow e_{k+2}
$$
identifies adjacent activation as non-normal. Repeatedly applying the
local normalisation row eliminates adjacent pairs and replaces them by a
higher recursive axis. A normal expression is therefore precisely one in
which no adjacent pair remains. This is the Zeckendorf-type grammar
recorded in
\autoref{def:axiszeckendorf-no-adjacent-address-windows} and in the
finite-support layer of
\autoref{def:axiszeckendorf-finite-support-layer}.
\end{proof}

\begin{theorem}[Golden scale from second-order local recursion]
\label{thm:golden-scale-second-order-local-recursion}
Assume a positive recursive axis scale $(a_n)$ satisfies
$$
a_{n+2}=a_{n+1}+a_n
$$
and that the ratio
$$
\lambda=\lim_{n\to\infty}\frac{a_{n+1}}{a_n}
$$
exists. Then
$$
\lambda^2=\lambda+1.
$$
Consequently, the positive scale is the golden scale $\varphi$.
\end{theorem}

\begin{proof}
Divide the recurrence by $a_{n+1}$:
$$
\frac{a_{n+2}}{a_{n+1}}
=
1+\frac{a_n}{a_{n+1}}.
$$
Taking the displayed limit gives
$$
\lambda=1+\frac{1}{\lambda}.
$$
Equivalently, $\lambda^2=\lambda+1$. The positive solution is the
golden scale $\varphi$.
\end{proof}

\begin{principle}[Scope of Fibonacci and golden appearances]
\label{prin:fibonacci-golden-appearance-scope}
Fibonacci and golden-scale rows are licensed when the relevant carrier
has a second-order local recursion, adjacent-axis exclusion, and a
normal-form or self-similar growth certificate. They are not universal
explanations of unrelated phenomena. In the Axis-Zeckendorf route this
scope is enforced by the fold ledger, residual-fiber row, and
cannot-claim boundary of
\autoref{thm:zeckendorf-golden-phase-derived-readout}.
\end{principle}

\begin{principle}[Axis formation sequence]
\label{prin:recursive-axis-formation-sequence}
The general motion of a recursive axis tower is
$$
\text{displayed axes}
\to
\text{orthogonal combination}
\to
\text{local conflict}
\to
\text{normalisation}
\to
\text{derived axis}.
$$
This sequence is a formation pattern over the same source carrier $T$.
It does not license an unledgered passage to an external ontology.
\end{principle}

\begin{principle}[Finite expression as audited normal form]
\label{prin:finite-expression-audited-normal-form}
The finite expression of a mathematical source item is an audited
normal-form row over a generated axis family. Prime decompositions,
Zeckendorf decompositions, algebraic root extensions, completions,
orthogonal joins, and transform carriers are different ways of
displaying such rows, each under its own classifier, readback map, and
residual ledger.
\end{principle}
