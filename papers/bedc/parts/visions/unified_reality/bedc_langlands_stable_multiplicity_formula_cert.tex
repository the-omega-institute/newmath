\section{BEDC--Langlands Stable Multiplicity Formula Certificate}
\label{sec:unified-reality-bedc-langlands-stable-multiplicity-formula-cert}

The general $G$ existence bridge of
\autoref{sec:unified-reality-bedc-langlands-general-g-existence-bridge}
uses a stable multiplicity formula as the step from a nonzero stable
parameter distribution to a nonempty automorphic packet. This section
isolates that step. The additional datum, absent from the singleton
$GL_n$ setting, is the finite component group attached to an Arthur or
Langlands parameter. Its finite Fourier theory is the packet-internal
detector for multiplicity and labeling residuals.

\begin{definition}[Arthur component group]
\label{def:unified-reality-arthur-component-group}
Let
$$
\psi:L_F\times SL_2(\mathbb C)\to{}^LG
$$
be a discrete Arthur parameter. Its component group is the finite row
$$
\mathcal S_\psi
:=
\pi_0\left(
Z_{\widehat G}(\operatorname{im}\psi)/Z(\widehat G)^{\Gamma_F}
\right).
$$
It records the finite internal symmetry of the packet attached to
$\psi$.
\end{definition}

\begin{definition}[Packet Fourier ledger]
\label{def:unified-reality-packet-fourier-ledger}
A packet labeling certificate for $\psi$ consists of a packet candidate
$$
\Pi_\psi(G)
$$
and a finite packet character datum, written in the basic case as a map
$$
\ell:\Pi_\psi(G)\to\widehat{\mathcal S_\psi}.
$$
For $\pi\in\Pi_\psi(G)$ write $\chi_\pi:=\ell(\pi)$. Failure of packet
labeling, component-character normalization, sign convention, or packet
membership is recorded in
$$
\mathsf{PacketLabelLedger}(\psi).
$$
\end{definition}

\begin{definition}[Packet Fourier distributions]
\label{def:unified-reality-packet-fourier-distributions}
For a test function $f$ and $s\in\mathcal S_\psi$, define the
Fourier-twisted packet distribution
$$
\Theta_{\psi,s}(f)
:=
\sum_{\pi\in\Pi_\psi(G)}
\chi_\pi(s)\Theta_\pi(f),
$$
where
$$
\Theta_\pi(f):=\operatorname{tr}\pi(f).
$$
The stable packet distribution is a declared finite linear combination
$$
S\Theta_\psi^G(f)
:=
\sum_{\pi\in\Pi_\psi(G)}
m_\psi(\pi)\Theta_\pi(f),
$$
with $m_\psi(\pi)$ determined by the multiplicity formula and recorded
in the multiplicity ledger.
\end{definition}

\begin{definition}[Component group three-properties rows]
\label{def:unified-reality-component-group-three-properties}
The time row of $\mathcal S_\psi$ is generated by the chain
$$
\operatorname{im}\psi
\longrightarrow
Z_{\widehat G}(\operatorname{im}\psi)
\longrightarrow
\pi_0.
$$
The unity row is finite Fourier orthogonality on $\mathcal S_\psi$:
$$
\frac{1}{|\mathcal S_\psi|}
\sum_{s\in\mathcal S_\psi}
\chi(s)\overline{\chi'(s)}
=
\delta_{\chi,\chi'}.
$$
The ledger row records packet labels, multiplicities, global signs,
transfer factors, endoscopic labels, and normalization choices.
\end{definition}

\begin{theorem}[Component-group Fourier detection]
\label{thm:unified-reality-component-group-fourier-detection}
Let $\mathcal S$ be a finite group. If two coefficient functions
$$
a,b:\widehat{\mathcal S}\to\mathbb C
$$
are distinct, then there exists $s\in\mathcal S$ such that
$$
\sum_{\chi\in\widehat{\mathcal S}}a(\chi)\chi(s)
\ne
\sum_{\chi\in\widehat{\mathcal S}}b(\chi)\chi(s).
$$
\end{theorem}
\begin{proof}
Let $c=a-b$. Since $a\ne b$, the function $c$ is nonzero. The finite
Fourier transform
$$
\widehat c(s)
:=
\sum_{\chi\in\widehat{\mathcal S}}c(\chi)\chi(s)
$$
is invertible on the finite character group. If $\widehat c(s)=0$ for
all $s\in\mathcal S$, then Fourier inversion gives $c=0$, a
contradiction. Hence some $s$ has $\widehat c(s)\ne0$.
\end{proof}

\begin{corollary}[Packet multiplicity mismatch is finite-detected]
\label{cor:unified-reality-packet-multiplicity-mismatch-detected}
If the actual spectral multiplicity function on a packet differs from
the multiplicity function predicted by the stable multiplicity formula,
then a component-group Fourier detector
$$
s\in\mathcal S_\psi
$$
detects the mismatch, or the mismatch is a packet ledger row.
\end{corollary}
\begin{proof}
Group packet members by their component character labels. The actual
multiplicities and predicted multiplicities give two coefficient
functions on $\widehat{\mathcal S_\psi}$. If these functions differ,
\autoref{thm:unified-reality-component-group-fourier-detection} gives
an element $s$ whose Fourier-twisted packet distribution distinguishes
them. If the packet labels or the grouping needed for this comparison
are unavailable, the failure is exactly a packet ledger row.
\end{proof}

\begin{definition}[Stable multiplicity formula certificate]
\label{def:unified-reality-stable-multiplicity-formula-cert}
A stable multiplicity formula certificate for $\psi$ consists of:
\begin{itemize}
\item a packet candidate $\Pi_\psi(G)$;
\item a packet labeling certificate over $\mathcal S_\psi$;
\item a multiplicity function
$$
m_\psi:\Pi_\psi(G)\to\mathbb Z_{\ge0};
$$
\item an identity on the discrete stable scope
$$
S\Theta_\psi^G(f)
=
\sum_{\pi\in\Pi_\psi(G)}
m_\psi(\pi)\Theta_\pi(f)
+
\mathsf{MultiplicityLedger}_\psi(f),
$$
where the ledger contains component-group data, signs, transfer factors,
endoscopy, continuous spectrum, boundary, and normalization rows.
\end{itemize}
Failure to supply any listed row is a
$\mathsf{MultiplicityDefectLedger}$ entry.
\end{definition}

\begin{definition}[No hidden multiplicity residual]
\label{def:unified-reality-no-hidden-multiplicity-residual}
The no-hidden multiplicity residual condition says that every failure
of packet membership, packet labeling, component-group Fourier
compatibility, multiplicity, stable character identity, or packet
normalization is finite-detected by a packet Fourier detector or is
recorded in $\mathsf{MultiplicityDefectLedger}$, and closed packets
carry no unresolved such row.
\end{definition}

\begin{theorem}[Stable distribution nonzero forces packet nonempty]
\label{thm:unified-reality-stable-distribution-nonzero-packet-nonempty}
Assume:
\begin{itemize}
\item $S\Theta_\psi^G(f_0)\ne0$ for some fixed test function $f_0$;
\item the stable multiplicity formula certificate holds;
\item continuous, boundary, endoscopic, and normalization ledger terms
are separated or explicitly ledgered;
\item the object is closed against unresolved ledger rows.
\end{itemize}
Then
$$
\Pi_\psi(G)\ne\varnothing.
$$
\end{theorem}
\begin{proof}
By the stable multiplicity formula,
$$
S\Theta_\psi^G(f_0)
=
\sum_{\pi\in\Pi_\psi(G)}
m_\psi(\pi)\Theta_\pi(f_0)
+
\mathsf{MultiplicityLedger}_\psi(f_0).
$$
The ledger terms are separated or explicitly recorded. If
$\Pi_\psi(G)$ were empty, the packet sum would be zero. The nonzero
value of $S\Theta_\psi^G(f_0)$ would then have to come entirely from a
ledger term: continuous spectrum, boundary, endoscopy, normalization, or
multiplicity defect. Carrying such a row unresolved contradicts
closedness, while omitting it violates the ledger property. Therefore a
closed nonzero stable distribution has a nonempty packet.
\end{proof}

\begin{theorem}[Missing packet forces ledger contradiction]
\label{thm:unified-reality-missing-packet-forces-ledger-contradiction}
Let $\psi$ be a closed three-properties parameter with
$$
S\Theta_\psi^G(f_0)\ne0.
$$
If $\Pi_\psi(G)=\varnothing$, then an unresolved ledger residual is
present.
\end{theorem}
\begin{proof}
With an empty packet, the nonzero stable distribution cannot be
accounted for by packet characters. The source of the nonzero readout
must be endoscopic transfer, continuous spectrum, Eisenstein boundary,
packet-label failure, multiplicity formula defect, or normalization
data. These are precisely ledger rows. If the row is carried, the object
is not closed; if it is omitted, the ledger property fails.
\end{proof}

\begin{definition}[Endoscopic packet transfer residual]
\label{def:unified-reality-endoscopic-packet-transfer-residual}
Let $\xi:{}^LH\to{}^LG$ be an endoscopic embedding and suppose
$$
\psi=\xi\circ\psi_H.
$$
For matching test functions $f^H\leftrightarrow f^G$, the endoscopic
packet transfer residual is
$$
R_{\mathrm{endo}}(f^G,f^H)
:=
S\Theta_\psi^G(f^G)-S\Theta_{\psi_H}^H(f^H).
$$
\end{definition}

\begin{theorem}[Endoscopic transfer residual detection]
\label{thm:unified-reality-endoscopic-transfer-residual-detection}
If $R_{\mathrm{endo}}$ is nonzero, then it is detected on a fixed
matching test window or enters the endoscopic tail ledger.
\end{theorem}
\begin{proof}
If some fixed matching pair $f^H\leftrightarrow f^G$ has
$$
R_{\mathrm{endo}}(f^G,f^H)\ne0,
$$
then the residual is finite-visible. If all fixed matching windows are
clean but transfer still fails globally, the failure lies in singular
orbital limits, ramification escape, transfer-factor normalization,
boundary, or continuous spectrum. These are exactly endoscopic tail
ledger rows.
\end{proof}

\begin{theorem}[Endoscopic induction with multiplicity]
\label{thm:unified-reality-endoscopic-induction-with-multiplicity}
Assume proper endoscopic groups have the existence bridge, endoscopic
transfer residuals are no-hidden, and the relevant stable multiplicity
formulae are closed. If
$$
\psi=\xi\circ\psi_H
$$
comes from a proper endoscopic group $H$, then the $G$-side packet
$\Pi_\psi(G)$ exists in the closed scope.
\end{theorem}
\begin{proof}
The induction hypothesis gives a nonempty $H$-side packet
$\Pi_{\psi_H}(H)$. Endoscopic transfer carries its stable distribution
to the $G$-side unless a transfer residual occurs. By
\autoref{thm:unified-reality-endoscopic-transfer-residual-detection},
such a residual is finite-visible or ledgered as endoscopic tail; both
alternatives contradict closedness when unresolved. Hence transfer
succeeds. The stable multiplicity formula on the $G$-side then records
the transferred packet with its packet labels and multiplicities.
\end{proof}

\begin{theorem}[General $G$ existence bridge, multiplicity form]
\label{thm:unified-reality-general-g-existence-bridge-multiplicity-form}
Assume:
\begin{itemize}
\item a parameter state certificate gives
$$
S\Theta_\psi^G(f_0)\ne0
$$
on a fixed window;
\item the stable multiplicity formula certificate holds;
\item endoscopic transfer and endoscopic induction certificates hold;
\item trace tail ledger rows are complete;
\item no-hidden multiplicity residual and closedness hold.
\end{itemize}
Then the closed parameter $\psi$ has
$$
\Pi_\psi(G)\ne\varnothing.
$$
\end{theorem}
\begin{proof}
If the primitive discrete contribution of $S\Theta_\psi^G(f_0)$ is
nonzero, then
\autoref{thm:unified-reality-stable-distribution-nonzero-packet-nonempty}
gives a nonempty $G$-packet. If the nonzero contribution is a proper
endoscopic contribution, then
\autoref{thm:unified-reality-endoscopic-induction-with-multiplicity}
gives the $G$-packet. If neither alternative holds, the nonzero readout
belongs to continuous spectrum, Eisenstein boundary, singular orbit,
normalization, or multiplicity-defect data. These are trace or
multiplicity ledger rows, and closedness excludes unresolved such rows.
\end{proof}

\begin{theorem}[Stable multiplicity reciprocity]
\label{thm:unified-reality-stable-multiplicity-reciprocity}
Under the hypotheses of
\autoref{thm:unified-reality-general-g-existence-bridge-multiplicity-form},
plus Tannakian separation, Chebotarev, and no-hidden Langlands residual,
there is $\pi\in\Pi_\psi(G)$ such that
$$
\lambda_{v,r}(\pi)
=
\operatorname{tr}r(\psi(\operatorname{Frob}_v))
$$
for almost all certified unramified places $v$ and all
finite-dimensional readouts $r:{}^LG\to GL(V_r)$ in scope.
\end{theorem}
\begin{proof}
The multiplicity-form bridge gives a nonempty packet. Choose
$\pi\in\Pi_\psi(G)$. Any unramified mismatch is detected by the finite
Tannakian detector $(v,r)$. If no such detector sees a mismatch,
Tannakian separation and Chebotarev fix the semisimple unramified
readout; any remaining discrepancy is ramified, endoscopic, continuous,
boundary, multiplicity, or normalization ledger data. No-hidden residual
and closedness exclude unresolved residuals.
\end{proof}

\begin{theorem}[Stable multiplicity certificate boundary]
\label{thm:unified-reality-stable-multiplicity-certificate-boundary}
The stable multiplicity formula certificate closes packet nonemptiness
from a nonzero stable distribution once packet labels, component-group
Fourier data, multiplicities, endoscopic transfer, and ledger rows are
supplied. It does not construct Arthur packets, prove the stable
multiplicity formula, or prove endoscopic transfer.
\end{theorem}
\begin{proof}
The arguments are finite Fourier, linear distribution, and ledger
arguments after the traditional packet and multiplicity data are
present. The construction of those data remains the independent
mathematical input.
\end{proof}
