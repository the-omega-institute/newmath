\section{BEDC--Langlands Tannakian Functorial Closure}
\label{sec:unified-reality-bedc-langlands-tannakian-functorial-closure}

The $GL_n$ converse bridge constructs automorphic realizations for
linear representations of an $L$-parameter. For a general reductive
group $G$, the finite readouts are the finite-dimensional
representations of ${}^LG$. The question is not whether each composed
parameter
$$
r\circ\phi:W'_F\to GL(V_r)
$$
has a $GL(V_r)$-realization, but whether the resulting family of
$GL$-objects is tensor-compatible and therefore comes from one
${}^LG$-valued packet.

\begin{definition}[Tannakian detector]
\label{def:unified-reality-tannakian-detector}
A Tannakian detector for $G$ is a pair
$$
(v,r)
$$
where $v$ is an unramified place in the certified scope and
$$
r:{}^LG\to GL(V_r)
$$
is a finite-dimensional representation. For a parameter $\phi$, the
parameter-side readout is
$$
a_{v,r}(\phi)
:=
\operatorname{tr}r(\phi(\operatorname{Frob}_v)).
$$
For a $G$-side automorphic packet $\Pi_G$ with Satake parameter
$s_v(\Pi_G)$, the automorphic readout is
$$
\lambda_{v,r}(\Pi_G)
:=
\operatorname{tr}r(s_v(\Pi_G)).
$$
The finite detector residual is
$$
D_{v,r}(\Pi_G,\phi)
:=
\lambda_{v,r}(\Pi_G)-a_{v,r}(\phi).
$$
\end{definition}

\begin{definition}[Representation compatibility rows]
\label{def:unified-reality-representation-compatibility-rows}
Let
$$
\{\Pi_r\}_{r\in\operatorname{Rep}({}^LG)}
$$
be a family of $GL(V_r)$-side automorphic objects obtained from the
composed parameters $r\circ\phi$. Its Tannakian compatibility rows are
the finite relations generated by direct sum, tensor product, and dual:
$$
\Pi_{r_1\oplus r_2}
\simeq
\Pi_{r_1}\boxplus\Pi_{r_2},
$$
$$
L(s,\Pi_{r_1\otimes r_2})
=
L(s,\Pi_{r_1}\times\Pi_{r_2}),
$$
and
$$
\Pi_{r^\vee}\simeq\Pi_r^\vee.
$$
Failures of these rows are denoted
$$
R_{\oplus}(r_1,r_2),\qquad
R_{\otimes}(r_1,r_2),\qquad
R_{\vee}(r),
$$
and form the Tannakian compatibility ledger.
\end{definition}

\begin{theorem}[Tannakian separation]
\label{thm:unified-reality-tannakian-functorial-separation}
Let $g,h\in{}^LG$ be semisimple. If
$$
\operatorname{tr}r(g)=\operatorname{tr}r(h)
$$
for every finite-dimensional representation $r$ of ${}^LG$, then
$g$ and $h$ have the same semisimple conjugacy class.
\end{theorem}
\begin{proof}
This is the detector separation theorem of
\autoref{thm:unified-reality-tannakian-detectors-separate-classes}.
Characters of finite-dimensional representations generate the
coordinate algebra of the semisimple conjugacy quotient
$$
{}^LG/\!/\operatorname{conj}.
$$
Equality of all character values therefore puts $g$ and $h$ at the same
point of that quotient. Since both elements are semisimple, this point
determines the semisimple conjugacy class.
\end{proof}

\begin{definition}[Tannakian generator]
\label{def:unified-reality-tannakian-generator}
A finite set
$$
\mathcal R_0=\{r_1,\ldots,r_m\}
$$
of finite-dimensional representations of ${}^LG$ is a Tannakian
generator when the representation ring is generated by $\mathcal R_0$
under $\oplus$, $\otimes$, and ${}^\vee$, and tensor-word characters in
these generators separate semisimple conjugacy classes.
\end{definition}

\begin{theorem}[Generator detection]
\label{thm:unified-reality-tannakian-generator-detection}
Let $\mathcal R_0$ be a Tannakian generator. If two semisimple elements
$g,h\in{}^LG$ have equal trace under every tensor word in
$\mathcal R_0$, then they have the same semisimple conjugacy class.
\end{theorem}
\begin{proof}
Every finite-dimensional character is an integral linear combination of
characters of tensor words built from $\mathcal R_0$, because
$\mathcal R_0$ generates the representation ring. Equality on all
tensor words therefore implies equality of
$\operatorname{tr}r(g)$ and $\operatorname{tr}r(h)$ for every
finite-dimensional representation $r$. The result follows from
\autoref{thm:unified-reality-tannakian-functorial-separation}.
\end{proof}

\begin{definition}[Tannakian residual dichotomy data]
\label{def:unified-reality-tannakian-residual-dichotomy-data}
A finite Tannakian residual is a failure of one displayed finite
representation relation, such as
$$
R_{\otimes}(r_1,r_2)\ne0,
$$
inside a fixed finite representation window. A representation-tail
residual is a compatibility failure invisible in every fixed finite
representation window but present along unbounded highest weight,
tensor depth, multiplicity, local factor complexity, or functorial
compatibility complexity. Its ledger is
$$
\mathsf{RepTailLedger}
:=
(\mathsf{HighestWeightEscape},\mathsf{TensorDepthEscape},
\mathsf{MultiplicityEscape},\mathsf{NonuniformLFactorBound},
\mathsf{FunctorialCompatibilityTail}).
$$
\end{definition}

\begin{theorem}[Tannakian residual dichotomy]
\label{thm:unified-reality-tannakian-residual-dichotomy}
If a family $\{\Pi_r\}_{r\in\operatorname{Rep}({}^LG)}$ of
$GL$-transfers does not come from one tensor-compatible
${}^LG$-readout system, then there is a finite Tannakian residual or a
representation-tail residual.
\end{theorem}
\begin{proof}
Failure to come from one tensor-compatible readout means that some
relation in the tensor category $\operatorname{Rep}({}^LG)$ is not
respected by the family $\{\Pi_r\}$. If the failing relation uses a
fixed finite expression in $\oplus$, $\otimes$, and ${}^\vee$, it is a
finite Tannakian residual. If every fixed finite expression is
compatible but no global tensor-compatible readout exists, the defect is
not located in any finite representation window. It can only appear
along unbounded representation complexity, which is exactly the
representation-tail residual.
\end{proof}

\begin{definition}[No hidden Tannakian residual]
\label{def:unified-reality-no-hidden-tannakian-residual}
The no-hidden Tannakian residual condition says that every failure of
representation compatibility is either detected by a finite
representation relation or is recorded in $\mathsf{RepTailLedger}$, and
that a closed accepted system carries no unresolved entry in either
place.
\end{definition}

\begin{theorem}[Tannakian closure]
\label{thm:unified-reality-tannakian-closure}
Let $\{\Pi_r\}$ be a family of $GL$-transfers with the BEDC
three-properties ledger, no-hidden Tannakian residual, and closedness.
Then the family determines a unique semisimple ${}^LG$-readout system on
the certified unramified scope.
\end{theorem}
\begin{proof}
If the family did not determine one tensor-compatible readout system,
\autoref{thm:unified-reality-tannakian-residual-dichotomy} would give a
finite Tannakian residual or a representation-tail residual. A finite
residual is detected by a finite representation relation. It must be
entered in the ledger, leaving the system not closed, or omitted,
violating the ledger property. A tail residual is handled in the same
way by $\mathsf{RepTailLedger}$. Both alternatives contradict the
closed no-hidden hypotheses. Thus the family is tensor-compatible.

For each certified unramified place $v$, the compatible character
system on $\operatorname{Rep}({}^LG)$ defines a point of
${}^LG/\!/\operatorname{conj}$. Uniqueness follows from
\autoref{thm:unified-reality-tannakian-functorial-separation}.
\end{proof}

\begin{theorem}[Reconstruction from $GL_N$ transfers]
\label{thm:unified-reality-reconstruction-from-gln-transfers}
Let $\mathcal R_0$ be a Tannakian generator. Assume that for every
tensor word $w$ in $\mathcal R_0$, the composed parameter
$$
w\circ\phi
$$
has a $GL(V_w)$-automorphic realization $\Pi_w$ supplied by the
$GL_N$ local and global closure bridges. If these transfers satisfy
no-hidden Tannakian residual and closedness, then for every certified
unramified place $v$ there is a unique semisimple class
$$
s_v\in{}^LG/\!/\operatorname{conj}
$$
such that
$$
\operatorname{tr}r(s_v)=\lambda_{v,r}(\Pi_r)
$$
for every finite-dimensional representation $r$ in the generated
readout scope.
\end{theorem}
\begin{proof}
The $GL_N$ closure bridges give the automorphic readouts for the
generators and their tensor words. By
\autoref{thm:unified-reality-tannakian-closure}, no closed compatible
system can have a finite or tail Tannakian residual. Therefore the
readouts form a tensor-compatible character system on
$\operatorname{Rep}({}^LG)$. Tannakian separation gives a unique
semisimple class $s_v$ realizing those character values.
\end{proof}

\begin{theorem}[General $G$ finite-readout closure]
\label{thm:unified-reality-general-g-finite-readout-closure}
In the setting of
\autoref{thm:unified-reality-reconstruction-from-gln-transfers}, the
reconstructed semisimple class agrees with the original parameter
readout:
$$
s_v=\phi(\operatorname{Frob}_v)
$$
in ${}^LG/\!/\operatorname{conj}$ for every certified unramified place
$v$. Hence for every finite-dimensional representation $r$ in the
readout scope,
$$
\lambda_{v,r}(\Pi_r)
=
\operatorname{tr}r(\phi(\operatorname{Frob}_v)).
$$
\end{theorem}
\begin{proof}
Each $\Pi_r$ was constructed from the composed $GL(V_r)$-parameter
$r\circ\phi$, so its unramified readout is
$$
\lambda_{v,r}(\Pi_r)
=
\operatorname{tr}r(\phi(\operatorname{Frob}_v)).
$$
Thus the reconstructed class $s_v$ and the class
$\phi(\operatorname{Frob}_v)$ have the same trace under every generated
finite-dimensional representation. By generator detection, they are the
same semisimple class.
\end{proof}

\begin{definition}[General $G$ existence bridge certificate]
\label{def:unified-reality-general-g-existence-bridge-cert}
A general $G$ existence bridge certificate says that a tensor-compatible
system of $GL_N$ transfers attached to a parameter $\phi$ either
descends to a $G$-side automorphic packet
$$
\Pi_\phi^G
$$
or enters the existence ledger
$$
\mathsf{ExistenceLedger}_G
:=
(\mathsf{MissingGPacket},\mathsf{EndoscopicPacketDefect},
\mathsf{ArthurPacketDefect},\mathsf{ContinuousSpectrumDefect},
\mathsf{BoundaryDefect},\mathsf{GeometricFunctorDefect}).
$$
The certificate may be supplied by stable trace formula data,
endoscopic classification, Arthur packet construction, a geometric
Langlands functor, or a cohomological realization on the declared scope.
\end{definition}

\begin{theorem}[General $G$ BEDC reciprocity]
\label{thm:unified-reality-general-g-bedc-reciprocity}
Assume:
\begin{itemize}
\item $GL_N$ transfers exist for a Tannakian generator and its tensor
words;
\item the family satisfies no-hidden Tannakian residual and closedness;
\item the general $G$ existence bridge certificate is present;
\item no-hidden Langlands residual and closedness hold for the resulting
$G$-packet.
\end{itemize}
Then there is a closed $G$-side automorphic packet $\Pi_\phi^G$ such
that, for almost all certified unramified $v$ and all finite-dimensional
$r:{}^LG\to GL(V_r)$ in the readout scope,
$$
\lambda_{v,r}(\Pi_\phi^G)
=
\operatorname{tr}r(\phi(\operatorname{Frob}_v)).
$$
\end{theorem}
\begin{proof}
By
\autoref{thm:unified-reality-general-g-finite-readout-closure}, the
$GL_N$ transfers are compatible with the single parameter readout
$\phi$. The general $G$ existence bridge certificate says that the
compatible system descends to a $G$-side packet or enters
$\mathsf{ExistenceLedger}_G$. An unresolved existence-ledger entry is
not compatible with closedness; omitting it violates the ledger
property. Hence the packet $\Pi_\phi^G$ exists in the closed case.

If a Hecke--Frobenius mismatch remains, the no-hidden Langlands
residual condition makes it a finite detector residual or a ledger row.
The first contradicts the finite readout closure; the second contradicts
closedness while unresolved. The displayed equality follows.
\end{proof}

\begin{theorem}[Tannakian functoriality closure]
\label{thm:unified-reality-tannakian-functoriality-closure}
Let
$$
\eta:{}^LH\to{}^LG
$$
be an $L$-group homomorphism and let $(\Pi_H,\phi_H)$ be a closed
$H$-side packet. Put
$$
\phi_G:=\eta\circ\phi_H.
$$
If $\phi_G$ satisfies the hypotheses of
\autoref{thm:unified-reality-general-g-bedc-reciprocity}, then there is
a closed $G$-side packet $\Pi_G$ satisfying
$$
\lambda_{v,r}(\Pi_G)
=
\operatorname{tr}r(\eta(\phi_H(\operatorname{Frob}_v)))
$$
for almost all certified unramified $v$ and all finite-dimensional
$r:{}^LG\to GL(V_r)$ in the readout scope.
\end{theorem}
\begin{proof}
Apply
\autoref{thm:unified-reality-general-g-bedc-reciprocity} to the
parameter $\phi_G=\eta\circ\phi_H$. The resulting $G$-packet has
readouts
$$
\lambda_{v,r}(\Pi_G)
=
\operatorname{tr}r(\phi_G(\operatorname{Frob}_v)).
$$
Substituting the definition of $\phi_G$ gives the displayed formula.
\end{proof}

\begin{corollary}[$GL_1$ as the minimal Tannakian model]
\label{cor:unified-reality-gl1-minimal-tannakian-model}
For ${}^LG=\mathbb C^\times$, the representation category is generated
by the characters
$$
z\mapsto z^k.
$$
Tannakian compatibility is the identity
$$
z^{k+\ell}=z^kz^\ell.
$$
For the quadratic character attached to $\mathbb Q(i)/\mathbb Q$, the
visible unramified detector
$$
\chi_{-4}(\operatorname{Frob}_p)=1
$$
is equivalent to $p=a^2+b^2$ for odd primes. Thus the two-square
readout is the minimal abelian shadow of the general Tannakian detector
system.
\end{corollary}
\begin{proof}
For $\mathbb C^\times$, every finite-dimensional semisimple
representation is a finite direct sum of characters $z^k$, and tensor
product adds exponents. The quadratic character $\chi_{-4}$ is the
class-field detector for the extension $\mathbb Q(i)/\mathbb Q$; its
unramified value $1$ is exactly the splitting condition for odd primes,
equivalently $p=a^2+b^2$.
\end{proof}

\begin{theorem}[Tannakian closure boundary]
\label{thm:unified-reality-tannakian-closure-boundary}
Tannakian functorial closure proves that compatible $GL_N$ readouts
determine the finite detector system of a general $G$-parameter, and
that incompatibility is finite-visible or ledgered as representation
tail. It does not by itself construct the missing $G$-automorphic
packet; that is exactly the role of the general $G$ existence bridge.
\end{theorem}
\begin{proof}
The arguments above are representation-category and ledger arguments:
they use $GL_N$ transfers, tensor compatibility, Tannakian separation,
and no-hidden residual discipline. The descent from compatible
$GL_N$ readouts to an actual $G$-side automorphic packet is a separate
existence statement, represented by
\autoref{def:unified-reality-general-g-existence-bridge-cert}.
\end{proof}
