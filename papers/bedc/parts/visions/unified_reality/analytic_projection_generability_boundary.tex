\section{Analytic Projection and Closed Generability Boundary}
\label{sec:unified-reality-analytic-projection-generability-boundary}

The analytic continuation row must be separated from the closed
generation row.  Analytic continuation recovers a projected analytic
object; it does not, by itself, certify that every analytic feature of
that projection is generated by the closed triadic dynamics used in the
RH-facing route.  The distinction is between projected analytic
existence and closed-system realizability.

\begin{definition}[Analytic projection carrier]
\label{def:unified-reality-analytic-projection-carrier}
The \emph{analytic projection carrier} for zeta is the carrier
$$
\mathcal A_{\zeta}
$$
whose rows are the analytic readouts produced by the Dirichlet-series
domain, Euler-product readout on its convergence side, Mellin or contour
representations, analytic continuation, pole ledger, functional equation,
and completed value projection.  A row in $\mathcal A_{\zeta}$ is
therefore a projection-side analytic datum.  It is licensed as an
analytic object, not automatically as a closed dynamical state.
\end{definition}

\begin{definition}[Closed generability carrier]
\label{def:unified-reality-closed-generability-carrier}
The \emph{closed generability carrier} is the subcarrier
$$
\mathcal G_{\zeta}\subseteq\mathcal A_{\zeta}
$$
consisting of analytic projection rows which also carry a closed
triadic generation certificate.  Such a certificate displays:
$$
\mathsf{ActiveCapture},
\qquad
\mathsf{CompactPhaseOrSolenoidFibre},
\qquad
\mathsf{RoutingCompatibility},
$$
$$
\mathsf{TriadicClosure},
\qquad
\mathsf{DefectExtraction},
\qquad
\mathsf{BoundaryStability}.
$$
Thus $\mathcal G_{\zeta}$ records the analytic rows which are not merely
visible in the completed projection, but are generated through finite
packets, compatible routing, legal hidden fibres, and unit-boundary
stability.
\end{definition}

\begin{principle}[Analytic continuation is not generability]
\label{prin:unified-reality-analytic-continuation-not-generability}
Membership in $\mathcal A_{\zeta}$ does not imply membership in
$\mathcal G_{\zeta}$.  Analytic continuation supplies the projected
analytic object and its uniqueness in the analytic category.  Closed
generability requires additional source rows: zero-relevant packet
capture, compatible finite routing, compact phase or solenoid fibre
placement, and a unit-boundary or energy-stability certificate.
\end{principle}

\begin{definition}[Non-generable analytic residue]
\label{def:unified-reality-non-generable-analytic-residue}
A row $x\in\mathcal A_{\zeta}$ has a \emph{non-generable analytic
residue} when it requires at least one of the following data without a
closed certificate:
$$
\text{a nontrivial real positive scale }e^{-\delta h},
$$
$$
\text{a hidden coordinate outside compact phase or prime-register fibre},
$$
$$
\text{a projected zero with no active Rouch\'e or local-degree capture},
$$
$$
\text{a finite routing table with no inverse-limit compatibility},
$$
or
$$
\text{a zero-relevant packet with no unit-boundary stability row}.
$$
Such a row may still belong to the analytic projection carrier.  The
point is only that it has not been shown to belong to the closed
generability carrier.
\end{definition}

\begin{theorem}[Real positive scale is not a closed compact residue]
\label{thm:unified-reality-real-positive-scale-not-closed-compact-residue}
Let a closed generability certificate for a local channel use only
unitary phase and compact solenoid prime-register fibres for its hidden
coordinates.  For
$$
s=\frac12+\delta+i\tau,
\qquad
U_a(s)=e^{-(s-1/2)h_a},
\qquad
h_a>0,
$$
the factor
$$
e^{-\delta h_a}
$$
can be supplied by that compact hidden carrier only when $\delta=0$.
\end{theorem}

\begin{proof}
The local channel factors as
$$
U_a(s)=e^{-\delta h_a}e^{-i\tau h_a}.
$$
The phase factor belongs to the unit-circle readout.  A compact solenoid
fibre can record compact prime-register data, but by
\autoref{lem:unified-reality-compact-phase-no-positive-scale-character}
it has no nontrivial continuous character into $\RealUp_{>0}$.  Hence the
real positive factor must be trivial:
$$
e^{-\delta h_a}=1.
$$
Since $h_a>0$, this gives $\delta=0$.
\end{proof}

\begin{definition}[Zeta zero generability row]
\label{def:unified-reality-zeta-zero-generability-row}
The \emph{zeta zero generability row} is the source assertion
$$
\mathsf{ZetaZeroGenerability}_{\zeta}
:=
\prod_{s:\ComplexUp}
\left(
\mathsf{ZetaZero}(s)
\times
\mathsf{InCritStrip}(s)
\to
s\in\mathcal G_{\zeta}
\right).
$$
It says that every critical-strip zero of the completed zeta projection
has a closed triadic generation certificate.  It is not supplied by
analytic continuation alone.
\end{definition}

\begin{definition}[Closed triadic no-real-scale row]
\label{def:unified-reality-closed-triadic-no-real-scale-row}
The \emph{closed triadic no-real-scale row} is the assertion
$$
\prod_{s:\ComplexUp}
\left(
s\in\mathcal G_{\zeta}
\to
\Re(s)=\frac12
\right),
$$
with the implication witnessed by defect extraction, cofinal growth, and
unit-boundary or compact-solenoid stability for the same zero-relevant
packet.
\end{definition}

\begin{theorem}[Generability and no-real-scale rows are conditionally sufficient]
\label{thm:unified-reality-generability-no-real-scale-conditionally-sufficient}
If
$$
\mathsf{ZetaZeroGenerability}_{\zeta}
$$
and the closed triadic no-real-scale row both hold, then the RH
fixed-half section exists.
\end{theorem}

\begin{proof}
Let $z:\mathcal Z_{\zeta}$ have coordinate $s$.  The zero generability
row gives
$$
s\in\mathcal G_{\zeta}.
$$
The no-real-scale row then gives
$$
\Re(s)=\frac12.
$$
Varying $z$ gives the fixed-half section, equivalently the RH section
identified by
\autoref{thm:unified-reality-rh-fixed-half-section-equivalence}.
\end{proof}

\begin{corollary}[The hard source row is generability]
\label{cor:unified-reality-hard-source-row-is-generability}
For the analytic-projection route, the hard source row is not the
exclusion of real scale inside an already closed packet.  It is the
construction of a closed packet for each completed zero:
$$
\mathsf{ZetaZeroGenerability}_{\zeta}.
$$
\end{corollary}

\begin{proof}
Once a zero row is in $\mathcal G_{\zeta}$, the compact hidden-fibre
argument of
\autoref{thm:unified-reality-real-positive-scale-not-closed-compact-residue}
or the energy-bound argument of
\autoref{thm:unified-reality-public-triadic-verification-sound-fixed-half}
excludes an off-half real scale.  What those arguments do not supply is
membership in $\mathcal G_{\zeta}$.  That membership requires active
capture, dominance or tail control, compatible routing, solenoid
placement, and boundary stability.  This is the same source load
isolated in
\autoref{thm:unified-reality-source-gap-remaining-theorem}.
\end{proof}

\begin{corollary}[Projection existence and generability remain separate]
\label{cor:unified-reality-projection-existence-generability-separate}
The existence of the analytic continuation of zeta, including its
completed functional-equation projection, does not discharge the
generability row.
\end{corollary}

\begin{proof}
The analytic continuation theorem places the completed zeta readout in
$\mathcal A_{\zeta}$.  By
\autoref{prin:unified-reality-analytic-continuation-not-generability},
membership in $\mathcal A_{\zeta}$ does not provide the finite packet,
routing, solenoid, capture, and boundary rows required for
$\mathcal G_{\zeta}$.  Therefore the generability row is an additional
source theorem, not a consequence of analytic continuation as such.
\end{proof}
