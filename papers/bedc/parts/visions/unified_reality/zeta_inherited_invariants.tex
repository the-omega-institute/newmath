\section{Inherited Invariants of the Zeta Endpoint}
\label{sec:unified-reality-zeta-inherited-invariants}

\autoref{sec:unified-reality-generalized-continuation-chain} separates
the generalized continuation chain from ordinary holomorphic
continuation. The present section records the carrier-level invariants
that reach the zeta endpoint through that chain. The point is not that
zeta contains a literal triangle, circle, Euclidean ellipse, Fourier
series, or Laplace transform as a visible subobject. The point is that
normalization, unit phase, frequency, damping, local-to-global assembly,
symmetry, cancellation, and defect-rigidity have transported zeta-side
readouts.

\begin{theorem}[Scale normalization transports to centered prime channels]
\label{thm:unified-reality-zeta-centered-prime-channel}
The Pythagorean scale-normalization row transports to the zeta centered
prime channel
$$
p^{-s}\mapsto p^{1/2-s}.
$$
\end{theorem}

\begin{proof}
In the Pythagorean carrier, a nonzero scale $c$ licenses the normalized
shape read
$$
(a,b,c)\mapsto(a/c,b/c),
$$
as in
\autoref{thm:unified-reality-triangle-circle-scale-fiber}. In the zeta
prime-local carrier, the raw channel is $p^{-s}$ and the fixed-half
center is $1/2$. Centering by that normalization produces
$$
p^{1/2}p^{-s}=p^{1/2-s}.
$$
This is a transported normalization row, not a literal Pythagorean
equation inside zeta.
\end{proof}

\begin{theorem}[Unit norm transports to the critical line]
\label{thm:unified-reality-zeta-unitary-critical-line}

\begin{proof}

\begin{theorem}[Fourier frequency transports to logarithmic prime clocks]
\label{thm:unified-reality-zeta-log-prime-frequency}
The Fourier phase row transports to the zeta prime-channel phase

\begin{proof}

The second factor is the Fourier phase row with clock $\log p$.
\end{proof}

\begin{theorem}[Off-critical bias produces finite prime-window defect]
\label{thm:unified-reality-zeta-off-critical-defect}
If $s=1/2+\delta+i\tau$, $\delta\#0$, and $S$ is a nonempty finite prime
window, then the prime-window defect
$$
D_S(s):=\sum_{p\in S}\log|p^{1/2-s}|
$$
is apart from zero.
\end{theorem}

\begin{proof}
For every $p\in S$,
$$
|p^{1/2-s}|=p^{-\delta}.
$$
Therefore
$$
D_S(s)
=
\sum_{p\in S}\log(p^{-\delta})
=
-\delta\sum_{p\in S}\log p.
$$
Since $S$ is nonempty and each displayed prime satisfies $p>1$, the sum
of logarithms is positive. Apartness of $\delta$ from zero gives
apartness of $D_S(s)$ from zero. This is the unified-reality read of
\autoref{thm:unified-reality-rh-off-half-creates-prime-defect}.
\end{proof}
