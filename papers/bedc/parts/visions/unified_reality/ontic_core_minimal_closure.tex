\section{Ontic Core and Minimal Closure}
\label{sec:unified-reality-ontic-core-minimal-closure}

\autoref{sec:unified-reality-zeta-ontic-projection} reads zeta as an
arithmetic-spectral projection of atom, time, and symmetry. If the
object-level examples are bracketed and only the ontic architecture is
kept, the triple becomes
$$
\mathsf{OnticCore}
:=
(\mathsf{ObsTime},\mathsf{DistAtom},\mathsf{SymTransport}).
$$
The three entries are observation as time, distinction as atom, and
persistence as symmetry transport. The third entry is not shape
symmetry alone. It is classifier-stable transport that preserves
distinction through observation-time.

\begin{definition}[Observation-time row]
\label{def:unified-reality-observation-time-row}
An \emph{observation-time row} is a finite readout event carried by a
history
$$
h:\Hist
$$
together with the continuation relation
$$
h\preceq h'
:\Longleftrightarrow
\Cont(h,h').
$$
In this row, time is not an external parameter. It is the readable order
of observations under continuation.
\end{definition}

\begin{principle}[Time is observable continuation]
\label{prin:unified-reality-time-observable-continuation}
Inside the BEDC carrier discipline, time is the continuation order of
observation histories. Without observed histories there is no internal
time readout; without continuation there are only isolated readings.
\end{principle}

\begin{definition}[Distinction-atom row]
\label{def:unified-reality-distinction-atom-row}
A \emph{distinction-atom row} is an irreducible stable distinction in a
displayed reality. It has the abstract form
$$
x\# y
$$
when the current classifier cannot identify the two rows, and the
atomic form
$$
\mathsf{Atom}(a)
$$
when $a$ is a minimal stable distinction under the generation law of
that reality. Number-theoretic primes are the multiplicative arithmetic
projection of this row, not its only instance.
\end{definition}

\begin{definition}[Symmetry-transport row]
\label{def:unified-reality-symmetry-transport-row}
A \emph{symmetry-transport row} is a transport
$$
g:X\to X
$$
that preserves classifier sameness and reflected distinction:
$$
x\sim_X y
\to
g(x)\sim_X g(y),
$$
and
$$
x\# y
\Longleftrightarrow
g(x)\# g(y).
$$
When $g$ also preserves continuation,
$$
h\preceq h'
\to
g(h)\preceq g(h'),
$$
it is an ontic symmetry transport. In BEDC terms this is the role played
by $\hsame$-respect, $\Cont$, classifier stability, $\StabilityCert$,
and $\NameCert$.
\end{definition}

\begin{theorem}[Symmetry is distinction preserved through observation-time]
\label{thm:unified-reality-symmetry-distinction-through-time}
The symmetry-transport row is precisely the preservation of
distinction-atom data along observation-time.
\end{theorem}

\begin{proof}
Observation-time supplies histories and continuation rows
$h\preceq h'$. Distinction-atom data supplies the classifier boundary
$x\# y$. A symmetry transport is admissible only when it sends
continuation rows to continuation rows and preserves the classifier
boundary:
$$
h\preceq h'
\to
g(h)\preceq g(h'),
$$
$$
x\# y
\Longleftrightarrow
g(x)\# g(y).
$$
Thus the content of symmetry is not visual resemblance. It is stable
transport of distinction through observation-time.
\end{proof}

\begin{corollary}[Observation, distinction, and persistence form a loop]
\label{cor:unified-reality-observation-distinction-persistence-loop}
The ontic core is the loop
$$
\mathsf{ObsTime}
\to
\mathsf{DistAtom}
\to
\mathsf{SymTransport}
\to
\mathsf{ObsTime}.
$$
Observation supplies readable time; distinction supplies observable
content; symmetry transport lets that content persist across time.
\end{corollary}

\begin{proof}
Without observation-time, distinction does not enter a generated
history. Without distinction, observation has no detectable content.
Without symmetry transport, a distinction in one observation cannot be
tracked through later observations. Hence the three rows are not
independent decorations; they close one carrier loop.
\end{proof}

\begin{definition}[Minimal ontic core]
\label{def:unified-reality-minimal-ontic-core}
The \emph{minimal ontic core} is
$$
\mathsf{OnticCore}
:=
(\mathsf{Observation},\mathsf{Distinction},\mathsf{Persistence})
$$
with the readings
$$
\mathsf{Observation}
=
\text{finite readout producing time},
$$
$$
\mathsf{Distinction}
=
\text{irreducible difference producing atom},
$$
and
$$
\mathsf{Persistence}
=
\text{transport preserving distinction across observation-time}.
$$
The persistence row is the ontic reading of symmetry.
\end{definition}

\begin{theorem}[Stable interfaces require the ontic core]
\label{thm:unified-reality-stable-interfaces-require-ontic-core}
A stable BEDC interface must display observation-time, distinction-atom,
and symmetry-transport ledgers before it can be consumed as a mature
named object.
\end{theorem}

\begin{proof}
The roadmap presents mature objects through a $\NameCert$ surface
$$
(\SourceSpec_N,\PatternSpec_N,\ClassifierSpec_N,\StabilityCert_N,
\LedgerPolicy_N).
$$
The source and pattern rows expose how the object is observed or
generated. The classifier row exposes sameness and distinction. The
stability certificate exposes preservation of the classifier under
transport. The ledger policy records what is hidden, scoped, or refused.
These are precisely the observation-time, distinction-atom, and
symmetry-transport roles of
\autoref{def:unified-reality-minimal-ontic-core}.
\end{proof}

\begin{corollary}[Kernel reading of the ontic core]
\label{cor:unified-reality-kernel-reading-ontic-core}
The BEDC kernel reads the ontic core through the spine
$$
\Mark \to \Hist \to \Ext,\Cont \to \Ask,\Sig \to \Pkg,\Gap \to
\NameCert.
$$
Here histories and continuations read observation-time, marks and
classifier gaps read distinction-atom, and $\hsame$-respect,
$\Cont$-respect, $\StabilityCert$, and $\NameCert$ read
symmetry-transport.
\end{corollary}

\begin{proof}
The displayed spine is the kernel-layer routing surface of the roadmap.
The earlier layers produce readable histories and continuation. The
classifier and gap layers keep distinctions from being collapsed. The
package and naming layers admit only transports whose stability and
ledger rows have been displayed.
\end{proof}

\begin{definition}[Ontic collapse of the core]
\label{def:unified-reality-ontic-core-collapse}
An \emph{ontic core collapse} is an unaudited step that erases one of
the three core rows:
$$
(\exists x,\ P(x))\to\sum_x P(x),
$$
$$
x\sim y\to [x]=[y],
$$
or
$$
(P\leftrightarrow Q)\to(P=Q).
$$
These are respectively collapse of observation-time provenance,
symmetry-transport fibers, and distinction-atom provenance.
\end{definition}

\begin{theorem}[The three excluded axioms collapse the ontic core]
\label{thm:unified-reality-three-excluded-axioms-collapse-ontic-core}
Read through the minimal ontic core, the three excluded dependencies
collapse the three core rows:
$$
\mathsf{Classical.choice}
\quad\text{collapses observation-time provenance},
$$
$$
\mathsf{Quot.sound}
\quad\text{collapses symmetry-transport fibers},
$$
and
$$
\mathsf{propext}
\quad\text{collapses distinction provenance}.
$$
\end{theorem}

\begin{proof}
This is the ontic-core reading of
\autoref{thm:unified-reality-excluded-axioms-collapse-modes} and
\autoref{thm:three-axioms-unification}. Choice supplies a witness
without the generating path. $\mathsf{Quot.sound}$ turns a classifier
relation into quotient equality without retaining the transport fiber.
$\mathsf{propext}$ turns bidirectional implication into proposition
equality without retaining the two directions as distinct provenance.
\end{proof}

\begin{corollary}[Axiom-purity protects observation, distinction, and persistence]
\label{cor:unified-reality-axiom-purity-protects-ontic-core}
The axiom-purity discipline protects the minimal ontic core by refusing
unaudited collapse of observation-time, distinction-atom, and
symmetry-transport rows.
\end{corollary}

\begin{proof}
By
\autoref{thm:unified-reality-three-excluded-axioms-collapse-ontic-core},
the excluded dependencies correspond exactly to the three collapse
modes. Refusing them preserves the ledgered core rather than compressing
it into an oracle value, quotient equality, or proposition equality.
\end{proof}

\begin{theorem}[Minimal closure of the ontic core]
\label{thm:unified-reality-minimal-closure-ontic-core}
Observation, distinction, and persistence form the minimal closure
needed for a stable object interface.
\end{theorem}

\begin{proof}
If observation and distinction are present without persistence, a
distinction can be seen but cannot be transported through later
observations; the result is not a stable object. If observation and
persistence are present without distinction, the continuation has no
detectable content. If distinction and persistence are present without
observation, there is no history or continuation surface on which the
distinction can be generated. Hence all three rows are required.
\end{proof}

\begin{theorem}[Ontic-core projection principle]
\label{thm:unified-reality-ontic-core-projection-principle}
Mature mathematical objects are projections of the minimal ontic core.
Zeta is one dense arithmetic-spectral projection; it is not the core
itself.
\end{theorem}

\begin{proof}
For $\NatUp$, observation-time is successor history, distinction is the
zero/successor split, and persistence is induction stability. For
$\RealUp$, observation-time is stream/modulus data, distinction is
apartness, and persistence is classifier stability. For the phase and
transform ladder, observation-time appears as phase or logarithmic
time, distinction as modes or prime-local factors, and persistence as
duality, unitary balance, or functional symmetry. The zeta projection
compresses these rows into prime factors, log-time Laplace readout, and
the mirror $J(s)=1-\overline{s}$, as in
\autoref{thm:unified-reality-zeta-carries-ontic-triple}. Thus zeta is a
projection of the core, not an exhaustive replacement for it.
\end{proof}
