\section{Ontic Triple Invariants and the Zeta Projection}
\label{sec:unified-reality-zeta-ontic-projection}

The inherited-invariant packet of
\autoref{sec:unified-reality-zeta-inherited-invariants} can be compressed
one level further. The recurring carrier data have three ontic roles:
atom, time, and symmetry. Zeta is not the total ontology of mathematics.
It is the arithmetic-spectral projection in which these three roles
become prime-local factors, logarithmic Laplace time, functional
symmetry, and a zero ledger. The RH-facing target is the zero-fiber
fixedness of this projection.

\begin{definition}[Ontic invariant packet]
\label{def:unified-reality-ontic-invariant-packet}
An \emph{ontic invariant packet} is a triple
$$
\mathcal I
:=
(\mathsf{Atom},\mathsf{Time},\mathsf{Sym}).
$$
The row $\mathsf{Atom}$ records irreducible generated distinction. The
row $\mathsf{Time}$ records generation order, continuation, filtration,
or spectral-frequency direction. The row $\mathsf{Sym}$ records
classifier preservation, duality, involution, or a fixed-surface
structure. Different mathematical realities need not share the same
objects; they may share the same ontic triple through different
projections.
\end{definition}

\begin{definition}[Atom row]
\label{def:unified-reality-ontic-atom-row}
For a reality interface $\mathcal R$, an \emph{atom row}
$$
\mathsf{Atom}_{\mathcal R}(a)
$$
asserts that $a$ is irreducible with respect to the displayed generation
law of $\mathcal R$, or that every permitted factorization of $a$ is
trivial in that reality. Typical reads include prime numbers in
multiplicative arithmetic, irreducible polynomials in a polynomial
reality, branch generators in a curve reality, and eigenmodes or
spectral branches in a spectral reality.
\end{definition}

\begin{definition}[Time row]
\label{def:unified-reality-ontic-time-row}
For a reality interface $\mathcal R$, a \emph{time row} is an ordered,
continued, filtered, or frequency parameter through which generated data
are read. Examples include $\Cont$ and history continuation in the BEDC
kernel, regular-modulus depth in located analysis, recurrence index
$m$, Fourier phase or frequency parameters, Laplace time $t$, and the
zeta logarithmic axis $\log n$.
\end{definition}

\begin{definition}[Symmetry row]
\label{def:unified-reality-ontic-symmetry-row}
For a reality interface $\mathcal R$, a \emph{symmetry row} is a
classifier-preserving involution, duality, reflection, or transport
operation
$$
J_{\mathcal R}:\mathcal R\to\mathcal R
$$
with fixed locus
$$
\mathsf{Fix}(J_{\mathcal R})
:=
\{x:J_{\mathcal R}(x)\sim x\}.
$$
The fixed locus is a balance surface only relative to the displayed
classifier and transport ledger.
\end{definition}

\begin{definition}[Zeta ontic projection carrier]
\label{def:unified-reality-zeta-ontic-projection-carrier}
The \emph{zeta ontic projection carrier} is the packet
$$
\mathcal O_{\zeta}^{\mathrm{ontic}}
:=
(
\mathsf{PrimeAtoms},
\mathsf{LogTime},
\mathsf{LaplaceReadout},
\mathsf{EulerProduct},
\mathsf{FunctionalSymmetry},
\mathsf{ZeroLedger}
).
$$
Its three ontic projections are
$$
\pi_{\mathrm{Atom}}(\mathcal O_{\zeta}^{\mathrm{ontic}})
=
\prod_p(1-p^{-s})^{-1},
$$
$$
\pi_{\mathrm{Time}}(\mathcal O_{\zeta}^{\mathrm{ontic}})
=
e^{-s\log n},
$$
and
$$
\pi_{\mathrm{Sym}}(\mathcal O_{\zeta}^{\mathrm{ontic}})
=
J(s)=1-\overline{s}.
$$
\end{definition}

\begin{theorem}[Zeta carries the ontic triple projection]
\label{thm:unified-reality-zeta-carries-ontic-triple}
The zeta ontic projection carrier carries atom, time, and symmetry
projection rows.
\end{theorem}

\begin{proof}
The atom projection is the Euler-product face
$$
\zeta(s)\sim\prod_p(1-p^{-s})^{-1},
$$
which exposes prime-local arithmetic atoms as in
\autoref{def:unified-reality-zeta-local-global-factorization}. The time
projection is the logarithmic Laplace row
$$
n^{-s}=e^{-s\log n},
$$
which is the arithmetic readout of
\autoref{thm:unified-reality-zeta-arithmetic-laplace-readout}. The
symmetry projection is the functional mirror
$$
J(s)=1-\overline{s},
$$
whose fixed surface is read in
\autoref{def:unified-reality-zeta-functional-symmetry-fixed-surface}.
Thus the same zeta carrier supports the three ontic roles.
\end{proof}

\begin{corollary}[Zeta is not only a value function]
\label{cor:unified-reality-zeta-not-only-value-function}
The readout $\zeta(s)$ is one projection of
$\mathcal O_{\zeta}^{\mathrm{ontic}}$, not the complete carrier. The
carrier also exposes prime atoms, logarithmic time, Euler-product
factorization, functional symmetry, and zero-ledger rows.
\end{corollary}

\begin{proof}
This is immediate from
\autoref{thm:unified-reality-zeta-carries-ontic-triple}. A function
value read does not by itself display the atom, time, symmetry, and
zero-ledger rows that make the zeta carrier relevant to the RH route.
\end{proof}

\begin{definition}[Prime atom and zero resonance]
\label{def:unified-reality-prime-atom-zero-resonance}
A \emph{prime atom} is a displayed row
$$
\mathsf{PrimeAtom}(p),
\qquad
p\in\PrimeUp,
$$
expressing irreducibility in the natural-number multiplicative reality.
A \emph{zero resonance} is a displayed row
$$
\mathsf{ZeroRes}(s):=\mathsf{ZetaZero}(s),
\qquad
s\in\ComplexUp.
$$
The rows live in different types. They may be coupled by the zeta
carrier, but they are not the same object.
\end{definition}

\begin{theorem}[Prime atoms and zero resonances are dual projections]
\label{thm:unified-reality-prime-zero-dual-projections}
Prime atoms and zero resonances are different typed projections of the
zeta carrier: primes are arithmetic-local atoms, while zeros are
spectral cancellation rows.
\end{theorem}

\begin{proof}
Prime atoms appear in the Euler-product face
$$
\prod_p(1-p^{-s})^{-1}.
$$
They belong to the arithmetic local-factor reality. Zero resonances
appear in the global value ledger
$$
\zeta(s)=0.
$$
They belong to the spectral cancellation reality described in
\autoref{thm:unified-reality-zeta-zero-global-cancellation-row}. Since
$p:\PrimeUp$ and $s:\ComplexUp$ have different carrier types and
classifiers, they cannot be identified. They are coupled only through
the common zeta carrier and its transform ledgers.
\end{proof}

\begin{corollary}[Prime-zero language is projection-relative]
\label{cor:unified-reality-prime-zero-language-projection-relative}
Statements relating primes and zeros must be read as carrier-coupling
statements, not as type-level identifications. The precise form is:
prime rows are arithmetic atoms; zero rows are spectral resonances; the
zeta carrier transports information between their realities.
\end{corollary}

\begin{proof}
This is the typed consequence of
\autoref{thm:unified-reality-prime-zero-dual-projections} and the
projection discipline of
\autoref{prin:unified-reality-no-shadow-replacement}.
\end{proof}

\begin{definition}[Ontic balance]
\label{def:unified-reality-ontic-balance}
For
$$
s=\frac12+\delta+i\tau,
$$
the centered prime channel is
$$
U_p(s):=p^{1/2-s}.
$$
The \emph{ontic balance} row is
$$
\mathsf{OnticBalance}(s)
:\Longleftrightarrow
\forall p\in\PrimeUp,\ |U_p(s)|\sim1.
$$
It is the atom-time-symmetry balance read of the zeta prime-channel
projection.
\end{definition}

\begin{theorem}[Ontic balance is the critical line surface]
\label{thm:unified-reality-ontic-balance-critical-line}
For a displayed prime row $p>1$,
$$
|U_p(s)|=1
\quad\Longleftrightarrow\quad
\Re(s)=\frac12.
$$
Thus $\mathsf{OnticBalance}(s)$ is the critical-line surface in the
prime-channel projection.
\end{theorem}

\begin{proof}
This is the same prime-unitary computation as
\autoref{thm:unified-reality-zeta-unitary-critical-line}. Writing
$s=\sigma+i\tau$ gives
$$
U_p(s)=p^{1/2-\sigma}e^{-i\tau\log p}.
$$
The phase factor has modulus one, so
$$
|U_p(s)|=p^{1/2-\sigma}.
$$
For $p>1$, this is $1$ exactly when $\sigma=1/2$.
\end{proof}

\begin{corollary}[Critical line as ontic balance surface]
\label{cor:unified-reality-critical-line-ontic-balance-surface}
The critical line is not defined by the zero predicate. It is the fixed
balance surface obtained from prime atoms, logarithmic time phase, and
functional symmetry.
\end{corollary}

\begin{proof}
The critical-line surface is the same as the ontic balance surface by
\autoref{thm:unified-reality-ontic-balance-critical-line}. The zero
predicate is a separate ledger read over the completed zeta carrier.
\end{proof}

\begin{definition}[Ontic collapse modes]
\label{def:unified-reality-ontic-collapse-modes}

\begin{theorem}[Excluded axioms are collapse modes]
\label{thm:unified-reality-excluded-axioms-collapse-modes}
Read through the ontic packet, $\mathsf{Classical.choice}$ is a
time/provenance collapse, $\mathsf{Quot.sound}$ is a symmetry/fiber
collapse, and $\mathsf{propext}$ is a distinction collapse.
\end{theorem}

\begin{proof}
The structural background is
\autoref{thm:three-axioms-unification}. Choice accepts a term-level
existential as if a witness and its generating path had been supplied,
so it collapses provenance time. $\mathsf{Quot.sound}$ accepts a
classifier relation as quotient equality, so it collapses the fiber
structure of symmetry. $\mathsf{propext}$ accepts bidirectional
transport between propositions as equality of proposition objects, so it
collapses predicate distinction. The axiom-purity gate described in
\autoref{rem:three-axioms-gate-meaning} refuses exactly these
unaudited closures at the public proof surface.
\end{proof}

\begin{corollary}[Axiom-purity protects the ontic ledgers]

BEDC's axiom-purity discipline preserves the three ledger distinctions:

\begin{proof}

\begin{definition}[Ontic defect]
\label{def:unified-reality-ontic-defect}
The \emph{ontic defect} row is the apartness form of failure of ontic
balance:
$$
\delta(s):=\Re(s)-\frac12,
\qquad
\mathsf{OnticDefect}(s):=\delta(s)\#0.
$$
For a nonempty finite prime window $S$, the defect readout is
$$
D_S(s):=\sum_{p\in S}\log|p^{1/2-s}|.
$$
\end{definition}

\begin{theorem}[Ontic defect is visible in finite prime windows]
\label{thm:unified-reality-ontic-defect-visible-prime-window}
If $\mathsf{OnticDefect}(s)$ holds and $S$ is a nonempty finite prime
window, then
$$
D_S(s)\#0.
$$
\end{theorem}

\begin{proof}
Write $s=1/2+\delta+i\tau$. Then
$$
|p^{1/2-s}|=p^{-\delta}.
$$
Therefore
$$
D_S(s)
=
\sum_{p\in S}\log(p^{-\delta})
=
-\delta\sum_{p\in S}\log p.
$$
If $S$ is nonempty and every displayed prime satisfies $p>1$, the
logarithmic sum is positive. Apartness of $\delta$ from zero gives
apartness of $D_S(s)$ from zero. This is the ontic reading of
\autoref{thm:unified-reality-zeta-off-critical-defect}.
\end{proof}

\begin{theorem}[Ontic incompatibility closes the section target]
\label{thm:unified-reality-ontic-incompatibility-closes-section-target}
Assume ontic defect-zero incompatibility and the located row that turns
absence of ontic defect into ontic balance. Then the zero-fiber
fixedness section target is obtained.
\end{theorem}

\begin{proof}
Let $s$ carry rows
$$
\mathsf{ZetaZero}(s),
\qquad
\mathsf{InCritStrip}(s).
$$
To produce the zero-fiber fixedness output it suffices to produce
$\mathsf{OnticBalance}(s)$. Ontic defect-zero incompatibility rules out
the simultaneous presence of the two zero rows and
$\mathsf{OnticDefect}(s)$. Hence the defect row is absent. The located
row converts this absence into $\mathsf{OnticBalance}(s)$. This is the
same proof skeleton as
\autoref{thm:unified-reality-rh-defect-incompatibility-rules-out-offline},
now expressed in atom-time-symmetry language.
\end{proof}

\begin{corollary}[The remaining ontic boundary]
\label{cor:unified-reality-remaining-ontic-boundary}
In the ontic projection language, the remaining RH-facing boundary is to
show that a zeta zero row in the critical strip cannot coexist with an
atom-time-symmetry defect inside the completed zeta carrier.
\end{corollary}

\begin{proof}

\autoref{thm:unified-reality-ontic-defect-visible-prime-window}. The

\autoref{def:unified-reality-ontic-defect-zero-incompatibility}. Thus
the boundary is a zero-relevant carrier compatibility problem, not the
mere observation that off-line prime-channel bias can be measured.
\end{proof}

\begin{theorem}[Spectral dominance warns against raw-defect collapse]
\label{thm:unified-reality-spectral-dominance-warns-defect-collapse}
The quartic spectral example shows why visible defect data cannot be
used as a zero-relevant boundary without an active dominance ledger.
\end{theorem}

\begin{proof}
In the quartic carrier, branch or discriminant data alone does not
determine the endpoint-zero boundary. The zero-relevant boundary passes
through maximal-modulus branch competition, as recorded in
\autoref{thm:unified-reality-quartic-dominance-finer-than-ramification}
and
\autoref{thm:unified-reality-spectral-curves-mediate-zero-reality}. The
zeta analogue is that a finite prime-window defect is only a visible
defect row until a completed-carrier incompatibility with the zero ledger
is supplied.
\end{proof}

\begin{theorem}[Zeta ontic projection principle]
\label{thm:unified-reality-zeta-ontic-projection-principle}
Zeta is the arithmetic-spectral projection of the ontic packet
$(\mathsf{Atom},\mathsf{Time},\mathsf{Sym})$. Its atom projection is the
prime-local Euler face, its time projection is the logarithmic
Laplace/Mellin readout, and its symmetry projection is the functional
mirror. The RH route can be read as the zero-fiber fixedness target for
this projection, with ontic defect-zero incompatibility as the remaining
zero-relevant boundary.
\end{theorem}

\begin{proof}
The three projection rows are given by
\autoref{thm:unified-reality-zeta-carries-ontic-triple}. Prime and zero
rows remain typed projections by
\autoref{thm:unified-reality-prime-zero-dual-projections}. The balance
surface is the critical line by
\autoref{thm:unified-reality-ontic-balance-critical-line}. The zero
fiber section form is
\autoref{thm:unified-reality-zero-fiber-fixedness-rh-form}. The
remaining incompatibility target is
\autoref{def:unified-reality-ontic-defect-zero-incompatibility}, and
the quartic warning against raw-defect collapse is
\autoref{thm:unified-reality-spectral-dominance-warns-defect-collapse}.
These rows give the ontic projection principle without claiming the
missing RH bridge.
\end{proof}
