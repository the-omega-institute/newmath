\section{No Far-End Absorption Boundary}
\label{sec:unified-reality-no-far-end-absorption-boundary}

The recursive spectral reading of zeta separates two meanings often
compressed into the symbol $\infty$.  The archimedean place is a positive
completion row: it supplies the Gamma normalization needed for the
completed zeta carrier and the functional-equation mirror.  The
apophatic far end is not such a row; it is the boundary role of an
unbounded finite-observation tower.  The RH-facing discipline is
therefore not to remove the archimedean row, but to refuse the
reification of the far end as a hidden point capable of absorbing finite
prime defects.

\begin{definition}[Two infinity roles in the zeta carrier]
\label{def:unified-reality-two-infinity-roles-zeta}
The zeta carrier uses two distinct infinity-facing roles:
$$
\infty_{\mathrm{place}}
:=
\text{archimedean completion row},
$$
and
$$
\mathsf{T}_{\mathrm{far}}
:=
\text{apophatic far-end role}.
$$
The first is represented analytically by the completion factor
$$
\pi^{-s/2}\Gamma(s/2)
$$
and by the pole-normalization rows entering the completed function.  The
second is a boundary name for the unbounded continuation of finite prime
windows, recurrence lengths, or observation layers.  It is not an
element of the zeta carrier.
\end{definition}

\begin{definition}[Finite prime-window recursive fold]
\label{def:unified-reality-finite-prime-window-recursive-fold}
For a finite prime window $S$, the raw finite Euler fold is
$$
Z_S(s)
:=
\prod_{p\in S}(1-p^{-s})^{-1}.
$$
If $q\notin S$ is a displayed prime row, the successor fold is
$$
Z_{S\cup\{q\}}(s)
=
Z_S(s)(1-q^{-s})^{-1}.
$$
Thus the raw Euler product is read as the compatible tower of finite
prime-window folds, not as the value of a completed point
$S=\PrimeUp$ inside the carrier.
\end{definition}

\begin{definition}[Archimedean completion row]
\label{def:unified-reality-archimedean-completion-row}
The \emph{archimedean completion row} for zeta is the analytic
normalization packet containing
$$
\pi^{-s/2}\Gamma(s/2),
\qquad
s(s-1),
$$
and the functional-equation ledger for the completed readout
$$
\xi(s)
=
\frac12s(s-1)\pi^{-s/2}\Gamma(s/2)\zeta(s).
$$
This row is part of the completed carrier.  It is not an infinite prime
and not an apophatic point.
\end{definition}

\begin{theorem}[Completion is not point reification]
\label{thm:unified-reality-completion-not-point-reification}
Passing from finite prime-window folds to the completed zeta carrier
requires the archimedean completion row, but it does not introduce
$\mathsf{T}_{\mathrm{far}}$ as a point of the carrier.
\end{theorem}

\begin{proof}
The finite prime-window tower is generated by
\autoref{def:unified-reality-finite-prime-window-recursive-fold}.  The
completed carrier additionally consumes the analytic normalization row
of \autoref{def:unified-reality-archimedean-completion-row}.  These are
positive data rows.  By
\autoref{def:unified-reality-recursive-spectral-apophatic-far-end} and
\autoref{thm:unified-reality-infinite-observation-time-not-spectral-point},
the far end of the tower is a boundary role for unbounded finite
readouts, not an internal object.  Therefore completion adds the
archimedean row and functional-equation ledger; it does not add an
extra point at infinity.
\end{proof}

\begin{definition}[Far-end absorption]
\label{def:unified-reality-far-end-absorption}
\emph{Far-end absorption} is the illicit inference pattern that treats
the unbounded tail, the archimedean row, or the apophatic far-end role as
a hidden term able to cancel a displayed finite prime-window defect
without a source-compatible zero-exclusion ledger.  In formulas, it is
the attempted passage from
$$
D_S(s)\#0
$$
to harmlessness inside the completed carrier without proving
$$
\mathsf{ZetaZero}(s)
\times
\mathsf{InCritStrip}(s)
\times
D_S(s)\#0
\to
\bot.
$$
\end{definition}

\begin{definition}[No far-end absorption]
\label{def:unified-reality-no-far-end-absorption}
The \emph{no far-end absorption} row for the zeta carrier is
$$
\mathsf{NoFarEndAbsorption}_{\zeta}
:=
\prod_{s:\ComplexUp}
\left(
\mathsf{ZetaZero}(s)
\times
\mathsf{InCritStrip}(s)
\times
\sum_{S:\mathsf{FinitePrimeWindow}}
\bigl(D_S(s)\#0\bigr)
\to
\bot
\right).
$$
It says that a finite prime-window defect cannot be hidden at the
completed carrier boundary, whether the hiding is described as Euler
tail, archimedean completion, functional-equation transport, or
apophatic far-end pressure.
\end{definition}

\begin{theorem}[No far-end absorption is prime-sieve odd exactness]
\label{thm:unified-reality-no-far-end-absorption-prime-sieve-exactness}
For the zeta route, no far-end absorption is the completion-boundary
form of prime-sieve odd exactness.
\end{theorem}

\begin{proof}
The finite defect row in
\autoref{def:unified-reality-no-far-end-absorption} is the same
existential finite prime-window defect used in
\autoref{def:unified-reality-prime-sieve-odd-exactness}.  The zero and
critical-strip hypotheses place that defect in the completed zeta
source.  Both definitions exclude coexistence of that source row with a
nonzero finite prime-window defect.  The difference is terminological:
prime-sieve odd exactness names the odd-defect mechanism, while no
far-end absorption names the completion-boundary discipline that forbids
the defect from being hidden by tail or far-end language.
\end{proof}

\begin{theorem}[Archimedean normalization is retained]
\label{thm:unified-reality-archimedean-normalization-retained}
No far-end absorption does not remove the archimedean completion row.
It only refuses to read that row as a hidden defect absorber.
\end{theorem}

\begin{proof}
The completed function row of
\autoref{def:unified-reality-archimedean-completion-row} is required for
the functional-equation mirror and for the completed carrier.  Without
that row, the fixed surface $\Re(s)=1/2$ would not be read as the
natural mirror surface of the completed object.  No far-end absorption
therefore preserves the archimedean row as a normalization and symmetry
ledger.  What it refuses is the additional, unledgered claim that this
row can cancel a finite prime-window defect while leaving a zero ledger
intact.
\end{proof}

\begin{corollary}[No far-end absorption is the RH-facing boundary]
\label{cor:unified-reality-no-far-end-absorption-rh-boundary}
Within the recursive zeta tower, the RH-facing boundary is the claim that
finite prime-window defect cannot be absorbed by completion.
\end{corollary}

\begin{proof}
If $s=1/2+\delta+i\tau$ and $\delta\#0$, then every nonempty finite prime
window reads
$$
D_S(s)
=
-\delta\sum_{p\in S}\log p
\#0
$$
by \autoref{thm:unified-reality-rh-off-half-creates-prime-defect}.  To
obtain the RH fixed-half section, this visible finite defect must
contradict the completed zero ledger.  That contradiction is exactly the
role of
\autoref{def:unified-reality-rh-defect-zero-incompatibility-route},
\autoref{def:unified-reality-prime-sieve-odd-exactness}, and the no-tail
absorption row of \autoref{prin:rh-no-tail-absorption-row}.  The
no-far-end-absorption formulation states the same requirement as a
boundary discipline for the recursive spectral tower.
\end{proof}

\begin{theorem}[Quartic carriers have finite closure, zeta has a far-end boundary]
\label{thm:unified-reality-quartic-finite-closure-zeta-far-end-boundary}
The quartic spectral carrier closes by finite certificate ledgers, while
the zeta carrier requires a non-point far-end completion boundary.
\end{theorem}

\begin{proof}
The fixed quartic carrier closes recurrence, spectral polynomial,
elliptic normalization, dominance, and endpoint-zero rows through the
finite certificate ledger of
\autoref{thm:unified-reality-quartic-finite-certificate-common-carrier}.
Its endpoint-zero theorem is scoped by fixed windows and finite local
branch data.  The zeta carrier instead builds finite prime-window folds
and then passes to analytic continuation, archimedean completion, and
functional-equation rows.  The latter passage has a far-end boundary:
finite prime defects remain visible locally, but their incompatibility
with zero rows requires the no-far-end-absorption ledger.
\end{proof}

\begin{theorem}[No-far-end formulation of the RH bridge]
\label{thm:unified-reality-no-far-end-formulation-rh-bridge}
The RH bridge can be stated as the refusal of hidden absorption of
finite prime defect at the apophatic far end of the zeta recursion tower.
\end{theorem}

\begin{proof}
By
\autoref{thm:unified-reality-no-far-end-absorption-prime-sieve-exactness},
this refusal is equivalent to prime-sieve odd exactness.  By
\autoref{thm:unified-reality-prime-sieve-odd-exactness-sufficient},
prime-sieve odd exactness, together with the located fixed-half collapse
row, supplies the RH fixed-half section.  Thus the no-far-end phrasing
does not add a separate theorem.  It names the exact boundary condition
needed to prevent a finite off-half prime defect from being hidden by
the completed carrier.
\end{proof}
