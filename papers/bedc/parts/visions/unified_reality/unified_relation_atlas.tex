\section{Unified Relation Atlas}
\label{sec:unified-reality-relation-atlas}

The preceding sections give local worked examples of the same general
discipline. A mathematical object is not treated as an isolated point.
It is treated as a carriered packet whose visible appearances are
reality projections. The atlas in this section is not a catalogue of all
mathematics. It is the relation schema governing the examples already in
view: common carrier, projection, generalized continuation, transport
ledger, and boundary ledger.

\begin{definition}[Unified relation packet]
\label{def:unified-reality-relation-packet}
A \emph{unified relation packet} is a tuple
$$
\mathcal O
:=
(\mathsf{Carrier},\mathsf{Classifier},(\pi_i)_{i\in I},
\mathsf{Transport},\mathsf{Ledger},\mathsf{Boundary})
$$
where the maps
$$
\pi_i:\mathcal O\to R_i
$$
read the same carrier in different realities. The reality targets may
include algebraic, geometric, analytic, spectral, arithmetic, measure,
and categorical interfaces:
$$
R_{\mathrm{alg}},
R_{\mathrm{geom}},
R_{\mathrm{analytic}},
R_{\mathrm{spectral}},
R_{\mathrm{arith}},
R_{\mathrm{measure}},
R_{\mathrm{cat}}.
$$
The classifier records when two carrier rows are the same for the
purpose of the packet. The transport rows record continuation between
realities. The ledger records which scale, branch, convergence, prime,
zero, or boundary data remain present after a projection.
\end{definition}
\concretizedIn{ch:concrete-instances-unified-relation-atlas-namecert}{The unified relation packet is realized as a finite carrier/classifier/projection/transport/ledger NameCert surface.}

\begin{principle}[Carriered relation before object isolation]
\label{prin:unified-reality-carriered-relation-before-object-isolation}
Within unified reality, a mature mathematical interface is read as a
licensed name over displayed carrier data. It is not used as an
uninspected primitive when a carrier, classifier, transport row, and
boundary ledger are the relevant mathematical content.
\end{principle}

\begin{definition}[Quadratic-norm atlas row]
\label{def:unified-reality-quadratic-norm-atlas-row}
The \emph{quadratic-norm row} begins with the Pythagorean carrier of
\autoref{sec:unified-reality-pythagorean-common-carrier}:
$$
a^2+b^2\sim c^2.
$$
After the nonzero scale row $c\#0$ is displayed, the normalized shape
projection reads
$$
x=\frac ac,
\qquad
y=\frac bc,
\qquad
x^2+y^2\sim1.
$$
In higher dimension the same norm layer has unit-sphere readout
$$
S^{n-1}:\quad x_1^2+\cdots+x_n^2=1
$$
and symmetry readout
$$
O(n):=\{A:A^TA=I\}.
$$
The circle and the sphere are unit norm layers. The orthogonal
interface is the automorphism read preserving that norm layer.
\end{definition}

\begin{theorem}[Quadratic normalization is a scale-to-shape transport]
\label{thm:unified-reality-quadratic-normalization-scale-shape}
The quadratic-norm row transports a Pythagorean scale carrier to a
unit-shape carrier, while the scale remains in the projection fiber.
\end{theorem}

\begin{proof}
The two-dimensional case is
\autoref{thm:unified-reality-triangle-circle-scale-fiber}. It sends
$(a,b,c)$ with $c\#0$ to $(a/c,b/c)$, preserving the normalized equation
and forgetting the visible value of $c$. The higher-dimensional sphere
and orthogonal rows do not change this ledger reading: they expose unit
norm and norm-preserving symmetry, not the original scale coordinate.
\end{proof}

\begin{definition}[Phase and transform atlas row]
\label{def:unified-reality-phase-transform-atlas-row}
The \emph{phase and transform row} reads the unit circle as phase,
Fourier analysis as character projection, Laplace analysis as real-biased
phase transport, and Mellin analysis as Laplace transport in logarithmic
coordinates. Its kernels are
$$
e^{in\theta},
\qquad
e^{-i\omega t},
\qquad
e^{-st},
\qquad
x^{-s}.
$$
For $s=\sigma+i\omega$ the Laplace kernel factors as
$$
e^{-st}=e^{-\sigma t}e^{-i\omega t}.
$$
For $x=e^t$ the Mellin kernel reads
$$
x^{-s}=e^{-s\log x}.
$$
\end{definition}

\begin{theorem}[Mellin is logarithmic Laplace transport]
\label{thm:unified-reality-mellin-logarithmic-laplace-transport}
The Mellin kernel $x^{-s}$ is the Laplace kernel on the logarithmic
coordinate of the multiplicative axis.
\end{theorem}

\begin{proof}
Set $t=\log x$. Then $x=e^t$, and hence
$$
x^{-s}=(e^t)^{-s}=e^{-st}=e^{-s\log x}.
$$
The logarithm transports multiplication to addition:
$$
\log(mn)=\log m+\log n.
$$
Thus multiplicative arithmetic data can be read as additive Laplace data
after the logarithmic transport row is displayed. This is the transport
principle already recorded in
\autoref{prin:unified-reality-logarithm-transport-multiplication-laplace}.
\end{proof}

\begin{definition}[Arithmetic spectral atlas row]
\label{def:unified-reality-arithmetic-spectral-atlas-row}
The \emph{arithmetic spectral row} reads a coefficient sequence
$(a_n)_{n\ge1}$ as the atomic logarithmic source
$$
\mu_a:=\sum_{n\ge1}a_n\delta_{\log n}
$$
with Dirichlet/Laplace readout
$$
\mathcal L[\mu_a](s)
=
\sum_{n\ge1}a_n n^{-s}.
$$
For $a_n=1$, this is the zeta Dirichlet face on its convergence window,
as in
\autoref{thm:unified-reality-zeta-arithmetic-laplace-readout}. When the
coefficient and convergence rows license multiplicative regrouping, the
same carrier has Euler-product face
$$
\prod_p(1-p^{-s})^{-1}.
$$
The Dirichlet face is the additive list reality. The Euler face is the
prime-local multiplicative reality.
\end{definition}

\begin{principle}[Zeta is a shared arithmetic carrier]
\label{prin:unified-reality-zeta-shared-arithmetic-carrier}
Zeta is not merely a sum and not merely a product. In this atlas it is a
shared arithmetic carrier whose Dirichlet, logarithmic Laplace, and
prime-local faces are compatible only under their displayed convergence,
factorization, and transport ledgers.
\end{principle}

\begin{definition}[Prime-channel boundary row]
\label{def:unified-reality-prime-channel-boundary-row}
For
$$
s=\frac12+\delta+i\tau
$$
the centered prime channel is
$$
p^{1/2-s}=p^{-\delta}e^{-i\tau\log p}.
$$
Its amplitude read is
$$
|p^{1/2-s}|=p^{-\delta}.
$$
Therefore the critical line is the prime-channel unitary surface,
as recorded in
\autoref{thm:unified-reality-critical-line-prime-unitary-surface}.
\end{definition}

\begin{theorem}[Spectral curves mediate recurrence and zero reality]
\label{thm:unified-reality-spectral-curves-mediate-zero-reality}
The quartic spectral row mediates recurrence algebra, matrix spectra,
elliptic branch geometry, dominance, and endpoint-zero reality without
identifying these projections as the same visible object.
\end{theorem}

\begin{proof}
This is the content of
\autoref{thm:unified-reality-quartic-spectral-common-carrier-principle}.
The recurrence, companion matrix, spectral polynomial, affine curve,
elliptic normalization, dominance row, and endpoint-zero law are
projections of the fixed quartic carrier. The carrier makes the
projections comparable; it does not erase the boundary ledgers attached
to each projection.
\end{proof}

\begin{definition}[Partition and zero-boundary row]
\label{def:unified-reality-partition-zero-boundary-row}

:=
\sum_{\omega\in\Omega}e^{-\beta E(\omega)+hM(\omega)}.

It is a weighted Laplace/Fourier readout of a state space. Zeros of such
a readout record phase cancellation or boundary behavior in the

theorem, requires its own model-specific carrier and proof ledger.
\end{definition}

\begin{definition}[Heat-kernel and spectral-zeta row]
\label{def:unified-reality-heat-spectral-zeta-row}
For a displayed operator with eigenvalue ledger $(\lambda_n)$, the heat
trace read is
$$
\mathrm{Tr}(e^{-t\Delta})
=
\sum_n e^{-t\lambda_n}.
$$
The spectral zeta read is
$$
\zeta_{\Delta}(s)
=
\sum_n\lambda_n^{-s}.
$$
Thus a geometric operator can project to a spectral measure, to its heat
Laplace readout, and to its Mellin-type spectral zeta readout. Analytic
continuation, pole structure, and determinant interpretations remain
separate boundary ledgers.
\end{definition}

\begin{theorem}[Relation atlas does not collapse projections]
\label{thm:unified-reality-relation-atlas-noncollapse}
No row in the relation atlas may replace the carrier by one of its
visible projections unless the corresponding classifier, transport,
fiber, and boundary ledgers are displayed.
\end{theorem}

\begin{proof}
This is the atlas form of
\autoref{prin:unified-reality-no-shadow-replacement}. The quadratic row
forgets scale under the circle projection. The Laplace and Mellin rows
add damping and logarithmic transport not present in pure Fourier phase.
The zeta row has both Dirichlet and Euler faces only with the relevant
convergence and factorization ledgers. The spectral-curve row separates
recurrence, matrix, branch, dominance, and endpoint-zero data. Therefore
projection visibility is never sufficient by itself to replace the
common carrier.
\end{proof}

\begin{theorem}[Unified relation atlas principle]
\label{thm:unified-reality-relation-atlas-principle}
Triangles, circles, spheres, orthogonal symmetries, Fourier modes,
Laplace kernels, Mellin kernels, zeta, spectral curves, elliptic
carriers, heat traces, $L$-functions, and RH boundary rows form a
compatible relation atlas when read as carrier projections connected by
transport ledgers. Their compatibility is not object-level identity; it
is common-carrier projection disciplined by noncollapse and boundary
accounting.
\end{theorem}

\begin{proof}
The common-carrier and generalized-continuation rules are
\autoref{def:unified-reality-common-carrier} and
\autoref{def:unified-reality-generalized-analytic-continuation}. The
Pythagorean scale-shape row is
\autoref{thm:unified-reality-triangle-circle-scale-fiber}. The
phase-transform-zeta chain is
\autoref{thm:unified-reality-phase-transform-zeta-ladder-principle}. The
quartic spectral chain is
\autoref{thm:unified-reality-quartic-spectral-common-carrier-principle}.
The RH boundary route is
\autoref{prin:unified-reality-rh-zero-rigidity-unitary-boundary}. The
cross-domain bridge layer is
\autoref{def:unified-reality-cross-domain-bridge-ledger}. Taken
together, these rows supply a relation atlas: transport between
realities is licensed by displayed ledgers, and every projection remains
boundary-relative.
\end{proof}
