\section{Finite Prime Tower as Completed Readout}
\label{sec:unified-reality-finite-prime-tower-completed-readout}

The no-far-end route can be stated as a completed-readout discipline for
the finite prime-window tower.  The zeta carrier is not a last prime
window and not an evaluation at an infinite prime set.  It is the
completed analytic readout of a recursively generated finite tower,
together with archimedean normalization, functional-equation transport,
and an apophatic boundary that cannot be consumed as a correction term.

\begin{definition}[Finite prime-window tower]
\label{def:unified-reality-finite-prime-window-tower}
The \emph{finite prime-window tower} is the directed family of finite
Euler folds
$$
E_S(s):=\prod_{p\in S}(1-p^{-s})^{-1}
$$
over nonempty finite prime windows $S$.  If $q\notin S$ is a displayed
prime row, the successor edge is
$$
E_{S\cup\{q\}}(s)
=
E_S(s)(1-q^{-s})^{-1}.
$$
Thus the tower is generated by finite recursive extension, as in
\autoref{def:unified-reality-finite-prime-window-recursive-fold}.
\end{definition}

\begin{definition}[Finite Hamiltonian prime layer]
\label{def:unified-reality-finite-hamiltonian-prime-layer}
For a nonempty finite prime window $S$ and a depth row $M\ge0$, the
\emph{finite Hamiltonian prime layer} is the finite occupation carrier
$$
\mathcal K_{S,M}
:=
\bigotimes_{p\in S}
\operatorname{span}\{|0\rangle_p,\ldots,|M\rangle_p\}.
$$
The local number operator is
$$
N_p|k\rangle_p=k|k\rangle_p,
$$
and the finite Hamiltonian is
$$
H_{S,M}:=\sum_{p\in S}(\log p)N_p.
$$
Its finite partition readout is
$$
Z_{S,M}(s)
:=
\operatorname{Tr}_{\mathcal K_{S,M}}\left(e^{-sH_{S,M}}\right)
=
\prod_{p\in S}
\left(1+p^{-s}+\cdots+p^{-Ms}\right).
$$
Thus the finite Euler fold has a spectral-dynamical refinement: prime
rows are occupation axes and $\log p$ is the displayed frequency weight.
\end{definition}

\begin{definition}[Centered finite Hamiltonian readout]
\label{def:unified-reality-centered-finite-hamiltonian-readout}
For
$$
s=\frac12+\delta+i\tau,
$$
the centered finite Hamiltonian readout is
$$
\widetilde Z_{S,M}(s)
:=
\operatorname{Tr}_{\mathcal K_{S,M}}
\left(e^{-(s-1/2)H_{S,M}}\right).
$$
Equivalently,
$$
e^{-(s-1/2)H_{S,M}}
=
e^{-\delta H_{S,M}}e^{-i\tau H_{S,M}}.
$$
The local centered prime channel is
$$
U_p(s):=p^{1/2-s}=p^{-\delta}e^{-i\tau\log p}.
$$
\end{definition}

\begin{theorem}[The critical line is the unitary Hamiltonian surface]
\label{thm:unified-reality-critical-line-unitary-hamiltonian-surface}
For the centered finite Hamiltonian readout, the fixed-half surface is
the unitary evolution surface.
\end{theorem}

\begin{proof}
If $\delta=0$, then
$$
e^{-(s-1/2)H_{S,M}}=e^{-i\tau H_{S,M}},
$$
and every local channel satisfies
$$
|U_p(s)|=1.
$$
If $\delta\#0$, the factor $e^{-\delta H_{S,M}}$ is a real
amplitude-bias row.  The local equality
$$
|U_p(s)|=p^{-\delta}
$$
is the same fixed-surface readout as
\autoref{thm:unified-reality-fixed-surface-cycle-readout-equivalence}.
Thus the centered Hamiltonian language reads $\Re(s)=1/2$ as the
absence of nonunitary radial bias.
\end{proof}

\begin{definition}[Interlayer Hamiltonian lift]
\label{def:unified-reality-interlayer-hamiltonian-lift}
An \emph{interlayer Hamiltonian lift} is one of the two finite extension
rows
$$
(S,M)\to(S\cup\{q\},M),
\qquad
(S,M)\to(S,M+1).
$$
For an additional displayed prime $q\notin S$, the Hamiltonian extension
has the form
$$
H_{S\cup\{q\},M}
=
H_{S,M}\otimes\mathbf 1_q
+
\mathbf 1_S\otimes(\log q)N_q.
$$
For a depth extension, the carrier admits the additional occupation row
$|M+1\rangle_p$ on each displayed prime axis.  These are finite layer
extensions, not an infinite prime point.
\end{definition}

\begin{definition}[Spectral zero island of a finite layer]
\label{def:unified-reality-spectral-zero-island-finite-layer}
Let $F_{S,M}$ be a finite-layer spectral readout obtained from the
Hamiltonian, sieve, or completed-tail ledger.  A \emph{spectral zero
island} is a local zero component, or a certified small-value component,
of the readout
$$
F_{S,M}(s).
$$
The island is a zero-relevant object only when its source, layer,
dominance or exactness ledger, and transition ledger are displayed.
This is the zeta analogue of the quartic dominance discipline of
\autoref{thm:unified-reality-quartic-marks-zero-relevance-boundary}.
\end{definition}

\begin{theorem}[Normalized defect is invariant under prime-layer lifts]
\label{thm:unified-reality-normalized-defect-invariant-prime-layer-lifts}
For
$$
s=\frac12+\delta+i\tau,
$$
the normalized prime-window defect of
\autoref{def:unified-reality-normalized-prime-window-defect} is
independent of the finite prime layer:
$$
\overline D_S(s)=-\delta.
$$
Consequently every interlayer lift preserving the same $s$ preserves
the normalized off-critical defect.
\end{theorem}

\begin{proof}
The displayed equality is exactly
\autoref{thm:unified-reality-normalized-defects-constant-over-windows}.
If $S\subseteq T$ are nonempty finite prime windows, then both
normalized readouts are equal to the same normal displacement:
$$
\overline D_S(s)=-\delta=\overline D_T(s).
$$
Depth extension does not change the prime-window normalization row
$D_S/L_S$.  Hence the nonzero normal defect is not erased by adding a
prime axis or by increasing finite occupation depth.
\end{proof}

\label{def:unified-reality-completed-readout-finite-prime-tower}

The archimedean component is the positive completion row of

\end{definition}

\begin{definition}[Completed-readout absorption error]
\label{def:unified-reality-completed-readout-absorption-error}
A \emph{completed-readout absorption error} is the passage from a

In schematic form it treats
$$
\mathsf{CofStablePrimeDefect}(s)
$$
as if it could be cancelled by the apophatic boundary of
$$
\mathsf{CompletedReadout}_{\zeta}.
$$
This is precisely the hidden-correction pattern excluded by
\autoref{def:unified-reality-no-hidden-far-end-correction}.
\end{definition}

\begin{definition}[Completed-readout exactness boundary]

\begin{theorem}[Finite rows of the completed-readout route are closed]
\label{thm:unified-reality-finite-rows-completed-readout-route-closed}
The finite rows of the completed-readout route are already displayed:
zero rows enter every finite sieve tail, and off-half displacement is a
cofinally stable normalized prime-window defect.
\end{theorem}

\begin{proof}
For the zero side,
\autoref{thm:unified-reality-zero-rows-enter-finite-sieve-tails} gives
$$
\mathsf{ZetaZero}(s)\times\mathsf{InCritStrip}(s)
\to
\mathsf{FiniteSieveTailZero}_{\zeta}(s).
$$
For the defect side,
\autoref{thm:unified-reality-normalized-defects-constant-over-windows}
gives
$$
\overline D_S(s)=-\delta(s)
$$
for every nonempty finite prime window, and
\autoref{cor:unified-reality-off-half-displacement-cofinally-visible}
turns $\delta(s)\#0$ into cofinal visibility of the normalized defect.
No appeal to an infinite prime point is used in either row.
\end{proof}

\begin{definition}[Off-critical island survival]
\label{def:unified-reality-off-critical-island-survival}
An \emph{off-critical island survival} row is a compatible family of
finite spectral zero islands through interlayer Hamiltonian lifts whose
normal coordinate satisfies
$$
\delta(s)\#0
$$
and whose completed readout is a zeta zero-source row.  It records the
claim that a nonunitary centered Hamiltonian island survives the finite
prime tower and enters the completed zero ledger.
\end{definition}

\begin{theorem}[Off-critical island survival is the stable-defect exactness boundary]
\label{thm:unified-reality-off-critical-island-survival-stable-defect-boundary}
Excluding off-critical island survival is the spectral-dynamical form of
the completed-readout exactness boundary.
\end{theorem}

\begin{proof}
If an off-critical island survives, then $\delta(s)\#0$.  By
\autoref{thm:unified-reality-normalized-defect-invariant-prime-layer-lifts},
the normalized finite prime-window defect is the same nonzero value
$-\delta(s)$ at every nonempty finite prime layer.  Thus the surviving
island carries a cofinally stable normalized defect.  If its completed
readout is also a zero-source row, the situation is exactly the
tail-zero stable-defect pattern named in
\autoref{def:unified-reality-completed-readout-exactness-boundary}.  The
remaining task is therefore not detection of the defect, but exclusion
of its coexistence with the completed zero ledger.
\end{proof}

\label{thm:unified-reality-completed-readout-exactness-normal-form}

\begin{proof}

then follows from

\end{proof}

\begin{corollary}[Completed readout does not close by far-end reification]
\label{cor:unified-reality-completed-readout-no-far-end-reification}
The completed readout of the finite prime-window tower may use analytic
continuation, archimedean completion, and functional-equation transport,
but it cannot close a stable finite defect by treating the apophatic far
end as a hidden absorber.
\end{corollary}

\begin{proof}
\autoref{thm:unified-reality-completion-not-point-reification} separates
positive completion rows from the apophatic far-end role.
\autoref{def:unified-reality-completed-readout-absorption-error}
identifies the forbidden inference pattern, and
\autoref{def:unified-reality-no-hidden-far-end-correction} rejects the
corresponding hidden correction row.  Therefore completed readout is a
carrier discipline, not a license to cancel a finite defect at an
undisplayed endpoint.
\end{proof}

\begin{corollary}[Quartic comparison]
\label{cor:unified-reality-quartic-comparison-completed-readout}
The quartic spectral carrier closes by finite certificate rows, while
the zeta completed readout closes the analogous RH route only after the
tail-zero stable-defect exactness boundary is discharged.
\end{corollary}

\begin{proof}
The quartic comparison is recorded in
\autoref{thm:unified-reality-quartic-finite-closure-zeta-far-end-boundary}:
fixed windows, finite branch ledgers, and dominance rows close the
quartic endpoint-zero law.  The zeta carrier instead has an unbounded
finite prime-window tower.  Its finite rows are visible by
\autoref{thm:unified-reality-finite-rows-completed-readout-route-closed},
but the zero-facing route still requires the completed-readout exactness
boundary of
\autoref{def:unified-reality-completed-readout-exactness-boundary}.
\end{proof}
