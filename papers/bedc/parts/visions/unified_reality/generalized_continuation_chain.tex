\section{Generalized Continuation Chain}
\label{sec:unified-reality-generalized-continuation-chain}

The phase-transform ladder of
\autoref{sec:unified-reality-phase-transform-zeta-ladder} must be read
with a precise boundary. It is a generalized analytic continuation
chain in the sense of
\autoref{def:unified-reality-generalized-analytic-continuation}; it is
not a claim that every arrow is ordinary holomorphic continuation of one
complex function across a larger domain. The chain transports structure
from algebraic and geometric realities into phase, spectral, transform,
and arithmetic realities. Only specific subrows, such as the
$\ZetaContUp$ application of
\autoref{ch:concrete-instances-zeta-continuation}, are ordinary
analytic-continuation rows.

\begin{definition}[Reality-continuation chain]
\label{def:unified-reality-reality-continuation-chain}
A \emph{reality-continuation chain} is a finite sequence of carriered
interfaces
$$
\mathcal R_0
\xrightarrow{\mathsf{GenCont}_{01}}
\mathcal R_1
\xrightarrow{\mathsf{GenCont}_{12}}
\cdots
\xrightarrow{\mathsf{GenCont}_{(k-1)k}}
\mathcal R_k
$$
where every edge is a generalized continuation packet of
\autoref{def:unified-reality-generalized-analytic-continuation}. The
edge may be algebraic normalization, geometric parametrization,
character decomposition, exponential-kernel transport, logarithmic
Laplace transport, or ordinary holomorphic continuation, provided the
source, classifier, projection, transport, and boundary ledgers are
displayed.
\end{definition}

\begin{definition}[Narrow analytic-continuation row]
\label{def:unified-reality-narrow-analytic-continuation-row}
A \emph{narrow analytic-continuation row} is an edge whose source and
target are holomorphic-domain readouts of one complex function, with
overlap, identity-theorem, uniqueness, pole, and branch ledgers
displayed. This is the ordinary $\AnalyticContUp$ situation discussed in
\autoref{rem:unified-reality-ordinary-analytic-continuation-special-case}
and realized for zeta by \autoref{thm:zeta-full-continuation}.
\end{definition}

\begin{principle}[Generalized continuation is wider than holomorphic continuation]
\label{prin:unified-reality-generalized-continuation-wider-than-holomorphic}
Every narrow analytic-continuation row is a generalized continuation row
when its carrier and ledgers are displayed. The converse is not valid:
normalization, phase parametrization, Fourier character projection,
Laplace damping, Mellin logarithmic transport, Euler product regrouping,
and spectral-curve normalization are generalized continuation rows but
are not, merely by that fact, holomorphic continuation steps.
\end{principle}

\begin{definition}[Phase-transform arithmetic continuation]
\label{def:unified-reality-phase-transform-arithmetic-continuation}
The \emph{phase-transform arithmetic continuation} is the chain
$$
\text{quadratic norm}
\to
\text{unit phase}
\to
\text{frequency decomposition}
\to
\text{damped frequency kernel}
\to
\text{arithmetic log-spectrum}.
$$
Its displayed formula rows are
$$
a^2+b^2=c^2,
\qquad
x^2+y^2=1,
\qquad
x+iy=e^{i\theta},
$$
$$
e^{-i\omega t},
\qquad
e^{-st}=e^{-\sigma t}e^{-i\omega t},
\qquad
n^{-s}=e^{-s\log n}.
$$
The zeta endpoint reads
$$
\zeta(s)
=
\sum_{n\ge1}n^{-s}
=
\sum_{n\ge1}e^{-s\log n}
$$
on the displayed convergence window.
\end{definition}

\begin{theorem}[The phase-transform chain is generalized continuation]
\label{thm:unified-reality-phase-transform-chain-generalized-continuation}
The phase-transform arithmetic continuation of
\autoref{def:unified-reality-phase-transform-arithmetic-continuation} is
a generalized continuation chain, not a stepwise holomorphic
continuation chain.
\end{theorem}

\begin{proof}
The Pythagorean-to-circle edge is scale normalization by
\autoref{thm:unified-reality-triangle-circle-scale-fiber}. The
circle-to-Fourier edge is character projection by
\autoref{prin:unified-reality-fourier-frequency-projection}. The
Fourier-to-Laplace edge adds the real amplitude-bias coordinate by
\autoref{thm:unified-reality-laplace-fourier-bias-direction}. The
Laplace-to-Mellin arithmetic edge is logarithmic transport by
\autoref{thm:unified-reality-mellin-logarithmic-laplace-transport}. The
endpoint zeta read is the atomic logarithmic Laplace readout of
\autoref{thm:unified-reality-zeta-arithmetic-laplace-readout}. These
edges transport carriered structure across realities. They are not all
holomorphic-domain extension operations of one complex function.
\end{proof}

\begin{definition}[Transported invariant]
\label{def:unified-reality-transported-invariant}
Let $\mathsf{GenCont}_{i\to j}$ be a generalized continuation packet. A
\emph{transported invariant} from $\mathcal R_i$ to $\mathcal R_j$ is a
target-side row $P_j$ equipped with a transport certificate from a
source-side row $P_i$ through the packet ledger. The target row need not
have the same formula as $P_i$. It must instead be the certified image
of the same carriered constraint under the displayed projection and
transport rows.
\end{definition}

\begin{definition}[Zeta transported-invariant packet]
\label{def:unified-reality-zeta-transported-invariant-packet}
The \emph{zeta transported-invariant packet} is the collection of zeta
rows corresponding to the phase-transform chain:
$$
\begin{array}{c|c}
\text{source structure} & \text{zeta-side transported row}\\
\hline
\text{scale normalization} & \text{centering at }1/2\\
\text{unit phase} & |p^{1/2-s}|=1\text{ on the critical line}\\
\text{frequency clock} & e^{-i\tau\log p}\\
\text{Laplace damping} & p^{-\delta}\text{ or }e^{-\delta x}\\
\text{spectral decomposition} & \text{Dirichlet coefficients and Euler factors}\\
\text{local-to-global assembly} & \text{prime-local product to zeta readout}\\
\text{symmetry} & \text{functional-equation mirror}\\
\text{zero boundary} & \zeta(s)=0\text{ as global cancellation}\\
\text{defect boundary} & \text{finite prime-window defect}
\end{array}
$$
Here $s=1/2+\delta+i\tau$, and the prime-channel row is
$$
p^{1/2-s}=p^{-\delta}e^{-i\tau\log p}.
$$
\end{definition}

\begin{principle}[Zeta inherits transported invariants, not literal shapes]
\label{prin:unified-reality-zeta-inherits-transported-invariants}
If zeta is read as the arithmetic log-spectrum endpoint of the
phase-transform chain, then zeta must carry transported versions of
normalization, unit phase, frequency, damping, local-to-global assembly,
symmetry, zero cancellation, and boundary rigidity. It need not contain
the literal equation $a^2+b^2=c^2$, a Euclidean circle, or an ordinary
Fourier series as object-level substructures.
\end{principle}

\begin{definition}[Zeta zero as global cancellation]
\label{def:unified-reality-zeta-zero-global-cancellation}
A \emph{zeta zero as global cancellation} is a zero row
$$
\zeta(s)=0
$$
read after the Dirichlet, Euler, continuation, and boundary ledgers have
assembled nonzero local phase and amplitude information into one global
zeta carrier. It is not a claim that each local term $n^{-s}$ or each
prime channel $p^{-s}$ vanishes.
\end{definition}

\begin{theorem}[Prime-unitary balance is the transported unit-phase row]
\label{thm:unified-reality-prime-unitary-transported-unit-phase}
For a displayed prime $p>1$, the unit-phase condition transported to
the centered prime channel is
$$
|p^{1/2-s}|=1
\quad\Longleftrightarrow\quad
\Re(s)=\frac12.
$$
\end{theorem}

\begin{proof}
This is
\autoref{thm:unified-reality-critical-line-prime-unitary-surface}. With
$s=1/2+\delta+i\tau$,
$$
p^{1/2-s}=p^{-\delta}e^{-i\tau\log p}
$$
and hence
$$
|p^{1/2-s}|=p^{-\delta}.
$$
Since $p>1$, this amplitude is $1$ exactly when $\delta=0$, which is the
critical-line equation $\Re(s)=1/2$.
\end{proof}

\begin{definition}[Defect-boundary transport]
\label{def:unified-reality-defect-boundary-transport}
The \emph{defect-boundary transport} is the RH-facing analogue of the
dominance boundary in the quartic spectral carrier. In the quartic row,
zero-relevant boundary data is not the discriminant alone; it is the
maximal-modulus branch competition recorded in
\autoref{thm:unified-reality-quartic-dominance-finer-than-ramification}.
In the zeta row, the corresponding boundary obligation is the passage
from finite prime-window defect to zero incompatibility, using
\autoref{def:number-theory-spine-prime-window-defect} and
\autoref{prin:number-theory-spine-defect-zero-incompatibility}.
\end{definition}

\begin{theorem}[RH is transported unitary zero-rigidity]
\label{thm:unified-reality-rh-transported-unitary-zero-rigidity}
Under the zeta transported-invariant packet and the required
defect-boundary transport, the RH target of
\autoref{def:number-theory-spine-rh} is the assertion that nontrivial
zeta zeros in the critical strip occur only on the transported unitary
balance surface.
\end{theorem}

\begin{proof}
The transported unitary surface is the critical line by
\autoref{thm:unified-reality-prime-unitary-transported-unit-phase}. The
zero rows are global cancellation rows in the sense of
\autoref{def:unified-reality-zeta-zero-global-cancellation}. The missing
analytic content is not the existence of the critical line; it is the
rigidity statement that off-unitary amplitude bias cannot coexist with a
zero row inside the completed zeta carrier. This is exactly the
defect-zero route isolated in
\autoref{def:unified-reality-rh-defect-zero-incompatibility-route} and
the section target of
\autoref{thm:unified-reality-rh-fixed-half-section-equivalence}.
\end{proof}

\begin{theorem}[Generalized continuation chain boundary principle]
\label{thm:unified-reality-generalized-continuation-chain-boundary-principle}
The chain from quadratic norm through phase, Fourier, Laplace/Mellin,
and zeta is valid only as a ledgered reality-continuation chain. It
licenses transported invariants; it does not license literal
identification of triangles, circles, Fourier series, Laplace kernels,
and zeta.
\end{theorem}

\begin{proof}
By
\autoref{prin:unified-reality-generalized-continuation-wider-than-holomorphic},
generalized continuation is broader than ordinary holomorphic
continuation but remains ledger-bound. By
\autoref{def:unified-reality-transported-invariant}, every inherited
property requires a transport certificate. By
\autoref{prin:unified-reality-zeta-inherits-transported-invariants}, the
zeta endpoint receives transported versions of phase, amplitude,
local-to-global, symmetry, zero, and defect-boundary rows, not literal
copies of the source shapes. Therefore the chain is a precise
continuation discipline rather than an analogy or an object-level
collapse.
\end{proof}
