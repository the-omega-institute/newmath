\section{Zero Recursor and Cover Boundary}
\label{sec:unified-reality-zero-recursor-cover-boundary}

The zero-signature route requires a type-level distinction before its
faithfulness rows can be read correctly.  A single zeta zero source is
not a constructor for the whole zero fibre.  Even a proof that one zero
admits an ontic-core expression does not supply a section over all zero
sources.  The constructive RH target is a total map out of the zero
source, and totality requires a cover or recursor for that source.

\begin{definition}[Zero source recursor]
\label{def:unified-reality-zero-source-recursor}
Let
$$
\mathcal Z_{\zeta}
:=
\sum_{s:\ComplexUp}
\left(
\mathsf{ZetaZero}(s)\times\mathsf{InCritStrip}(s)
\right)
$$
be the critical-strip zero source.  A \emph{zero source recursor} is a
packet
$$
\mathsf{ZeroRecursor}_{\zeta}
:=
(
\mathsf{Seed},
\mathsf{Step},
\mathsf{SymTransport},
\mathsf{Cover},
\mathsf{FarEndUnity}
)
$$
where $\mathsf{Seed}$ supplies a generated zero-source row,
$\mathsf{Step}$ supplies the permitted observation-time or spectral-time
continuation rows, $\mathsf{SymTransport}$ transports generated rows
through the mirror $J(s)=1-\overline{s}$, $\mathsf{Cover}$ states that
every row of $\mathcal Z_{\zeta}$ is generated by the recursor, and
$\mathsf{FarEndUnity}$ supplies far-end unity for generated zero rows.
\end{definition}

\begin{definition}[Core expression for zero sources]
\label{def:unified-reality-zero-core-expression}
A zero-source row $z:\mathcal Z_{\zeta}$ has a \emph{core expression}
when its zero data are read only through the displayed ontic-core rows
$$
(\mathsf{ObsTime},\mathsf{DistAtom},\mathsf{SymTransport})
$$
of the zeta carrier.  The row is written
$$
\mathsf{CoreExpr}_{\zeta}(z).
$$
It records expression through the permitted observation-time,
distinction-atom, and symmetry-transport ledgers; it is not a claim that
the whole zero source has been generated.
\end{definition}

\begin{principle}[Seed expression is not zero-source construction]
\label{prin:unified-reality-seed-expression-not-zero-source-construction}
A seed row
$$
z_0:\mathcal Z_{\zeta},
\qquad
\mathsf{CoreExpr}_{\zeta}(z_0)
$$
does not imply
$$
\prod_{z:\mathcal Z_{\zeta}}
\mathsf{CoreExpr}_{\zeta}(z).
$$
The latter is a dependent section over the zero source.  It requires a
cover or recursor whose generation rows reach every zero-source row.
\end{principle}

\begin{definition}[Zero-source cover]
\label{def:unified-reality-zero-source-cover}
The cover row of a zero source recursor is a displayed surjective
generation map
$$
\mathsf{Generate}_{\zeta}:\mathsf{Param}_{\zeta}\to\mathcal Z_{\zeta}
$$
together with
$$
\mathsf{Cover}_{\zeta}
:=
\prod_{z:\mathcal Z_{\zeta}}
\sum_{a:\mathsf{Param}_{\zeta}}
\left(
\mathsf{Generate}_{\zeta}(a)=z
\right).
$$
It is the row that turns a generated family into a construction principle
for the whole zero source.
\end{definition}

\begin{theorem}[Covered recursor transports core expression]
\label{thm:unified-reality-covered-recursor-transports-core-expression}
Assume a zero source recursor whose seed, step, and mirror transport rows
preserve core expression, and whose cover row reaches every zero source.
Then every critical-strip zero source has a core expression:
$$
\prod_{z:\mathcal Z_{\zeta}}
\mathsf{CoreExpr}_{\zeta}(z).
$$
\end{theorem}

\begin{proof}
Let $z:\mathcal Z_{\zeta}$.  The cover row supplies a parameter
$a:\mathsf{Param}_{\zeta}$ with
$$
\mathsf{Generate}_{\zeta}(a)=z.
$$
Starting from the seed row and applying the displayed step and mirror
transport rows along the generating path for $a$, the recursor preserves
$\mathsf{CoreExpr}_{\zeta}$ at each generated zero row.  Transporting
along the cover equality gives $\mathsf{CoreExpr}_{\zeta}(z)$.
\end{proof}

\begin{theorem}[Covered far-end-unified recursor gives the RH section]
\label{thm:unified-reality-covered-far-end-unified-recursor-gives-rh-section}
Assume a zero source recursor whose cover reaches every zero-source row
and whose far-end unity row holds for every generated zero.  Then the RH
fixed-half section is obtained.
\end{theorem}

\begin{proof}
Let $z:\mathcal Z_{\zeta}$ have complex coordinate $s$.  By cover, $z$
is generated by some parameter of the zero source recursor.  The
far-end-unity row of the recursor supplies
$$
\mathsf{FarEndUnity}_{\zeta}(s).
$$
By
\autoref{thm:unified-reality-far-end-unity-fixed-half-surface}, this is
equivalent to $\delta(s)=0$, hence to $\Re(s)=1/2$.  The fixed-half
section equivalence of
\autoref{thm:unified-reality-rh-fixed-half-section-equivalence} gives
the RH section.
\end{proof}

\begin{theorem}[Zero recursor cover isolates the same hard point]
\label{thm:unified-reality-zero-recursor-cover-isolates-hard-point}
The zero-recursor route reduces the RH-facing task to two zeta-specific
rows: cover of the whole zero source and far-end unity for generated
zero rows.
\end{theorem}

\begin{proof}
\autoref{prin:unified-reality-seed-expression-not-zero-source-construction}
shows that a single expressed zero cannot supply the dependent RH
section.  The cover row of
\autoref{def:unified-reality-zero-source-cover} upgrades a generated
family to a total zero-source construction.  The far-end-unity row then
converts each generated zero to the fixed-half surface by
\autoref{thm:unified-reality-covered-far-end-unified-recursor-gives-rh-section}.
Thus the missing content is not the shared vocabulary of the zero type,
but the total cover and the refusal of hidden far-end defect absorption.
The latter is the same analytic rigidity already isolated as
\autoref{def:unified-reality-rh-defect-zero-incompatibility-route} and
\autoref{def:unified-reality-no-off-balanced-tail-zero}.
\end{proof}

\begin{corollary}[Zero type sameness is weaker than zero recursor totality]
\label{cor:unified-reality-zero-type-sameness-weaker-than-recursor-totality}
The statement that zero rows have the same type, or that one zero row is
core-expressible, is strictly weaker than a covered zero recursor.
\end{corollary}

\begin{proof}
Type sameness supplies a common codomain for rows of
$\mathcal Z_{\zeta}$.  It does not supply a generating path to an
arbitrary element of that type.  A covered zero recursor supplies that
path through $\mathsf{Generate}_{\zeta}$ and $\mathsf{Cover}_{\zeta}$.
Only the latter can produce the dependent section required by the
constructive RH target.
\end{proof}
