\section{BEDC--Langlands General $G$ Existence Bridge}
\label{sec:unified-reality-bedc-langlands-general-g-existence-bridge}

The Tannakian functorial closure of
\autoref{sec:unified-reality-bedc-langlands-tannakian-functorial-closure}
reduces the finite readout problem for a general reductive group $G$ to
compatible $GL_N$ transfers. It does not by itself construct the
$G$-side automorphic packet. The existence bridge for general $G$ is the
stable-trace and endoscopic part of the Langlands machine: a nonzero
stable parameter readout must appear as a primitive $G$-packet, as a
proper endoscopic transfer, or as a declared trace tail ledger row.

\begin{definition}[General $G$ parameter object]
\label{def:unified-reality-general-g-parameter-object}
Let $F$ be a global field and let $G/F$ be a connected reductive group.
A general $G$ parameter object is an Arthur-type row
$$
\psi:L_F\times SL_2(\mathbb C)\to{}^LG,
$$
with ordinary Langlands parameters
$$
\phi:W'_F\to{}^LG
$$
included by taking the $SL_2$ factor to be trivial. Its BEDC ledger is
$$
\Lambda_\psi
:=
(S_{\mathrm{ram}},N_v,\epsilon_v,\gamma_v,
\mathcal S_\psi,\mathsf{Endoscopy},\mathsf{ContinuousSpectrum},
\mathsf{Boundary},\mathsf{Normalization}),
$$
where
$$
\mathcal S_\psi
:=
\pi_0(Z_{\widehat G}(\operatorname{im}\psi)).
$$
The time row is generated by Frobenius, local parameters, Arthur
$SL_2$, and trace-window inclusions; the unity row is the stable
semisimple or tempered readout; the ledger row is $\Lambda_\psi$.
\end{definition}

\begin{definition}[$G$-side automorphic realization residual]
\label{def:unified-reality-general-g-automorphic-realization-residual}
A $G$-side automorphic packet $\Pi_\psi(G)$ realizes $\psi$ when, for
almost all unramified places $v$, some $\pi\in\Pi_\psi(G)$ has
$$
\lambda_{v,r}(\pi)
=
\operatorname{tr}r(\psi(\operatorname{Frob}_v))
$$
for every finite-dimensional representation $r:{}^LG\to GL(V_r)$ in
the certified readout scope. If no such $G$-packet is present, the
missing realization residual is
$$
E_G(\psi)
:=
\mathsf{MissingGAutomorphicPacket}(\psi).
$$
\end{definition}

\begin{definition}[Stable parameter distribution certificate]
\label{def:unified-reality-general-g-stable-parameter-distribution}
A stable parameter distribution certificate for $\psi$ assigns to every
fixed Hecke window $\mathcal W$ a stable functional
$$
S\Theta_\psi^G:\mathcal H_{\mathcal W}(G)\to\mathbb C
$$
and a parameter projector $f_{\psi,\mathcal W}^G$ such that
$$
S\Theta_\psi^G(f_{\psi,\mathcal W}^G)\ne0
$$
on the declared nonzero scope. At unramified places this is Satake
evaluation at $\psi(\operatorname{Frob}_v)$; at ramified places it uses
the local packet stable character and its local ledger.
\end{definition}

\begin{definition}[Stable multiplicity formula certificate]
\label{def:unified-reality-general-g-stable-multiplicity-formula}
A stable multiplicity formula certificate for a discrete parameter
$\psi$ gives a packet candidate $\Pi_\psi(G)$ and a component-group
pairing so that the target stable distribution has the form
$$
S\Theta_\psi^G(f)
=
\sum_{\pi\in\Pi_\psi(G)}m(\pi)\Theta_\pi(f)
+
\mathsf{LedgerTerms}_\psi(f),
$$
where $\mathsf{LedgerTerms}_\psi$ consists only of continuous spectrum,
Eisenstein residual, endoscopic transfer, boundary, multiplicity
normalization, and transfer-factor rows.
\end{definition}

\begin{definition}[Endoscopic transfer certificate]
\label{def:unified-reality-general-g-endoscopic-transfer-cert}
An endoscopic datum is a row
$$
\mathfrak e=(H,\xi,s,\Delta)
$$
with an $L$-embedding
$$
\xi:{}^LH\to{}^LG.
$$
An endoscopic transfer certificate assigns matching test functions
$$
f^H\longleftrightarrow f^G
$$
such that stable orbital integrals match after the declared transfer
factor and the spectral stable distributions satisfy
$$
S\Theta_{\psi_H}^H(f^H)
=
S\Theta_{\xi\circ\psi_H}^G(f^G)
+
\mathsf{EndoscopicLedger}(f).
$$
Transfer-factor, normalization, nonregular, ramified, and matching
failures are rows of $\mathsf{EndoscopicLedger}$.
\end{definition}

\begin{definition}[Stabilized trace formula certificate]
\label{def:unified-reality-general-g-stabilized-trace-formula-cert}
A stabilized trace formula certificate for $G$ records, for admitted
test functions $f^G$, an identity
$$
I_{\mathrm{disc}}^G(f^G)
=
\sum_{\mathfrak e=(H,\xi)}
\iota(G,H)SI_{\mathrm{disc}}^H(f^H)
+
\mathsf{TraceLedger}(f^G),
$$
where $\mathsf{TraceLedger}$ contains continuous spectrum, Eisenstein
terms, boundary, singular orbital integrals, ramification,
normalization, and transfer-factor data. Its finite/tail residual
discipline is the trace defect dichotomy of
\autoref{thm:unified-reality-trace-defect-dichotomy}.
\end{definition}

\begin{definition}[Endoscopic induction certificate]
\label{def:unified-reality-general-g-endoscopic-induction-cert}
Endoscopic complexity is ordered by a well-founded tuple such as
$$
\operatorname{ec}(G)
:=
(\operatorname{ssrank}G,\dim G,\#\pi_0(Z(\widehat G))).
$$
An endoscopic induction certificate says that every proper elliptic
endoscopic group $H$ appearing in the stabilized trace formula has
smaller complexity than $G$, and that the general existence bridge is
already available for $H$ on the declared scope.
\end{definition}

\begin{definition}[General trace tail ledger]
\label{def:unified-reality-general-g-trace-tail-ledger}
The general $G$ trace tail ledger is
$$
\mathsf{GeneralTraceTailLedger}_G
:=
(\mathsf{ContinuousSpectrum},\mathsf{EisensteinResidual},
\mathsf{Boundary},\mathsf{SingularOrbitalIntegral},
\mathsf{TransferNormalization},\mathsf{RamificationEscape},
\mathsf{EndoscopicTail}).
$$
A trace-tail ledger certificate says that any nonzero stable readout not
accounted for by a primitive packet or a proper endoscopic packet enters
this ledger.
\end{definition}

\begin{theorem}[Endoscopic residual dichotomy]
\label{thm:unified-reality-general-g-endoscopic-residual-dichotomy}
If an endoscopic transfer certificate fails for an admitted transfer
row, then the failure is either a finite transfer defect on a fixed test
window or an endoscopic tail defect.
\end{theorem}
\begin{proof}
If the mismatch is visible for a fixed pair of matching test functions
$f^H\leftrightarrow f^G$, then the stable orbital-integral or spectral
transfer equality fails on a finite window. That is a finite transfer
defect. If every fixed window transfer matches but the transfer fails
along singular orbit limits, ramification growth, transfer-factor
normalization limits, or cofinal window growth, the defect is not
finite-window data. It is an endoscopic tail defect and belongs to the
endoscopic tail ledger.
\end{proof}

\begin{corollary}[Closed objects have no endoscopic transfer failure]
\label{cor:unified-reality-general-g-no-endoscopic-transfer-failure}
A closed BEDC trace object with complete endoscopic ledger discipline
cannot carry an unresolved endoscopic transfer failure.
\end{corollary}
\begin{proof}
By
\autoref{thm:unified-reality-general-g-endoscopic-residual-dichotomy},
any failure is finite-visible or a tail row. A finite failure must be
entered in the ledger or omitted; the first leaves the object not
closed, and the second violates the ledger property. The same argument
applies to a tail row.
\end{proof}

\begin{theorem}[Stabilized trace residual dichotomy]
\label{thm:unified-reality-general-g-stabilized-trace-residual-dichotomy}
If the stabilized trace formula identity for $G$ fails on the declared
scope, then the residual is visible on a fixed trace window or belongs
to $\mathsf{GeneralTraceTailLedger}_G$.
\end{theorem}
\begin{proof}
This is the stable form of
\autoref{thm:unified-reality-trace-defect-dichotomy}. A nonzero
restricted residual on some finite trace window is finite-visible. If
every finite window is clean but the stabilized identity fails globally,
the defect can only lie in continuous spectrum, Eisenstein boundary,
singular orbital integrals, ramification escape, transfer
normalization, or endoscopic tail behavior, all of which are entries of
$\mathsf{GeneralTraceTailLedger}_G$.
\end{proof}

\begin{theorem}[General $G$ existence bridge certificate]
\label{thm:unified-reality-general-g-existence-bridge-cert}
Let $\psi^\uparrow$ be a legal discrete $G$-parameter object. Assume:
\begin{itemize}
\item local packet and local state certificates hold at every displayed
place;
\item a stable parameter distribution certificate gives a nonzero
fixed-window readout;
\item the stabilized trace formula certificate holds;
\item endoscopic transfer certificates hold for all proper endoscopic
data in the trace identity;
\item the stable multiplicity formula certificate holds for primitive
$G$-discrete contributions;
\item the endoscopic induction certificate holds;
\item the general trace tail ledger is complete;
\item closed accepted objects carry no unresolved trace, endoscopic, or
existence ledger rows.
\end{itemize}
Then $\psi^\uparrow$ has a $G$-side automorphic packet realization, or
its absence is an explicit ledger row. In the closed case, the packet
exists.
\end{theorem}
\begin{proof}
Choose a fixed window and a test function
$$
f_{\psi,\mathcal W}^G
$$
with
$$
S\Theta_\psi^G(f_{\psi,\mathcal W}^G)\ne0.
$$
The stabilized trace formula and endoscopic transfer certificates place
this nonzero stable readout into exactly three classes.

First, the primitive $G$-discrete part may be nonzero. The stable
multiplicity formula then expresses that primitive stable distribution
as a sum over $\Pi_\psi(G)$ plus declared ledger terms. If
$\Pi_\psi(G)$ were empty and the primitive readout were nonzero, the
nonzero contribution would have to be a ledger term. A closed object
cannot carry it unresolved, and omission violates the ledger property.
Thus the primitive nonzero case gives a nonempty $G$-packet.

Second, the stable readout may come from a proper endoscopic group
$H$. By the endoscopic induction certificate, the existence bridge for
$H$ supplies an $H$-side packet for the corresponding parameter
$\psi_H$. The endoscopic transfer certificate carries that packet to
the $G$-side unless a transfer residual occurs. By
\autoref{cor:unified-reality-general-g-no-endoscopic-transfer-failure},
an unresolved transfer residual is incompatible with closedness. Thus
the endoscopic case gives a $G$-side packet in the closed scope.

Third, the readout may be neither primitive discrete nor proper
endoscopic discrete. Then it lies in continuous spectrum, Eisenstein
residual, boundary, singular orbital-integral, normalization,
ramification, or endoscopic-tail data. These are exactly the rows of
$\mathsf{GeneralTraceTailLedger}_G$. A closed object cannot carry them
unresolved.

The three alternatives exhaust the stabilized spectral decomposition.
Therefore a closed legal parameter has a $G$-side automorphic packet.
Without closedness, failure is precisely a ledger row.
\end{proof}

\begin{corollary}[General $G$ reciprocity from the existence bridge]
\label{cor:unified-reality-general-g-reciprocity-existence-bridge}
Under the hypotheses of
\autoref{thm:unified-reality-general-g-existence-bridge-cert}, if
no-hidden Langlands residual holds, then there is a closed packet
$\Pi_\psi(G)$ such that for almost all certified unramified places $v$
and all finite-dimensional $r:{}^LG\to GL(V_r)$ in the readout scope,
$$
\lambda_{v,r}(\pi)
=
\operatorname{tr}r(\psi(\operatorname{Frob}_v))
$$
for some $\pi\in\Pi_\psi(G)$.
\end{corollary}
\begin{proof}
The existence bridge gives a nonempty $G$-packet in the closed case. If
a finite Hecke--Frobenius mismatch remains, it is detected by the
Tannakian detector $(v,r)$ of
\autoref{def:unified-reality-tannakian-detector}. If no finite detector
sees it, then by Tannakian separation and Chebotarev the unramified
semisimple readout is already fixed; any remaining obstruction is a
ramified, endoscopic, continuous-spectrum, boundary, multiplicity, or
normalization ledger row. No-hidden residual and closedness exclude
both unresolved alternatives.
\end{proof}

\begin{corollary}[General $G$ functoriality from the existence bridge]
\label{cor:unified-reality-general-g-functoriality-existence-bridge}
Let
$$
\eta:{}^LH\to{}^LG
$$
be an $L$-group homomorphism and let $(\Pi_H,\psi_H)$ be a closed
$H$-side packet. Put
$$
\psi_G:=\eta\circ\psi_H.
$$
If $\psi_G$ satisfies the hypotheses of
\autoref{thm:unified-reality-general-g-existence-bridge-cert}, then
there is a $G$-side packet $\Pi_G$ such that
$$
\lambda_{v,r}(\Pi_G)
=
\operatorname{tr}r(\eta(\psi_H(\operatorname{Frob}_v)))
$$
on the certified unramified finite detector scope.
\end{corollary}
\begin{proof}
The parameter $\psi_G=\eta\circ\psi_H$ is a legal $G$-parameter on the
declared scope. The existence bridge gives a $G$-side packet, and the
preceding reciprocity corollary gives its finite readout equality with
$\psi_G$. Substituting the definition of $\psi_G$ yields the displayed
formula.
\end{proof}

\begin{theorem}[General $G$ BEDC--Langlands closure]
\label{thm:unified-reality-general-g-bedc-langlands-closure}
On any reductive-group scope carrying local packets, parameter states,
stabilized trace formula, endoscopic transfer, stable multiplicity,
endoscopic induction, Tannakian separation, Chebotarev, no-hidden
Langlands residual, trace-tail ledger discipline, and closedness,
Langlands reciprocity and functoriality hold on the certified scope.
\end{theorem}
\begin{proof}
Existence is supplied by
\autoref{thm:unified-reality-general-g-existence-bridge-cert}. Once the
$G$-packet exists, the detector equality is forced by
\autoref{cor:unified-reality-general-g-reciprocity-existence-bridge}.
Functoriality is obtained by applying the same existence and no-hidden
residual argument to $\eta\circ\psi_H$ as in
\autoref{cor:unified-reality-general-g-functoriality-existence-bridge}.
Uniqueness of the semisimple parameter readout follows from the
Tannakian and Chebotarev rows of
\autoref{thm:unified-reality-chebotarev-tannakian-uniqueness}.
\end{proof}

\begin{theorem}[General $G$ existence bridge boundary]
\label{thm:unified-reality-general-g-existence-bridge-boundary}
The general $G$ existence bridge is a conditional BEDC closure theorem.
It consumes stable trace formula, endoscopic transfer, stable
multiplicity, local packet, parameter-state, induction, and tail-ledger
certificates. It does not construct those certificates or remove the
traditional analytic and endoscopic obligations.
\end{theorem}
\begin{proof}
Every positive implication above invokes one of the listed
certificates. The argument proves that, once they are present, absence
of a $G$-packet cannot remain hidden: it is primitive discrete,
proper-endoscopic, or ledgered as trace tail. The construction of the
certificates themselves is outside this bridge.
\end{proof}
