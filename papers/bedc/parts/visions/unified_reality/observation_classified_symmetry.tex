\section{Observation-Classified Symmetry on the Critical Surface}
\label{sec:unified-reality-observation-classified-symmetry}

\autoref{sec:unified-reality-zeta-ontic-projection} reads zeta through
prime atoms, logarithmic time, functional symmetry, and a zero ledger.
\autoref{sec:unified-reality-ontic-core-minimal-closure} then extracts the
minimal ontic core as observation-time, distinction-atom, and
symmetry-transport.  The critical line is the place where these two views
meet: it is not a literal boundary between constructive and classical
reasoning, nor a side-by-side separation of discrete and continuous objects.
It is the observation-classified fixed surface of the zeta mirror, with
prime-local and Laplace readouts giving compatible tests of the same
classifier.

\begin{definition}[Critical mirror classifier]
\label{def:unified-reality-critical-mirror-classifier}
For a complex history
$$
s=\sigma+i\tau,
$$
write
$$
J(s):=1-\overline{s},
\qquad
\delta(s):=\Re(s)-\frac12.
$$
The critical mirror classifier is the three-way readout
$$
\mathsf{HalfClass}_{\zeta}(s)
:=
\begin{cases}
\mathsf{LeftBias}, & \delta(s)<0,\\
\mathsf{Fixed}, & \delta(s)=0,\\
\mathsf{RightBias}, & \delta(s)>0.
\end{cases}
$$
It classifies a point by its observed displacement from the mirror fixed
surface.
\end{definition}

\begin{theorem}[The critical line is the mirror fixed surface]
\label{thm:unified-reality-critical-line-mirror-fixed-surface}
For the mirror $J(s)=1-\overline{s}$,
$$
J(s)=s
\quad\Longleftrightarrow\quad
\delta(s)=0
\quad\Longleftrightarrow\quad
\Re(s)=\frac12.
$$
Thus the line $\Re(s)=1/2$ is supplied by symmetry before any zero row is
read.
\end{theorem}

\begin{proof}
Writing $s=\sigma+i\tau$ gives
$$
J(s)=1-\overline{\sigma+i\tau}
=1-\sigma+i\tau.
$$
Hence $J(s)=s$ is equivalent to
$$
1-\sigma+i\tau=\sigma+i\tau,
$$
which is equivalent to $\sigma=1/2$.  Since
$\delta(s)=\sigma-1/2$, this is the same as $\delta(s)=0$.  The zero
predicate is not used in this calculation.
\end{proof}

\begin{definition}[Observation-classified symmetry]
\label{def:unified-reality-observation-classified-symmetry}
An \emph{observation-classified symmetry} is a packet
$$
(\mathcal X,J,\delta)
$$
where $J:\mathcal X\to\mathcal X$ is an involutive symmetry and
$\delta:\mathcal X\to\mathbb R$ is an observation row satisfying
$$
\delta(Jx)=-\delta(x).
$$
The observation row separates the $J$-orbits into fixed and split
classes:
$$
\mathsf{FixedOrbit}_{J}(x):\Longleftrightarrow \delta(x)=0,
$$
and
$$
\mathsf{SplitOrbit}_{J}(x):\Longleftrightarrow \delta(x)\#0.
$$
For zeta, this packet is obtained from
$$
\mathcal X=\ComplexUp,
\qquad
J(s)=1-\overline{s},
\qquad
\delta(s)=\Re(s)-\frac12.
$$
\end{definition}

\begin{theorem}[Zeta mirror orbits are classified by the half displacement]
\label{thm:unified-reality-zeta-mirror-orbits-classified}
The zeta mirror satisfies
$$
\delta(J(s))=-\delta(s).
$$
Consequently the observation row $\delta(s)$ separates mirror orbits into
the fixed class $\delta(s)=0$ and the split class $\delta(s)\#0$.
\end{theorem}

\begin{proof}
By definition,
$$
J(s)=1-\overline{s}.
$$
If $s=\sigma+i\tau$, then $J(s)=1-\sigma+i\tau$.  Hence
$$
\delta(J(s))
=
\Re(J(s))-\frac12
=
1-\sigma-\frac12
=
-\left(\sigma-\frac12\right)
=-\delta(s).
$$
The fixed class is the zero fiber of $\delta$, and the split class is
its apartness fiber.
\end{proof}

\begin{definition}[Critical-surface readout packet]
\label{def:unified-reality-critical-surface-readout-packet}
The critical-surface readout packet of the zeta carrier is
$$
\mathsf{CritRead}_{\zeta}(s)
:=
\bigl(
\mathsf{MirrorFix}(s),
\mathsf{PrimeUnit}(s),
\mathsf{NoLaplaceBias}(s)
\bigr),
$$
where
$$
\mathsf{MirrorFix}(s):\Longleftrightarrow s=J(s),
$$
$$
\mathsf{PrimeUnit}(s):\Longleftrightarrow
\forall p\in\PrimeUp,\ |p^{1/2-s}|=1,
$$
and, for logarithmic time $x>0$,
$$
\mathsf{NoLaplaceBias}(s):\Longleftrightarrow
e^{-\delta(s)x}=1.
$$
The three rows are respectively the symmetry readout, the prime-local
readout, and the Laplace readout of the same centered surface.
\end{definition}

\begin{theorem}[Prime and Laplace reads detect the same fixed surface]
\label{thm:unified-reality-prime-laplace-detect-same-fixed-surface}
For $s=1/2+\delta+i\tau$, the following rows are equivalent readings of
the centered surface:
$$
s=J(s),
\qquad
\delta=0,
\qquad
\forall p\in\PrimeUp,\ |p^{1/2-s}|=1,
\qquad
\forall x>0,\ e^{-\delta x}=1.
$$
\end{theorem}

\begin{proof}
The equivalence of $s=J(s)$ and $\delta=0$ is
\autoref{thm:unified-reality-critical-line-mirror-fixed-surface}.  For
each displayed prime row,
$$
p^{1/2-s}=p^{-\delta}e^{-i\tau\log p},
$$
so its modulus is $p^{-\delta}$.  This is one for every displayed
prime exactly when $\delta=0$, as in
\autoref{thm:unified-reality-ontic-balance-critical-line}.  The
Laplace row decomposes as
$$
e^{-s\log n}
=
e^{-(1/2+\delta)\log n}e^{-i\tau\log n}.
$$
The factor $e^{-\delta x}$ is the additional bias along positive
logarithmic time $x$.  It is neutral for every displayed positive
logarithmic time exactly on the zero fiber of $\delta$.
\end{proof}

\begin{corollary}[The critical line is a balance surface, not a side boundary]
\label{cor:unified-reality-critical-line-balance-not-side-boundary}
The line $\Re(s)=1/2$ is not the literal boundary between discrete and
continuous projections.  The discrete prime readout and the analytic
Laplace-continuation readout coexist in the zeta carrier, and the
critical line is their centered balance classifier.
\end{corollary}

\begin{proof}
The prime-local row reads the centered channel
$$
p^{1/2-s},
$$
while the Laplace row reads
$$
n^{-s}=e^{-s\log n}.
$$
Both rows are present as projections of the zeta carrier.  By
\autoref{thm:unified-reality-prime-laplace-detect-same-fixed-surface},
their centered classifiers agree exactly on $\delta=0$.  The
absolute-convergence boundary of the Dirichlet and Euler product rows is a
different ledger, namely the $\Re(s)>1$ domain of
\autoref{prop:rh-euler-product-prime-local-gluing}.
\end{proof}

\begin{definition}[Prime-window observation of the split class]
\label{def:unified-reality-prime-window-observation-split-class}
For a nonempty finite prime window $S$, the observation of the split
class is the defect row
$$
D_S(s)
:=
\sum_{p\in S}\log|p^{1/2-s}|.
$$
If $s=1/2+\delta+i\tau$, then
$$
D_S(s)
=
-\delta\sum_{p\in S}\log p.
$$
\end{definition}

\begin{theorem}[Prime windows observe split mirror orbits]
\label{thm:unified-reality-prime-windows-observe-split-orbits}
If $\mathsf{SplitOrbit}_{J}(s)$ holds and $S$ is a nonempty finite prime
window, then the prime-window observation satisfies
$$
D_S(s)\#0.
$$
\end{theorem}

\begin{proof}
For the zeta mirror, $\mathsf{SplitOrbit}_{J}(s)$ is
$\delta(s)\#0$ by
\autoref{def:unified-reality-observation-classified-symmetry}.  The
defect formula of
\autoref{def:unified-reality-prime-window-observation-split-class} gives
$$
D_S(s)
=
-\delta(s)\sum_{p\in S}\log p.
$$
The logarithmic sum is positive for a nonempty displayed prime window.
Thus apartness of $\delta(s)$ from zero gives apartness of $D_S(s)$
from zero, matching
\autoref{thm:unified-reality-ontic-defect-visible-prime-window}.
\end{proof}

\begin{definition}[Zero-fixedness target]
\label{def:unified-reality-zero-fixedness-target}
The \emph{zero-fixedness target} is the section
$$
\mathsf{ZeroFixed}_{\zeta}
:=
\prod_{s:\ComplexUp}
\left(
\mathsf{ZetaZero}(s)\times\mathsf{InCritStrip}(s)
\to
s=J(s)
\right).
$$
Equivalently, it asks that every critical-strip zero observation belongs
to the fixed mirror class rather than the split mirror class.
\end{definition}

\begin{theorem}[Zero-fixedness is the orbit form of the RH target]
\label{thm:unified-reality-zero-fixedness-orbit-form-rh}
The zero-fixedness target is the observation-classified orbit form of
the BEDC RH target.
\end{theorem}

\begin{proof}
The BEDC RH target is the fixed-half section sending a row
$$
\mathsf{ZetaZero}(s)\times\mathsf{InCritStrip}(s)
$$
to an $\mathsf{OnCritLine}(s)$ witness, as in
\autoref{thm:unified-reality-rh-fixed-half-section-equivalence}.  By
\autoref{thm:unified-reality-critical-line-mirror-fixed-surface}, the
same conclusion may be read as $s=J(s)$.  Thus the fixed-half section is
equivalently the assertion that zero rows in the critical strip occupy
the fixed mirror class.
\end{proof}

\begin{corollary}[Mirror closure alone does not supply zero-fixedness]
\label{cor:unified-reality-mirror-closure-not-zero-fixedness}
Transport of zero rows along the mirror $J$ is weaker than
zero-fixedness.
\end{corollary}

\begin{proof}
Mirror closure supplies a companion zero row at $J(s)$ when the
functional-equation and non-escape ledgers are present.  It does not
identify $s$ with $J(s)$.  The remaining row is precisely the passage
from a possible split mirror orbit to the fixed mirror orbit, as
isolated in \autoref{cor:rh-mirror-closure-weaker-than-rh} and
\autoref{def:unified-reality-zero-fixedness-target}.
\end{proof}

\begin{theorem}[Observation-classified critical-surface principle]
\label{thm:unified-reality-observation-classified-critical-surface-principle}
The zeta critical surface is the common observation classifier of three
compatible rows:
$$
\text{functional mirror fixedness},
\qquad
\text{prime-local unit amplitude},
\qquad
\text{absence of extra Laplace bias}.
$$
The RH-facing obligation is not to construct the surface, but to show
that the critical-strip zero ledger factors through its fixed class.
\end{theorem}

\begin{proof}
The three compatible rows are assembled in
\autoref{thm:unified-reality-prime-laplace-detect-same-fixed-surface}.
Prime windows observe off-fixed displacement by
\autoref{thm:unified-reality-prime-windows-observe-split-orbits}.  The
zero-facing target is the fixed-class section of
\autoref{thm:unified-reality-zero-fixedness-orbit-form-rh}.  Therefore
the critical surface is already determined as a classifier; the
remaining boundary is the zero-relevant factorization through that
classifier.
\end{proof}
