\section{Algebraic Extension Directions and Information Dimension}
\label{sec:unified-reality-algebraic-extension-information-dimension}

The Pythagorean example isolates the role of the scale row: when
$(1,1,\sqrt{2})$ is read over $\mathbb Q$, the full carrier cannot stay
inside the base-field layer. It is read through the extension
$\mathbb Q(\sqrt{2})$, and the normalized shape coordinates consume the
same extension ledger. This section records the general algebraic form
of that phenomenon. An algebraic element outside a base field is not an
anomalous point of the base carrier; it is a finite-dimensional
extension direction over the base field, certified by its minimal
polynomial and its power-basis ledger.

The hierarchy is compatible with the L8 algebra layer of the project
roadmap: field extensions, algebraic closure, Galois extensions,
splitting fields, separability, and transcendence are mature algebra
interfaces downstream of arithmetic and upstream of the geometry and
analysis surfaces that consume them.

\begin{definition}[Field-extension reality]
\label{def:unified-reality-field-extension-reality}
A \emph{field-extension reality} is a displayed embedding
$$
\iota:F\hookrightarrow E
$$
between field-like reality interfaces in the sense of
\autoref{def:unified-reality-field-like-interface}, together with
classifier-preservation ledgers for $0,1,+,-,\cdot$, apartness,
licensed inverses, and the induced scalar action
$$
F\times E\to E.
$$
The notation $E/F$ refers to the whole packet: base interface, extension
interface, embedding, scalar action, operation compatibility, and
readback ledger.
\end{definition}

\begin{definition}[Generated simple extension]
\label{def:unified-reality-generated-simple-extension}
Let $E/F$ be a field-extension reality and let $\alpha\in E$. The
\emph{simple extension generated by $\alpha$} is the least displayed
sub-reality of $E$ containing $\iota(F)$ and $\alpha$ and closed under
the licensed field operations. It is denoted
$$
F(\alpha).
$$
Its constructive rows are represented by rational expressions
$$
\frac{p(\alpha)}{q(\alpha)},
\qquad
p,q\in F[X],
\qquad
q(\alpha)\#0,
$$
with ledger entries for the expression, denominator apartness,
classifier equivalence of expressions, operation closure, and readback
into $E$.
\end{definition}

\begin{definition}[Algebraic element and minimal-polynomial certificate]
\label{def:unified-reality-algebraic-minimal-polynomial-certificate}
An element $\alpha\in E$ is \emph{algebraic over $F$} when there is a
nonzero polynomial $p(X)\in F[X]$ with
$$
p(\alpha)\sim_E0.
$$
A \emph{minimal-polynomial certificate} for $\alpha$ is a polynomial
$m_\alpha(X)\in F[X]$ such that
$$
m_\alpha\neq0,\qquad
m_\alpha\ \text{is monic},\qquad
m_\alpha(\alpha)\sim_E0,
$$
and every polynomial $p$ satisfying $p(\alpha)\sim_E0$ is divisible by
$m_\alpha$ in $F[X]$. Its degree is written
$$
\deg_F(\alpha):=\deg m_\alpha.
$$
\end{definition}

\begin{definition}[Algebraic extension direction]
\label{def:unified-reality-algebraic-extension-direction}
For an algebraic element $\alpha$ over $F$, the \emph{algebraic
extension direction generated by $\alpha$} is the simple extension
$$
Z_\alpha^{\mathrm{alg}}:=F(\alpha).
$$
Its algebraic information dimension over $F$ is
$$
\dim_F Z_\alpha^{\mathrm{alg}}
:=
[F(\alpha):F],
$$
the vector-space dimension of the extension carrier over the base
carrier.
\end{definition}

\begin{definition}[Power-basis candidate]
\label{def:unified-reality-power-basis-candidate}
If $\alpha$ has minimal-polynomial degree $d$, its \emph{power-basis
candidate} is
$$
\mathcal B_\alpha
:=
(1,\alpha,\alpha^2,\ldots,\alpha^{d-1}).
$$
\end{definition}

\begin{theorem}[Power-basis independence]
\label{thm:unified-reality-power-basis-independence}
Let $m_\alpha$ be a minimal-polynomial certificate of degree $d$. Then
$$
1,\alpha,\alpha^2,\ldots,\alpha^{d-1}
$$
are linearly independent over $F$.
\end{theorem}

\begin{proof}
Suppose there is a nontrivial linear relation
$$
a_0+a_1\alpha+\cdots+a_{d-1}\alpha^{d-1}\sim_E0,
\qquad
a_i\in F.
$$
Let
$$
p(X)=a_0+a_1X+\cdots+a_{d-1}X^{d-1}.
$$
Then $p(\alpha)\sim_E0$, $p\neq0$, and $\deg p<d$. The
minimal-polynomial certificate says $m_\alpha\mid p$, which is
impossible for a nonzero $p$ of smaller degree. Hence every displayed
linear relation is trivial.
\end{proof}

\begin{theorem}[Power-basis spanning]
\label{thm:unified-reality-power-basis-spanning}
If $\deg m_\alpha=d$, every element of $F(\alpha)$ is classified with a
linear combination
$$
b_0+b_1\alpha+\cdots+b_{d-1}\alpha^{d-1},
\qquad
b_i\in F.
$$
\end{theorem}

\begin{proof}
For any polynomial $p(X)\in F[X]$, divide by $m_\alpha$:
$$
p(X)=q(X)m_\alpha(X)+r(X),
\qquad
\deg r<d.
$$
Evaluating at $\alpha$ gives $p(\alpha)\sim_Er(\alpha)$ because
$m_\alpha(\alpha)\sim_E0$. Thus polynomial expressions reduce to
linear combinations of the displayed power-basis rows.

For a general rational expression $p(\alpha)/q(\alpha)$ with
$q(\alpha)\#0$, the denominator row says $q$ does not vanish at
$\alpha$. The minimal-polynomial certificate therefore gives the
coprimality ledger of $q$ with $m_\alpha$ in $F[X]$. A Euclidean ledger
supplies $u,v\in F[X]$ with
$$
u(X)q(X)+v(X)m_\alpha(X)=1.
$$
After evaluation, $u(\alpha)q(\alpha)\sim_E1$, so
$q(\alpha)^{-1}\sim_Eu(\alpha)$. Hence
$$
\frac{p(\alpha)}{q(\alpha)}
\sim_E
p(\alpha)u(\alpha),
$$
and the product again reduces to a combination of powers below $d$.
\end{proof}

\begin{theorem}[Algebraic extension dimension]
\label{thm:unified-reality-algebraic-extension-dimension}
If $\alpha$ is algebraic over $F$ and has minimal-polynomial degree
$d$, then
$$
[F(\alpha):F]=d.
$$
Moreover $\mathcal B_\alpha=(1,\alpha,\ldots,\alpha^{d-1})$ is a
displayed $F$-basis of $F(\alpha)$.
\end{theorem}

\begin{proof}
\autoref{thm:unified-reality-power-basis-independence} gives linear
independence, and
\autoref{thm:unified-reality-power-basis-spanning} gives spanning. Thus
the power-basis candidate is a basis, and its length is $d$.
\end{proof}

\begin{definition}[Algebraic noncollapse]
\label{def:unified-reality-algebraic-noncollapse}
For an algebraic element $\alpha\in E$ over $F$, define
$$
\mathsf{AlgNonCollapse}_F(\alpha)
:=
\alpha\notin F.
$$
\end{definition}

\begin{theorem}[Algebraic noncollapse and degree]
\label{thm:unified-reality-algebraic-noncollapse-degree}
For an algebraic element $\alpha$ over $F$,
$$
\mathsf{AlgNonCollapse}_F(\alpha)
\Longleftrightarrow
\deg_F(\alpha)>1.
$$
\end{theorem}

\begin{proof}
If $\deg_F(\alpha)>1$, then $\alpha$ cannot lie in $F$, since an
element of $F$ has the degree-one annihilator $X-\alpha$. Conversely, if
$\alpha\notin F$ and $\deg_F(\alpha)=1$, the monic minimal polynomial
has the form $X-a$ for some $a\in F$, forcing $\alpha\sim_Ea$ and hence
$\alpha\in F$, a contradiction.
\end{proof}

\begin{corollary}[Algebraic information dimension detects noncollapse]
\label{cor:unified-reality-algebraic-information-dimension-noncollapse}
For algebraic $\alpha$ over $F$,
$$
\mathsf{AlgNonCollapse}_F(\alpha)
\Longleftrightarrow
[F(\alpha):F]>1.
$$
\end{corollary}

\begin{proof}
Combine
\autoref{thm:unified-reality-algebraic-noncollapse-degree} with
\autoref{thm:unified-reality-algebraic-extension-dimension}.
\end{proof}

\begin{example}[The \texorpdfstring{$\mathbb Q(\sqrt{2})$}{Q(sqrt2)} extension]
\label{ex:unified-reality-q-sqrt-two-extension}
Let $F=\mathbb Q$ and $\alpha=\sqrt{2}$. The polynomial
$$
m_\alpha(X)=X^2-2
$$
is irreducible over $\mathbb Q$ and has degree $2$. Therefore
$$
[\mathbb Q(\sqrt{2}):\mathbb Q]=2
$$
and the power basis is
$$
(1,\sqrt{2}).
$$
Thus
$$
\mathbb Q(\sqrt{2})
=
\{a+b\sqrt{2}:a,b\in\mathbb Q\},
$$
with equality read as the extension classifier supplied by the
minimal-polynomial and reduction ledger.
\end{example}

\begin{corollary}[Pythagorean irrational scale has dimension two]
\label{cor:unified-reality-pythagorean-irrational-scale-dimension}
The Pythagorean packet
$$
(1,1,\sqrt{2})
$$
from \autoref{ex:unified-reality-pythagorean-sqrt-two} is read over the
two-dimensional extension reality
$$
\mathbb Q(\sqrt{2})/\mathbb Q.
$$
Its normalized shape point
$$
(1/\sqrt{2},1/\sqrt{2})
=
(\sqrt{2}/2,\sqrt{2}/2)
$$
uses the same extension carrier.
\end{corollary}

\begin{proof}
By \autoref{ex:unified-reality-q-sqrt-two-extension}, the extension
dimension is two. The normalized coordinate statement is the special
case of
\autoref{thm:unified-reality-pythagorean-extension-lift-shape} with
$a=b=1$ and $c=\sqrt{2}$.
\end{proof}

\begin{definition}[Extension projection graph]
\label{def:unified-reality-extension-projection-graph}
Let $F\subset E$ be a field-extension reality and let
$P\in\mathsf{Pyth}_E$. The \emph{extension projection graph}
$$
\mathsf{ExtProj}(P;F,E)
$$
records, for each displayed projection of $P$, whether the readout lies
in the base layer $F$ or only in the extension layer $E$. For example,
when $P=(1,1,\sqrt{2})$ over
$\mathbb Q\subset\mathbb Q(\sqrt{2})$, the leg readout lies in
$\mathbb Q^2$, the scale readout lies in
$\mathbb Q(\sqrt{2})\setminus\mathbb Q$, and the normalized shape
readout lies in
$\mathbb Q(\sqrt{2})^2$ rather than $\mathbb Q^2$.
\end{definition}

\begin{theorem}[Extension projection consistency]
\label{thm:unified-reality-extension-projection-consistency}
Let $F\subset E$ be a field-extension reality, and let
$$
P=(a,b,c;\nu_c;h_P)\in\mathsf{Pyth}_E
$$
with $a,b\in F$, $c\in E\setminus F$, and $a\#0$, $b\#0$. Then the
algebraic equation, the scale readout, and the normalized shape readout
are all read through the same extension ledger $F\subset E$: the
equation closes in $E$, the scale readout lies outside $F$, and the
normalized shape readout lies outside $F^2$.
\end{theorem}

\begin{proof}
The equation closes in $E$ because $P$ carries the row
$h_P:a^2+b^2\sim_Ec^2$. The scale readout is $c$, which is outside $F$
by hypothesis. The normalized shape coordinates are outside $F$ by
\autoref{thm:unified-reality-pythagorean-extension-lift-shape}. All
three readouts are projected from the same packet $P$ and the same
embedding ledger $F\subset E$.
\end{proof}

\begin{definition}[Algebraic extension continuation]
\label{def:unified-reality-algebraic-extension-continuation}
The finite algebraic continuation from $F$ to $F(\alpha)$ is the packet
$$
\mathsf{GenCont}_{F\to F(\alpha)}
:=
(F(\alpha),\iota,\alpha,m_\alpha,\mathcal B_\alpha,\mathsf{Led}_\alpha),
$$
where $\iota:F\hookrightarrow F(\alpha)$ is the embedding,
$m_\alpha$ is the minimal-polynomial certificate,
$\mathcal B_\alpha=(1,\alpha,\ldots,\alpha^{d-1})$ is the power basis,
and $\mathsf{Led}_\alpha$ records reduction, multiplication, licensed
division, classifier transport, and readback into the ambient extension.
\end{definition}

\begin{theorem}[Algebraic extensions are finite continuation packets]
\label{thm:unified-reality-algebraic-extension-finite-continuation}
If $\alpha$ is algebraic over $F$ with minimal-polynomial degree $d$,
then $F\to F(\alpha)$ is a finite certificate-style generalized
continuation in the sense of
\autoref{def:unified-reality-generalized-analytic-continuation}.
\end{theorem}

\begin{proof}
The minimal-polynomial certificate is finite, and
\autoref{thm:unified-reality-algebraic-extension-dimension} supplies the
finite power basis. Multiplication is computed by reducing powers
$\alpha^n$ with $n\ge d$ using the relation
$m_\alpha(\alpha)\sim0$. Addition and negation are coordinatewise.
Licensed division uses denominator apartness and the Euclidean ledger in
$F[X]$ as in
\autoref{thm:unified-reality-power-basis-spanning}. Thus the carrier,
operations, classifiers, stability rows, and readback are all displayed
by finite data.
\end{proof}

\begin{definition}[Algebraic information dimension]
\label{def:unified-reality-algebraic-information-dimension}
For algebraic $\alpha$ over $F$, define
$$
\dim_{\mathrm{info}}^{\mathrm{alg}}(\alpha/F)
:=
[F(\alpha):F].
$$
\end{definition}

\begin{theorem}[Algebraic extension direction principle]
\label{thm:unified-reality-algebraic-extension-direction-principle}
Let $\alpha$ be algebraic over $F$. If $\alpha\in F$, then the
generated extension has algebraic information dimension $1$. If
$\alpha\notin F$, then $F(\alpha)$ is a nontrivial finite-dimensional
extension over $F$, its dimension is $\deg_F(\alpha)$, and its carrier is
given by a finite continuation packet. In a Pythagorean carrier, an
extension-scale row $c\notin F$ lifts both the scale projection and the
normalized shape projection through the same extension reality.
\end{theorem}

\begin{proof}
The dimension statement is
\autoref{thm:unified-reality-algebraic-extension-dimension}. The
noncollapse criterion is
\autoref{thm:unified-reality-algebraic-noncollapse-degree}. The finite
continuation packet is
\autoref{thm:unified-reality-algebraic-extension-finite-continuation}.
The Pythagorean projection statement is
\autoref{thm:unified-reality-extension-projection-consistency}.
\end{proof}
