\section{Infinite Prime Torus and Far-End Compatibility}
\label{sec:unified-reality-infinite-prime-torus-far-end-compatibility}

The circular image must be calibrated before it is used for the zeta
carrier.  A circle may arise from a genuine periodic quotient, but it
may also arise from a far-end closure of a line.  The zeta prime carrier
does not have a nonzero time at which all prime phases return to their
initial positions.  Its infinite closure is instead an interface of
compatible finite prime-window readings.  The far end is therefore not a
period point; it is the cofinal compatibility socket of the infinite
prime torus.

\begin{definition}[Periodic circle and far-end circle]
\label{def:unified-reality-periodic-circle-far-end-circle}
The \emph{periodic circle} is the quotient readout
$$
\RealUp/2\pi\IntUp\cong S^1,
$$
where the identification is a genuine period row
$$
\theta\sim\theta+2\pi.
$$
The \emph{far-end circle} is the closure readout
$$
\RealUp\cup\{\infty\}\cong S^1,
$$
where $\infty$ names the closure point of the unbounded line rather than
a time period of the flow.
\end{definition}

\begin{definition}[Finite prime torus projection system]
\label{def:unified-reality-finite-prime-torus-projection-system}
For a nonempty finite prime window $S$, let
$$
\mathbb T_S:=\prod_{p\in S}S^1.
$$
If $S\subseteq T$, the projection
$$
\pi_{T,S}:\mathbb T_T\to\mathbb T_S
$$
forgets the coordinates in $T\setminus S$.  The finite prime tori and
forgetful maps form the finite-window projection system of the zeta
prime phase carrier.
\end{definition}

\begin{definition}[Infinite prime torus]
\label{def:unified-reality-infinite-prime-torus}
The \emph{infinite prime torus} is the compatible finite-window carrier
$$
\mathbb T_{\PrimeUp}
:=
\varprojlim_{S\Subset\PrimeUp}
\mathbb T_S.
$$
Equivalently, it is the cofinal product
$$
\mathbb T_{\PrimeUp}
=
\prod_{p\in\PrimeUp}S^1
$$
read only through its finite projections
$$
\pi_S:\mathbb T_{\PrimeUp}\to\mathbb T_S.
$$
Thus a finite prime window is not an independent small object; it is a
finite readout of the infinite prime torus.
\end{definition}

\begin{theorem}[Prime phase flow lands in the inverse-limit torus]
\label{thm:unified-reality-prime-phase-flow-lands-in-inverse-limit-torus}
The phase profile
$$
\Phi_{\PrimeUp}(\tau)
=
\left(e^{-i\tau\log p}\right)_{p\in\PrimeUp}
$$
is a point of the infinite prime torus, and its finite projection is
$$
\pi_S\bigl(\Phi_{\PrimeUp}(\tau)\bigr)=\Phi_S(\tau).
$$
\end{theorem}

\begin{proof}
For each prime $p$, the coordinate $e^{-i\tau\log p}$ lies in $S^1$.
Thus the family lies in the product
$\prod_{p\in\PrimeUp}S^1$.  For any inclusion $S\subseteq T$, forgetting
the coordinates outside $S$ sends the finite profile on $T$ to the
finite profile on $S$.  Hence the finite profiles are compatible under
the projection maps of
\autoref{def:unified-reality-finite-prime-torus-projection-system}, and
the cofinal family is an element of the inverse-limit torus.
\end{proof}

\begin{theorem}[The infinite prime torus has no common period point]
\label{thm:unified-reality-infinite-prime-torus-no-common-period-point}
There is no nonzero real time $T$ such that
$$
\Phi_{\PrimeUp}(\tau+T)=\Phi_{\PrimeUp}(\tau)
$$
for every time $\tau$.
\end{theorem}

\begin{proof}
If such a $T$ existed, then every finite prime window containing two
distinct primes would have the same nonzero common period.  This is
excluded by
\autoref{thm:unified-reality-finite-prime-phase-windows-no-common-period}.
Therefore the infinite prime torus is not a periodic circle with a
single return time for all prime coordinates.
\end{proof}

\begin{definition}[Prime far-end compatibility]
\label{def:unified-reality-prime-far-end-compatibility}
The \emph{prime far-end compatibility} row is the assertion that an
infinite prime-phase state is specified by a compatible family
$$
(x_S)_{S\Subset\PrimeUp},
\qquad
x_S\in\mathbb T_S,
$$
such that for every $S\subseteq T$,
$$
\pi_{T,S}(x_T)=x_S.
$$
The far end of the prime tower is this compatibility interface.  It is
not an additional prime, a final finite window, or a hidden coordinate.
\end{definition}

\begin{theorem}[Far-end compatibility is not phase periodicity]
\label{thm:unified-reality-far-end-compatibility-not-phase-periodicity}
For the zeta prime phase carrier, far-end closure means compatibility of
all finite torus projections, not exact return of all prime phases to a
common time origin.
\end{theorem}

\begin{proof}
Compatibility is the inverse-limit condition of
\autoref{def:unified-reality-prime-far-end-compatibility}.  Exact common
return would supply a nonzero period for every finite window, hence a
nonzero common period for the infinite prime torus.  The latter is
excluded by
\autoref{thm:unified-reality-infinite-prime-torus-no-common-period-point}.
Thus the far-end closure role is closer to far-end completion than to a
periodic quotient.
\end{proof}

\begin{theorem}[Stable normal defect is the common finite-window readout]
\label{thm:unified-reality-stable-normal-defect-common-window-readout}
For
$$
s=\frac12+\delta+i\tau,
$$
every nonempty finite prime window satisfies
$$
\overline D_S(s)=-\delta.
$$
Consequently the finite windows need not agree in phase, but they do
agree on the normalized normal defect.
\end{theorem}

\begin{proof}
The normalized defect calculation is exactly
\autoref{thm:unified-reality-normalized-defects-constant-over-windows}.
It uses the radial row
$$
\log |p^{1/2-s}|=-\delta\log p
$$
and divides by the finite logarithmic weight
$\sum_{p\in S}\log p$.  The phase coordinates
$e^{-i\tau\log p}$ need not be periodic across the window.  The stable
cofinal datum is instead the common normalized normal defect $-\delta$.
\end{proof}

\begin{corollary}[Off-half rows are visible in every finite torus readout]
\label{cor:unified-reality-off-half-visible-every-finite-torus-readout}
If $\delta\#0$, then every nonempty finite prime window reads a nonzero
normalized normal defect and positive normalized second-order energy.
\end{corollary}

\begin{proof}
By
\autoref{thm:unified-reality-stable-normal-defect-common-window-readout},
$\overline D_S(s)=-\delta$ for every nonempty finite window.  Hence
$\delta\#0$ gives a nonzero normalized defect in every finite projection.
Squaring gives the positive normalized energy
$\delta^2$, as in
\autoref{thm:unified-reality-prime-adic-parity-windows-read-one-normal-coordinate}
for the parity-refined windows and in
\autoref{def:unified-reality-even-defect-energy} for prime-window energy.
\end{proof}

\begin{corollary}[The zeta far end cannot be a hidden defect absorber]
\label{cor:unified-reality-zeta-far-end-not-hidden-defect-absorber}
The infinite prime torus far end cannot be used to absorb a nonzero
normal defect that is visible in every finite window.
\end{corollary}

\begin{proof}
The far end is the compatibility interface of
\autoref{def:unified-reality-prime-far-end-compatibility}, not an element
with an additional coordinate.  If $\delta\#0$, then
\autoref{cor:unified-reality-off-half-visible-every-finite-torus-readout}
shows that every finite projection reads the same nonzero normalized
normal defect, while
\autoref{thm:unified-reality-finite-to-infinite-energy-divergence}
shows that the unnormalized positive energy is unbounded cofinally.
Absorbing this row at the far end would reify the compatibility socket as
a hidden correction point, contrary to
\autoref{def:unified-reality-no-hidden-energy-correction-far-end} and
\autoref{thm:unified-reality-completion-not-point-reification}.
\end{proof}

\begin{corollary}[RH as unit infinite-torus zero boundary]
\label{cor:unified-reality-rh-unit-infinite-torus-zero-boundary}
The RH-facing geometric form is that completed zeta zero rows in the
critical strip are supported on the unit infinite prime torus, not on a
radially deformed prime-torus layer.
\end{corollary}

\begin{proof}
On the fixed-half surface, the prime phase-radial readback lies in
$\mathbb T_{\PrimeUp}$ by
\autoref{thm:unified-reality-prime-phase-flow-lands-in-inverse-limit-torus}.
Off the fixed-half surface, the same finite phase profile is multiplied
by the radial row $(p^{-\delta})_{p\in\PrimeUp}$.  The latter is detected
by every finite normalized defect readout and by the cofinal energy tower
of
\autoref{sec:unified-reality-finite-to-infinite-phase-energy-tower}.  To
prove the RH section is therefore to prove the zero-relevance boundary:
completed zero rows cannot live on the radially deformed layer.
\end{proof}
