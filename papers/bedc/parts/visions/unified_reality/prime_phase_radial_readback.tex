\section{Prime Phase-Radial Readback}
\label{sec:unified-reality-prime-phase-radial-readback}

\autoref{sec:unified-reality-zeta-ontic-projection} separates the
complex value of zeta from the carrier rows that produce it.  The
reverse readback of a complex input $s$ exposes those carrier rows as
prime-indexed phase and radial coordinates.  This section records that
readback without treating the complex value $\zeta(s)$ as the complete
object.

\begin{definition}[Prime phase-radial readback]
\label{def:unified-reality-prime-phase-radial-readback}
For
$$
s=\frac12+\delta+i\tau,
$$
the \emph{prime phase-radial readback} of $s$ is the compatible family
of finite prime-window coordinates
$$
\mathcal U_S(s)
:=
\bigl(U_p(s)\bigr)_{p\in S},
\qquad
U_p(s):=p^{1/2-s}=p^{-\delta}e^{-i\tau\log p}.
$$
As $S$ ranges over nonempty finite prime windows, these rows form the
prime-coordinate readback
$$
\mathcal U_{\zeta}(s)
:=
\bigl(p^{1/2-s}\bigr)_{p\in\PrimeUp}.
$$
This readback is not the complex value $\zeta(s)$.  It is the
prime-indexed phase and radial carrier whose trace, Euler-product, and
completed readouts project down to the complex zeta value.
\end{definition}

\begin{definition}[Prime torus and radial deformation body]
\label{def:unified-reality-prime-torus-radial-deformation-body}
The \emph{prime torus} is the finite-window compatible phase body
$$
\mathbb T_{\PrimeUp}
:=
\prod_{p\in\PrimeUp}S^1,
$$
read through its finite faces
$$
\mathbb T_S:=\prod_{p\in S}S^1.
$$
For a normal displacement $\delta$, the corresponding radial deformation
body has finite faces
$$
\mathbb T_{S,\delta}
:=
\prod_{p\in S}p^{-\delta}S^1.
$$
Thus $\delta=0$ is the pure phase torus, while $\delta\#0$ is a common
normal radial deformation across all displayed prime axes.
\end{definition}

\begin{theorem}[The critical surface is the prime torus]
\label{thm:unified-reality-critical-surface-prime-torus}
For the prime phase-radial readback,
$$
\Re(s)=\frac12
\quad\Longleftrightarrow\quad
\mathcal U_S(s)\in\mathbb T_S
\text{ for every nonempty finite prime window }S.
$$
\end{theorem}

\begin{proof}
Write $s=1/2+\delta+i\tau$.  For each displayed prime $p$,
$$
U_p(s)=p^{-\delta}e^{-i\tau\log p}.
$$
If $\Re(s)=1/2$, then $\delta=0$, so
$$
U_p(s)=e^{-i\tau\log p}\in S^1
$$
for every $p$.  Hence every finite readback $\mathcal U_S(s)$ lies in
$\mathbb T_S$.  Conversely, if every finite face lies on the prime
torus, choose any displayed prime $p$.  The singleton face gives
$$
|U_p(s)|=1.
$$
By \autoref{thm:unified-reality-ontic-balance-critical-line}, this is
equivalent to $\Re(s)=1/2$.
\end{proof}

\begin{theorem}[Zeta zeros are projected global cancellations]
\label{thm:unified-reality-zeta-zeros-projected-global-cancellations}
A zeta zero is not a vanishing of a prime coordinate.  It is a global
cancellation row in the trace, product, continuation, and completed
projection of the prime phase-radial carrier.
\end{theorem}

\begin{proof}
For every displayed prime $p$ and every complex row $s$, the local
channel $p^{-s}$ and the centered channel $p^{1/2-s}$ are nonzero
channel rows.  The zero ledger is instead the global row
$$
\zeta(s)=0
$$
or, in the completed carrier, the corresponding completed zero row.
Thus zerohood belongs to the projected readout of all prime channels
together with continuation and completion ledgers.  It is not a local
coordinate zero.  The RH-facing statement is therefore that such global
cancellation, when it occurs in the critical strip, is supported on the
prime torus rather than on a nonunitary radial deformation body.
\end{proof}

\begin{definition}[Prime phase profile]
\label{def:unified-reality-prime-phase-profile}
For a real time coordinate $\tau$, the \emph{prime phase profile} is the
compatible family of finite-window phase rows
$$
\Phi_S(\tau)
:=
\left(e^{-i\tau\log p}\right)_{p\in S}
\in
\prod_{p\in S}S^1,
$$
and the cofinal profile
$$
\Phi_{\PrimeUp}(\tau)
:=
\left(e^{-i\tau\log p}\right)_{p\in\PrimeUp}
\in
\mathbb T_{\PrimeUp}.
$$
It records the simultaneous phase positions of all prime logarithmic
frequencies at the same observation time $\tau$.
\end{definition}

\begin{definition}[Global prime phase closure]
\label{def:unified-reality-global-prime-phase-closure}
A \emph{global prime phase closure} at time $\tau$ is a completed
zero-ledger row whose critical-line carrier is the prime phase profile
$$
\Phi_{\PrimeUp}(\tau)
$$
together with the analytic continuation and completion rows of the zeta
carrier.  It is not a row assigning the zero to a finite set of prime
coordinates.  The displayed local phase rows remain nonzero unit
coordinates; closure belongs to the global completed projection.
\end{definition}

\begin{theorem}[Critical-line zeros are all-prime phase closures]
\label{thm:unified-reality-critical-line-zeros-all-prime-phase-closures}
If
$$
s=\frac12+i\gamma
$$
is a completed zeta zero row, then its prime-coordinate carrier is the
all-prime phase profile
$$
\Phi_{\PrimeUp}(\gamma)
=
\left(e^{-i\gamma\log p}\right)_{p\in\PrimeUp},
$$
and its zerohood is a global closure of that profile through the
completed zeta projection.
\end{theorem}

\begin{proof}
When $s=1/2+i\gamma$, the normal displacement in
\autoref{def:unified-reality-prime-phase-radial-readback} is
$\delta=0$.  Hence every centered prime channel is
$$
U_p(s)=e^{-i\gamma\log p}\in S^1.
$$
Thus the prime-coordinate carrier is exactly
$\Phi_{\PrimeUp}(\gamma)$.  By
\autoref{thm:unified-reality-zeta-zeros-projected-global-cancellations},
the zero row is not a local coordinate zero; it is the completed
projection of the whole prime phase-radial carrier.  Therefore a
critical-line zero is an all-prime phase closure at the time $\gamma$.
\end{proof}

\begin{corollary}[Different zero times give different phase fingerprints]
\label{cor:unified-reality-different-zero-times-phase-fingerprints}
The zero-height coordinate is read as a prime phase fingerprint: two
critical-line zero rows at different times correspond to different
cofinal prime phase profiles.
\end{corollary}

\begin{proof}
If two phase profiles agree on all displayed primes, then for every
prime $p$ the phase ratio satisfies
$$
e^{-i(\gamma_1-\gamma_2)\log p}=1.
$$
For any two distinct primes this gives integer winding rows
$$
(\gamma_1-\gamma_2)\log p=2\pi m_p,
\qquad
(\gamma_1-\gamma_2)\log q=2\pi m_q.
$$
Eliminating the common time difference gives a rational relation between
$\log p$ and $\log q$.  By
\autoref{thm:unified-reality-prime-logarithmic-directions-independent},
the relation is trivial.  Hence the time difference is zero.  Thus a
different zero time cannot have the same cofinal prime phase profile.
\end{proof}

\begin{corollary}[Prime frequencies do not cancel as coordinates]
\label{cor:unified-reality-prime-frequencies-do-not-cancel-as-coordinates}
Global phase closure can cancel in the completed complex readout, but it
does not identify the prime logarithmic frequencies with one another or
remove a prime coordinate from the carrier.
\end{corollary}

\begin{proof}
The completed value may vanish because the whole phase profile is
projected through Euler, continuation, and completion ledgers.  That is
the global cancellation row of
\autoref{thm:unified-reality-zeta-zeros-projected-global-cancellations}.
The prime frequencies themselves remain the logarithmic directions
$\log p$.  Their finite rational independence is
\autoref{thm:unified-reality-prime-logarithmic-directions-independent},
so no finite algebraic relation among displayed prime frequencies
collapses one prime direction into the others.
\end{proof}

\begin{theorem}[Radial deformation is not phase closure]
\label{thm:unified-reality-radial-deformation-not-phase-closure}
For
$$
s=\frac12+\delta+i\tau,
$$
the phase profile
$$
\Phi_{\PrimeUp}(\tau)
$$
and the radial deformation
$$
\bigl(p^{-\delta}\bigr)_{p\in\PrimeUp}
$$
are separate rows of the prime phase-radial carrier.  Phase cancellation
in the completed readout does not cancel the positive square energy of a
nonzero radial deformation.
\end{theorem}

\begin{proof}
The centered channel splits as
$$
U_p(s)=p^{-\delta}e^{-i\tau\log p}.
$$
The unit phase term is the $p$-coordinate of
$\Phi_{\PrimeUp}(\tau)$.  The radial term is measured by
$$
d_p(s):=\log |U_p(s)|=-\delta\log p.
$$
Under the mirror $J$, this signed row changes sign, as recorded in
\autoref{thm:unified-reality-prime-defect-mirror-odd}.  Its square
$$
e_p(s)=d_p(s)^2=\delta^2(\log p)^2
$$
is mirror-even by
\autoref{thm:unified-reality-prime-square-energy-mirror-even}.  Thus
phase closure may account for a vanishing completed value, but it does
not supply a negative term for the positive radial energy.
\end{proof}

\begin{corollary}[Off-half zero rows would carry all-prime radial energy]
\label{cor:unified-reality-off-half-zero-rows-all-prime-radial-energy}
If a critical-strip zero row had $\delta\#0$, then every displayed prime
coordinate would carry nonzero radial defect, and the global prime
normal energy would be infinite.
\end{corollary}

\begin{proof}
For every displayed prime,
\autoref{thm:unified-reality-radial-deformation-not-phase-closure} gives
$$
d_p(s)=-\delta\log p.
$$
Since $\delta\#0$ and $\log p>0$, each prime row detects the same
off-half normal displacement.  Summing the positive square energies over
finite prime windows gives the extended global energy of
\autoref{def:unified-reality-prime-normal-square-energy}.  By
\autoref{thm:unified-reality-global-prime-square-energy-diverges-off-half},
that energy is infinite.
\end{proof}

\begin{corollary}[The no-hair target is zero relevance of radial energy]
\label{cor:unified-reality-no-hair-target-zero-relevance-radial-energy}
The remaining RH-facing target is not the detection of an off-half
radial deformation.  It is the zero-relevance claim that a completed
zeta zero cannot carry the all-prime positive radial energy of
\autoref{cor:unified-reality-off-half-zero-rows-all-prime-radial-energy}.
\end{corollary}

\begin{proof}
Detection is already finite and prime-local: each displayed prime reads
$d_p(s)=-\delta\log p$, and every finite prime window reads the same
normal displacement after normalization.  The completed zero ledger must
still be linked to that detected obstruction.  This is the no-hair
boundary of
\autoref{def:unified-reality-zeta-no-hair-boundary} and the cofinal
socket form of
\autoref{def:unified-reality-zero-far-end-energy-unity-target}.
\end{proof}
