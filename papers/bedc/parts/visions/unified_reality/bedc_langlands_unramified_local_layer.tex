\section{BEDC--Langlands Unramified Local Layer}
\label{sec:unified-reality-bedc-langlands-unramified-local-layer}

The finite detector $(v,r)$ used throughout the global
BEDC--Langlands packet is a local object. It is licensed by the
unramified Satake correspondence: spherical Hecke eigenvalues at an
unramified nonarchimedean place are the same data as a semisimple
conjugacy class on the dual side. This section isolates that local
certificate before the global trace-window projector is used.

\begin{definition}[Unramified local spherical datum]
\label{def:unified-reality-unramified-local-spherical-datum}
Let $F_v$ be a nonarchimedean local field with residue cardinality
$q_v$. Let $G/F_v$ be an unramified connected reductive group, and let
$$
K_v\subset G(F_v)
$$
be a hyperspecial compact subgroup. The local spherical Hecke algebra is
$$
\mathcal H_v(G,K_v)
:=
C_c(K_v\backslash G(F_v)/K_v),
$$
with convolution product. An unramified local automorphic row is an
irreducible smooth representation $\pi_v$ with
$$
\pi_v^{K_v}\ne0.
$$
\end{definition}

\begin{definition}[Local BEDC rows]
\label{def:unified-reality-local-bedc-rows}
The local time row is generated by the Cartan decomposition
$$
G(F_v)
=
\bigsqcup_{\lambda\in X_\ast(T)^+}
K_v\lambda(\varpi_v)K_v.
$$
Each dominant coweight $\lambda$ determines a Hecke time step
$T_{v,\lambda}$. The local unity row is the spherical Hecke character
$$
\chi_{\pi_v}:\mathcal H_v(G,K_v)\to\mathbb C
$$
or, equivalently, its Satake parameter readout. The local ledger row
records every departure from the unramified spherical setting:
ramification, failure of $K_v$-fixed vectors, non-hyperspecial level,
normalization choices including $q_v^{1/2}$, central character rows, and
bad reduction.
\end{definition}

\begin{definition}[Unramified local parameter]
\label{def:unified-reality-unramified-local-parameter}
Let $\widehat G$ be the dual group. In the unramified local setting, the
local $L$-group has the form
$$
{}^LG=\widehat G\rtimes\langle\operatorname{Frob}_v\rangle.
$$
An unramified local parameter is represented by a semisimple conjugacy
class
$$
s_{\phi,v}\in\widehat G/\!/\mathrm{conj},
$$
equivalently by
$$
\phi_v(\operatorname{Frob}_v)
=
s_{\phi,v}\rtimes\operatorname{Frob}_v
$$
under the declared normalization.
\end{definition}

\begin{definition}[Local Hecke--Frobenius residual]
\label{def:unified-reality-local-hecke-frobenius-residual}
Let $\pi_v$ be spherical and let $\phi_v$ be unramified. For every
finite-dimensional representation $r:{}^LG\to GL(V_r)$, define
$$
D_{v,r}(\pi_v,\phi_v)
:=
\operatorname{tr}r(s(\pi_v))
-
\operatorname{tr}r(\phi_v(\operatorname{Frob}_v)).
$$
The local detector $r$ matches exactly when
$$
D_{v,r}(\pi_v,\phi_v)=0.
$$
\end{definition}

\begin{theorem}[Unramified Satake correspondence]
\label{thm:unified-reality-unramified-satake-correspondence}
For the unramified local spherical datum, algebra homomorphisms
$$
\mathcal H_v(G,K_v)\to\mathbb C
$$
are equivalent to semisimple conjugacy classes in the dual quotient:
$$
\operatorname{Hom}_{\mathrm{alg}}(\mathcal H_v(G,K_v),\mathbb C)
\cong
\widehat G/\!/\mathrm{conj}.
$$
Thus a spherical irreducible representation $\pi_v$ has a Satake
parameter
$$
s(\pi_v)\in\widehat G/\!/\mathrm{conj}.
$$
\end{theorem}
\begin{proof}
The Satake isomorphism gives the normalized identification
$$
\mathcal H_v(G,K_v)
\cong
\mathbb C[X_\ast(T)]^W,
$$
with the $q_v^{1/2}$ convention recorded in the local ledger. The ring
$\mathbb C[X_\ast(T)]^W$ is the coordinate ring of the semisimple
conjugacy quotient of $\widehat G$. Algebra characters of the Hecke
algebra are therefore points of that affine quotient, which are exactly
semisimple Satake parameters.
\end{proof}

\begin{corollary}[Local finite detectors separate Satake parameters]
\label{cor:unified-reality-local-detectors-separate-satake}
If two spherical local rows $\pi_v$ and $\pi_v'$ satisfy
$$
\operatorname{tr}r(s(\pi_v))
=
\operatorname{tr}r(s(\pi_v'))
$$
for every finite-dimensional representation $r$ of ${}^LG$, then
$$
s(\pi_v)=s(\pi_v')
$$
in $\widehat G/\!/\mathrm{conj}$.
\end{corollary}
\begin{proof}
Finite-dimensional characters separate semisimple conjugacy classes of
the complex reductive dual group. Hence equality of all trace readouts
places the two Satake parameters at the same point of the semisimple
quotient.
\end{proof}

\begin{theorem}[Unramified local no-hidden residual]
\label{thm:unified-reality-unramified-local-no-hidden-residual}
Let $(\pi_v,\phi_v)$ be an unramified local BEDC--Langlands row with the
local three-properties discipline. If the row is closed against
unresolved local ledger entries, then
$$
D_{v,r}(\pi_v,\phi_v)=0
$$
for every finite-dimensional representation $r$.
\end{theorem}
\begin{proof}
If $D_{v,r}(\pi_v,\phi_v)\ne0$ for some $r$, then the finite local
detector $r$ has directly found a Hecke--Frobenius mismatch. Ledger
discipline gives only two possibilities: the mismatch is recorded as a
local residual row, or it is omitted. The first alternative leaves the
object not closed until the row is resolved; the second violates the
ledger property. Both contradict closedness with the three-properties
discipline. Thus every local finite detector matches.
\end{proof}

\begin{corollary}[Unramified local Langlands closure]
\label{cor:unified-reality-unramified-local-langlands-closure}
In the unramified spherical setting, a closed local BEDC--Langlands row
is equivalent to equality of Satake and Frobenius parameters:
$$
s(\pi_v)=s_{\phi,v}
$$
in $\widehat G/\!/\mathrm{conj}$. Equivalently,
$$
\operatorname{tr}r(s(\pi_v))
=
\operatorname{tr}r(\phi_v(\operatorname{Frob}_v))
$$
for every finite-dimensional $r$.
\end{corollary}
\begin{proof}
Equality in the quotient gives equality of all finite-dimensional trace
readouts. Conversely, equality of all readouts separates the semisimple
quotient by
\autoref{cor:unified-reality-local-detectors-separate-satake}. The
no-hidden residual theorem excludes any closed local row with a
remaining detected mismatch.
\end{proof}

\begin{definition}[Local finite packet]
\label{def:unified-reality-local-finite-packet}
Fix a finite displayed set of spherical local rows
$$
\mathcal P_v=\{\pi_{v,1},\ldots,\pi_{v,M}\}.
$$
Write
$$
s_i:=s(\pi_{v,i}),
\qquad
s_{\phi}:=s_{\phi,v}.
$$
This is a fixed local packet list; no growing-window assertion is made
inside the local projector construction.
\end{definition}

\begin{theorem}[Local Hecke projector certificate]
\label{thm:unified-reality-local-hecke-projector-cert}
For a finite local packet $\mathcal P_v$ and target unramified parameter
$s_{\phi}$, there is
$$
f_{\phi,v}\in\mathcal H_v(G,K_v)
$$
such that
$$
s_i\ne s_{\phi}
\quad\Longrightarrow\quad
\operatorname{tr}\pi_{v,i}(f_{\phi,v})=0,
$$
and
$$
s_i=s_{\phi}
\quad\Longrightarrow\quad
\operatorname{tr}\pi_{v,i}(f_{\phi,v})\ne0
$$
outside explicitly declared local cancellation rows.
\end{theorem}
\begin{proof}
By the Satake correspondence,
$$
\mathcal H_v(G,K_v)
\cong
\mathbb C[\widehat G/\!/\mathrm{conj}].
$$
The finite set of points $\{s_1,\ldots,s_M,s_{\phi}\}$ lies in an
affine quotient. For each nonmatching $s_i\ne s_{\phi}$, the coordinate
ring separates the two closed points, so there is a function $F_i$ with
$$
F_i(s_{\phi})=1,
\qquad
F_i(s_i)=0.
$$
The product
$$
F_{\phi,v}:=\prod_{s_i\ne s_{\phi}}F_i
$$
is $1$ at $s_{\phi}$ and $0$ at every displayed nonmatching local
parameter. Reading $F_{\phi,v}$ through Satake gives the desired Hecke
function $f_{\phi,v}$. On a spherical eigenrow, the trace is evaluation
at the Satake parameter.
\end{proof}

\begin{theorem}[Global detector as a product of local detectors]
\label{thm:unified-reality-global-detector-product-local}
Let
$$
S=\{v_1,\ldots,v_m\}
$$
be a finite unramified window. For each $v\in S$, choose the local
projector $f_{\phi,v}$ of
\autoref{thm:unified-reality-local-hecke-projector-cert}. Then the
finite tensor
$$
f_{\phi,S}:=\bigotimes_{v\in S}f_{\phi,v}
$$
belongs to
$$
\bigotimes_{v\in S}\mathcal H_v(G,K_v)
$$
and kills every displayed global row whose Satake fingerprint fails to
match $\phi$ at some place of $S$.
\end{theorem}
\begin{proof}
The restricted tensor product of the local spherical Hecke algebras is
the finite-window Hecke algebra on $S$. If a displayed global packet
fails to match the target at some $v\in S$, then the corresponding local
factor has zero trace against $f_{\phi,v}$. The tensor trace is therefore
zero. If all local Satake fingerprints match, each local factor is
nonzero outside declared cancellation rows, so the finite tensor is the
global fixed-window projector used by
\autoref{thm:unified-reality-finite-hecke-projector-cert}.
\end{proof}

\begin{theorem}[Fixed-window trace realization through local Satake]
\label{thm:unified-reality-fixed-window-trace-local-satake}
Assume every $v\in S$ is unramified for $\phi$ and for the displayed
candidate packets. Suppose the local projectors $f_{\phi,v}$ are
constructed as above, the stable trace-window certificate holds for
$$
f_{\phi,S}=\bigotimes_{v\in S}f_{\phi,v},
$$
the parameter stable distribution is nonzero on $f_{\phi,S}$, and all
non-target continuous, endoscopic, boundary, ramified, and normalization
rows are separated or ledgered. Then some automorphic packet $\Pi$
matches $\phi$ on every place of $S$:
$$
s_v(\Pi)=\phi(\operatorname{Frob}_v)
\qquad(v\in S).
$$
\end{theorem}
\begin{proof}
The stable trace-window certificate transports the nonzero parameter
readout to a nonzero spectral value. After the non-target ledger rows
are separated, some automorphic packet has nonzero trace against
$f_{\phi,S}$. By
\autoref{thm:unified-reality-global-detector-product-local}, a packet
with any local mismatch in $S$ is killed by the tensor projector. Hence
the packet with nonzero trace matches the target Satake parameter at
every place of the finite window.
\end{proof}

\begin{theorem}[Local-to-global residual dichotomy]
\label{thm:unified-reality-local-global-residual-dichotomy}
Let $(\Pi,\phi)$ be a global BEDC--Langlands row. Any residual mismatch
has one of two forms: either there is an unramified place $v$ and a
finite-dimensional representation $r$ such that
$$
D_{v,r}(\Pi,\phi)\ne0,
$$
or the residual belongs to a ramified, archimedean, endoscopic,
continuous-spectrum, boundary, nilpotent-support, or normalization
ledger row.
\end{theorem}
\begin{proof}
If the residual appears at an unramified spherical place, the local
Satake layer applies. A mismatch of semisimple local parameters is
detected by some finite-dimensional representation by
\autoref{cor:unified-reality-local-detectors-separate-satake}. This is
the first alternative. If no unramified spherical detector sees the
residual, then it lies outside the local layer: ramification,
archimedean data, endoscopy, continuous spectrum, boundary terms,
nilpotent or singular support, or normalization. Those are exactly the
global ledger rows.
\end{proof}

\begin{theorem}[Local-to-global BEDC Langlands closure]
\label{thm:unified-reality-local-global-bedc-langlands-closure}
Assume the unramified local layer above, fixed-window trace realization
on a cofinal window tower, cofinal automorphic tail closure, complete
ledger discipline, and closedness against unresolved ledger rows. Then
the resulting global packet satisfies
$$
\lambda_{v,r}(\Pi_{\phi})
=
\operatorname{tr}r(\phi(\operatorname{Frob}_v))
$$
for every certified unramified finite detector $(v,r)$.
\end{theorem}
\begin{proof}
The unramified local layer licenses each finite detector $(v,r)$ and
the finite-window projectors built from them. Fixed-window trace
realization supplies a realizing packet on each window. The cofinal
window-to-global bridge gives either a global packet or an automorphic
tail ledger row; closedness excludes unresolved tail rows. If a residual
remained after the global packet is obtained, the local-to-global
residual dichotomy would make it either a finite detector mismatch or a
ledger row. A finite detector mismatch must be recorded and hence is not
closed until resolved; an unrecorded one violates ledger discipline.
The same applies to the ledger alternative. Therefore no residual
remains on the certified unramified scope.
\end{proof}

\begin{corollary}[Quadratic detector as the abelian local shadow]
\label{cor:unified-reality-quadratic-detector-local-shadow}
For $G=GL_1$ and $K=\mathbb Q(i)$, the detector
$$
p\longmapsto\chi_{-4}(\operatorname{Frob}_p)
$$
is the abelian local form of the general detector
$$
(v,r)\longmapsto\operatorname{tr}r(\phi(\operatorname{Frob}_v)).
$$
Thus the predicate $p=a^2+b^2$ is a visible local Frobenius detector in
the abelian base case.
\end{corollary}
\begin{proof}
For $GL_1$, the dual group is $\mathbb C^\times$ and finite-dimensional
representations are characters. The Satake parameter at an unramified
prime is therefore a single scalar. In the quadratic extension
$\mathbb Q(i)/\mathbb Q$, the scalar $\chi_{-4}(\operatorname{Frob}_p)$
equals $1$ exactly at split primes, which are the odd primes represented
by $a^2+b^2$ by
\autoref{thm:unified-reality-gl1-quadratic-splitting-detector}.
\end{proof}

\begin{theorem}[Unramified local boundary]
\label{thm:unified-reality-unramified-local-boundary}
The unramified local layer proves the legality and separation of finite
Hecke--Frobenius detectors at spherical places. It does not prove the
stable trace formula, construct parameter stable distributions, settle
ramified local Langlands, or remove endoscopic, continuous-spectrum,
boundary, nilpotent-support, or archimedean ledger rows.
\end{theorem}
\begin{proof}
Every result in this section consumes the unramified spherical datum and
the Satake correspondence. The non-spherical, ramified, endoscopic,
continuous, boundary, nilpotent-support, and archimedean terms are
excluded from the local proof and explicitly placed in the ledger. The
global existence step is supplied separately by the trace-window or
geometric existence bridge.
\end{proof}
