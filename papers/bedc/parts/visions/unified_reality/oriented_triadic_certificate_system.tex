\section{Oriented Triadic Certificate System}
\label{sec:unified-reality-oriented-triadic-certificate-system}

The oriented tower construction can be read as a certificate system.  It
does not assert the unconditional RH statement.  It records the precise
conditional form: a completed critical-strip zero placed inside a closed
oriented triadic tower has no admissible real positive scale direction.
The mathematical burden is therefore the construction of the certificate
for every completed zero, not the downstream energy contradiction.

\begin{definition}[Ontic triadic rows]
\label{def:unified-reality-ontic-triadic-rows}
The \emph{ontic triadic rows} are
$$
\Omega_{\mathrm{tri}}
:=
(\mathsf P,\mathsf U,\mathsf T),
$$
where $\mathsf P$ is the prime-register or discrete-label row,
$\mathsf U$ is the unit-boundary, parity, compactness, or Lee--Yang row,
and $\mathsf T$ is the phase-time or recursive-routing row.  A closed
triadic packet carries a closure relation
$$
\mathcal C(\mathsf P,\mathsf U,\mathsf T)=0
$$
up to legal gauge.  Legal gauge may come from the unit phase circle or
from prime-adic register fibres.  A nontrivial real positive scale is
not a legal gauge row.
\end{definition}

\begin{definition}[Prime ledger and logarithmic height]
\label{def:unified-reality-prime-ledger-logarithmic-height}
The \emph{prime ledger} is the free finite-support integer register
$$
\mathcal L_{\PrimeUp}:=\bigoplus_{p\in\PrimeUp}\mathbb Z e_p.
$$
For a positive integer
$$
n=\prod_p p^{v_p(n)},
$$
the ledger row is
$$
\nu(n):=\sum_p v_p(n)e_p,
$$
and the logarithmic height row is
$$
H(n):=\log n=\sum_p v_p(n)\log p.
$$
More generally a finite ledger vector
$$
x=\sum_p x_pe_p
$$
has height
$$
H(x):=\sum_p x_p\log p.
$$
\end{definition}

\begin{lemma}[Prime ledger height is separated]
\label{lem:unified-reality-prime-ledger-height-separated}
For a finite ledger vector $x\in\mathcal L_{\PrimeUp}$,
$$
H(x)=0
\quad\Longrightarrow\quad
x=0.
$$
\end{lemma}

\begin{proof}
Write
$$
x=\sum_{i=1}^{m}x_{p_i}e_{p_i}.
$$
If $H(x)=0$, then
$$
\sum_{i=1}^{m}x_{p_i}\log p_i=0.
$$
Exponentiating gives
$$
\prod_{i=1}^{m}p_i^{x_{p_i}}=1.
$$
By unique factorization, every exponent $x_{p_i}$ is zero.  Hence
$x=0$.
\end{proof}

\begin{definition}[Ontic logarithmic Hamiltonian]
\label{def:unified-reality-ontic-logarithmic-hamiltonian}
The \emph{ontic logarithmic Hamiltonian} on the integer basis is the
diagonal readout
$$
H_{\mathrm{ont}}|n\rangle=(\log n)|n\rangle
$$
with domain
$$
\operatorname{Dom}(H_{\mathrm{ont}})
:=
\left\{
\psi=(\psi_n)\in\ell^2(\NatUp_{\ge1}):
\sum_{n\ge1}(\log n)^2|\psi_n|^2<\infty
\right\}.
$$
Its unitary phase evolution row is
$$
U(\gamma)=e^{-i\gamma H_{\mathrm{ont}}}.
$$
This Hamiltonian row is a spectral readout for the logarithmic height,
not a primitive physical postulate.
\end{definition}

\begin{definition}[Oriented triadic certificate]
\label{def:unified-reality-oriented-triadic-certificate}
For a critical-strip coordinate
$$
s_0=\frac12+\delta+i\tau,
$$
an \emph{oriented triadic certificate} over $s_0$ is a finite-window
tower
$$
\mathcal W_{s_0}
:=
(F_n,T_n,D_n,A_n,S_n,\Sigma_{S_n},X_n,\mathcal R_n,\pi_{n+1,n},
\Omega_n,W_n,B_{s_0})_n
$$
whose rows are:
$$
\mathsf{ActiveCapture},
\quad
\mathsf{EulerLogSolenoidBridge},
\quad
\mathsf{FiniteTriadicClosure},
$$
$$
\mathsf{DefectExtraction},
\quad
\mathsf{CofinalGrowth},
\quad
\mathsf{IndependentBoundaryStability}.
$$
The certificate is the fully displayed form of
\autoref{def:unified-reality-oriented-tower-construction-rows}.  It
keeps source data, compatibility data, and boundary data in the same
packet.
\end{definition}

\begin{definition}[Certificate row meanings]
\label{def:unified-reality-oriented-certificate-row-meanings}
The six certificate rows mean the following.
The active-capture row supplies a zero island $D_n$ with
$$
\Xi=F_n+T_n,
\qquad
|T_n|<|F_n|
\quad
\text{on }\partial D_n,
$$
or an equivalent local-degree capture row.  The Euler-log solenoid bridge
places the visible Euler-log phase and the prime-register fibre in one
phase carrier.  The finite triadic closure row closes
$(\mathsf P,\mathsf U,\mathsf T)$ modulo legal prime-adic gauge.  The
defect-extraction row gives
$$
\Omega_n(s_0)=(-\delta h_a)_{a\in A_n}
$$
for positive heights $h_a>0$.  The cofinal-growth row gives
$$
W_n:=\sum_{a\in A_n}h_a^2\to\infty.
$$
The independent-boundary-stability row gives a finite constant
$$
B_{s_0}<\infty
$$
with
$$
|\Omega_n(s_0)|^2\le B_{s_0}
$$
for the same zero-relevant tower.
\end{definition}

\begin{proposition}[Defect extraction gives square energy]
\label{prop:unified-reality-oriented-certificate-defect-square-energy}
For an oriented triadic certificate and
$$
s_0=\frac12+\delta+i\tau,
$$
the defect row gives
$$
|\Omega_n(s_0)|^2=\delta^2W_n.
$$
\end{proposition}

\begin{proof}
For every active node $a\in A_n$, the local channel is
$$
U_a(s_0)=e^{-(s_0-1/2)h_a}
=
e^{-\delta h_a}e^{-i\tau h_a}.
$$
Thus
$$
\log|U_a(s_0)|=-\delta h_a.
$$
By definition,
$$
|\Omega_n(s_0)|^2
=
\sum_{a\in A_n}(-\delta h_a)^2
=
\delta^2\sum_{a\in A_n}h_a^2
=
\delta^2W_n.
$$
\end{proof}

\begin{proposition}[Parity oddness of the normal defect]
\label{prop:unified-reality-oriented-certificate-normal-defect-parity}
For the mirror involution
$$
J(s)=1-\overline{s},
$$
if $s=1/2+\delta+i\tau$, then $J(s)=1/2-\delta+i\tau$ and every active
node satisfies
$$
d_a(Js)=-d_a(s).
$$
\end{proposition}

\begin{proof}
The mirror computation gives
$$
J\left(\frac12+\delta+i\tau\right)
=
1-\left(\frac12+\delta-i\tau\right)
=
\frac12-\delta+i\tau.
$$
Since $d_a(s)=-\delta h_a$, replacing $\delta$ by $-\delta$ gives
$d_a(Js)=\delta h_a=-d_a(s)$.
\end{proof}

\begin{proposition}[Active capture preserves the zero count]
\label{prop:unified-reality-active-capture-preserves-zero-count}
If the active-capture row holds on a zero island $D_n$, then $F_n$ and
$\Xi$ have the same number of zeros in $D_n$, counted with
multiplicity.
\end{proposition}

\begin{proof}
The capture row gives
$$
\Xi=F_n+T_n
$$
and
$$
|T_n|<|F_n|
\quad
\text{on }\partial D_n.
$$
By Rouch\'e's theorem, $F_n$ and $F_n+T_n=\Xi$ have the same zero count
inside $D_n$, counted with multiplicity.
\end{proof}

\begin{proposition}[A split-prime tower is a cofinal-growth template]
\label{prop:unified-reality-split-prime-tower-cofinal-growth-template}
Let $q$ be a rational prime and let $(K_n)$ be a family of number fields
with
$$
[K_n:\mathbb Q]\to\infty.
$$
If $q$ splits completely in $K_n$ as
$$
q\mathcal O_{K_n}
=
\mathfrak q_{n,1}\cdots\mathfrak q_{n,d_n},
\qquad
d_n=[K_n:\mathbb Q],
$$
and each split-prime node has height $\log q$, then the associated
square weight satisfies
$$
W_n=d_n(\log q)^2\to\infty.
$$
\end{proposition}

\begin{proof}
There are $d_n$ displayed split-prime nodes, each with height $\log q$.
Therefore
$$
W_n=\sum_{i=1}^{d_n}(\log q)^2=d_n(\log q)^2.
$$
Since $d_n\to\infty$ and $\log q>0$, the weights are cofinal.
\end{proof}

\begin{theorem}[An oriented triadic certificate forces the fixed-half row]
\label{thm:unified-reality-oriented-triadic-certificate-forces-fixed-half}
If a critical-strip zero coordinate
$$
s_0=\frac12+\delta+i\tau
$$
has an oriented triadic certificate, then
$$
\delta=0.
$$
\end{theorem}

\begin{proof}
By
\autoref{prop:unified-reality-oriented-certificate-defect-square-energy},
the certificate gives
$$
|\Omega_n(s_0)|^2=\delta^2W_n.
$$
Assume $\delta\#0$.  Then $\delta^2>0$, and the cofinal-growth row
$W_n\to\infty$ gives
$$
|\Omega_n(s_0)|^2\to\infty.
$$
This contradicts independent boundary stability,
$$
|\Omega_n(s_0)|^2\le B_{s_0}<\infty.
$$
Hence $\neg(\delta\#0)$.  Located fixed-half collapse gives
$\delta=0$.
\end{proof}

\begin{corollary}[Certificate existence is conditionally sufficient]
\label{cor:unified-reality-oriented-certificate-existence-conditionally-sufficient}
If every completed critical-strip zeta zero has an oriented triadic
certificate, then the RH fixed-half section exists.
\end{corollary}

\begin{proof}
Let $z:\mathcal Z_{\zeta}$ have coordinate
$$
s=\frac12+\delta+i\tau.
$$
By hypothesis, $s$ has an oriented triadic certificate.  By
\autoref{thm:unified-reality-oriented-triadic-certificate-forces-fixed-half},
we have $\delta=0$, so $\Re(s)=1/2$.  Varying $z$ gives the fixed-half
section, equivalently the RH section of
\autoref{thm:unified-reality-rh-fixed-half-section-equivalence}.
\end{proof}

\begin{corollary}[Off-half zeros are not closed-triadic generated]
\label{cor:unified-reality-off-half-zeros-not-closed-triadic-generated}
If
$$
s=\frac12+\delta+i\tau
$$
is a critical-strip zero coordinate with $\delta\#0$, then $s$ cannot
carry an oriented triadic certificate satisfying cofinal growth and
independent boundary stability.
\end{corollary}

\begin{proof}
If such a certificate existed, then
\autoref{thm:unified-reality-oriented-triadic-certificate-forces-fixed-half}
would give $\delta=0$, contradicting $\delta\#0$.
\end{proof}

\begin{definition}[Source obligations for the certificate route]
\label{def:unified-reality-oriented-certificate-source-obligations}
The remaining source obligations for the oriented certificate route are:
$$
\mathsf{CanonicalActiveCapture},
\qquad
\mathsf{EulerLogSolenoidBridge},
\qquad
\mathsf{FiniteTriadicClosure},
$$
$$
\mathsf{DefectExtraction},
\qquad
\mathsf{ActiveCofinalGrowth},
\qquad
\mathsf{IndependentBoundaryStability}.
$$
The defect-extraction row is the direct calculation of
\autoref{prop:unified-reality-oriented-certificate-defect-square-energy}.
The other rows are source theorems: they must be proved for the same
zero-relevant tower and may not be inferred merely from the zero
predicate.
\end{definition}

\begin{theorem}[The source obligations are the noncircular content]
\label{thm:unified-reality-oriented-certificate-source-obligations-noncircular-content}
The conditional implication from an oriented triadic certificate to the
fixed-half row does not discharge the source obligations of
\autoref{def:unified-reality-oriented-certificate-source-obligations}.
\end{theorem}

\begin{proof}
The implication uses the certificate rows after they have been supplied.
It consumes active capture to ensure zero relevance, cofinal growth to
make an off-half defect unbounded, and independent boundary stability to
bound the same defect.  It does not construct the active endpoint packet,
the Euler-log solenoid bridge, the finite triadic closure law, the active
growth tower, or the Lee--Yang, unitary, cosine, or dominance stability
input.  Therefore the proof is noncircular exactly when these rows are
proved independently as source data, as also isolated in
\autoref{cor:unified-reality-hard-source-row-is-generability}.
\end{proof}
