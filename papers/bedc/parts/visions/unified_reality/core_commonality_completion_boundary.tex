\section{Core Commonality and Completion Boundaries}
\label{sec:unified-reality-core-commonality-completion-boundary}

The unified-reality programme is a theory of commonality, but the word
``common'' has two distinct roles.  The first role is ontic: every
licensed BEDC object must expose observation-time, distinction-atom, and
symmetry-persistence rows.  The second role is object-specific:
one may ask whether those rows are complete and faithful for all public
observables of a given carrier.  The first role is the unifying
language of the theory.  The second role is a proof obligation, and in
the zeta-zero case it is strong enough to imply the RH section.

\begin{definition}[Core commonality basis]
\label{def:unified-reality-core-commonality-basis}
The \emph{core commonality basis} is
$$
\mathsf{Core}
:=
(\mathsf{ObsTime},\mathsf{DistAtom},\mathsf{SymPersist}),
$$
where $\mathsf{ObsTime}$ records observation, history, and continuation;
$\mathsf{DistAtom}$ records irreducible distinction and atom data; and
$\mathsf{SymPersist}$ records classifier-stable transport preserving
distinction through observation-time.
\end{definition}

\begin{definition}[Core signature of a licensed object]
\label{def:unified-reality-core-signature-licensed-object}
For a licensed object $X^{\uparrow}$, a \emph{core signature} is a
displayed readout
$$
\Sigma_X:
X^{\uparrow}
\to
\mathsf{ObsTime}_X
\times
\mathsf{DistAtom}_X
\times
\mathsf{SymPersist}_X.
$$
It records that $X$ can be observed through histories, has stable
distinctions, and carries those distinctions through permitted
transports.
\end{definition}

\begin{definition}[Core completeness]
\label{def:unified-reality-core-completeness}
A licensed object $X^{\uparrow}$ is \emph{core-complete} when every
displayed public observable
$$
F:X^{\uparrow}\to R
$$
in the public carrier class factors through the core signature:
$$
F=\overline F\circ\Sigma_X
$$
for some readout
$$
\overline F:
\mathsf{ObsTime}_X
\times
\mathsf{DistAtom}_X
\times
\mathsf{SymPersist}_X
\to R.
$$
This does not deny the existence of derived observables such as order,
area, derivative, spectrum, zero, or defect.  It says that no such
observable is admitted as an independent fourth ontic primitive.
\end{definition}

\begin{definition}[Independent fourth observable]
\label{def:unified-reality-independent-fourth-observable}
An \emph{independent fourth observable} for $X^{\uparrow}$ is a public
observable $Q:X^{\uparrow}\to R_Q$ for which no factorization through
$\Sigma_X$ is supplied and which is not explicitly marked as a boundary,
bridge, or refused external operation.
\end{definition}

\begin{theorem}[Core completeness excludes independent fourth observables]
\label{thm:unified-reality-core-completeness-excludes-fourth-observables}
If $X^{\uparrow}$ is core-complete, then every displayed public
observable of $X$ is either a core-signature readout or is outside the
core surface as an explicit bridge, boundary, or refusal.  In
particular, no independent fourth observable is admitted inside the core
interface.
\end{theorem}

\begin{proof}
Core completeness gives a factorization
$$
F=\overline F\circ\Sigma_X
$$
for every public observable $F$ in the core class.  Therefore such an
$F$ is a readout of the core signature rather than an independent
primitive.  If a displayed row does not factor through the core
signature, the only licensed alternatives are to place it in a bridge
or boundary ledger, or to refuse it at the public core surface.  This is
the anti-primitive discipline of
\autoref{thm:unified-reality-stable-interfaces-require-ontic-core} and
\autoref{cor:unified-reality-axiom-purity-object-induction-invariant}.
\end{proof}

\begin{corollary}[Fourth commonality is a theory-extension signal]
\label{cor:unified-reality-fourth-commonality-theory-extension-signal}
If a genuinely independent public observable $Q$ is required for a
carrier, then either $Q$ is a derived core readout, a bridge or boundary
row, or the three-row core is not complete for that carrier.
\end{corollary}

\begin{proof}
By
\autoref{thm:unified-reality-core-completeness-excludes-fourth-observables},
a core-complete carrier leaves no place for an independent fourth
observable inside its public core interface.  Thus a proposed $Q$ must
be reclassified as a derived readout, placed in a bridge or boundary
ledger, refused as an unaudited primitive, or treated as evidence that
the core-completeness claim for the carrier has failed.
\end{proof}

\begin{definition}[Zeta zero core-completeness target]
\label{def:unified-reality-zeta-zero-core-completeness-target}
The \emph{zeta zero core-completeness target} is
$$
\mathsf{CoreComplete}(\mathcal Z_{\zeta}),
$$
where the source carrier $\mathcal Z_{\zeta}$ is the critical-strip zero
source and $\Sigma_Z$ is the zero ontic signature of
\autoref{def:unified-reality-zero-ontic-signature}.  It asserts that
every public zero observable factors through the atom-type, spectral
time, and symmetry-persistence signature of the zero source.
\end{definition}

\begin{definition}[Normal displacement as a candidate fourth observable]
\label{def:unified-reality-normal-displacement-candidate-fourth}
For the zeta mirror, the normal displacement is
$$
\delta(s):=\Re(s)-\frac12.
$$
It is a candidate fourth observable for the zero source because it
separates the mirror companions
$$
s=\frac12+\delta+i\gamma,
\qquad
J(s)=\frac12-\delta+i\gamma
$$
by the sign rule
$$
\delta(J(s))=-\delta(s),
$$
while their zero atom type, spectral time, and mirror orbit signature
are the same.
\end{definition}

\begin{theorem}[Zeta zero core completeness forces the fixed surface]
\label{thm:unified-reality-zeta-zero-core-completeness-forces-fixed-surface}
Assume mirror closure, mirror-even zero signature, core completeness of
the zeta zero source, and that $\delta$ is a public zero observable.
Then every critical-strip zero lies on the fixed surface
$$
\Re(s)=\frac12.
$$
\end{theorem}

\begin{proof}
Core completeness of $\mathcal Z_{\zeta}$ gives a factorization
$$
\delta(s)=f(\Sigma_Z(s)).
$$
Mirror-evenness gives
$$
\Sigma_Z(J(s))=\Sigma_Z(s),
$$
so
$$
\delta(J(s))=f(\Sigma_Z(J(s)))=f(\Sigma_Z(s))=\delta(s).
$$
But the direct mirror calculation in
\autoref{def:unified-reality-normal-displacement-candidate-fourth} gives
$$
\delta(J(s))=-\delta(s).
$$
Hence $\delta(s)=0$ in the characteristic-zero real readout.  This is
the fixed surface $\Re(s)=1/2$.
\end{proof}

\begin{corollary}[Core completeness gives the RH witness shape]
\label{cor:unified-reality-core-completeness-gives-rh-witness-shape}
Under the hypotheses of
\autoref{thm:unified-reality-zeta-zero-core-completeness-forces-fixed-surface},
the zeta zero source admits the RH fixed-half section.
\end{corollary}

\begin{proof}
The theorem assigns the fixed-half conclusion to every critical-strip
zero source row.  By
\autoref{thm:unified-reality-rh-fixed-half-section-equivalence}, this is
exactly the RH fixed-half section, i.e. the constructive witness shape
needed for the BEDC RH target.
\end{proof}

\begin{theorem}[Core expression is distinct from core completeness]
\label{thm:unified-reality-core-expression-distinct-from-completeness}
The claim that every licensed object carries the core commonality basis
is weaker than the claim that a specific object is core-complete.
\end{theorem}

\begin{proof}
\autoref{thm:unified-reality-object-induction-yields-ontic-section}
supplies the core rows to licensed objects by object induction.  This
only gives the existence of observation-time, distinction-atom, and
symmetry-persistence ledgers.  Core completeness additionally asserts
that every public observable factors through those ledgers.  A symmetric
two-point carrier can have observation, distinction, and symmetry while
still carrying a non-fixed mirror orbit.  Thus core expression does not
collapse all symmetry orbits to fixed orbits; completeness and
faithfulness are the additional claims.
\end{proof}

\begin{theorem}[No-fourth-ledger is the zeta core-completeness boundary]
\label{thm:unified-reality-no-fourth-ledger-core-completeness-boundary}
For the zeta zero source, the no-fourth-ledger row is the boundary form
of core completeness with respect to the normal displacement.
\end{theorem}

\begin{proof}
If $\delta(s)\#0$ occurred on a zero source row, then
\autoref{thm:unified-reality-split-mirror-zero-prime-defect} supplies a
finite prime-window defect.  This defect is a public witness of
off-fixed displacement unless the zero carrier excludes it.  The
no-fourth-ledger row of
\autoref{def:unified-reality-no-fourth-ledger-row} expresses precisely
that exclusion: $\delta$ cannot stand as an independent observable
outside the zero signature.  Therefore the no-fourth-ledger row is the
normal-displacement instance of core completeness.
\end{proof}

\begin{corollary}[Defect-zero incompatibility is the analytic form]
\label{cor:unified-reality-defect-zero-incompatibility-analytic-form}
The analytic form of the zeta core-completeness boundary is
defect-zero incompatibility.
\end{corollary}

\begin{proof}
The visible off-fixed displacement is read by finite prime-window
defects.  To prove that the zero source is core-complete with respect to
this displacement, one must show that such a defect cannot coexist with
the completed zeta zero ledger.  This is exactly the role of
\autoref{def:unified-reality-rh-defect-zero-incompatibility-route} and
\autoref{def:unified-reality-reality-induction-hard-point}.
\end{proof}

\begin{theorem}[Core commonality completion principle]
\label{thm:unified-reality-core-commonality-completion-principle}
The unified-reality programme supplies the common ontic basis for
licensed objects.  RH requires the stronger zeta-zero statement that the
zero source is complete and faithful for that basis.
\end{theorem}

\begin{proof}
The common basis is
\autoref{def:unified-reality-core-commonality-basis}.  Its object-level
transport is
\autoref{thm:unified-reality-object-induction-yields-ontic-section}.
The zeta zero completion target is
\autoref{def:unified-reality-zeta-zero-core-completeness-target}.  If
that target holds for the normal displacement, then
\autoref{thm:unified-reality-zeta-zero-core-completeness-forces-fixed-surface}
and
\autoref{cor:unified-reality-core-completeness-gives-rh-witness-shape}
give the RH section.  If it fails, the failure is an independent
observable in the zero carrier, which is exactly the fourth-ledger
boundary identified by
\autoref{thm:unified-reality-no-fourth-ledger-core-completeness-boundary}.
\end{proof}
