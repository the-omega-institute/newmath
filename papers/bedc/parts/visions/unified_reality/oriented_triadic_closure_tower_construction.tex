\section{Oriented Triadic Closure Tower Construction}
\label{sec:unified-reality-oriented-triadic-closure-tower-construction}

The triadic closure program can be sharpened from a route statement into
an object-construction problem.  The object to construct is an oriented
triadic closure tower over a completed zeta zero.  Its layers are not a
proof by themselves; they are the certificate rows which make the
conditional RH implication noncircular.

\begin{definition}[Oriented triadic closure tower object]
\label{def:unified-reality-zeta-oriented-triadic-closure-tower}
For a critical-strip zero coordinate
$$
s=\frac12+\delta+i\tau,
$$
a \emph{zeta oriented triadic closure tower} is a packet
$$
\mathfrak T_s
:=
(X_\infty,\mathcal R_\infty,J,x_s,A_n,h_a,\Omega_n,\mathcal B_s)
$$
with the following rows:
$$
X_\infty=\varprojlim X_n,
\qquad
\mathcal R_\infty:X_\infty\to X_\infty,
$$
$$
J^2=\operatorname{id},
\qquad
J\mathcal R_\infty J=\mathcal R_\infty^{-1},
$$
and a distinguished zero packet $x_s\in X_\infty$.  Each finite level
has active nodes $A_n$, positive heights $h_a>0$, and normal-defect
vector
$$
\Omega_n(s):=(\log|e^{-(s-1/2)h_a}|)_{a\in A_n}.
$$
The row $\mathcal B_s$ denotes the zero-compatible boundary or energy
bound attached to the same packet.
\end{definition}

\begin{definition}[Finite routing tower]
\label{def:unified-reality-finite-routing-tower}
A \emph{finite routing tower} is a system
$$
X_1\leftarrow X_2\leftarrow X_3\leftarrow\cdots
$$
with projections
$$
\pi_{n+1,n}:X_{n+1}\to X_n
$$
and bijective routing maps
$$
\mathcal R_n:X_n\to X_n
$$
such that
$$
\pi_{n+1,n}\circ\mathcal R_{n+1}
=
\mathcal R_n\circ\pi_{n+1,n}.
$$
The inverse limit carries the induced homeomorphism
$$
\mathcal R_\infty:X_\infty\to X_\infty.
$$
Thus the routing row is not a hidden oracle; it is the inverse limit of
finite auditable successor tables.
\end{definition}

\begin{definition}[Euler-log solenoid bridge]
\label{def:unified-reality-euler-log-solenoid-bridge}
For a finite or selected prime family $S$, an \emph{Euler-log solenoid
bridge} is a common phase carrier containing the Euler-log phase orbit
$$
\Lambda_S(\tau)
:=
\overline{\{(e^{-it\log p})_{p\in S}:t\in\mathbb R\}}
\subseteq\mathbb T^S
$$
and the solenoid phase carrier
$$
\Sigma_S=\widehat{\mathbb Z[S^{-1}]},
$$
with a displayed comparison row between the visible unit-circle phase
and the prime-register fibre
$$
\prod_{p\in S}\mathbb Z_p.
$$
The bridge is a source obligation: the multiplicative Euler-log phase
does not automatically equal the additive localization phase of the
solenoid carrier.
\end{definition}

\begin{definition}[Construction rows for the oriented tower]
\label{def:unified-reality-oriented-tower-construction-rows}
An oriented triadic closure tower is constructed by six rows:
$$
\mathsf{ActiveCapture},
\quad
\mathsf{EulerLogSolenoidBridge},
\quad
\mathsf{TriadicClosure},
$$
$$
\mathsf{DefectExtraction},
\quad
\mathsf{CofinalGrowth},
\quad
\mathsf{BoundaryStability}.
$$
They mean respectively:
the packet captures the completed zero by Rouch\'e or winding; the
Euler-log phase and solenoid prime-register fibre sit in one phase
container; prime charge, unit boundary, and time routing close modulo
legal prime-adic gauge; the normal defect is extracted as
$-\delta h_a$; the square weight
$$
W_n:=\sum_{a\in A_n}h_a^2
$$
is cofinal; and the same zero-compatible packet carries a bound
$$
|\Omega_n(s)|^2\le B_s<\infty
$$
or a stronger unit-boundary row.
\end{definition}

\begin{theorem}[Defect extraction for oriented towers]
\label{thm:unified-reality-oriented-tower-defect-extraction}
For $s=1/2+\delta+i\tau$ and every active node $a$,
$$
\log|e^{-(s-1/2)h_a}|=-\delta h_a.
$$
Consequently
$$
|\Omega_n(s)|^2
=
\delta^2\sum_{a\in A_n}h_a^2.
$$
\end{theorem}

\begin{proof}
Since $s-1/2=\delta+i\tau$,
$$
e^{-(s-1/2)h_a}
=
e^{-\delta h_a}e^{-i\tau h_a}.
$$
Taking the logarithm of the modulus gives $-\delta h_a$.  Squaring and
summing over $A_n$ gives the energy identity.
\end{proof}

\begin{theorem}[Oriented triadic closure towers are conditionally sufficient]
\label{thm:unified-reality-oriented-triadic-closure-towers-conditionally-sufficient}
If every critical-strip zeta zero carries a zeta oriented triadic closure
tower whose construction rows hold, then the RH fixed-half section
exists.
\end{theorem}

\begin{proof}
Let $z:\mathcal Z_{\zeta}$ have coordinate
$$
s=\frac12+\delta+i\tau.
$$
Let $\mathfrak T_s$ be the associated oriented tower.  By
\autoref{thm:unified-reality-oriented-tower-defect-extraction},
$$
|\Omega_n(s)|^2=\delta^2W_n.
$$
If $\delta\#0$, then cofinal growth $W_n\to\infty$ gives
$$
|\Omega_n(s)|^2\to\infty.
$$
This contradicts boundary stability
$$
|\Omega_n(s)|^2\le B_s<\infty.
$$
Hence $\neg(\delta\#0)$.  Located fixed-half collapse gives
$\delta=0$, so $\Re(s)=1/2$.  The fixed-half section follows from
\autoref{thm:unified-reality-rh-fixed-half-section-equivalence}.
\end{proof}

\begin{corollary}[Parity closure gives the same fixed-half conclusion]
\label{cor:unified-reality-oriented-tower-parity-closure-same-conclusion}
If the oriented tower also satisfies
$$
Jx_s\in\overline{\{\mathcal R_\infty^m x_s:m\in\mathbb Z\}},
$$
and its defect readout is routing-invariant and parity-odd, then
$\delta=0$.
\end{corollary}

\begin{proof}
This is exactly the abstract argument of
\autoref{thm:unified-reality-parity-closure-kills-odd-defect} applied to
the routing space $X_\infty$ and to the defect readout
$\Delta(x_s)=\delta$.
\end{proof}

\begin{definition}[Construction frontier]
\label{def:unified-reality-oriented-tower-construction-frontier}
The \emph{construction frontier} for the oriented triadic closure tower
is the pair
$$
\mathsf{ActiveCapture}
\quad\text{and}\quad
\mathsf{BoundaryStability}.
$$
Active capture asks for a canonical dominant endpoint packet associated
to each completed zero.  Boundary stability asks for a Lee--Yang,
unitary, solenoid-compactness, or equivalent bound for that same packet.
The Euler-log solenoid bridge and routing closure are additional source
obligations, but the analytic difficulty concentrates in these two rows.
\end{definition}

\begin{corollary}[The construction target is not the zero statement]
\label{cor:unified-reality-oriented-tower-target-not-zero-statement}
Constructing $\mathfrak T_s$ is stronger data than naming the fixed-half
surface and weaker than an unconditional RH theorem.
\end{corollary}

\begin{proof}
The tower includes source data: finite routing tables, phase containers,
prime-register fibres, active endpoint capture, and a stability bound.
None of these is implied by writing $\Re(s)=1/2$.  Conversely, the tower
is introduced as a conditional certificate: until the construction rows
are supplied for every completed zero, it remains a proof target rather
than an unconditional theorem.
\end{proof}
