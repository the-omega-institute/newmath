\section{Generation Invariant Exclusion Route}
\label{sec:unified-reality-generation-invariant-exclusion-route}

The RHRoute formal bridge may also be read as an invariant argument in
the style of a formal word system.  The target is not to guess a direct
analytic implication from a zero row to the critical line.  The target is
to display a generation system, identify an invariant subcarrier, prove
that every legal rule preserves that subcarrier, and then prove that
every completed zero enters the generated carrier through an independent
source theorem.

\begin{definition}[Invariant generation system]
\label{def:unified-reality-invariant-generation-system}
An \emph{invariant generation system} for a zero carrier is a tuple
$$
\mathfrak G
:=
\left(
\Sigma,
\mathcal W,
\mathcal A,
\mathcal R,
\mathcal T,
\mathcal I
\right)
$$
where $\Sigma$ is a signature, $\mathcal W$ is the ambient carrier of
legal packets, $\mathcal A\subseteq\mathcal W$ is the source family,
$\mathcal R$ is the family of generation rules, $\mathcal T$ is the
closed carrier generated from $\mathcal A$ by $\mathcal R$, and
$$
\mathcal I:\mathcal W\to M
$$
is an invariant readout.  The system is \emph{invariant-closed} when
every rule in $\mathcal R$ preserves $\mathcal I$ on its legal input
rows.
\end{definition}

\begin{theorem}[Invariant exclusion]
\label{thm:unified-reality-invariant-exclusion}
Let $\mathfrak G$ be invariant-closed.  If a target packet $w_\ast$ has
an invariant value not attained by any source packet in $\mathcal A$,
then
$$
w_\ast\notin\mathcal T.
$$
\end{theorem}

\begin{proof}
Every element of $\mathcal T$ is obtained from a source packet by a
finite chain of generation rules.  Since each rule preserves
$\mathcal I$, the invariant value of a generated packet agrees with the
value of some source packet.  The stated separation of invariant values
therefore prevents $w_\ast$ from lying in $\mathcal T$.
\end{proof}

\begin{definition}[Fixed-half invariant for RHRoute packets]
\label{def:unified-reality-fixed-half-invariant-rhroute-packets}
For an RHRoute formal bridge packet $P$ over a critical-strip coordinate
$s$, the \emph{fixed-half invariant} is the normal readout
$$
\mathcal I_{\mathrm{half}}(P)
:=
\delta(s)
:=
\Re(s)-\frac12.
$$
The fixed subcarrier is
$$
\mathcal W^{J}
:=
\left\{
P\in\mathcal W:
\mathcal I_{\mathrm{half}}(P)=0
\right\}.
$$
Equivalently, $\mathcal W^J$ is the subcarrier on which the mirror
readback gives $J$-fixedness and hence the critical-line row.
\end{definition}

\begin{definition}[RH-free generation discipline]
\label{def:unified-reality-rh-free-generation-discipline}
A generation discipline for RHRoute packets is \emph{RH-free} when its
rules are allowed to use finite prime-window data, mirror transport,
classifier transport, analytic gluing, compatible finite-window closure,
and route replay, but are not allowed to use $\mathsf{OnCritLine}(s)$ or
$\delta(s)=0$ as an input rule.  The purpose of the discipline is to
separate a genuine invariant-preservation proof from a hidden insertion
of the desired fixed-half output.
\end{definition}

\begin{definition}[Fixed-half closed generation]
\label{def:unified-reality-fixed-half-closed-generation}
An RH-free generation discipline is \emph{fixed-half closed} when
$$
P\in\mathcal W^J
\quad\Longrightarrow\quad
R(P)\in\mathcal W^J
$$
for every legal rule $R$ and every legal input packet $P$ of that rule.
For multi-input rules, the condition is that all inputs lie in
$\mathcal W^J$ and the output lies in $\mathcal W^J$.  This is the
formal invariant row corresponding to fixed-half closure.
\end{definition}

\begin{theorem}[Fixed-half generation excludes off-half outputs]
\label{thm:unified-reality-fixed-half-generation-excludes-off-half}
Let an RH-free generation discipline be fixed-half closed, and suppose
its source packets lie in $\mathcal W^J$.  Then every generated packet
has $\delta=0$.  In particular, no packet with
$$
s=\frac12+r+i\gamma,
\qquad
r\#0,
$$
is generated by that discipline.
\end{theorem}

\begin{proof}
Apply \autoref{thm:unified-reality-invariant-exclusion} with
$\mathcal I=\mathcal I_{\mathrm{half}}$.  The source packets have
invariant value $0$, and every legal rule preserves the fixed subcarrier.
Hence every generated packet has normal readout $0$.  A packet with
$r\#0$ has $\mathcal I_{\mathrm{half}}=r\#0$, so it cannot be in the
generated carrier.
\end{proof}

\begin{definition}[Generated-zero coverage row]
\label{def:unified-reality-generated-zero-coverage-row}
The \emph{generated-zero coverage row} for the completed zeta carrier is
$$
\mathsf{GeneratedZeroCoverage}_{\zeta}
:=
\prod_{s:\ComplexUp}
\left(
\mathsf{CritZero}(s)
\to
\sum_{P:\mathcal T_{\zeta}}
\mathsf{SameZeroReadout}(P,s)
\right),
$$
where $\mathcal T_{\zeta}$ is the generated RHRoute packet carrier.  The
row says that every completed critical-strip zero is covered by a
generated packet with the same zero readout.  It is the invariant-system
form of active capture and zero presentation completeness.
\end{definition}

\begin{theorem}[Generation coverage and fixed-half closure are sufficient]
\label{thm:unified-reality-generation-coverage-fixed-half-sufficient}
Assume:
$$
\mathsf{GeneratedZeroCoverage}_{\zeta},
$$
and assume the RHRoute generation discipline is RH-free and fixed-half
closed with fixed-half source packets.  Then the RH fixed-half section
exists.
\end{theorem}

\begin{proof}
Let $z:\mathcal Z_{\zeta}$ have coordinate $s$.  Generated-zero coverage
supplies a generated packet $P$ with the same zero readout as $s$.  By
\autoref{thm:unified-reality-fixed-half-generation-excludes-off-half},
$P$ has $\delta=0$.  Transporting this row along the same-zero readout
gives
$$
\delta(s)=0.
$$
The mirror-fixed algebra reads this as $J$-fixedness, hence as
$\mathsf{OnCritLine}(s)$.  Varying $z$ gives the fixed-half section.
\end{proof}

\begin{definition}[Energy form of the invariant]
\label{def:unified-reality-energy-form-generation-invariant}
Let a generated packet tower over $s=1/2+r+i\gamma$ have active nodes
$a\in A_n$ with heights $h_a>0$ and local channels
$$
U_a(s)
:=
e^{-(s-1/2)h_a}.
$$
Define
$$
d_a(s):=\log|U_a(s)|=-r h_a,
$$
$$
\Omega_n(s):=(d_a(s))_{a\in A_n},
\qquad
W_n:=\sum_{a\in A_n}h_a^2.
$$
The energy invariant is the square norm
$$
\left|\Omega_n(s)\right|^2
:=
\sum_{a\in A_n}d_a(s)^2.
$$
\end{definition}

\begin{lemma}[Energy reads the normal coordinate]
\label{lem:unified-reality-energy-reads-normal-coordinate-generation}
For every finite active window $A_n$,
$$
\left|\Omega_n(s)\right|^2=r^2W_n.
$$
\end{lemma}

\begin{proof}
By definition, $d_a(s)=-rh_a$.  Therefore
$$
\left|\Omega_n(s)\right|^2
=
\sum_{a\in A_n}(-rh_a)^2
=
r^2\sum_{a\in A_n}h_a^2
=
r^2W_n.
$$
\end{proof}

\begin{theorem}[Energy-bounded generation excludes off-half towers]
\label{thm:unified-reality-energy-bounded-generation-excludes-off-half}
Suppose a generated packet tower over $s=1/2+r+i\gamma$ satisfies
cofinal growth
$$
W_n\to\infty
$$
and a boundary-stability bound
$$
\left|\Omega_n(s)\right|^2\le B_s
$$
for a finite bound $B_s$ independent of $n$.  Then $r=0$.
\end{theorem}

\begin{proof}
If $r\#0$, then $r^2>0$.  By
\autoref{lem:unified-reality-energy-reads-normal-coordinate-generation},
$$
\left|\Omega_n(s)\right|^2=r^2W_n.
$$
The cofinal growth row forces the right side to be unbounded, while the
boundary-stability row gives a uniform finite bound.  This contradiction
excludes $r\#0$, and located fixed-half collapse gives $r=0$.
\end{proof}

\begin{corollary}[The invariant proof load is two bridge rows]
\label{cor:unified-reality-invariant-proof-load-two-bridge-rows}
The invariant route closes after two zeta-specific bridge rows are
supplied:
$$
\mathsf{ActiveCapture}_{\zeta}
\quad\text{and}\quad
\mathsf{BoundaryStability}_{\zeta}.
$$
\end{corollary}

\begin{proof}
Active capture supplies generated-zero coverage by constructing a
zero-relevant generated packet for each completed zero.  Boundary
stability supplies fixed-half closure, or in the tower reading a uniform
energy bound on the same zero-relevant packet family.  The invariant
argument itself is then formal: generated packets preserve the
fixed-half invariant, and off-half packets have a different invariant
value or unbounded positive energy.  Thus the remaining proof load is
source construction and independent boundary stability, not another
restatement of the critical-line surface.
\end{proof}
