\section{Fixed-Prime Split Resonance Tower}
\label{sec:unified-reality-fixed-prime-split-resonance-tower}

The solenoid source gap can be sharpened by separating two tasks that are
often conflated.  The zero-relevant resonance family need not begin as a
window over all rational primes.  A more economical source pattern fixes
one rational prime $q$ attached to the zero row, lets that prime split
through a tower of arithmetic carriers, and reads the resulting local
prime rows as the cofinal resonance support.  This mirrors the
unit-distance tower pattern: one split rational prime can generate many
local prime rows while a global discriminant bound keeps the tower
auditable.

\begin{definition}[Fixed split-prime tower]
\label{def:unified-reality-fixed-split-prime-tower}
Let $q:\PrimeUp$ be a displayed rational prime.  A \emph{fixed
split-prime tower} for $q$ is a chain of number-field rows
$$
K_0\subset K_1\subset K_2\subset\cdots
$$
with degree rows
$$
f_n:=[K_n:\mathbb Q],
$$
such that
$$
f_n\to\infty,
$$
and $q$ splits completely in each $K_n$:
$$
q\mathcal O_{K_n}
=
\mathfrak q_{n,1}\cdots \mathfrak q_{n,f_n}.
$$
The tower is \emph{globally controlled} when it carries a uniform root
discriminant bound
$$
\operatorname{rd}(K_n)
:=
|\operatorname{Disc}(K_n)|^{1/f_n}
\le C
$$
for a constant $C$ independent of $n$.
\end{definition}

\begin{definition}[$q$-split resonance nodes]
\label{def:unified-reality-q-split-resonance-nodes}
For a fixed split-prime tower and a depth row $j:\NatUp$, the displayed
local resonance nodes at level $n$ are
$$
a=(\mathfrak q_{n,i},j),
\qquad
1\le i\le f_n.
$$
Their Hamiltonian levels are
$$
h_a:=2^j\log N\mathfrak q_{n,i}.
$$
Since $q$ splits completely,
$$
N\mathfrak q_{n,i}=q,
$$
so the level readout is
$$
h_a=2^j\log q.
$$
A finite $q$-split resonance window at level $n$ is a finite set
$$
A_n\subset
\{\mathfrak q_{n,1},\ldots,\mathfrak q_{n,f_n}\}\times\NatUp.
$$
\end{definition}

\begin{definition}[Positive-density split support]
\label{def:unified-reality-positive-density-split-support}
A family of $q$-split resonance windows $(A_n)$ has
\emph{positive-density split support} when there are constants $c>0$ and
$j_0:\NatUp$ such that, for all sufficiently large $n$, at least
$c f_n$ nodes of the form
$$
(\mathfrak q_{n,i},j_0)
$$
belong to $A_n$.
\end{definition}

\begin{definition}[$q$-split normal energy]
\label{def:unified-reality-q-split-normal-energy}
For
$$
s=\frac12+\delta+i\tau,
$$
the local channel of a $q$-split node $a$ is
$$
U_a(s):=\exp(-(s-1/2)h_a)
=
e^{-\delta h_a}e^{-i\tau h_a}.
$$
The normal logarithmic defect is
$$
d_a(s):=\log |U_a(s)|=-\delta h_a,
$$
and the $q$-split window energy is
$$
\mathcal E_{A_n}^{(q)}(s)
:=
\sum_{a\in A_n}d_a(s)^2.
$$
\end{definition}

\begin{theorem}[One split prime detects off-half displacement]
\label{thm:unified-reality-one-split-prime-detects-off-half}
Let $(K_n)$ be a fixed split-prime tower for $q$, and let $(A_n)$ have
positive-density split support.  If
$$
s=\frac12+\delta+i\tau
$$
and $\delta\#0$, then
$$
\mathcal E_{A_n}^{(q)}(s)\to\infty
$$
along the tower.
\end{theorem}

\begin{proof}
Choose the depth $j_0$ and density constant $c$ from
\autoref{def:unified-reality-positive-density-split-support}.  For every
displayed node
$$
a=(\mathfrak q_{n,i},j_0),
$$
complete splitting gives
$$
h_a=2^{j_0}\log q.
$$
Hence
$$
d_a(s)^2
=
\delta^2(2^{j_0}\log q)^2.
$$
For all sufficiently large $n$, at least $c f_n$ such nodes belong to
$A_n$, so
$$
\mathcal E_{A_n}^{(q)}(s)
\ge
c f_n\delta^2(2^{j_0}\log q)^2.
$$
The tower has $f_n\to\infty$, and the coefficient is positive when
$\delta\#0$.  Therefore the displayed energy is unbounded and tends to
the far end.
\end{proof}

\begin{definition}[$q$-solenoid resonance compatibility]
\label{def:unified-reality-q-solenoid-resonance-compatibility}
Let $q$ be a fixed rational prime.  The associated one-prime solenoid is
$$
\Sigma_q:=\widehat{\mathbb Z[q^{-1}]}
$$
with projection
$$
\pi_q:\Sigma_q\twoheadrightarrow\mathbb T
$$
and hidden fibre
$$
K_q:=\ker(\pi_q)\cong\mathbb Z_q.
$$
A $q$-split resonance tower is \emph{$q$-solenoid-compatible} for $s$
when its phase data factors through $\pi_q$, its hidden registers are
carried by the $q$-adic fibre rows generated by the split local primes,
and it introduces no additional $\RealUp_{>0}$ scale coordinate.
\end{definition}

\begin{theorem}[$q$-solenoid compatibility has unit boundary]
\label{thm:unified-reality-q-solenoid-compatibility-unit-boundary}
If a stable zero-relevant $q$-split resonance tower is
$q$-solenoid-compatible for
$$
s=\frac12+\delta+i\tau,
$$
then $\delta=0$.
\end{theorem}

\begin{proof}
For every displayed node $a$ in the tower,
$$
|U_a(s)|=e^{-\delta h_a}
$$
is a real positive scale readout.  By $q$-solenoid compatibility, such a
readout must arise from the compact solenoid phase carrier rather than
from a noncompact scale coordinate.  The compact carrier has no
nontrivial continuous positive scale character by
\autoref{lem:unified-reality-compact-phase-no-positive-scale-character}.
Thus $|U_a(s)|=1$ for each zero-relevant displayed node.  Since
$h_a>0$, the equality $e^{-\delta h_a}=1$ gives $\delta=0$.
\end{proof}

\begin{definition}[Zeta fixed-prime tower Lee--Yang bound]
\label{def:unified-reality-zeta-fixed-prime-tower-lee-yang-bound}
The \emph{zeta fixed-prime tower Lee--Yang bound} is the source row
$$
\mathsf{ZetaFixedPrimeTowerLeeYang}_{\zeta}
$$
asserting that every critical-strip zero coordinate $s$ carries a packet
consisting of:
$$
q:\PrimeUp,
\qquad
\mathsf{FixedSplitPrimeTower}(q),
\qquad
(A_n),
$$
with positive-density split support, a stable local decomposition
$$
\Xi(s')=F_n(s')+T_n(s')
$$
on a zero island around $s$, a Rouch\'e boundary gap
$$
|T_n|<|F_n|,
$$
a $q$-solenoid-compatible phase lift, and a uniform normal-energy bound
$$
\sup_n\mathcal E_{A_n}^{(q)}(s)<\infty.
$$
\end{definition}

\begin{theorem}[The fixed-prime tower bound is conditionally sufficient]
\label{thm:unified-reality-fixed-prime-tower-bound-conditionally-sufficient}
If
$$
\mathsf{ZetaFixedPrimeTowerLeeYang}_{\zeta}
$$
holds, then the RH fixed-half section exists.
\end{theorem}

\begin{proof}
Let $z:\mathcal Z_{\zeta}$ have complex coordinate
$$
s=\frac12+\delta+i\tau.
$$
The fixed-prime tower Lee--Yang bound supplies a rational prime $q$, a
fixed split-prime tower, positive-density split support, and the bound
$$
\sup_n\mathcal E_{A_n}^{(q)}(s)<\infty.
$$
If $\delta\#0$, then
\autoref{thm:unified-reality-one-split-prime-detects-off-half} gives
$$
\mathcal E_{A_n}^{(q)}(s)\to\infty,
$$
contradicting the displayed bound.  Thus $\neg(\delta\#0)$, and located
fixed-half collapse gives $\delta=0$.  Hence $\Re(s)=1/2$, and the
fixed-half section follows from
\autoref{thm:unified-reality-rh-fixed-half-section-equivalence}.
\end{proof}

\begin{corollary}[Fixed-prime towers refine the source gap]
\label{cor:unified-reality-fixed-prime-towers-refine-source-gap}
The fixed-prime tower route refines the dominant solenoid source gap by
replacing an all-prime starting window with a single zero-relevant split
prime whose tower generates the needed cofinal support.
\end{corollary}

\begin{proof}
\autoref{thm:unified-reality-source-gap-remaining-theorem} identifies
the remaining theorem as the construction of a dominant solenoid source
from the completed zero row.  The fixed-prime tower route proposes a
more explicit source shape: first select the rational prime $q$, then
construct the split tower, the positive-density node family, the local
Rouch\'e gap, the $q$-solenoid phase lift, and the far-end energy bound.
The divergence theorem above shows why this is enough once the source is
constructed: one split rational prime already yields unbounded normal
energy off the fixed-half surface.
\end{proof}
