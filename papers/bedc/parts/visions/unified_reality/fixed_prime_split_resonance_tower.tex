\section{Fixed-Prime Split Resonance Tower}
\label{sec:unified-reality-fixed-prime-split-resonance-tower}

The solenoid source gap can be sharpened by separating two tasks that are
often conflated.  The zero-relevant resonance family need not begin as a
window over all rational primes.  A more economical source pattern fixes
one rational prime $q$ attached to the zero row, lets that prime split
through a tower of arithmetic carriers, and reads the resulting local
prime rows as the cofinal resonance support.  This mirrors the
unit-distance tower pattern: one split rational prime can generate many
local prime rows while a global discriminant bound keeps the tower
auditable.

\begin{definition}[Fixed split-prime tower]
\label{def:unified-reality-fixed-split-prime-tower}
Let $q:\PrimeUp$ be a displayed rational prime.  A \emph{fixed
split-prime tower} for $q$ is a chain of number-field rows
$$
K_0\subset K_1\subset K_2\subset\cdots
$$
with degree rows
$$
f_n:=[K_n:\mathbb Q],
$$
such that
$$
f_n\to\infty,
$$
and $q$ splits completely in each $K_n$:
$$
q\mathcal O_{K_n}
=
\mathfrak q_{n,1}\cdots \mathfrak q_{n,f_n}.
$$
The tower is \emph{globally controlled} when it carries a uniform root
discriminant bound
$$
\operatorname{rd}(K_n)
:=
|\operatorname{Disc}(K_n)|^{1/f_n}
\le C
$$
for a constant $C$ independent of $n$.
\end{definition}

\begin{definition}[$q$-split resonance nodes]
\label{def:unified-reality-q-split-resonance-nodes}
For a fixed split-prime tower and a depth row $j:\NatUp$, the displayed
local resonance nodes at level $n$ are
$$
a=(\mathfrak q_{n,i},j),
\qquad
1\le i\le f_n.
$$
Their Hamiltonian levels are
$$
h_a:=2^j\log N\mathfrak q_{n,i}.
$$
Since $q$ splits completely,
$$
N\mathfrak q_{n,i}=q,
$$
so the level readout is
$$
h_a=2^j\log q.
$$
A finite $q$-split resonance window at level $n$ is a finite set
$$
A_n\subset
\{\mathfrak q_{n,1},\ldots,\mathfrak q_{n,f_n}\}\times\NatUp.
$$
\end{definition}

\begin{definition}[Positive-density split support]
\label{def:unified-reality-positive-density-split-support}
A family of $q$-split resonance windows $(A_n)$ has
\emph{positive-density split support} when there are constants $c>0$ and
$j_0:\NatUp$ such that, for all sufficiently large $n$, at least
$c f_n$ nodes of the form
$$
(\mathfrak q_{n,i},j_0)
$$
belong to $A_n$.
\end{definition}

\begin{definition}[$q$-split normal energy]
\label{def:unified-reality-q-split-normal-energy}
For
$$
s=\frac12+\delta+i\tau,
$$
the local channel of a $q$-split node $a$ is
$$
U_a(s):=\exp(-(s-1/2)h_a)
=
e^{-\delta h_a}e^{-i\tau h_a}.
$$
The normal logarithmic defect is
$$
d_a(s):=\log |U_a(s)|=-\delta h_a,
$$
and the $q$-split window energy is
$$
\mathcal E_{A_n}^{(q)}(s)
:=
\sum_{a\in A_n}d_a(s)^2.
$$
\end{definition}

\begin{theorem}[One split prime detects off-half displacement]
\label{thm:unified-reality-one-split-prime-detects-off-half}
Let $(K_n)$ be a fixed split-prime tower for $q$, and let $(A_n)$ have
positive-density split support.  If
$$
s=\frac12+\delta+i\tau
$$
and $\delta\#0$, then
$$
\mathcal E_{A_n}^{(q)}(s)\to\infty
$$
along the tower.
\end{theorem}

\begin{proof}
Choose the depth $j_0$ and density constant $c$ from
\autoref{def:unified-reality-positive-density-split-support}.  For every
displayed node
$$
a=(\mathfrak q_{n,i},j_0),
$$
complete splitting gives
$$
h_a=2^{j_0}\log q.
$$
Hence
$$
d_a(s)^2
=
\delta^2(2^{j_0}\log q)^2.
$$
For all sufficiently large $n$, at least $c f_n$ such nodes belong to
$A_n$, so
$$
\mathcal E_{A_n}^{(q)}(s)
\ge
c f_n\delta^2(2^{j_0}\log q)^2.
$$
The tower has $f_n\to\infty$, and the coefficient is positive when
$\delta\#0$.  Therefore the displayed energy is unbounded and tends to
the far end.
\end{proof}

\begin{definition}[$q$-solenoid resonance compatibility]
\label{def:unified-reality-q-solenoid-resonance-compatibility}
Let $q$ be a fixed rational prime.  The associated one-prime solenoid is
$$
\Sigma_q:=\widehat{\mathbb Z[q^{-1}]}
$$
with projection
$$
\pi_q:\Sigma_q\twoheadrightarrow\mathbb T
$$
and hidden fibre
$$
K_q:=\ker(\pi_q)\cong\mathbb Z_q.
$$
A $q$-split resonance tower is \emph{$q$-solenoid-compatible} for $s$
when its phase data factors through $\pi_q$, its hidden registers are
carried by the $q$-adic fibre rows generated by the split local primes,
and it introduces no additional $\RealUp_{>0}$ scale coordinate.
\end{definition}

\begin{theorem}[$q$-solenoid compatibility has unit boundary]
\label{thm:unified-reality-q-solenoid-compatibility-unit-boundary}
If a stable zero-relevant $q$-split resonance tower is
$q$-solenoid-compatible for
$$
s=\frac12+\delta+i\tau,
$$
then $\delta=0$.
\end{theorem}

\begin{proof}
For every displayed node $a$ in the tower,
$$
|U_a(s)|=e^{-\delta h_a}
$$
is a real positive scale readout.  By $q$-solenoid compatibility, such a
readout must arise from the compact solenoid phase carrier rather than
from a noncompact scale coordinate.  The compact carrier has no
nontrivial continuous positive scale character by
\autoref{lem:unified-reality-compact-phase-no-positive-scale-character}.
Thus $|U_a(s)|=1$ for each zero-relevant displayed node.  Since
$h_a>0$, the equality $e^{-\delta h_a}=1$ gives $\delta=0$.
\end{proof}

\begin{definition}[Resonance counting lower row]
\label{def:unified-reality-resonance-counting-lower-row}
For a fixed split-prime tower and resonance windows $(A_n)$, the
\emph{resonance counting function} is
$$
N_{\rho}(n):=|A_n|.
$$
In the pure split-prime case at a fixed depth, the tower count is
$$
N_q(n):=f_n=[K_n:\mathbb Q],
$$
because complete splitting gives one displayed local prime row
$\mathfrak q_{n,i}$ for each embedding degree slot.  A resonance family
has a \emph{counting lower row} when there are constants $c>0$ and
$j_0:\NatUp$ such that
$$
N_{\rho}(n)\ge cN_q(n)
$$
along the displayed fixed-depth split nodes for all sufficiently large
$n$.
\end{definition}

\begin{theorem}[Counting lower rows force off-half energy growth]
\label{thm:unified-reality-counting-lower-rows-force-off-half-energy-growth}
Let a $q$-split resonance family have a counting lower row.  If
$$
s=\frac12+\delta+i\tau
$$
and $\delta\#0$, then there is a constant $c_q>0$ such that, for all
sufficiently large $n$,
$$
\mathcal E_{A_n}^{(q)}(s)\ge c_q\,N_q(n).
$$
In particular, if $N_q(n)\to\infty$, then
$$
\mathcal E_{A_n}^{(q)}(s)\to\infty.
$$
\end{theorem}

\begin{proof}
The counting lower row supplies a fixed depth $j_0$ and a positive
density constant $c$.  Each displayed split node at that depth has
$$
h_a=2^{j_0}\log q,
$$
so
$$
d_a(s)^2=\delta^2(2^{j_0}\log q)^2.
$$
Thus the contribution of the counted split nodes is at least
$$
cN_q(n)\delta^2(2^{j_0}\log q)^2.
$$
Set
$$
c_q:=c\delta^2(2^{j_0}\log q)^2.
$$
Since $q$ is prime and $\delta\#0$, this constant is positive.  The
displayed lower bound follows, and divergence follows when
$N_q(n)\to\infty$.
\end{proof}

\begin{definition}[Solenoid upper row]
\label{def:unified-reality-solenoid-upper-row}
For a zero-relevant $q$-split resonance family, a \emph{solenoid upper
row} is a packet consisting of:
$$
\mathsf{QSolenoidCompatible}(\rho,s),
\qquad
\mathsf{TailBound}(\rho),
\qquad
\mathsf{RoucheGap}(\rho),
$$
and a finite constant $B_s$ with
$$
\forall n,\qquad
\mathcal E_{A_n}^{(q)}(s)\le B_s.
$$
The compatibility row places the hidden registers in the compact
$q$-adic fibre of $\Sigma_q$.  The tail and Rouch\'e rows ensure that
the same resonance family is zero-relevant for the completed readout,
rather than merely visible in a finite projection.
\end{definition}

\begin{definition}[Zeta prime-count solenoid bound]
\label{def:unified-reality-zeta-prime-count-solenoid-bound}
The \emph{zeta prime-count solenoid bound} is the source row
$$
\mathsf{ZetaPrimeCountSolenoidBound}_{\zeta}
$$
asserting that every critical-strip zero coordinate $s$ carries a
zero-relevant fixed-prime resonance packet consisting of a rational prime
$q$, a fixed split-prime tower for $q$, a stable dominant resonance
family $(A_n)$, a counting lower row, and a solenoid upper row.
\end{definition}

\begin{theorem}[Prime-count lower and solenoid upper rows are conditionally sufficient]
\label{thm:unified-reality-prime-count-solenoid-bound-conditionally-sufficient}
If
$$
\mathsf{ZetaPrimeCountSolenoidBound}_{\zeta}
$$
holds, then the RH fixed-half section exists.
\end{theorem}

\begin{proof}
Let $z:\mathcal Z_{\zeta}$ have complex coordinate
$$
s=\frac12+\delta+i\tau.
$$
The prime-count solenoid bound supplies a fixed rational prime $q$, a
split tower, a zero-relevant resonance family, a counting lower row, and
a solenoid upper row.  The upper row gives a finite constant $B_s$ with
$$
\mathcal E_{A_n}^{(q)}(s)\le B_s
$$
for all $n$.  Suppose $\delta\#0$.  By
\autoref{thm:unified-reality-counting-lower-rows-force-off-half-energy-growth},
the counting lower row gives
$$
\mathcal E_{A_n}^{(q)}(s)\to\infty,
$$
because the split-prime tower has $N_q(n)\to\infty$.  This contradicts
the uniform upper bound.  Hence $\neg(\delta\#0)$, and located
fixed-half collapse gives $\delta=0$.  Therefore $\Re(s)=1/2$, and the
fixed-half section follows from
\autoref{thm:unified-reality-rh-fixed-half-section-equivalence}.
\end{proof}

\begin{corollary}[The prime-count solenoid row is a source theorem]
\label{cor:unified-reality-prime-count-solenoid-row-source-theorem}
The zeta prime-count solenoid bound is not a fixed-half alias row.
\end{corollary}

\begin{proof}
The row does not assert directly that a zero has $\delta=0$, zero normal
energy, or no normal hair.  It asserts the existence of a zero-relevant
source packet with two independent pieces of structure: a counting lower
row supplied by split-prime growth, and a solenoid upper row supplied by
compact fibre compatibility together with tail and Rouch\'e control.
Only after these two rows are combined does off-half displacement become
impossible.  Thus the row has the noncircular source shape isolated in
\autoref{thm:unified-reality-source-mechanisms-are-noncircular-content}.
\end{proof}

\begin{corollary}[Strong-triangle dominance explains the upper row]
\label{cor:unified-reality-strong-triangle-dominance-explains-upper-row}
The $p$-adic strong triangle inequality supplies the finite dominance
template behind the solenoid upper row.
\end{corollary}

\begin{proof}
For a non-archimedean absolute value, if one term is strictly larger
than all others, then the finite sum cannot vanish, as recorded in
\autoref{thm:unified-reality-nonarchimedean-dominance-prevents-finite-silence}.
Thus zero-relevant finite cancellation must occur at a boundary where
dominant levels tie rather than at an arbitrary projected cancellation.
This is the same proof-role shape as the quartic maximal-modulus
boundary in
\autoref{cor:unified-reality-quartic-carrier-finite-stability-template}.
The solenoid upper row imports this dominance discipline into the
fixed-prime tower setting: the compact $q$-adic fibre supplies legal
hidden registers, while the Rouch\'e and tail rows certify that the
dominant finite family controls the completed zero island.
\end{proof}

\begin{definition}[Zeta fixed-prime tower Lee--Yang bound]
\label{def:unified-reality-zeta-fixed-prime-tower-lee-yang-bound}
The \emph{zeta fixed-prime tower Lee--Yang bound} is the source row

\Xi(s')=F_n(s')+T_n(s')

\sup_n\mathcal E_{A_n}^{(q)}(s)<\infty.

\begin{theorem}[The fixed-prime tower bound is conditionally sufficient]
\label{thm:unified-reality-fixed-prime-tower-bound-conditionally-sufficient}
If
$$
\mathsf{ZetaFixedPrimeTowerLeeYang}_{\zeta}
$$

\begin{proof}

\begin{corollary}[Fixed-prime towers refine the source gap]
\label{cor:unified-reality-fixed-prime-towers-refine-source-gap}
The fixed-prime tower route refines the dominant solenoid source gap by
replacing an all-prime starting window with a single zero-relevant split
prime whose tower generates the needed cofinal support.
\end{corollary}

\begin{proof}
\autoref{thm:unified-reality-source-gap-remaining-theorem} identifies
the remaining theorem as the construction of a dominant solenoid source
from the completed zero row.  The fixed-prime tower route proposes a
more explicit source shape: first select the rational prime $q$, then
construct the split tower, the positive-density node family, the local
Rouch\'e gap, the $q$-solenoid phase lift, and the far-end energy bound.
The divergence theorem above shows why this is enough once the source is
constructed: one split rational prime already yields unbounded normal
energy off the fixed-half surface.
\end{proof}
