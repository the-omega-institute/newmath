\section{Far-End Energy Unity Socket}
\label{sec:unified-reality-far-end-energy-unity-socket}

The prime-adic parity energy window makes the non-point reading of the
far end explicit.  The far end is not a last node of
$\PrimeUp\times\NatUp$, not a final value of the tower, and not a hidden
register in which a visible normal defect can be cancelled.  It is the
cofinal socket through which all finite prime-adic parity windows are
read together.  Consequently an energy row vanishes at the far end only
when the finite-window energies become cofinally zero.  No additional
term at $\infty$ is available.

\begin{definition}[Prime-adic parity far-end socket]
\label{def:unified-reality-prime-adic-parity-far-end-socket}
The \emph{prime-adic parity far-end socket} is the cofinal boundary role
of the finite-window directed system
$$
A\subset\PrimeUp\times\NatUp,
\qquad
A\text{ finite and nonempty}.
$$
A cofinal energy condition on this socket is a row of the form
$$
\prod_{\varepsilon>0}
\sum_{A_0:\mathsf{PAdicParityWindow}_{\ne\varnothing}}
\prod_{A\supseteq A_0}
\left(
E_A^{\mathrm{pad}}(s)<\varepsilon
\right),
$$
where $A$ ranges over finite nonempty prime-adic parity windows.  The
symbol $\infty$ or $\mathsf{T}_{\mathrm{far}}$ names this cofinal
compatibility role; it does not name an element $A_\infty$.
\end{definition}

\begin{definition}[Far-end energy unity]
\label{def:unified-reality-far-end-energy-unity}
For a zeta readout $s$, define the \emph{far-end energy unity} row by
$$
\mathsf{FarEndEnergyUnity}_{\zeta}(s)
:=
\prod_{\varepsilon>0}
\sum_{A_0:\mathsf{PAdicParityWindow}_{\ne\varnothing}}
\prod_{A\supseteq A_0}
\left(
E_A^{\mathrm{pad}}(s)<\varepsilon
\right).
$$
It says that the normalized second-order prime-adic parity energy tends
to zero cofinally through finite windows.  It is not a statement that
there exists an infinite window where a correction term is evaluated.
\end{definition}

\begin{definition}[No hidden energy correction at the far end]
\label{def:unified-reality-no-hidden-energy-correction-far-end}
The \emph{no hidden energy correction} discipline is the refusal of an
undisplayed decomposition
$$
E_{\infty}^{\mathrm{pad}}(s)
=
E_A^{\mathrm{pad}}(s)+C_{\infty}(s)
$$
with $C_{\infty}(s)$ treated as a boundary term able to cancel
$E_A^{\mathrm{pad}}(s)$.  In the BEDC reading, the far end contributes
only the cofinal quantifier pattern of
\autoref{def:unified-reality-far-end-energy-unity}.  Any actual
correction row must appear as positive carrier data, as in the
archimedean completion row of
\autoref{def:unified-reality-archimedean-completion-row}; it cannot be
borrowed from the apophatic socket.
\end{definition}

\begin{theorem}[Far-end energy unity is fixed-half energy]
\label{thm:unified-reality-far-end-energy-unity-fixed-half-energy}
For
$$
s=\frac12+\delta+i\tau,
$$
the following rows are equivalent:
$$
\mathsf{FarEndEnergyUnity}_{\zeta}(s),
\qquad
\delta^2=0.
$$
Consequently, in the located real readout, far-end energy unity is
equivalent to
$$
\Re(s)=\frac12.
$$
\end{theorem}

\begin{proof}
By
\autoref{thm:unified-reality-prime-adic-parity-windows-read-one-normal-coordinate},
every finite nonempty prime-adic parity window satisfies
$$
E_A^{\mathrm{pad}}(s)=\delta^2.
$$
If $\delta^2=0$, then $E_A^{\mathrm{pad}}(s)=0$ for every finite
window, so the cofinal condition of
\autoref{def:unified-reality-far-end-energy-unity} holds.

Conversely, suppose
$\mathsf{FarEndEnergyUnity}_{\zeta}(s)$ holds.  If
$\delta^2\#0$, choose $\varepsilon$ below the positive row
$\delta^2$ in the located real readout.  The cofinal condition would
then require all sufficiently large finite windows to satisfy
$$
E_A^{\mathrm{pad}}(s)<\varepsilon.
$$
But every such window has
$$
E_A^{\mathrm{pad}}(s)=\delta^2,
$$
contradicting the choice of $\varepsilon$.  Hence $\delta^2=0$.
Located collapse gives $\delta=0$, and
\autoref{thm:unified-reality-critical-line-mirror-fixed-surface} reads
this as $\Re(s)=1/2$.
\end{proof}

\begin{definition}[Unnormalized parity far-end energy]
\label{def:unified-reality-unnormalized-parity-far-end-energy}
For a prime-adic parity window $A$, define the unnormalized parity energy
by
$$
\mathcal E_A^{\mathrm{pad}}(s)
:=
\sum_{(p,j)\in A}
\left(\log\left|U_{p,j}^{\mathrm{pad}}(s)\right|\right)^2.
$$
The corresponding far-end readout is the extended supremum over finite
windows:
$$
\mathcal E_{\mathrm{far}}^{\mathrm{pad}}(s)
:=
\sup_{A}
\mathcal E_A^{\mathrm{pad}}(s).
$$
\end{definition}

\begin{theorem}[Unnormalized parity energy diverges off half]
\label{thm:unified-reality-unnormalized-parity-energy-diverges-off-half}
If $s=1/2+\delta+i\tau$ and $\delta\#0$, then
$$
\mathcal E_{\mathrm{far}}^{\mathrm{pad}}(s)=\infty.
$$
\end{theorem}

\begin{proof}
For each node $(p,j)$,
$$
U_{p,j}^{\mathrm{pad}}(s)
=
p^{-2^j\delta}e^{-i2^j\tau\log p},
$$
so
$$
\log\left|U_{p,j}^{\mathrm{pad}}(s)\right|
=
-2^j\delta\log p.
$$
Therefore
$$
\mathcal E_A^{\mathrm{pad}}(s)
=
\delta^2
\sum_{(p,j)\in A}
\left(2^j\log p\right)^2.
$$
The directed carrier $\PrimeUp\times\NatUp$ has cofinally unbounded
finite node windows, and every displayed summand
$\left(2^j\log p\right)^2$ is positive.  If $\delta\#0$, multiplication
by $\delta^2$ preserves unboundedness in the extended readout.  Hence the
far-end supremum is infinite.
\end{proof}

\begin{definition}[Zero far-end energy unity target]
\label{def:unified-reality-zero-far-end-energy-unity-target}
The \emph{zero far-end energy unity target} is
$$
\mathsf{ZeroFarEndEnergyUnity}_{\zeta}
:=
\prod_{s:\ComplexUp}
\left(
\mathsf{ZetaZero}(s)
\times
\mathsf{InCritStrip}(s)
\to
\mathsf{FarEndEnergyUnity}_{\zeta}(s)
\right).
$$
It says that a completed critical-strip zero is a far-end
energy-unified boundary state.
\end{definition}

\begin{theorem}[Zero far-end energy unity is conditionally sufficient]
\label{thm:unified-reality-zero-far-end-energy-unity-conditionally-sufficient}
If $\mathsf{ZeroFarEndEnergyUnity}_{\zeta}$ holds, then the RH fixed-half
section exists.
\end{theorem}

\begin{proof}
Let $z:\mathcal Z_{\zeta}$ have complex coordinate $s$.  The zero row and
critical-strip row of $z$ give
$$
\mathsf{FarEndEnergyUnity}_{\zeta}(s)
$$
by
\autoref{def:unified-reality-zero-far-end-energy-unity-target}.  By
\autoref{thm:unified-reality-far-end-energy-unity-fixed-half-energy},
this implies $\Re(s)=1/2$.  Therefore $z$ has the fixed-half lift
required by
\autoref{def:unified-reality-rh-input-zero-source}, and the section
equivalence of
\autoref{thm:unified-reality-rh-fixed-half-section-equivalence} gives the
RH fixed-half section.
\end{proof}

\begin{corollary}[Energy unity is the socket form of no-hair]
\label{cor:unified-reality-energy-unity-socket-no-hair}
The zero far-end energy unity target is the cofinal finite-window form of
the zeta no-hair boundary.
\end{corollary}

\begin{proof}
\autoref{def:unified-reality-zeta-no-hair-boundary} states that a
critical-strip zero cannot carry global normal prime energy.  The present
socket formulation states the same obstruction through normalized
prime-adic parity energies: the finite-window energy family has the
constant value $\delta^2$, so cofinal vanishing is possible exactly on
the fixed-half surface.  The unnormalized form of
\autoref{thm:unified-reality-unnormalized-parity-energy-diverges-off-half}
recovers the infinite-energy reading of
\autoref{thm:unified-reality-global-prime-square-energy-diverges-off-half}
with binary depth included.
\end{proof}

\begin{corollary}[No point at infinity can absorb parity energy]
\label{cor:unified-reality-no-point-at-infinity-absorbs-parity-energy}
An off-half row cannot be made far-end energy-unified by adding a
correction at $\infty$.
\end{corollary}

\begin{proof}
By
\autoref{def:unified-reality-prime-adic-parity-far-end-socket}, the far
end is a cofinal socket for finite windows, not an element.  By
\autoref{thm:unified-reality-far-end-energy-unity-fixed-half-energy}, the
finite-window energy family is cofinally zero exactly when
$\delta^2=0$.  Thus an off-half row has stable positive normalized
energy in every finite window and unbounded unnormalized energy by
\autoref{thm:unified-reality-unnormalized-parity-energy-diverges-off-half}.
The prohibited hidden term of
\autoref{def:unified-reality-no-hidden-energy-correction-far-end} would
therefore be precisely a point-reification of the apophatic far end,
which is excluded by
\autoref{thm:unified-reality-completion-not-point-reification}.
\end{proof}
