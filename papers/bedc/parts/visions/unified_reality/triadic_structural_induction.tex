\section{Triadic Structural Induction}
\label{sec:unified-reality-triadic-structural-induction}

\autoref{sec:unified-reality-object-reality-induction-rh-route} organizes
licensed objects by their construction history, and
\autoref{sec:unified-reality-triadic-ontic-invariant-readback} reads the
RHRoute invariant through prime ledger, unit boundary, and recursive
readback. The present section records the common induction principle
behind those two uses. The induction is not over natural-number indices
and not over every named object in an ambient universe. It is structural
induction over objects generated by BEDC constructors and carrying
construction certificates.

\begin{definition}[Triadic base packet]
\label{def:unified-reality-triadic-base-packet}
The \emph{triadic base packet} is the triple
$$
\mathfrak B_{\triangle}
:=
\left(
\mathsf P_0,\mathsf U_0,\mathsf T_0
\right)
$$
with the following roles:
$$
\mathsf P_0
:=
\text{minimal distinction, finite source ledger, and prime register},
$$
$$
\mathsf U_0
:=
\text{unit phase boundary and orthogonal positive readout},
$$
and
$$
\mathsf T_0
:=
\text{append-only observation generation and finite witness order}.
$$
The packet is a role declaration, not an additional primitive carrier. Its
concrete rows are supplied by the kernel distinction and history
interfaces, the unit-circle and Pythagorean readouts of
\autoref{sec:unified-reality-pythagorean-common-carrier}, and the
prime-register routes used by the zeta-facing chapters.
\end{definition}

\begin{definition}[Triadic construction invariant]
\label{def:unified-reality-triadic-construction-invariant}
For a licensed BEDC object $X$, the predicate
$$
\mathcal I_{\triangle}(X)
$$
means that $X$ carries three compatible certificate rows
$$
\mathcal I_{\mathsf P}(X),
\qquad
\mathcal I_{\mathsf U}(X),
\qquad
\mathcal I_{\mathsf T}(X).
$$
The source row $\mathcal I_{\mathsf P}(X)$ records generated provenance,
irreducible decomposition, and ledger auditability. The unit row
$\mathcal I_{\mathsf U}(X)$ records unit-boundary closure, positive
orthogonal readout, or legal gauge. The time row
$\mathcal I_{\mathsf T}(X)$ records recursive generation, append-only
transport, and finite visibility of failures. The predicate is the
compatibility of the three rows, not their independent presence.
\end{definition}

\begin{principle}[Certificate-bounded domain of the induction]
\label{prin:unified-reality-certificate-bounded-domain-induction}
The structural induction governed by
$\mathcal I_{\triangle}$ ranges only over objects generated by the
declared BEDC construction grammar and admitted by a certificate. A
mathematical phrase not accompanied by such a construction certificate
does not inherit $\mathcal I_{\triangle}$ merely by being named.
\end{principle}

\subsection{Circle--Triangle--Register Grammar}
\label{subsec:unified-reality-circle-triangle-register-grammar}

\begin{definition}[BEDC null surface]
\label{def:unified-reality-bedc-null-surface}
The \emph{BEDC null surface}
$$
\varnothing_{\mathrm{BEDC}}
$$
is the pre-object state in which no distinction row, history row,
coordinate row, or ledger row has been displayed. It is not used as a
set-theoretic carrier. It is the boundary from which a certified object
can begin only when an apartness event is displayed:
$$
B_0\#B_1.
$$
The corresponding history rows are
$$
\Hist_0=(B_0),
\qquad
\Hist_1=(B_0,B_1).
$$
\end{definition}

\begin{definition}[Initial distinction split]
\label{def:unified-reality-initial-distinction-split}
An initial distinction split over the BEDC null surface is a packet
$$
\mathfrak D_1
:=
\bigl(B_0,B_1,\Hist_0,\Hist_1,\mathcal H_0,\mathcal H_1,E_1,\Lambda_1\bigr)
$$
such that the ledger row contains the earlier mark,
$$
\Hist_0\subseteq\Hist_1,
$$
and the Hilbert readout row splits as
$$
\mathcal H_1
=
\mathcal H_0\oplus E_1,
\qquad
E_1\perp\mathcal H_0.
$$
Thus ledger containment and Hilbert orthogonality are not competing
claims. The displayed norm readout is
$$
\|B_1\|^2
=
\|B_0\|^2+\|E_1\|^2
$$
at the resolution supplied by the packet, and $\Lambda_1$ records the
residual row $E_1$.
\end{definition}

\begin{definition}[Circle--triangle--register grammar]
\label{def:unified-reality-circle-triangle-register-grammar}
The \emph{circle--triangle--register grammar} is the least certified
construction grammar generated from an initial distinction split by the
following three constructor families.

The circle constructor attaches a unit phase row
$$
\mathsf{Circle}(\theta)=e^{i\theta}\in\mathbb T
$$
with
$$
|e^{i\theta}|=1.
$$
It supplies a unit-boundary row and no positive-real radial scale row.

The triangle-readout constructor attaches an orthogonal decomposition
row
$$
v=Pv+\eta,
\qquad
\eta\perp\operatorname{im}(P),
$$
with readout
$$
\|v\|^2
=
\|Pv\|^2+\|\eta\|^2.
$$
It supplies the positive residual-square row.

The register constructor sends every displayed residual $r$ that is used
by the object to an explicit ledger entry
$$
\mathsf{Register}(r)\in\Lambda.
$$
Finite arithmetic residuals may be addressed through the
Prime--Zeckendorf--Goedel register of
\autoref{def:rh-prime-zeckendorf-goedel-register}; the register
constructor does not allow a used residual to disappear by being
encodable.
\end{definition}

\begin{definition}[Circle--triangle--register generated class]
\label{def:unified-reality-circle-triangle-register-generated-class}
Let
$$
\mathcal C_{\circ\triangle\Lambda}
$$
be the finite generated class obtained from packets satisfying
\autoref{def:unified-reality-initial-distinction-split} by repeated
application of the circle, triangle-readout, and register constructors
of \autoref{def:unified-reality-circle-triangle-register-grammar}. A
limit point of this class belongs to the same generated domain only
after it supplies a limit closure row in the sense of
\autoref{def:unified-reality-triadic-limit-closure-row}.
\end{definition}

\begin{theorem}[Circle--triangle--register generation induction]
\label{thm:unified-reality-circle-triangle-register-generation-induction}
Assume the initial distinction split satisfies
$$
\mathcal I_{\triangle}(\mathfrak D_1),
$$
and assume that each displayed circle, triangle-readout, and register
constructor is triadic-preserving. Then every finitely generated object
$$
X\in\mathcal C_{\circ\triangle\Lambda}
$$
satisfies
$$
\mathcal I_{\triangle}(X).
$$
\end{theorem}

\begin{proof}
Proceed by structural induction on the construction tree of $X$. The
base case is the displayed initial distinction split
\autoref{def:unified-reality-initial-distinction-split}, which carries
the source ledger, unit-boundary refusal of radial scale, and generated
witness order required by $\mathcal I_{\triangle}$. For a constructor
node, the induction hypotheses give $\mathcal I_{\triangle}$ for all
inputs. The constructor hypothesis says that the applied circle,
triangle-readout, or register constructor is triadic-preserving in the
sense of \autoref{def:unified-reality-triadic-constructor-preservation}.
Therefore the output also satisfies $\mathcal I_{\triangle}$. This
covers every finite construction tree in
$\mathcal C_{\circ\triangle\Lambda}$.
\end{proof}

\begin{definition}[Triadic constructor preservation]
\label{def:unified-reality-triadic-constructor-preservation}
Let
$$
F(X_1,\ldots,X_n)
$$
be a legal constructor in the BEDC grammar. The constructor is
\emph{triadic-preserving} when it supplies a transport row
$$
\mathcal I_{\triangle}(X_1),\ldots,
\mathcal I_{\triangle}(X_n)
\Longrightarrow
\mathcal I_{\triangle}(F(X_1,\ldots,X_n)).
$$
The transport must display how source ledgers combine or restrict, how
unit-boundary or positivity rows are preserved, and how the resulting
object remains generated through finite or explicitly convergent witness
data.
\end{definition}

\begin{theorem}[Triadic structural induction]
\label{thm:unified-reality-triadic-structural-induction}
Let $\mathcal C_{\triangle}$ be the class generated from the triadic base
packet by a family of legal constructors. If the base packet satisfies
$\mathcal I_{\triangle}$ and every constructor in the family is
triadic-preserving, then every finitely generated
$X\in\mathcal C_{\triangle}$ satisfies
$$
\mathcal I_{\triangle}(X).
$$
\end{theorem}

\begin{proof}
Proceed by structural induction on the construction tree of $X$. For a
base leaf, the claim is the base-packet hypothesis. For a constructor
node
$$
X=F(X_1,\ldots,X_n),
$$
the induction hypotheses give $\mathcal I_{\triangle}(X_i)$ for each
input. Since $F$ is triadic-preserving,
\autoref{def:unified-reality-triadic-constructor-preservation} supplies
$\mathcal I_{\triangle}(X)$. This covers every finite construction tree
in the generated class.
\end{proof}

\begin{definition}[Limit closure row]
\label{def:unified-reality-triadic-limit-closure-row}
A generated tower
$$
X_0\to X_1\to X_2\to\cdots
$$
has a \emph{triadic limit closure row} when it supplies a limit object
$X_{\infty}$, compatible transports of
$\mathcal I_{\triangle}(X_m)$ along the tower, and a closedness
certificate
$$
\left(
\prod_m\mathcal I_{\triangle}(X_m)
\right)
\Longrightarrow
\mathcal I_{\triangle}(X_{\infty})
$$
in the declared topology or convergence discipline.
\end{definition}

\begin{theorem}[Triadic induction with certified limits]
\label{thm:unified-reality-triadic-induction-certified-limits}
The conclusion of
\autoref{thm:unified-reality-triadic-structural-induction} extends to
limit objects only when the construction supplies the limit closure row
of \autoref{def:unified-reality-triadic-limit-closure-row}.
\end{theorem}

\begin{proof}
The finite stages are covered by
\autoref{thm:unified-reality-triadic-structural-induction}. The displayed
limit closure row is precisely the additional rule transporting the
invariant from all finite stages to $X_{\infty}$. Without that row,
finite preservation gives no license to infer that an uninspected limit
preserves source ledger, unit positivity, or finite witness visibility.
\end{proof}

\section{Constructor Risk Surfaces}
\label{sec:unified-reality-triadic-constructor-risk-surfaces}

\begin{principle}[Product preservation]
\label{prin:unified-reality-triadic-product-preservation}
Products are triadic-preserving only when the product certificate
displays ledger pairing, positivity of the product or sum norm readout,
and a compatible interleaving of the two observation orders. A product
that forgets either provenance row or changes the witness order without
transport data is not a triadic-preserving constructor.
\end{principle}

\begin{principle}[Quotient and projection preservation]
\label{prin:unified-reality-triadic-quotient-preservation}
Quotients and projections are high-risk constructors because they hide
distinctions. They are triadic-preserving only when the hidden
distinction is retained as a legal fibre, gauge, or ledger row. The
solenoid projection
$$
\Sigma_S\to\mathbb T
$$
is legal in this sense because the hidden row is the prime-register
fibre of \autoref{def:rh-solenoid-phase-register}. A projection that
needs an unrecorded positive-real scale fibre is not a pure unit-phase
constructor.
\end{principle}

\begin{principle}[Limit preservation]
\label{prin:unified-reality-triadic-limit-preservation}
An infinite tower preserves the triadic invariant only by a closedness
row, not by the absence of a discovered counterexample. For positive
kernels, a typical legal row is finite-window convergence
$$
K_m\to K
$$
together with preservation of every finite Gram inequality in the
limit. This is a construction certificate; it is stronger than merely
having checked many finite levels.
\end{principle}

\begin{principle}[Analytic continuation preservation]
\label{prin:unified-reality-triadic-analytic-continuation-preservation}
Analytic continuation is triadic-preserving only when it displays local
source agreement, the connected continuation domain, branch choices, and
pole exclusion. Local Taylor agreement alone does not preserve
$\mathcal I_{\triangle}$, because a missing branch or pole row can hide
exactly the information that the source and unit certificates are meant
to audit.
\end{principle}

\section{Zeta Constructor Preservation Boundary}
\label{sec:unified-reality-zeta-constructor-preservation-boundary}

\begin{definition}[ZetaBEDCConstruction]
\label{def:unified-reality-zeta-bedc-construction}
The obligation
$$
\mathsf{ZetaBEDCConstruction}
$$
asserts that the completed zeta carrier, its zero-relevant observations,
and its Pick or moment positivity readouts are generated by legal BEDC
constructors preserving $\mathcal I_{\triangle}$. Its constructor
surface includes Euler product source rows, prime-power ledger rows,
Gamma and rational-boundary rows, completed normalization,
functional-equation transport, analytic continuation, logarithmic
derivative readout, Pick or Stieltjes transform, and finite or convergent
positive-kernel towers.
\end{definition}

\begin{definition}[ConstructorPreservationForXi]
\label{def:unified-reality-constructor-preservation-for-xi}
The obligation
$$
\mathsf{ConstructorPreservationForXi}
$$
is the preservation half of
$\mathsf{ZetaBEDCConstruction}$. It asks that each constructor used in
the completed zeta route be triadic-preserving in the sense of
\autoref{def:unified-reality-triadic-constructor-preservation}. In
particular, it must account for the negative sign in the prime side of
$\zeta'/\zeta$, the archimedean Gamma correction, completed
normalization, analytic continuation, and limit passage for finite
positive-kernel or moment towers.
\end{definition}

\begin{principle}[Zeta constructor preservation is the induction target]
\label{prin:unified-reality-zeta-constructor-preservation-target}
Once $\mathsf{ZetaBEDCConstruction}$ is established, the general
structural induction of
\autoref{thm:unified-reality-triadic-structural-induction} supplies the
triadic invariant on the completed zeta carrier. What remains
zeta-specific is not another restatement of RH, but the constructor
preservation proof for Euler product, Gamma completion, continuation,
logarithmic derivative, Pick transform, and finite tower convergence.
\end{principle}

\begin{theorem}[Zeta BEDC construction feeds the source-positive frontier]
\label{thm:unified-reality-zeta-bedc-construction-feeds-source-positive-frontier}
Assume $\mathsf{ZetaBEDCConstruction}$ and a source-built Pick
factorization for the completed zeta kernel. Then the source-positive RH
frontier has the positive witness required by
\autoref{thm:rh-source-positive-frontier-conditionally-sufficient}.
\end{theorem}

\begin{proof}
The zeta construction row places the completed zeta carrier and its
zero-relevant observations inside the certified BEDC construction
grammar. By
\autoref{thm:unified-reality-triadic-structural-induction}, its generated
objects carry $\mathcal I_{\triangle}$, including source ledger, legal
unit-boundary or positivity row, and finite or convergent witness
discipline. The assumed source-built Pick factorization is exactly the
positive-kernel witness named in
\autoref{def:rh-source-pick-hilbert-factorization}. Hence the hypotheses
needed for the source-positive frontier are supplied, subject only to
the standard zero-exclusion transport stated there.
\end{proof}

\begin{principle}[Status of triadic structural induction]
\label{prin:unified-reality-triadic-structural-induction-status}
Triadic structural induction explains why BEDC-generated certified
objects inherit source ledger, unit-boundary positivity, and generated
witness rows. It does not assert that an arbitrary named object is
generated, and it does not replace the analytic proof that completed
zeta's constructors preserve the invariant. For the RH route, the
remaining construction problem is
$\mathsf{ConstructorPreservationForXi}$ together with coverage of the
completed zero ledger by the generated carrier.
\end{principle}
