\section{BEDC--Langlands Stable Trace Formula Certificate}
\label{sec:unified-reality-bedc-langlands-stable-trace-formula-cert}

The stable trace formula is treated here as a BEDC closure certificate.
The claim is not that the analytic technology is avoided. The claim is
that, once the trace object is presented with its finite windows and
tail ledgers, a failure of the trace identity has only two possible
forms: it is visible on a fixed finite trace window, or it is a trace
tail row. A closed trace object can carry neither unresolved form.

\begin{definition}[Trace object]
\label{def:unified-reality-trace-object}
Let $G$ be a reductive group over a global field $F$, and let
$$
f\in C_c^\infty(G(\mathbb A_F))
$$
be an admitted test function. A trace object is the row
$$
\operatorname{TF}^{\uparrow}(G,F)
:=
(I_{\mathrm{spec}},I_{\mathrm{geom}},\Lambda_{\mathrm{TF}}),
$$
where
$$
I_{\mathrm{spec}}(f)
=
\sum_{\Pi}m(\Pi)\operatorname{tr}\Pi(f)
+I_{\mathrm{cont}}(f)
$$
and
$$
I_{\mathrm{geom}}(f)
=
\sum_{\gamma}a(\gamma)O_\gamma(f)
+I_{\mathrm{sing}}(f).
$$
The trace ledger $\Lambda_{\mathrm{TF}}$ records ramification,
normalization, orbital singularities, endoscopy, continuous spectrum,
boundary terms, transfer factors, and multiplicity rows.
\end{definition}

\begin{definition}[Trace three-properties rows]
\label{def:unified-reality-trace-three-properties-rows}
The time row of the trace object is generated by convolution,
Hecke-time operators, and trace-window inclusions
$$
\mathcal W_1\subset\mathcal W_2\subset\cdots.
$$
The unity row says that spectral and geometric expansions are two
readouts of the same trace pairing. The ledger row says that every
continuous, endoscopic, boundary, singular, ramified, multiplicity, or
normalization residual is recorded before any closed trace identity is
claimed.
\end{definition}

\begin{definition}[Trace residual]
\label{def:unified-reality-trace-residual}
The trace residual is the linear functional
$$
R_{\mathrm{TF}}(f)
:=
I_{\mathrm{spec}}(f)-I_{\mathrm{geom}}(f).
$$
A stable trace formula certificate is closed on a scope only if
$$
R_{\mathrm{TF}}(f)=0
$$
for every admitted test function on that scope, after all ledger rows
have been separated, cancelled, or discharged.
\end{definition}

\begin{definition}[Finite trace window]
\label{def:unified-reality-finite-trace-window}
A finite trace window is a tuple
$$
\mathcal W=(S,B,C),
$$
where $S$ is a finite place set, $B$ is a spectral complexity bound, and
$C$ is a geometric orbital complexity bound. Its admitted test space is
$$
\mathcal T_{\mathcal W}\subset C_c^\infty(G(\mathbb A_F)).
$$
The window carries a finite spectral list
$$
\mathsf{Spec}_{\mathcal W}=\{\Pi_1,\ldots,\Pi_M\}
$$
and a finite geometric list
$$
\mathsf{Geom}_{\mathcal W}=\{\gamma_1,\ldots,\gamma_N\},
$$
after all terms outside the window have been placed in the trace ledger.
\end{definition}

\begin{theorem}[Finite trace residual detection]
\label{thm:unified-reality-finite-trace-residual-detection}
If the restricted residual
$$
R_{\mathrm{TF}}\vert_{\mathcal T_{\mathcal W}}
$$
is nonzero on a fixed finite trace window, then there is
$$
f\in\mathcal T_{\mathcal W}
$$
with
$$
R_{\mathrm{TF}}(f)\ne0.
$$
\end{theorem}
\begin{proof}
The restriction of $R_{\mathrm{TF}}$ to $\mathcal T_{\mathcal W}$ is a
linear functional. A nonzero linear functional is nonzero on at least one
vector of its domain. That vector is the required test function.
\end{proof}

\begin{definition}[Finite trace matrix residual]
\label{def:unified-reality-finite-trace-matrix-residual}
On a finite trace window, the spectral and geometric sides have the
finite forms
$$
I_{\mathrm{spec}}(f)
=
\sum_{i=1}^{M}m_i\Theta_i(f),
\qquad
\Theta_i(f):=\operatorname{tr}\Pi_i(f),
$$
and
$$
I_{\mathrm{geom}}(f)
=
\sum_{j=1}^{N}a_jO_j(f).
$$
Thus the finite trace residual is
$$
R_{\mathrm{TF}}(f)
=
\sum_{i=1}^{M}m_i\Theta_i(f)
-
\sum_{j=1}^{N}a_jO_j(f).
$$
\end{definition}

\begin{definition}[Trace defect dichotomy rows]
\label{def:unified-reality-trace-defect-dichotomy-rows}
A finite trace defect is a window $\mathcal W$ such that
$$
R_{\mathrm{TF}}\vert_{\mathcal T_{\mathcal W}}\ne0.
$$
A trace tail defect is a residual that vanishes on every fixed finite
trace window but remains nonzero in a growing-window or limiting
operation. Tail rows include continuous spectrum, Eisenstein boundary,
endoscopic transfer, singular orbital integrals, ramification escape,
and measure or transfer-factor normalization.
\end{definition}

\begin{theorem}[Trace defect dichotomy]
\label{thm:unified-reality-trace-defect-dichotomy}
If
$$
R_{\mathrm{TF}}\ne0,
$$
then the trace object has either a finite trace defect or a trace tail
defect.
\end{theorem}
\begin{proof}
If there is a finite trace window $\mathcal W$ with nonzero restricted
residual, then the first alternative holds. Otherwise the residual
vanishes on every fixed finite trace window. Since the total residual is
assumed nonzero, it cannot be represented by any fixed finite window. It
therefore lies in a growing-window or limiting row: continuous spectrum,
Eisenstein boundary, endoscopy, singular orbits, ramification escape, or
normalization. This is precisely the trace tail alternative.
\end{proof}

\begin{definition}[Trace tail ledger certificate]
\label{def:unified-reality-trace-tail-ledger-certificate}
A trace tail ledger certificate says that every trace tail defect is
recorded in
$$
\operatorname{TraceTailLedger}
=
(\mathsf{ContinuousSpectrum},\mathsf{EisensteinBoundary},
\mathsf{Endoscopy},\mathsf{SingularOrbit},
\mathsf{Ramification},\mathsf{MeasureNormalization}).
$$
The row is closed only when the recorded defect is discharged or
separated from the claimed stable identity.
\end{definition}

\begin{theorem}[No hidden trace defect]
\label{thm:unified-reality-no-hidden-trace-defect}
If a trace object carries the trace three-properties rows, a complete
trace tail ledger certificate, and closedness against unresolved trace
ledger rows, then
$$
R_{\mathrm{TF}}=0.
$$
\end{theorem}
\begin{proof}
Assume $R_{\mathrm{TF}}\ne0$. By
\autoref{thm:unified-reality-trace-defect-dichotomy}, there is a finite
trace defect or a trace tail defect.

In the finite case,
\autoref{thm:unified-reality-finite-trace-residual-detection} supplies a
fixed test function $f$ with $R_{\mathrm{TF}}(f)\ne0$. This is a detected
trace residual. Ledger discipline requires it to be recorded; carrying
it unresolved violates closedness, and omitting it violates the ledger
row.

In the tail case, the trace tail ledger certificate requires the defect
to be recorded in $\operatorname{TraceTailLedger}$. Again, carrying it
unresolved violates closedness, while omitting it violates the ledger
row. Both alternatives contradict the closed trace object. Hence the
residual is zero.
\end{proof}

\begin{theorem}[Compact trace identity]
\label{thm:unified-reality-compact-trace-identity}
Let $X=\Gamma\backslash G$ be a compact quotient of a locally compact
group by a discrete cocompact subgroup, and let
$$
f\in C_c^\infty(G).
$$
For the right regular operator
$$
R(f)\varphi(x)=\int_G f(g)\varphi(xg)\,dg,
$$
one has
$$
\operatorname{Tr}R(f)
=
\sum_{\pi}m(\pi)\operatorname{tr}\pi(f)
=
\sum_{[\gamma]}
\operatorname{vol}(\Gamma_\gamma\backslash G_\gamma)O_\gamma(f).
$$
\end{theorem}
\begin{proof}
The kernel of $R(f)$ is
$$
K_f(x,y)=\sum_{\gamma\in\Gamma}f(x^{-1}\gamma y).
$$
Since $X$ is compact and $f$ is compactly supported, $R(f)$ is trace
class. The spectral decomposition of $L^2(X)$ gives
$$
\operatorname{Tr}R(f)
=
\sum_{\pi}m(\pi)\operatorname{tr}\pi(f).
$$
The same trace is the diagonal integral of the kernel:
$$
\operatorname{Tr}R(f)
=
\int_{\Gamma\backslash G}K_f(x,x)\,dx.
$$
Substituting the kernel, exchanging the finite-on-support sum with the
integral, and grouping by conjugacy classes gives
$$
\int_{\Gamma\backslash G}
\sum_{\gamma\in\Gamma}f(x^{-1}\gamma x)\,dx
=
\sum_{[\gamma]}
\operatorname{vol}(\Gamma_\gamma\backslash G_\gamma)O_\gamma(f).
$$
Thus the spectral and geometric readouts are equal to the same operator
trace.
\end{proof}

\begin{corollary}[Compact trace object closure]
\label{cor:unified-reality-compact-trace-object-closure}
In the compact quotient setting of
\autoref{thm:unified-reality-compact-trace-identity},
$$
R_{\mathrm{TF}}(f)=0
$$
for every admitted test function $f$.
\end{corollary}
\begin{proof}
Both the spectral and geometric sides are equal to
$\operatorname{Tr}R(f)$, so their difference is zero.
\end{proof}

\begin{definition}[Stable trace formula certificate]
\label{def:unified-reality-stable-trace-formula-cert-closure}
A stable trace formula certificate is the package consisting of:
\begin{itemize}
\item a compact trace identity base or a justified truncated kernel
identity;
\item a complete ledger for continuous spectrum, Eisenstein boundary,
endoscopy, singular orbital integrals, ramification, transfer factors,
and normalization;
\item stable orbital integral matching on the declared scope;
\item no unresolved trace residual after the ledger rows are discharged
or separated.
\end{itemize}
\end{definition}

\begin{theorem}[Stable trace formula as no-hidden trace defect]
\label{thm:unified-reality-stable-trace-formula-no-hidden-defect}
Within the BEDC trace object interface, the stable trace formula
certificate is equivalent to the compact trace identity base together
with no hidden trace defect and complete trace-tail ledger discipline on
the declared scope.
\end{theorem}
\begin{proof}
If the stable trace formula certificate holds, then every continuous,
Eisenstein, endoscopic, singular, ramified, transfer-factor, and
normalization residual has been recorded and discharged or separated.
Thus no hidden trace defect remains.

Conversely, assume the compact trace identity base, no hidden trace
defect, and complete trace-tail ledger discipline on the declared scope.
Any nonzero trace residual would be finite visible or tail visible by
\autoref{thm:unified-reality-trace-defect-dichotomy}. The finite case is
excluded by no hidden trace defect, and the tail case is excluded by the
complete ledger and closedness. Hence the residual vanishes on the
declared stable scope, which is the stable trace formula certificate.
\end{proof}

\begin{theorem}[Trace closure implies Langlands existence]
\label{thm:unified-reality-trace-closure-implies-langlands-existence}
Assume a parameter state certificate, finite Hecke projectors, the stable
trace formula certificate, cofinal automorphic tail closure, no-hidden
Langlands residual, and closedness against unresolved ledger rows. Then
the parameter packet $\phi^{\uparrow}$ has a global automorphic
realization on the certified unramified finite detector scope.
\end{theorem}
\begin{proof}
Fix a finite Hecke window $\mathcal W$. The parameter state certificate
gives a projector $f_{\phi,\mathcal W}$ with nonzero parameter-side
readout. The stable trace formula certificate transports this nonzero
readout to the spectral side. The finite Hecke projector then forces a
matching automorphic packet in the fixed window. Applying this to a
cofinal tower gives a cofinal realization sequence. By the automorphic
tail closure theorem, the sequence either globalizes to an automorphic
packet or enters the automorphic tail ledger. Closedness excludes an
unresolved tail row, so a global packet remains. No-hidden Langlands
residual then excludes any remaining Hecke--Frobenius mismatch.
\end{proof}

\begin{theorem}[BEDC--Langlands trace closure theorem]
\label{thm:unified-reality-bedc-langlands-trace-closure}
On a Langlands scope carrying unramified local Satake closure, finite
Hecke projectors, compact trace identity base, no hidden trace defect,
complete trace-tail ledger discipline, parameter state certificates,
cofinal automorphic tail closure, no-hidden Langlands residual, and
closedness, Langlands reciprocity holds on the certified unramified
finite detector scope.
\end{theorem}
\begin{proof}
The compact trace identity base and no-hidden trace defect give the
stable trace formula certificate by
\autoref{thm:unified-reality-stable-trace-formula-no-hidden-defect}.
Unramified local Satake closure licenses the finite detectors, and the
finite Hecke projector supplies fixed-window isolation. Parameter state
certificates provide nonzero parameter-side readouts. Therefore
\autoref{thm:unified-reality-trace-closure-implies-langlands-existence}
gives a global automorphic packet. No-hidden Langlands residual removes
all remaining mismatch, giving
$$
\lambda_{v,r}(\Pi_\phi)
=
\operatorname{tr}r(\phi(\operatorname{Frob}_v))
$$
for every certified unramified finite detector $(v,r)$.
\end{proof}

\begin{theorem}[Stable trace certificate boundary]
\label{thm:unified-reality-stable-trace-certificate-boundary}
The stable trace formula certificate described here is a BEDC closure
form of the trace formula. It identifies every possible failure as
finite visible or ledgered tail visible, and it proves the compact
kernel trace identity base. It does not replace the analytic work needed
to discharge continuous spectrum, Eisenstein series, endoscopy, singular
orbital integrals, transfer factors, ramification, or normalization in a
specific noncompact case.
\end{theorem}
\begin{proof}
The compact quotient proof uses only the ordinary trace-class kernel
identity. All noncompact and unstable terms are explicitly named as
ledger rows. The certificate is closed only when those rows are
discharged or separated on the declared scope; otherwise they remain
obligations rather than hidden equalities.
\end{proof}
