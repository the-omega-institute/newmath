\section{Harmonic Triadic Closure Map}
\label{sec:unified-reality-harmonic-triadic-closure-map}

The frequency, chord, solenoid, winding, Rouch\'e, Lee--Yang, and
dominance routes are not separate replacements for the RH-facing
boundary problem.  They occupy different rows of one harmonic triadic
closure map.  Prime logarithms supply the frequency ledger, phase
evolution supplies resonance closure, winding supplies integer local
degree, active capture ties a finite packet to an actual zero, and a
unit-boundary row forbids an additional real positive scale.

\begin{definition}[Harmonic triadic closure packet]
\label{def:unified-reality-harmonic-triadic-closure-packet}
For a critical-strip coordinate
$$
s=\frac12+\delta+i\tau,
$$
a \emph{harmonic triadic closure packet} is a zero-relevant packet
$$
\mathfrak H_s
:=
(A_n,h_a,U_a,\mathsf F,\mathsf W,\mathsf C,\mathsf B)
$$
with finite active windows $A_n$, positive Hamiltonian heights $h_a>0$,
local channels
$$
U_a(s):=e^{-(s-1/2)h_a},
$$
and four certificate rows:
$$
\mathsf F=\mathsf{FrequencyUniqueness},
\qquad
\mathsf W=\mathsf{IntegerWinding},
$$
$$
\mathsf C=\mathsf{ActiveCapture},
\qquad
\mathsf B=\mathsf{UnitBoundaryStability}.
$$
The rows mean respectively that the displayed frequencies have an
auditable prime ledger, the local zero count is an integer winding or
local degree, the packet captures the completed zero rather than a
spurious finite silence, and the same packet carries a unitary,
Lee--Yang, compact-solenoid, or equivalent normal-scale bound.
\end{definition}

\begin{definition}[Harmonic off-boundary defect]
\label{def:unified-reality-harmonic-off-boundary-defect}
For a finite active window $A_n$, the \emph{harmonic off-boundary defect}
is
$$
\Omega_{A_n}(s)
:=
\left(\log|U_a(s)|\right)_{a\in A_n}.
$$
The square weight of the same window is
$$
W_{A_n}:=\sum_{a\in A_n}h_a^2.
$$
\end{definition}

\begin{theorem}[Harmonic defect reads the normal coordinate]
\label{thm:unified-reality-harmonic-defect-reads-normal-coordinate}
For
$$
s=\frac12+\delta+i\tau,
$$
every harmonic active window satisfies
$$
\Omega_{A_n}(s)=(-\delta h_a)_{a\in A_n},
$$
and hence
$$
|\Omega_{A_n}(s)|^2=\delta^2W_{A_n}.
$$
\end{theorem}

\begin{proof}
Since
$$
e^{-(s-1/2)h_a}
=
e^{-\delta h_a}e^{-i\tau h_a},
$$
the logarithmic modulus of the local channel is $-\delta h_a$.  Squaring
and summing over $A_n$ gives the displayed energy identity.
\end{proof}

\begin{definition}[Three integer rows]
\label{def:unified-reality-harmonic-three-integer-rows}
The harmonic packet separates three integer rows:
$$
\mathsf{PrimeExponentInteger},
\qquad
\mathsf{LocalWindingInteger},
\qquad
\mathsf{HalfPeriodInteger}.
$$
The first is the unique prime-exponent ledger in an integer factorization.
The second is the zero multiplicity or local mapping degree read by the
argument principle.  The third is the half-period index in a two-arm
interference packet, where a local cancellation row has phase
$$
\arg(A/B)=(2k-1)\pi.
$$
These integer rows certify discrete closure data.  They do not by
themselves certify the unit boundary for the real positive factor
$e^{-\delta h_a}$.
\end{definition}

\begin{lemma}[Integer winding does not remove real scale]
\label{lem:unified-reality-integer-winding-does-not-remove-real-scale}
The existence of an integer winding row for a packet does not imply
$$
|U_a(s)|=1
$$
for its active channels.
\end{lemma}

\begin{proof}
The winding row is a statement about the local degree of the projected
complex map.  For a zero of order $m$, the argument of the projected
function changes by $2\pi m$ around a small boundary loop.  The local
channel, however, still factors as
$$
U_a(s)=e^{-\delta h_a}e^{-i\tau h_a}.
$$
The integer $m$ constrains the projected phase ledger.  It does not
force the real positive factor $e^{-\delta h_a}$ to equal $1$.  That
constraint belongs to the unit-boundary row.
\end{proof}

\begin{theorem}[Prime harmonic levels are ledger-separated]
\label{thm:unified-reality-prime-harmonic-levels-ledger-separated}
For prime-adic parity nodes
$$
a=(p,j),
\qquad
b=(q,k),
$$
with heights
$$
h_a=2^j\log p,
\qquad
h_b=2^k\log q,
$$
equality $h_a=h_b$ implies $a=b$.
\end{theorem}

\begin{proof}
This is exactly
\autoref{thm:unified-reality-prime-hamiltonian-levels-separated}.  The
case $p=q$ reduces to $2^j=2^k$.  The case $p\ne q$ would give a
nontrivial rational linear relation among $\log p$ and $\log q$,
contradicting the unique-factorization consequence recorded in
\autoref{thm:unified-reality-prime-logarithmic-directions-independent}.
\end{proof}

\begin{definition}[Proof roles in the harmonic map]
\label{def:unified-reality-harmonic-proof-roles}
The harmonic closure map assigns the standard analytic and arithmetic
tools to the following proof roles:
$$
\begin{gathered}
\text{Fourier and spectral decomposition}
\rightsquigarrow
\text{frequency expansion},\\
\text{unique factorization}
\rightsquigarrow
\text{prime ledger separation},\\
\text{Kronecker--Weyl and Dirichlet approximation}
\rightsquigarrow
\text{finite phase recurrence},\\
\text{pigeonhole}
\rightsquigarrow
\text{finite projection collision},\\
\text{argument principle}
\rightsquigarrow
\text{integer winding},\\
\text{Rouch\'e}
\rightsquigarrow
\text{active zero capture},\\
\text{Lee--Yang, BKW, and dominance}
\rightsquigarrow
\text{unit-boundary stability},\\
\text{Pontryagin duality and solenoids}
\rightsquigarrow

\end{gathered}
$$
No single row substitutes for the others.  In particular, winding and
Rouch\'e certify capture, while Lee--Yang or dominance supplies the
normal-scale obstruction.
\end{definition}

\begin{theorem}[Harmonic triadic closure is conditionally sufficient]
\label{thm:unified-reality-harmonic-triadic-closure-conditionally-sufficient}
Suppose every completed critical-strip zero carries a harmonic triadic
closure packet whose active windows satisfy cofinal growth
$$
W_{A_n}\to\infty
$$
and whose unit-boundary stability row gives a finite normal-energy bound
$$
|\Omega_{A_n}(s)|^2\le B_s
$$
for all displayed windows.  Then the RH fixed-half section exists.
\end{theorem}

\begin{proof}
Let $z:\mathcal Z_{\zeta}$ have coordinate
$$
s=\frac12+\delta+i\tau.
$$
The harmonic packet attached to $z$ satisfies
$$
|\Omega_{A_n}(s)|^2=\delta^2W_{A_n}
$$
by
\autoref{thm:unified-reality-harmonic-defect-reads-normal-coordinate}.
If $\delta\#0$, cofinal growth gives
$$
|\Omega_{A_n}(s)|^2\to\infty,
$$
contradicting the finite bound supplied by the unit-boundary stability
row.  Hence $\neg(\delta\#0)$.  Located fixed-half collapse gives
$\delta=0$, and the RH fixed-half section follows through
\autoref{thm:unified-reality-rh-fixed-half-section-equivalence}.
\end{proof}

\begin{corollary}[Capture rows and boundary rows have distinct jobs]
\label{cor:unified-reality-capture-boundary-distinct-jobs}
Argument-principle and Rouch\'e rows can certify that a local packet
counts the completed zero, but they do not by themselves exclude the
factor $e^{-\delta h_a}$.  Lee--Yang, BKW, compact solenoid, unitary, or
dominance rows are the candidates for excluding that real positive
scale.
\end{corollary}

\begin{proof}
\autoref{lem:unified-reality-integer-winding-does-not-remove-real-scale}
separates integer winding from the modulus condition.  Rouch\'e
stability preserves zero counts under a boundary inequality; it is an
active-capture row.  The contradiction in
\autoref{thm:unified-reality-harmonic-triadic-closure-conditionally-sufficient}
uses the separate finite-energy or unit-boundary bound.  Thus capture
and boundary stability are independent proof roles.
\end{proof}

\begin{corollary}[Solenoid fibres do not carry the real scale row]
\label{cor:unified-reality-harmonic-solenoid-fibres-no-real-scale-row}
A solenoid row can carry visible phase and hidden prime-register data,
but a nontrivial real positive scale
$$
e^{-\delta h_a}
$$
is not a solenoid fibre coordinate.
\end{corollary}

\begin{proof}
The solenoid row is compact and phase-like.  By
\autoref{lem:unified-reality-compact-phase-no-positive-scale-character},
a compact phase carrier has no nontrivial continuous character into
$\RealUp_{>0}$.  Therefore the factor $e^{-\delta h_a}$ must be killed
by the boundary row, not stored as hidden solenoid data.
\end{proof}
