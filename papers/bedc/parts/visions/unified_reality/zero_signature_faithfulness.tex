\section{Zero Signature Faithfulness and the Mirror Slot Principle}
\label{sec:unified-reality-zero-signature-faithfulness}

\autoref{sec:unified-reality-object-reality-induction-rh-route} isolates
the remaining RH-facing hard point as zeta-specific rigidity of the
zero fibre.  The same hard point can be expressed as a signature
faithfulness principle.  A critical-strip zero has a mirror companion
under the functional equation.  If the zero carrier has no hidden
observable beyond its ontic signature, then a zero and its mirror
companion occupy the same signature slot.  Faithfulness of that slot
forces them to be the same zero, hence to lie on the fixed surface.

\begin{definition}[Zero source and mirror companion]
\label{def:unified-reality-zero-source-mirror-companion}
The critical-strip zero source is
$$
\mathcal Z_{\zeta}
:=
\sum_{s:\ComplexUp}
\left(
\mathsf{ZetaZero}(s)\times\mathsf{InCritStrip}(s)
\right).
$$
For $s=\frac12+\delta+i\gamma$, the mirror is
$$
J(s):=1-\overline{s}
=
\frac12-\delta+i\gamma.
$$
The mirror companion row is the transport, supplied by the
functional-equation and non-escape ledgers, from a zero-source row at
$s$ to a zero-source row at $J(s)$.
\end{definition}

\begin{definition}[Zero ontic signature]
\label{def:unified-reality-zero-ontic-signature}
The \emph{zero ontic signature} is the packet
$$
\Sigma_Z(s)
:=
\bigl(
\mathsf{AtomType}_Z(s),
\mathsf{SpecTime}_Z(s),
\mathsf{MirrorSlot}_Z(s)
\bigr),
$$
where $\mathsf{AtomType}_Z(s)$ records the zero as an irreducible
spectral atom row, $\mathsf{SpecTime}_Z(s)$ records the spectral time
coordinate $\Im(s)$, and $\mathsf{MirrorSlot}_Z(s)$ records the
unordered mirror orbit generated by $s$ and $J(s)$.
\end{definition}

\begin{definition}[Mirror-even signature]
\label{def:unified-reality-mirror-even-zero-signature}
A zero signature is \emph{mirror-even} when the mirror companion has the
same signature:
$$
\Sigma_Z(J(s))=\Sigma_Z(s).
$$
For the signature of
\autoref{def:unified-reality-zero-ontic-signature}, this means that the
mirror companion has the same zero-atom type, the same imaginary
spectral time, and the same mirror slot.
\end{definition}

\begin{definition}[Zero signature faithfulness]
\label{def:unified-reality-zero-signature-faithfulness}
The zero source has \emph{signature faithfulness} when the signature slot
is faithful to zero identity:
$$
\mathsf{ZeroSignatureFaithful}_{\zeta}
:=
\prod_{s,t:\mathcal Z_{\zeta}}
\left(
\Sigma_Z(s)=\Sigma_Z(t)
\to
s=t
\right).
$$
Equivalently, the zero carrier has no hidden fourth observable that can
separate two zero-source rows after their atom type, spectral time, and
mirror slot agree.
\end{definition}

\begin{theorem}[Signature faithfulness implies zero-fixedness]
\label{thm:unified-reality-signature-faithfulness-implies-zero-fixedness}
Assume mirror closure for critical-strip zero rows, mirror-even zero
signature, and zero signature faithfulness.  Then the zero-fixedness
target holds:
$$
\prod_{s:\ComplexUp}
\left(
\mathsf{ZetaZero}(s)\times\mathsf{InCritStrip}(s)
\to
s=J(s)
\right).
$$
\end{theorem}

\begin{proof}
Let $s$ carry rows $\mathsf{ZetaZero}(s)$ and
$\mathsf{InCritStrip}(s)$.  Mirror closure supplies the companion
zero-source row at $J(s)$.  Mirror-evenness gives
$$
\Sigma_Z(J(s))=\Sigma_Z(s).
$$
Signature faithfulness applied to the two zero-source rows gives
$$
J(s)=s.
$$
This is exactly the zero-fixedness conclusion of
\autoref{def:unified-reality-zero-fixedness-target}.
\end{proof}

\begin{corollary}[Signature faithfulness gives the RH section]
\label{cor:unified-reality-signature-faithfulness-gives-rh-section}
Under the hypotheses of
\autoref{thm:unified-reality-signature-faithfulness-implies-zero-fixedness},
the BEDC RH fixed-half section is obtained.
\end{corollary}

\begin{proof}
By
\autoref{thm:unified-reality-signature-faithfulness-implies-zero-fixedness},
every critical-strip zero row satisfies $s=J(s)$.  By
\autoref{thm:unified-reality-critical-line-mirror-fixed-surface}, this
is equivalent to $\Re(s)=1/2$, hence to the critical-line output row.
The fixed-half section equivalence of
\autoref{thm:unified-reality-rh-fixed-half-section-equivalence} gives
the RH section target.
\end{proof}

\begin{definition}[Mirror-odd observable]
\label{def:unified-reality-mirror-odd-observable}
Let $R$ be an additive characteristic-zero readout carrier.  A zero
observable
$$
F:\mathcal Z_{\zeta}\to R
$$
is \emph{mirror-odd} when
$$
F(J(s))=-F(s).
$$
The normal displacement
$$
\delta(s):=\Re(s)-\frac12
$$
is the basic mirror-odd observable for the zeta mirror.
\end{definition}

\begin{definition}[Zero observable factorization]
\label{def:unified-reality-zero-observable-factorization}
The zero source has \emph{observable factorization through the zero
signature} when every zero observable $F$ in the displayed observable
class factors through $\Sigma_Z$:
$$
F(s)=f(\Sigma_Z(s))
$$
for some signature-level readout $f$.
\end{definition}

\begin{theorem}[Mirror-odd observables vanish under signature factorization]
\label{thm:unified-reality-mirror-odd-observable-vanishes}
Assume mirror-even zero signature and zero observable factorization
through that signature.  Then every mirror-odd zero observable in the
factorizing class is zero.
\end{theorem}

\begin{proof}
Let $F$ be mirror-odd and suppose
$$
F(s)=f(\Sigma_Z(s)).
$$
Mirror-evenness gives
$$
\Sigma_Z(J(s))=\Sigma_Z(s).
$$
Therefore
$$
F(J(s))
=
f(\Sigma_Z(J(s)))
=
f(\Sigma_Z(s))
=
F(s).
$$
Since $F$ is mirror-odd, also $F(J(s))=-F(s)$.  Hence
$$
F(s)=-F(s).
$$
In characteristic zero this forces $F(s)=0$.
\end{proof}

\begin{corollary}[Normal displacement vanishes under zero signature closure]
\label{cor:unified-reality-normal-displacement-vanishes-signature-closure}
If the normal displacement
$$
\delta(s)=\Re(s)-\frac12
$$
is a zero observable that factors through a mirror-even faithful zero
signature, then $\delta(s)=0$ for every critical-strip zero-source row.
\end{corollary}

\begin{proof}
By
\autoref{def:unified-reality-mirror-odd-observable}, $\delta$ is
mirror-odd.  Applying
\autoref{thm:unified-reality-mirror-odd-observable-vanishes} to
$F=\delta$ gives $\delta(s)=0$.  This is the fixed-half condition.
\end{proof}

\begin{definition}[No atomic mirror splitting]
\label{def:unified-reality-no-atomic-mirror-splitting}
The \emph{no atomic mirror splitting} row is
$$
\mathsf{NoAtomicMirrorSplitting}_{\zeta}
:=
\prod_s
\left(
\mathsf{ZeroAtom}(s)\times s\#J(s)
\to
\bot
\right).
$$
It says that a zeta zero atom is symmetry-absolute rather than a paired
atom separated by a hidden mirror-odd observable.
\end{definition}

\begin{theorem}[Signature faithfulness rules out atomic mirror splitting]
\label{thm:unified-reality-signature-faithfulness-rules-out-mirror-splitting}
Mirror-even zero signature and zero signature faithfulness rule out
atomic mirror splitting.
\end{theorem}

\begin{proof}
Suppose a zero atom at $s$ carried $s\#J(s)$.  Mirror closure places the
mirror companion in the same zero source.  Mirror-evenness gives equal
zero signatures for $s$ and $J(s)$, and signature faithfulness gives
$s=J(s)$.  This contradicts $s\#J(s)$.
\end{proof}

\begin{theorem}[Split mirror zeros create prime-window defects]
\label{thm:unified-reality-split-mirror-zero-prime-defect}
If a critical-strip zero source has $s\#J(s)$, then it carries the
normal defect $\delta(s)\#0$ and every nonempty finite prime window reads
a nonzero prime-window defect.
\end{theorem}

\begin{proof}
For the mirror $J(s)=1-\overline{s}$, the equality $s=J(s)$ is
equivalent to $\delta(s)=0$ by
\autoref{thm:unified-reality-critical-line-mirror-fixed-surface}.  Thus
apartness $s\#J(s)$ gives the off-fixed defect row
$\delta(s)\#0$ in the located comparison.  The finite-window defect
calculation is
\autoref{thm:unified-reality-prime-windows-observe-split-orbits}.
\end{proof}

\begin{corollary}[Signature faithfulness is a zero-relevance form of defect rigidity]
\label{cor:unified-reality-signature-faithfulness-zero-relevance-defect-rigidity}
Zero signature faithfulness turns prime-visible mirror splitting into a
zero-relevant contradiction.  In that sense it is a signature form of
ontic defect-zero incompatibility.
\end{corollary}

\begin{proof}
By
\autoref{thm:unified-reality-split-mirror-zero-prime-defect}, a split
mirror zero produces a prime-window defect.  By
\autoref{thm:unified-reality-signature-faithfulness-rules-out-mirror-splitting},
the same split row contradicts mirror-even signature faithfulness.  This
is the same logical role played by
\autoref{def:unified-reality-reality-induction-hard-point}: a visible
off-balance defect cannot coexist with a zero row in the completed zeta
carrier.
\end{proof}

\begin{theorem}[Mirror slot principle]
\label{thm:unified-reality-mirror-slot-principle}
The mirror slot principle is the implication
$$
\text{mirror closure}
+
\text{mirror-even zero signature}
+
\text{zero signature faithfulness}
\Rightarrow
\text{zero-fixedness}.
$$
It is a generalized slot-uniqueness principle for the zero carrier, not
a finite counting argument.
\end{theorem}

\begin{proof}
The implication is exactly
\autoref{thm:unified-reality-signature-faithfulness-implies-zero-fixedness}.
The word slot refers to the signature fibre of $\Sigma_Z$, not to a
finite pigeonhole set.  The principle says that one signature fibre
cannot contain two mirror-separated zero atoms when the zero signature is
faithful.
\end{proof}

\begin{corollary}[The signature hard point]
\label{cor:unified-reality-signature-hard-point}
The unclosed content of the signature route is
$$
\mathsf{ZeroSignatureFaithful}_{\zeta}
$$
or, equivalently for the normal displacement route, factorization of
zero observables through a mirror-even zero signature.
\end{corollary}

\begin{proof}
Mirror closure and the algebraic identity
$\delta(J(s))=-\delta(s)$ are structural rows already displayed by the
functional-equation mirror.  What remains is the assertion that the zero
signature has no hidden coordinate capable of separating $s$ from
$J(s)$.  That assertion is exactly
\autoref{def:unified-reality-zero-signature-faithfulness} or the
factorization row of
\autoref{def:unified-reality-zero-observable-factorization}.  Without
it, a mirror companion may be a paired zero rather than the same zero.
\end{proof}
