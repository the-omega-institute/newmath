\section{Phase, Transform, and Zeta Ladders}
\label{sec:unified-reality-phase-transform-zeta-ladder}

The unified-reality perspective also explains why trigonometry, the
circle, Fourier modes, Laplace kernels, elliptic curves, and zeta
functions appear in one mathematical corridor. They are not the same
object. They are compatible projections of carriers that preserve
quadratic norm, phase, exponential transport, spectral decomposition,
branch structure, and zero ledgers at different interfaces.

The guiding ladder is
$$
\text{triangle}
\to
\text{circle}
\to
\text{phase}
\to
\text{Fourier}
\to
\text{Laplace/Mellin}
\to
\text{zeta}.
$$
The spectral-curve branch is
$$
\text{recurrence}
\to
\text{spectral polynomial}
\to
\text{elliptic normalization}
\to
\text{branch/period/zero reality}.
$$
Both ladders meet at the same discipline: a visible readout may expose
part of a carrier, but the noncollapse ledger records which scale,
phase, branch, convergence, prime, or zero rows remain hidden.

\begin{definition}[Quadratic phase carrier]
\label{def:unified-reality-quadratic-phase-carrier}
The \emph{quadratic phase carrier} is the carrier row behind the
Pythagorean and unit-circle projections of
\autoref{sec:unified-reality-pythagorean-common-carrier}. It contains a
Pythagorean row
$$
a^2+b^2\sim c^2
$$
and, when $c\#0$, the normalized phase read
$$
x=\frac ac,
\qquad
y=\frac bc,
\qquad
x^2+y^2\sim1.
$$
The unit circle is the shape projection. The scale $c$ remains in the
fiber and is not recoverable from the circle point alone.
\end{definition}

\begin{theorem}[Triangle-to-circle projection preserves phase and loses scale]
\label{thm:unified-reality-triangle-circle-scale-fiber}
The passage from a nondegenerate Pythagorean carrier to a unit-circle
row preserves the normalized phase but forgets the scale fiber.
\end{theorem}

\begin{proof}
This is the normalization read of
\autoref{thm:unified-reality-pythagorean-normalization} and the
lossiness statement of
\autoref{cor:unified-reality-unit-circle-readout-not-complete}. If
$c\#0$, the projection sends $(a,b,c)$ to $(a/c,b/c)$. The equation
$a^2+b^2\sim c^2$ transports to $(a/c)^2+(b/c)^2\sim1$. The value of
$c$ is not a coordinate of the circle readout.
\end{proof}

\begin{definition}[Circle phase and Fourier reality]
\label{def:unified-reality-circle-fourier-reality}
The \emph{circle phase reality} reads the unit circle through a phase
parameter
$$
e^{i\theta}
=
\cos\theta+i\sin\theta.
$$
The \emph{Fourier reality} reads functions on the circle through the
character rows
$$
e^{in\theta},
\qquad
n\in\mathbb Z,
$$
and the spectral expansion schema
$$
f(\theta)\sim\sum_{n\in\mathbb Z}\widehat f_n e^{in\theta},
$$
whenever the relevant convergence, classifier, and coefficient ledgers
are displayed.
\end{definition}

\begin{principle}[Fourier is the frequency projection of circle phase]
\label{prin:unified-reality-fourier-frequency-projection}
Fourier analysis is not detached from the circle. It is the frequency
projection of the same phase carrier: trigonometric coordinates read
phase position, while character rows read spectral modes.
\end{principle}

\begin{definition}[Laplace extension of Fourier phase]
\label{def:unified-reality-laplace-extension-fourier-phase}
The \emph{Laplace extension} of Fourier phase replaces the pure phase
kernel
$$
e^{-i\omega t}
$$
by the damped exponential kernel
$$
e^{-st},
\qquad
s=\sigma+i\omega,
$$
so that
$$
e^{-st}=e^{-\sigma t}e^{-i\omega t}.
$$
The imaginary part is the Fourier frequency read. The real part is the
growth or damping read.
\end{definition}

\begin{theorem}[Laplace is Fourier with a real bias direction]
\label{thm:unified-reality-laplace-fourier-bias-direction}
The Laplace kernel extends the Fourier phase kernel by adding a real
amplitude-bias direction.
\end{theorem}

\begin{proof}
The displayed factorization
$$
e^{-(\sigma+i\omega)t}=e^{-\sigma t}e^{-i\omega t}
$$
separates the real damping row from the pure oscillatory phase row. The
kernel is therefore a two-coordinate reality: phase frequency and real
growth/decay.
\end{proof}

\begin{definition}[Arithmetic Laplace carrier]
\label{def:unified-reality-arithmetic-laplace-carrier}
For a coefficient sequence $(a_n)_{n\ge1}$, the \emph{arithmetic
Laplace carrier} is the atomic source
$$
\mu_a:=\sum_{n\ge1}a_n\delta_{\log n}
$$
with readout
$$
\mathcal L[\mu_a](s)
:=
\int e^{-st}\,d\mu_a(t)
=
\sum_{n\ge1}a_n n^{-s}
$$
on the displayed convergence window. For zeta, $a_n=1$ and
\autoref{prop:rh-zeta-self-similar-laplace-readout} gives the same
discrete Laplace read.
\end{definition}

\begin{theorem}[Zeta is an arithmetic Laplace readout]
\label{thm:unified-reality-zeta-arithmetic-laplace-readout}
On its convergence window, the Dirichlet face of $\zeta$ is the discrete
Laplace readout of the atomic source
$$
\mu_{\zeta}:=\sum_{n\ge1}\delta_{\log n}.
$$
\end{theorem}

\begin{proof}
For every $n\ge1$,
$$
n^{-s}=e^{-s\log n}.
$$
Therefore
$$
\sum_{n\ge1}n^{-s}
=
\int e^{-st}\,d\mu_{\zeta}(t)
$$
on any convergence window where the sum and readout are licensed. This
is the unified-reality restatement of
\autoref{prop:rh-zeta-self-similar-laplace-readout}.
\end{proof}

\begin{definition}[Multiplicative-prime face of zeta]
\label{def:unified-reality-zeta-multiplicative-prime-face}
The \emph{multiplicative-prime face} of zeta is the Euler-product read
$$
\zeta(s)\sim\prod_p(1-p^{-s})^{-1}
$$
on a displayed right-half-plane window. It records that the additive
Dirichlet/Laplace face is compatible with the multiplicative
factorization of integers into primes.
\end{definition}

\begin{principle}[Logarithm transports multiplication to Laplace addition]
\label{prin:unified-reality-logarithm-transport-multiplication-laplace}
The logarithm is the bridge from arithmetic multiplication to the
Laplace additive axis:
$$
\log(mn)=\log m+\log n.
$$
The zeta carrier consumes this bridge when it reads multiplicative prime
factorization as an additive atomic source on the $\log n$ axis.
\end{principle}

\begin{definition}[Prime-channel unitary balance]
\label{def:unified-reality-prime-channel-unitary-balance}
For $s=1/2+\delta+i\tau$ and a displayed prime $p$, the centered prime
channel is
$$
p^{1/2-s}=p^{-\delta}e^{-i\tau\log p}.
$$
It is unitary exactly when the amplitude read satisfies
$$
|p^{1/2-s}|=1.
$$
This is the unified-reality form of
\autoref{def:rh-prime-local-channels} and
\autoref{def:rh-prime-unitary-balance}.
\end{definition}

\begin{theorem}[The critical line is the prime-channel unitary surface]
\label{thm:unified-reality-critical-line-prime-unitary-surface}
For a displayed prime $p>1$,
$$
|p^{1/2-s}|=1
\quad\Longleftrightarrow\quad
\Re(s)=\frac12.
$$
\end{theorem}

\begin{proof}
Writing $s=1/2+\delta+i\tau$ gives
$$
|p^{1/2-s}|=p^{-\delta}.
$$
Since $p>1$, this amplitude is $1$ exactly when $\delta=0$, namely
$\Re(s)=1/2$. This is the prime-channel read of
\autoref{thm:rh-prime-unitary-balance-critical-line}.
\end{proof}

\begin{definition}[Elliptic spectral branch carrier]
\label{def:unified-reality-elliptic-spectral-branch-carrier}
The \emph{elliptic spectral branch carrier} is the geometric branch of
the quartic example of
\autoref{sec:unified-reality-quartic-spectral-common-carrier}. It reads
the recurrence and companion-matrix carrier through the normalized curve

E:\quad Y^2=4\lambda^3-4\lambda+1,
\qquad
y=\lambda^2-\frac12+\frac12Y.
$$
It carries branch, period, real-projection, dominance, and endpoint-zero
ledgers.
\end{definition}

\begin{principle}[Elliptic curves are branched phase carriers]
\label{prin:unified-reality-elliptic-branched-phase-carriers}
In this ladder, an elliptic curve is not an ordinary Euclidean ellipse.
It is a genus-one branched phase carrier: the circle gives a single
phase period, while the elliptic carrier exposes branch structure and
two-period geometry. Theta and modular expansions are therefore the
elliptic analogue of Fourier frequency reads on the circle whenever the
corresponding ledgers are displayed.
\end{principle}

\begin{theorem}[Two ladders meet in spectral-zero reality]
\label{thm:unified-reality-transform-and-spectral-ladders-meet}
The transform ladder and the elliptic spectral ladder meet at
zero-reality interfaces: transform kernels organize arithmetic and
analytic readouts, while spectral curves organize branch and dominance
readouts. Both require noncollapse ledgers before zero statements can be
transported.
\end{theorem}

\begin{proof}
On the transform side,
\autoref{thm:unified-reality-zeta-arithmetic-laplace-readout} and
\autoref{thm:unified-reality-critical-line-prime-unitary-surface} show
how the zeta carrier reads arithmetic data through Laplace and
prime-channel phase. On the spectral side,
\autoref{thm:unified-reality-quartic-spectral-common-carrier-principle}
shows how recurrence algebra, elliptic normalization, dominance, and
endpoint zeros are projections of one fixed quartic carrier. In both
cases, a visible readout is insufficient by itself: the zeta route needs
the zero-defect rigidity boundary of
\autoref{sec:unified-reality-rh-spectral-rigidity-route}, while the
quartic route needs dominance and fixed-window ledgers. Thus the ladders
meet as common-carrier disciplines for zero reality, not as a single
untyped identification.
\end{proof}

\begin{theorem}[Phase-transform-zeta ladder principle]
\label{thm:unified-reality-phase-transform-zeta-ladder-principle}
Triangles, circles, Fourier modes, Laplace kernels, zeta, and elliptic
spectral curves form a compatible carrier ladder:
$$
\text{geometric shape}
\leftrightarrow
\text{algebraic equation}
\leftrightarrow
\text{phase and transform kernel}
\leftrightarrow
\text{spectral decomposition}
\leftrightarrow
\text{zero reality}.
$$
The compatibility is ledgered projection and continuation, not
object-level collapse.
\end{theorem}

\begin{proof}
The triangle-to-circle part is
\autoref{thm:unified-reality-triangle-circle-scale-fiber}. The
circle-to-Fourier and Fourier-to-Laplace parts are
\autoref{prin:unified-reality-fourier-frequency-projection} and
\autoref{thm:unified-reality-laplace-fourier-bias-direction}. The
Laplace-to-zeta part is
\autoref{thm:unified-reality-zeta-arithmetic-laplace-readout}, with the
prime face governed by
\autoref{prin:unified-reality-logarithm-transport-multiplication-laplace}.
The elliptic spectral branch is
\autoref{def:unified-reality-elliptic-spectral-branch-carrier}. Finally,
\autoref{thm:unified-reality-transform-and-spectral-ladders-meet}
records the zero-reality interface and its noncollapse boundary.
\end{proof}
