\chapter{Ask--Sig--Pkg--Gap--NameCert Kernel Spine}
\label{ch:visions-ask-sig-pkg-gap-namecert-spine}
\origin{human}

The lower spine of \autoref{ch:visions-mark-hist-ext-cont-spine} supplies
raw marks, generated histories, one-step extension, and continuation.  The
finite kernel then reaches its certificate-facing boundary by adding asking,
signature formation, signature sameness, package tokens, gap ledgers, exact
globalization, and naming certificates:
$$
\Mark
\to
\Hist
\to
(\ExtR,\ContR)
\to
(\Ask,\Sig)
\to
\sameSig
\to
(\Pkg,\Gap)
\to
\NameCert{}.
$$
This chapter records the L3--L5 spine only.  It does not introduce mature
mathematical objects, completion objects, folds, phase spaces, zeta carriers,
or zero ledgers.  Those later interfaces may use this spine only through
displayed certificate fields.

\section{Asking and signatures}
\label{sec:ask-sig-spine-asking-signatures}

\begin{definition}[Probe names and bundles]
\label{def:ask-sig-spine-probe-bundles}
A probe name $\pi:\PN$ is an internal name that may be asked against a
generated history.  It is not an external function.  Probe bundles are the
closed carrier generated by
$$
\Bnil:\Bundle,
\qquad
\Bcons(\pi,\Pi):\Bundle
$$
for $\pi:\PN$ and $\Pi:\Bundle$.
\end{definition}

\begin{definition}[Ask event]
\label{def:ask-sig-spine-ask-event}
An ask event is a generated relation
$$
\Ask(\pi,h,m,\delta)
$$
where $\pi:\PN$, $h:\Hist$, $m:\Mark$, and $\delta$ is the event evidence.
It records that probe $\pi$ asked against source history $h$ emits mark $m$
with evidence $\delta$.  The evidence is a ledger item for the event, not a
numeric bias or an external truth value.
\end{definition}

\begin{definition}[Ask policy]
\label{def:ask-sig-spine-ask-policy}
For a bundle $\Pi$ and source domain $D:\Hist\to\mathsf{Prop}$, an asking
policy $\AskPol(\Pi,D)$ supplies the following internal obligations:
totality of ask events for admitted histories and bundle members,
determinacy of the emitted mark for the same probe and source history, and
respect for $\hsame$ on admitted source histories.
\end{definition}

\begin{definition}[Signature event]
\label{def:ask-sig-spine-signature-event}
The signature event
$$
\Sig(\Pi,h,s,\Delta)
$$
means that bundle $\Pi$ applied to source history $h$ generates answer
history $s$ with evidence ledger $\Delta$.  The empty bundle generates the
empty answer history:
$$
\Sig(\Bnil,h,\emp,\Delta_0).
$$
The cons rule is generated from a head ask event, a tail signature, and one
extension step:
$$
\Ask(\pi,h,m,\delta),
\qquad
\Sig(\Pi,h,s,\Delta),
\qquad
\ExtR(m,s,r)
$$
imply
$$
\Sig(\Bcons(\pi,\Pi),h,r,\delta\star\Delta).
$$
\end{definition}

\begin{theorem}[Signature inversion and determinacy]
\label{thm:ask-sig-spine-signature-inversion-determinacy}
The L3 signature relation satisfies the following finite-kernel facts.
\begin{enumerate}
  \item Empty-bundle signatures have answer history $\hsame$ to $\emp$.
  \item Cons-bundle signatures have answer history not $\hsame$ to $\emp$.
  \item If $\AskPol(\Pi,D)$, $h\in D$, and
  $$
  \Sig(\Pi,h,s,\Delta),
  \qquad
  \Sig(\Pi,h,t,\Theta),
  $$
  then $\hsame(s,t)$.
\end{enumerate}
\end{theorem}

\begin{proof}
The first clause is inversion of the empty-signature rule.  The second
clause inverts the cons-signature rule; its last step is an extension, whose
result has $\Ezero$ or $\Eone$ head and therefore cannot be $\hsame$ to
$\emp$ by history no-confusion.  The determinacy clause is induction on
$\Pi$.  The empty case reduces to the first clause and $\hsame$ equivalence.
In the cons case, invert both signatures.  Ask-policy determinacy gives
$\msame$ of the two head marks, the induction hypothesis gives $\hsame$ of
the two tail answer histories, and extension respects internal sameness.
\end{proof}

\begin{definition}[Signature sameness]
\label{def:ask-sig-spine-signature-sameness}
For a bundle $\Pi$, two histories are signature-same when their generated
answer histories are $\hsame$:
$$
\sameSig_{\Pi}(h,k)
\quad:\Longleftrightarrow\quad
\exists s,t,\Delta,\Theta.\;
\Sig(\Pi,h,s,\Delta)
\wedge
\Sig(\Pi,k,t,\Theta)
\wedge
\hsame(s,t).
$$
\end{definition}

\begin{theorem}[Signature sameness is the L3 classifier]
\label{thm:ask-sig-spine-samesig-classifier}
Assume $\AskPol(\Pi,D)$.  On admitted histories, $\sameSig_{\Pi}$ is
reflexive, symmetric, and transitive; the transitive step through a middle
history $k$ uses the hypothesis $k\in D$.
\end{theorem}

\begin{proof}
Reflexivity uses ask-policy totality to generate one signature for the
admitted history and then uses $\hsame$ reflexivity.  Symmetry swaps the two
signature witnesses and uses $\hsame$ symmetry.  For transitivity, unpack the
two signature-sameness witnesses.  They contain two generated signatures from
the admitted middle history.  Signature determinacy identifies the two middle
answer histories by $\hsame$, and $\hsame$ transitivity connects the left
answer history to the right answer history.
\end{proof}

\section{Packages, gaps, and exact globalization}
\label{sec:ask-sig-spine-packages-gaps}

\begin{definition}[Package token and package policy]
\label{def:ask-sig-spine-package-policy}
The package relation
$$
\Pkg(\Pi,s,p)
$$
records that answer history $s$ under bundle $\Pi$ is represented by visible
package token $p$.  A package policy $\PkgPol(\Pi)$ supplies token existence,
extensionality from $\hsame(s,t)$ to $\psame(p,q)$ for tokens introduced by
$s$ and $t$, source reflection of $\psame(p,q)$ back in generated answer histories,
and visibility of the token interface without exposing the full source
history as a public object.
\end{definition}

\begin{theorem}[Packages classify signatures]
\label{thm:ask-sig-spine-packages-classify-signatures}
Assume $\PkgPol(\Pi)$.  If
$$
\Pkg(\Pi,s,p),
\qquad
\Pkg(\Pi,t,q),
$$
then $\hsame(s,t)$ implies $\psame(p,q)$.  Conversely, $\psame(p,q)$ is
backed by package witnesses for answer histories internally related by
$\hsame$.
\end{theorem}

\begin{proof}
The forward direction is the extensionality field of the package policy.
The converse is the source-reflection field.  Therefore package-token behavior is a
licensed theorem of the package policy, not a primitive quotient shortcut.
\end{proof}

\begin{definition}[Gap membership and gap policy]
\label{def:ask-sig-spine-gap-policy}
The gap membership relation
$$
\Gap(\Pi,D,p,h)
$$
records that admitted source history $h$ lies in the ledger behind visible
package token $p$.  A gap policy $\GapPol(\Pi,D)$ supplies coverage of
admitted histories, separation of two package tokens that cover the same
source history, and signature determination: every gap membership is backed
by a generated signature and a package-token witness.
\end{definition}

\begin{theorem}[Gap coverage and separation]
\label{thm:ask-sig-spine-gap-coverage-separation}
Assume $\AskPol(\Pi,D)$, $\PkgPol(\Pi)$, and $\GapPol(\Pi,D)$.  If $h\in D$,
then there is a package token $p$ such that $\Gap(\Pi,D,p,h)$.  If
$$
\Gap(\Pi,D,p,h),
\qquad
\Gap(\Pi,D,q,h),
$$
then $\psame(p,q)$.
\end{theorem}

\begin{proof}
Coverage is the coverage field of the gap policy, with ask totality and
package existence supplying the generated signature and token when the
policy is unfolded.  For separation, unfold the two gap memberships.  They
produce two signatures from the same source history and two package-token
witnesses.  Signature determinacy gives $\hsame$ of the two answer histories,
and package extensionality gives $\psame(p,q)$.
\end{proof}

\begin{principle}[Compression carries memory]
\label{prin:ask-sig-spine-compression-memory}
Packaging, hiding, projecting, sealing, or otherwise compressing generated
histories into a visible token is admissible only with a gap ledger.  Without
the ledger, the visible token has no internal record of the source histories
compressed into it.
\end{principle}

\concretizedIn{ch:concrete-instances-ask-sig-gap-route-namecert}{AskSigGapRouteUp realizes the Ask--Sig--Pkg--Gap exact-globalization spine as finite NameCert route data.}

\begin{definition}[Generated signature sameness witness]
\label{def:ask-sig-spine-generated-samesig}
The witness object $\GeneratedSameSig(\Pi,h,k)$ consists of answer histories
$s,t$, evidence ledgers $\Delta,\Theta$, generated signatures
$$
\Sig(\Pi,h,s,\Delta),
\qquad
\Sig(\Pi,k,t,\Theta),
$$
and the internal sameness row $\hsame(s,t)$.  Its inhabitedness is the
checker-facing form of the signature-classification statement.
\end{definition}

\begin{definition}[Exact globalize base]
\label{def:ask-sig-spine-exact-globalize-base}
An exact finite globalization base $\ExactGlobalizeBase(\Pi,D)$ consists of
coverage, base soundness, and base completeness:
$$
h\in D
\Longrightarrow
\exists p.\;\Gap(\Pi,D,p,h),
$$
$$
\Gap(\Pi,D,p,h)
\wedge
\Gap(\Pi,D,q,k)
\wedge
\GeneratedSameSig(\Pi,h,k)
\Longrightarrow
\psame^{\mathrm{base}}_{\Pi}(p,q),
$$
and
$$
\Gap(\Pi,D,p,h)
\wedge
\Gap(\Pi,D,q,k)
\wedge
\psame^{\mathrm{base}}_{\Pi}(p,q)
\Longrightarrow
\mathsf{Nonempty}(\GeneratedSameSig(\Pi,h,k)).
$$
\end{definition}

\begin{theorem}[Exact globalize base classification]
\label{thm:ask-sig-spine-exact-globalize-classification}
If $\ExactGlobalizeBase(\Pi,D)$ and the two displayed package tokens carry
gap witnesses
$$
\Gap(\Pi,D,p,h),
\qquad
\Gap(\Pi,D,q,k),
$$
then
$$
\psame^{\mathrm{base}}_{\Pi}(p,q)
\Longleftrightarrow
\mathsf{Nonempty}(\GeneratedSameSig(\Pi,h,k)).
$$
\end{theorem}

\begin{proof}
The forward implication is base completeness.  The reverse implication
unpacks the inhabited $\GeneratedSameSig(\Pi,h,k)$ witness and applies base
soundness.  Coverage remains part of the exact globalization structure
because every admitted history must have a visible package ledger, even
though this local equivalence only needs the two displayed gap witnesses.
\end{proof}

\section{Name certification}
\label{sec:ask-sig-spine-namecert}

\begin{definition}[NameCert fields]
\label{def:ask-sig-spine-namecert-fields}
For a proposed derived interface $N$, the paper naming certificate is the
five-field packet
$$
\NameCert(N)
=
(\SourceSpec_N,\PatternSpec_N,\ClassifierSpec_N,\StabCert_N,\LedgerPolicy_N).
$$
The source field records the generated material from which $N$ is licensed.
The pattern field records the stable behavior observed in that source.  The
classifier field records how source material is grouped, normally through a
bundle, asking policy, package policy, and gap policy at the finite-kernel
level.  The stability field records closure, reuse, descent, composition, or
seal behavior.  The ledger policy records what the name hides and how that
hidden material remains tracked.
\end{definition}

\begin{theorem}[Derived interfaces require NameCert]
\label{thm:ask-sig-spine-derived-interface-admission}
A derived interface $N$ is admitted only through $\NameCert(N)$.
\end{theorem}

\begin{proof}
The finite kernel has no mature-object primitive beyond its generated
judgments and certificate records.  A derived name therefore cannot enter by
host familiarity, notation, quotient collapse, or semantic intention.  It
enters only when source, pattern, classifier, stability, and ledger fields
are all supplied.
\end{proof}

\begin{definition}[Certified existence]
\label{def:ask-sig-spine-certified-existence}
For a proposed object or interface name $X$, BEDC-certified existence is
the judgment
$$
\mathsf{Exists}^{\uparrow}(X)
:=
\sum_{N:\NameCert(X)}
\mathsf{Admits}(N).
$$
The admission row records that the source, pattern, classifier,
stability, and ledger fields of $N$ are available to downstream
consumers. Thus existence is a construction certificate for the named
interface, not prior possession of an untyped object.
\end{definition}

\begin{principle}[Existence is certificate-bearing admission]
\label{prin:ask-sig-spine-existence-certificate-bearing}
In BEDC, a mathematical name is usable only as far as its construction
certificate exposes the relevant source, classifier, boundary, transport,
and ledger rows. A zeta carrier, zero packet, Hilbert-style surface,
solenoid phase carrier, Pick kernel, completion, quotient, or limit may be
used only after the corresponding certificate has named what is generated,
what is hidden, and how the hidden material remains auditable.
\end{principle}

\begin{theorem}[Certified existence is the object-layer admission rule]
\label{thm:ask-sig-spine-certified-existence-admission}
Every object-layer use of a derived interface factors through
$\mathsf{Exists}^{\uparrow}(X)$.
\end{theorem}

\begin{proof}
By \autoref{thm:ask-sig-spine-derived-interface-admission}, a derived
interface $X$ enters the finite kernel only through $\NameCert(X)$. The
additional admission row in
\autoref{def:ask-sig-spine-certified-existence} records that the fields of
that certificate are available for the intended object-layer use. Thus any
use of $X$ factors through the displayed pair consisting of the naming
certificate and its admission row.
\end{proof}

\begin{principle}[Ledger policy for hidden material]
\label{prin:ask-sig-spine-ledger-policy-hidden-material}
Any derived interface that hides a process, tail, source fiber, phase row,
gap row, or compatibility thread must expose a $\LedgerPolicy$ witness.
This applies in particular to completion-like, fold-like, phase-like,
geometric, algebraic, analytic, zeta-facing, and zero-ledger-facing
interfaces.
\end{principle}

\begin{theorem}[L3--L5 finite-kernel closure]
\label{thm:ask-sig-spine-l3-l5-closure}
Given $\AskPol(\Pi,D)$, $\PkgPol(\Pi)$, $\GapPol(\Pi,D)$, and the exact
globalize fields where claimed, the L3--L5 spine supplies:
\begin{enumerate}
  \item signature determinacy for each admitted source history;
  \item $\sameSig_{\Pi}$ as the internal signature classifier on admitted
  histories;
  \item package classification of answer histories;
  \item gap coverage for admitted source histories;
  \item gap separation for same-source package overlaps;
  \item exact local classification by $\ExactGlobalizeBase(\Pi,D)$;
  \item certificate admission for derived interfaces through $\NameCert$.
\end{enumerate}
\end{theorem}

\begin{proof}
The first clause is \autoref{thm:ask-sig-spine-signature-inversion-determinacy}.
The second is \autoref{thm:ask-sig-spine-samesig-classifier}.  The third is
\autoref{thm:ask-sig-spine-packages-classify-signatures}.  The fourth and
fifth are \autoref{thm:ask-sig-spine-gap-coverage-separation}.  The sixth is
\autoref{thm:ask-sig-spine-exact-globalize-classification}.  The final clause
is \autoref{thm:ask-sig-spine-derived-interface-admission}.
\end{proof}

\begin{principle}[Finite-data boundary]
\label{prin:ask-sig-spine-finite-data-boundary}
The next layer begins only after the L0--L5 spine has supplied these
policies and ledgers.  Finite data interfaces such as
$\mathsf{Bool}^{\uparrow}$, $\mathsf{Option}^{\uparrow}$,
$\mathsf{Sum}^{\uparrow}$, $\mathsf{Prod}^{\uparrow}$, and
$\mathsf{List}^{\uparrow}$ are therefore certificate consumers, not
additional kernel primitives.
\end{principle}
