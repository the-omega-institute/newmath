\chapter{Mark--Hist--Ext--Cont Kernel Spine}
\label{ch:visions-mark-hist-ext-cont-spine}
\origin{human}

The finite BEDC kernel begins below packages, names, signatures, and
ordinary mathematical objects.  Its first visible surface is the passage
from raw marks to generated histories and then to one-step extension and
continuation.  This chapter records only that initial spine:
$$
\mathsf{BMark}
\longrightarrow
\mathsf{BHist}
\longrightarrow
(\mathsf{Ext},\mathsf{Cont}).
$$
Later layers such as $\mathsf{Ask}$, $\mathsf{Sig}$, $\mathsf{sameSig}$,
$\mathsf{Pkg}$, $\mathsf{Gap}$, and $\NameCert{}$ depend on this spine but
are not part of the present surface.

\section{Raw marks}
\label{sec:mark-hist-ext-cont-raw-marks}

\begin{definition}[Raw mark carrier]
\label{def:kernel-spine-raw-mark-carrier}
The raw mark carrier is the closed two-constructor carrier
$$
\mathsf{BMark}:=\{\mathsf{b0},\mathsf{b1}\}_{\mathrm{gen}}.
$$
The two constructors are primitive visible emissions.  They are not the
numbers $0$ and $1$, not truth values, and not elements drawn from an
ambient set.
\end{definition}

\begin{definition}[Mark sameness]
\label{def:kernel-spine-mark-sameness}
The mark-sameness judgment
$$
\mathsf{msame}:\mathsf{BMark}\to\mathsf{BMark}\to\mathsf{Prop}
$$
is generated only by the two diagonal rules
$$
\mathsf{msame}(\mathsf{b0},\mathsf{b0}),
\qquad
\mathsf{msame}(\mathsf{b1},\mathsf{b1}).
$$
No rule generates cross-mark sameness.
\end{definition}

\begin{principle}[Closed generation at the mark layer]
\label{prin:kernel-spine-mark-closed-generation}
At the mark layer, a judgment is internal only when it is generated by the
displayed constructors and rules.  There is no hidden conversion, hidden
coercion, ambient extensionality rule, or external equality that identifies
$\mathsf{b0}$ with $\mathsf{b1}$.
\end{principle}

\begin{theorem}[Mark sameness is diagonal]
\label{thm:kernel-spine-mark-sameness-diagonal}
Mark sameness is reflexive, symmetric, and transitive on generated marks,
and the cross judgments
$$
\mathsf{msame}(\mathsf{b0},\mathsf{b1}),
\qquad
\mathsf{msame}(\mathsf{b1},\mathsf{b0})
$$
are not generated.
\end{theorem}

\begin{proof}
Closed generation gives only two mark cases.  The diagonal judgments supply
reflexivity.  Symmetry and transitivity never leave the diagonal cases,
because any premise of mark sameness is already one of those cases.  The
cross judgments are not conclusions of either displayed rule, hence are
not generated.
\end{proof}

\section{Histories}
\label{sec:mark-hist-ext-cont-histories}

\begin{definition}[History carrier]
\label{def:kernel-spine-history-carrier}
The history carrier is the closed generated carrier
$$
\mathsf{BHist}
$$
with constructors
$$
\mathsf{Empty}:\mathsf{BHist},
\qquad
\mathsf{e0}:\mathsf{BHist}\to\mathsf{BHist},
\qquad
\mathsf{e1}:\mathsf{BHist}\to\mathsf{BHist}.
$$
The empty constructor is the empty history.  The constructors
$\mathsf{e0}$ and $\mathsf{e1}$ append one visible $\mathsf{b0}$ or
$\mathsf{b1}$ emission to an existing generated history.
\end{definition}

\begin{definition}[History sameness]
\label{def:kernel-spine-history-sameness}
The history-sameness judgment
$$
\mathsf{hsame}:\mathsf{BHist}\to\mathsf{BHist}\to\mathsf{Prop}
$$
is generated by the empty rule
$$
\mathsf{hsame}(\mathsf{Empty},\mathsf{Empty})
$$
and the two constructor-congruence rules
$$
\mathsf{hsame}(h,k)
\Longrightarrow
\mathsf{hsame}(\mathsf{e0}(h),\mathsf{e0}(k)),
$$
$$
\mathsf{hsame}(h,k)
\Longrightarrow
\mathsf{hsame}(\mathsf{e1}(h),\mathsf{e1}(k)).
$$
There is no rule identifying an empty history with an extended history,
and no rule identifying an $\mathsf{e0}$-headed history with an
$\mathsf{e1}$-headed history.
\end{definition}

\begin{theorem}[History no-confusion]
\label{thm:kernel-spine-history-no-confusion}
For generated histories $h,k$, the following judgments are not generated:
$$
\mathsf{hsame}(\mathsf{Empty},\mathsf{e0}(h)),
\qquad
\mathsf{hsame}(\mathsf{Empty},\mathsf{e1}(h)),
$$
$$
\mathsf{hsame}(\mathsf{e0}(h),\mathsf{e1}(k)),
\qquad
\mathsf{hsame}(\mathsf{e1}(h),\mathsf{e0}(k)).
$$
\end{theorem}

\begin{proof}
Inspect the displayed generation rules for $\mathsf{hsame}$.  The empty
rule has empty source and target.  The $\mathsf{e0}$ rule has
$\mathsf{e0}$ on both sides.  The $\mathsf{e1}$ rule has $\mathsf{e1}$ on
both sides.  None of these rules has a cross-constructor conclusion.
\end{proof}

\begin{corollary}[Histories record finite distinction sequences]
\label{cor:kernel-spine-histories-record-sequences}
Every generated history records a finite sequence of raw mark emissions,
and $\mathsf{hsame}$ preserves the generated constructor shape of that
sequence.
\end{corollary}

\begin{proof}
Define the sequence readout recursively by sending $\mathsf{Empty}$ to the
empty word, $\mathsf{e0}(h)$ to the readout of $h$ followed by a
$\mathsf{b0}$ emission, and $\mathsf{e1}(h)$ to the readout of $h$ followed
by a $\mathsf{b1}$ emission.  The no-confusion theorem shows that
$\mathsf{hsame}$ cannot cross the empty, $\mathsf{e0}$, and $\mathsf{e1}$
constructors.
\end{proof}

\section{Extension and continuation}
\label{sec:mark-hist-ext-cont-extension-continuation}

\begin{definition}[One-step extension]
\label{def:kernel-spine-extension}
The one-step extension relation is
$$
\mathsf{Ext}:\mathsf{BHist}\to\mathsf{BMark}\to\mathsf{BHist}\to
\mathsf{Prop}.
$$
Its generation rules are
$$
\mathsf{Ext}(h,\mathsf{b0},\mathsf{e0}(h)),
\qquad
\mathsf{Ext}(h,\mathsf{b1},\mathsf{e1}(h)).
$$
Thus $\mathsf{Ext}(h,m,r)$ records that the history $r$ is obtained from
$h$ by one displayed mark emission.
\end{definition}

\begin{theorem}[Extension result characterization]
\label{thm:kernel-spine-extension-result-characterization}
For generated $m,h,t$, the extension judgment ending in an
$\mathsf{e0}$-history has the form
$$
\mathsf{Ext}(h,m,\mathsf{e0}(t))
\Longleftrightarrow
\mathsf{msame}(m,\mathsf{b0})
\wedge
\mathsf{hsame}(h,t),
$$
and the extension judgment ending in an $\mathsf{e1}$-history has the form
$$
\mathsf{Ext}(h,m,\mathsf{e1}(t))
\Longleftrightarrow
\mathsf{msame}(m,\mathsf{b1})
\wedge
\mathsf{hsame}(h,t).
$$
\end{theorem}

\begin{proof}
For the first equivalence, inspect the generation rules for
$\mathsf{Ext}$.  A result with head $\mathsf{e0}$ can only come from the
$\mathsf{b0}$ rule, and its tail is the source history up to
$\mathsf{hsame}$.  Conversely, the $\mathsf{b0}$ rule gives
$\mathsf{Ext}(h,\mathsf{b0},\mathsf{e0}(h))$, and the displayed sameness
rows transport the mark and source history.  The $\mathsf{e1}$ case is the
same argument with $\mathsf{b1}$ and $\mathsf{e1}$.
\end{proof}

\begin{theorem}[Same-source cross-mark extensions are separated]
\label{thm:kernel-spine-cross-mark-extension-separation}
If
$$
\mathsf{Ext}(h,\mathsf{b0},r_0),
\qquad
\mathsf{Ext}(h,\mathsf{b1},r_1),
$$
then $\mathsf{hsame}(r_0,r_1)$ is not generated.
\end{theorem}

\begin{proof}
The first extension has result $\mathsf{hsame}$-classified with
$\mathsf{e0}(h)$.  The second has result $\mathsf{hsame}$-classified with
$\mathsf{e1}(h)$.  If $\mathsf{hsame}(r_0,r_1)$ were generated, transitivity
would generate $\mathsf{hsame}(\mathsf{e0}(h),\mathsf{e1}(h))$, contradicting
\autoref{thm:kernel-spine-history-no-confusion}.
\end{proof}

\begin{definition}[Continuation]
\label{def:kernel-spine-continuation}
The continuation relation is
$$
\mathsf{Cont}:\mathsf{BHist}\to\mathsf{BHist}\to\mathsf{BHist}\to
\mathsf{Prop}.
$$
It records the BEDC-internal form of history concatenation.  Its basic
interface includes empty-continuation rows, one-step compatibility with
$\mathsf{e0}$ and $\mathsf{e1}$, unit rows, and associativity up to
$\mathsf{hsame}$.  These rows are L2 obligations over the same
$\mathsf{BHist}$ carrier; they do not introduce another visible object
layer.
\end{definition}

\begin{theorem}[Continuation unit and associativity obligations]
\label{thm:kernel-spine-continuation-unit-assoc-obligations}
Continuation has empty-history unit rows
$$
\mathsf{Cont}(\mathsf{Empty},h,h),
\qquad
\mathsf{Cont}(h,\mathsf{Empty},h),
$$
and threefold continuation is associative up to $\mathsf{hsame}$: if
$$
\mathsf{Cont}(a,b,u),
\qquad
\mathsf{Cont}(u,c,v),
$$
$$
\mathsf{Cont}(b,c,w),
\qquad
\mathsf{Cont}(a,w,z),
$$
then $\mathsf{hsame}(v,z)$ is generated.
\end{theorem}

\begin{proof}
The unit rows are the empty-history clauses of continuation.  For
associativity, both displayed composites represent the same generated
history obtained by first reading $a$, then $b$, then $c$.  The
constructor recursion defining continuation therefore classifies the two
results by the same $\mathsf{BHist}$ shape, so $\mathsf{hsame}(v,z)$ is
generated.
\end{proof}

\begin{definition}[Initial kernel spine]
\label{def:initial-kernel-spine}
The initial BEDC kernel spine is the tuple
$$
\mathcal T_{\leq 2}
:=
(\mathsf{BMark},\mathsf{msame},
\mathsf{BHist},\mathsf{hsame},
\mathsf{Ext},\mathsf{Cont}).
$$
\end{definition}

\begin{theorem}[Initial kernel spine separation]
\label{thm:initial-kernel-spine-separation}
The initial kernel spine separates raw marks, separates history
constructors, and separates same-source $\mathsf{b0}$ and $\mathsf{b1}$
extension results.
\end{theorem}

\begin{proof}
Raw marks are separated by
\autoref{thm:kernel-spine-mark-sameness-diagonal}.  History constructors
are separated by \autoref{thm:kernel-spine-history-no-confusion}.
Same-source cross-mark extensions are separated by
\autoref{thm:kernel-spine-cross-mark-extension-separation}.
\end{proof}

\begin{corollary}[Minimal binary recursive readout]
\label{cor:kernel-spine-minimal-binary-recursive-readout}
The initial kernel spine supports the recursive readout
$$
P_{\mathsf{B}}:\mathsf{BHist}\to
\{\mathsf{b0},\mathsf{b1}\}^{<\omega}
$$
defined by
$$
P_{\mathsf{B}}(\mathsf{Empty})=\epsilon,
\qquad
P_{\mathsf{B}}(\mathsf{e0}(h))=P_{\mathsf{B}}(h)\mathsf{b0},
\qquad
P_{\mathsf{B}}(\mathsf{e1}(h))=P_{\mathsf{B}}(h)\mathsf{b1}.
$$
\end{corollary}

\begin{proof}
The recursion follows the three constructors of $\mathsf{BHist}$.
\autoref{thm:kernel-spine-history-no-confusion} supplies uniqueness of the
visible last step, and \autoref{thm:kernel-spine-mark-sameness-diagonal}
supplies the two separated terminal emissions.
\end{proof}

\begin{principle}[Kernel spine as the self-similar starting surface]
\label{prin:kernel-spine-self-similar-starting-surface}
The pair $\mathsf{b0},\mathsf{b1}$ supplies the minimal visible distinction
surface, and the constructors $\mathsf{e0},\mathsf{e1}$ supply the recursive
history surface.  This is the bottom support for later window, Zeckendorf,
prime-coordinate, zeta-carrier, and zero-ledger routes.  Higher layers may
compress, package, or project these histories only after they expose the
source fiber and the ledger rows that license the compression.
\end{principle}

\begin{principle}[Next layer boundary]
\label{prin:kernel-spine-next-layer-boundary}
The next layer begins only after this spine, with
$$
\mathsf{Ask}
\to
\mathsf{Sig}
\to
\mathsf{sameSig}
\to
\mathsf{Pkg}
\to
\mathsf{Gap}
\to
\NameCert{}.
$$
Its role is not to add primitive mathematical objects to the kernel, but
to expose the source ledger for visible objects obtained by asking,
signature formation, packaging, gap accounting, and certification.
\end{principle}
