\begin{definition}[Golden scale eigenvalue]
\label{def:rh-golden-scale-eigenvalue}
Let
$$
w:=s-\frac{1}{2},
\qquad
\varphi:=\frac{1+\sqrt{5}}{2},
\qquad
q:=\varphi^{w}.
$$
The golden scale eigenvalue of $s$ is the centered scale readout $q$.
It is located only as a scale-coordinate packet over the centered
critical-strip route; it is not an independent zero generator, a branch
choice for $\zeta$, or a replacement for the zeta zero ledger.
\end{definition}

\begin{principle}[Unitary scale closure reduction]
\label{prin:rh-unitary-scale-closure-reduction}
For a critical-strip zero packet $s$, write $w=s-1/2$ and
$q=\varphi^w$. Since $\varphi>1$,
$$
|q|
=
|\varphi^w|
=
\varphi^{\Re(w)}.
$$
Therefore
$$
\Re(s)=\frac{1}{2}
\quad\Longleftrightarrow\quad
\Re(w)=0
\quad\Longleftrightarrow\quad
|q|=1.
$$
The unitary scale closure route is the conditional reduction
$$
\mathsf{NoIndependentRealScaleLedger}(s)
\Longrightarrow
\mathsf{UnitaryScale}(q)
\Longrightarrow
|q|=1
\Longrightarrow
\mathsf{OnCritLine}(s).
$$
If every certified non-trivial zeta zero in the critical strip satisfies
the displayed no-scale-to-unitary obligation, then the BEDC-RH witness
function of \autoref{def:rh-proof-obligation-witness-function} is
inhabited.
\end{principle}

\begin{principle}[Argument-principle bypass is only a route choice]
\label{prin:rh-unitary-scale-closure-argument-principle-bypass}
This route does not count zeros by contour integration, does not use the
argument principle as a zero-enumeration engine, and does not require a
classical winding-number decision over an unbounded contour family. It
instead asks for a scale-closure certificate on already certified zero
packets. The bypass is methodological: it avoids the constructive
obstruction of turning global argument variation into a total zero ledger,
but it does not remove the need to prove the scale-closure obligation.
\end{principle}

\begin{remark}[Connes--Consani half-density reading]
\label{rem:rh-unitary-scale-closure-half-density}
The centered coordinate $w=s-1/2$ matches the half-density normalization
in scale-action readings of the zeta symmetry: the real half-shift is
absorbed into the carrier before the residual scale action is inspected.
In this language, the nontrivial part of the scale action is the phase row
$\varphi^{it}$ when $w=it$. The BEDC route uses that analogy only as a
scale-normalized readout of the ZetaCosmosClosure ledger; it does not
import an external spectral triple, trace formula, or classical
cohomology theorem as a proof object.
\end{remark}

\begin{principle}[Symmetry alone is not enough]
\label{prin:rh-unitary-scale-closure-symmetry-no-go}
The functional equation, conjugation symmetry, and the mirror fixed surface
do not by themselves prove $\mathsf{RH}$. An even real polynomial can have
a symmetric zero set with off-axis real scale, so symmetry alone cannot
force all zeros onto a unit circle after centering. The no-independent-real
scale ledger is the genuine extra input in this route. Without the
obligation
$$
\mathsf{NoIndependentRealScaleLedger}
\Longrightarrow
\mathsf{UnitaryScale},
$$
the identity $|\varphi^w|=\varphi^{\Re(w)}$ is only a readout of the
critical line, not a proof that zero packets land there.
\end{principle}

\begin{definition}[G\"omb\"oc-style homogeneous container row]
\label{def:rh-gomboc-homogeneous-container-row}
A G\"omb\"oc-style homogeneous container row for the completed zeta
carrier is a source-side packet
$$
\mathsf{ZetaGombocContainer}_{\zeta}
$$
with three displayed fields.  The homogeneity field says that the zero
packet carries no hidden mass, weight, or density parameter beyond the
source, prime, phase, tail, and defect ledgers already present in the
carrier.  In the centered channel this forbids an unrecorded
$\mathbb R_{>0}$ factor acting as a radial counterweight.  The equilibrium
field says that the legal balance surface of the container is the unit
scale surface
$$
|q|=1.
$$
The source-field says that the balance is produced by the internal
container rows themselves, not by selecting a zero and then attaching a
compensating scale ledger to it.  In particular, the row is global: it is
a constraint on the completed zeta carrier as a shape, not a family of
local supports built separately around individual zero packets.
\end{definition}

\begin{principle}[Analytic G\"omb\"oc reading]
\label{prin:rh-analytic-gomboc-reading}
The G\"omb\"oc-style reading is a geometric discipline for the no-scale
obligation in \autoref{prin:rh-unitary-scale-closure-reduction}.  A
homogeneous convex body with a single stable balance point represents, at
the level of analogy, a container whose righting behaviour is carried by
shape rather than by hidden weights.  The zeta translation is the row
$$
\mathsf{ZetaGombocContainer}_{\zeta}
\Longrightarrow
\mathsf{NoIndependentRealScaleLedger}
\Longrightarrow
\mathsf{UnitaryScale}(q).
$$
Thus an off-half readout is not interpreted as a second equilibrium of
the same homogeneous container.  It is interpreted as a demand for a
radial counterweight, namely an extra real-positive scale ledger.  The
row is still conditional: it names the geometric content of the
no-scale-to-unitary implication, but it does not prove that every zeta
zero packet carries such a container.
\end{principle}

\begin{definition}[Analytic G\"omb\"oc source shape]
\label{def:rh-analytic-gomboc-source-shape}
An analytic G\"omb\"oc source shape for $\xi$ is a source-side global
presentation of the completed zeta carrier in one of the following
positive forms:
$$
K_{\xi}(z,w)=\langle\Phi_{\xi}(z),\Phi_{\xi}(w)\rangle_{\mathcal H}
$$
as in \autoref{def:rh-source-pick-hilbert-factorization}, the positive
real row
$$
\operatorname{Re}\frac{\xi'(s)}{\xi(s)}\ge0
\qquad
\left(\operatorname{Re}(s)>\frac12\right)
$$
of \autoref{def:rh-xi-log-derivative-herglotz-row}, or a positive
spectral representation of the same log-derivative as in
\autoref{def:rh-positive-spectral-measure-for-xi}.  The representation
must be constructed before zero locations are consumed.  A representation
assembled around the already-read zero divisor is not an analytic
G\"omb\"oc source shape.
\end{definition}

\begin{principle}[Source-shape proof form]
\label{prin:rh-gomboc-source-shape-proof-form}
The G\"omb\"oc row changes the proof burden from local repair to global
shape.  A local container placed around a candidate zero would be a
supporting device for that readout; it would not certify that the
completed carrier itself is homogeneous.  By contrast, a source-side Pick
factorization, Herglotz positive-real row, or positive spectral measure
is a shape theorem for $\xi$ as a whole.  In that form an off-half zero
cannot be hidden as a local exception: by
\autoref{thm:rh-right-side-zero-one-point-pick-witness} and
\autoref{thm:rh-xi-herglotz-row-fixed-half-readout}, it becomes a finite
negative Pick witness or a negative scale-flow witness against the same
global positive structure.
\end{principle}

\begin{theorem}[G\"omb\"oc container gives the unitary scale reduction]
\label{thm:rh-gomboc-container-unitary-scale-reduction}
Assume that every certified non-trivial zeta zero in the critical strip
carries $\mathsf{ZetaGombocContainer}_{\zeta}$ and that the container row
supplies the implication
$$
\mathsf{ZetaGombocContainer}_{\zeta}
\to
\mathsf{UnitaryScale}(q)
$$
for the centered golden scale readout $q=\varphi^{s-1/2}$.  Then every
certified non-trivial zeta zero in the critical strip satisfies the
critical-line readout.
\end{theorem}

\begin{proof}
Fix a certified non-trivial zeta zero $s$ in the critical strip and set
$$
q=\varphi^{s-1/2}.
$$
The assumed container row gives $\mathsf{UnitaryScale}(q)$, hence
$|q|=1$.  By
\autoref{prin:rh-unitary-scale-closure-reduction}, the identity
$$
|q|=\varphi^{\Re(s-1/2)}
$$
then forces $\Re(s)=1/2$.  This is precisely the ordinary
$\mathsf{OnCritLine}(s)$ readout.
\end{proof}

\begin{definition}[Route obligations]
\label{def:rh-unitary-scale-closure-obligations}
The unitary scale closure route has five obligations.

\begin{enumerate}
\item The centered coordinate $w=s-1/2$ must be carried by the same
critical-strip zero packet as $s$.
\item The golden scale eigenvalue $q=\varphi^w$ must be a located
scale-readout with the modulus identity
$$
|q|=\varphi^{\Re(w)}.
$$
\item The ZetaCosmosClosure no-independent-real-scale ledger must imply
unitarity of the scale readout for every certified non-trivial zero packet.
\item The optional G\"omb\"oc-style container row, when used, must be
source-side and homogeneous: it may package the no-scale-to-unitary
implication through a Pick, Herglotz, or positive spectral source shape,
but it may not be built by adding a compensating radial ledger after a
zero has been selected.
\item The unitary readout must transport back to the ordinary
$\mathsf{OnCritLine}(s)$ witness used by the BEDC-RH statement.
\end{enumerate}
\end{definition}
