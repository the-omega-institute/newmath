\chapter{Arithmetic Spectral String Route}
\label{ch:visions-rh-arithmetic-spectral-string-route}
\origin{ai}

\autoref{prin:rh-conservative-oscillator-reading-gap} reads the eta
packet as a many-oscillator phase sum, while
\autoref{def:rh-zero-certificate-tower} reads a completed zero through a
compatible tower of finite certificates. The present chapter records the
same row in string-like spectral language. The word ``string'' here is a
mode-organization term: one oscillator displays one mode, a finite packet
displays finitely many modes, and the arithmetic spectral carrier
organizes the compatible tower of logarithmic modes. No spacetime
target, quantized physical spectrum, or imported physical formalism is
assumed.

\begin{definition}[Arithmetic spectral string carrier]
\label{def:rh-arithmetic-spectral-string-carrier}
An \emph{arithmetic spectral string carrier} is a displayed normal-mode
readout
$$
X(t)=\sum_{n\ge1}a_n e^{-it h_n}
$$
on a licensed summability or certificate window, with logarithmic
heights
$$
h_n=\log n.
$$
The eta critical-line carrier is the instance
$$
a_n=(-1)^{n-1}n^{-1/2},
\qquad
X_{\eta}(t)=\sum_{n\ge1}(-1)^{n-1}n^{-1/2}e^{-it\log n}.
$$
The prime ledger is the factorization row
$$
\log n=\sum_p v_p(n)\log p,
$$
so the displayed modes are arithmetic normal modes generated by prime
frequency registers. The carrier is therefore string-like because it
organizes a compatible normal-mode family, not because it imports an
external physical string formalism.
\end{definition}

\begin{principle}[String nodes are phase cancellations]
\label{prin:rh-string-nodes-phase-cancellations}
A node of the conservative arithmetic spectral string is a zero readout
of the phase carrier:
$$
X(t)=0.
$$
For the eta instance this is the critical-line cancellation
$$
\eta(1/2+it)=0
$$
whenever the displayed readout is licensed. A projected cancellation of
an amplitude-biased readout is not yet a conservative string node. The
certificate must also show that the active channels close by phase
inside the unit boundary.
\end{principle}

\begin{definition}[Centered string channel]
\label{def:rh-centered-string-channel}
For
$$
s=\frac12+r+it
$$
and an active logarithmic height $h_a>0$, the centered string channel is
$$
U_a(s):=e^{-(s-1/2)h_a}
=
e^{-r h_a}e^{-it h_a}.
$$
The factor $e^{-it h_a}$ is the conservative phase mode. The factor
$e^{-r h_a}$ is a real positive radial scale mode. Thus an off-half
readout carries a second nonunitary coordinate unless the certificate
supplies a legal scale ledger, which the closed arithmetic string row
does not permit.
\end{definition}

\begin{definition}[Unitary arithmetic string flow]
\label{def:rh-unitary-arithmetic-string-flow}
A \emph{unitary arithmetic string flow} is a phase evolution
$$
U(t)=e^{-itH}
$$
on a displayed normal-mode carrier with
$$
H e_n=(\log n)e_n
$$
on the mode basis under consideration, and with
$$
U(t)^*U(t)=I.
$$
The same Hamiltonian readout gives every active logarithmic mode the
single time parameter $t$. A radial factor $e^{-rH}$ is not part of this
unitary flow; it is a damping or gain row.
\end{definition}

\begin{theorem}[Unitary string admissibility excludes radial scale]
\label{thm:rh-unitary-string-admissibility-excludes-radial-scale}
Let a string certificate have limit readout
$$
s_{\infty}=\frac12+r+it.
$$
If some active height $h_a>0$ satisfies the unit-boundary row
$$
|e^{-(s_{\infty}-1/2)h_a}|=1,
$$
then $r=0$.
\end{theorem}

\begin{proof}
The centered channel factorization gives
$$
|e^{-(s_{\infty}-1/2)h_a}|=e^{-r h_a}.
$$
The unit-boundary row therefore says
$$
e^{-r h_a}=1.
$$
Since the real exponential is injective and $h_a>0$, this implies
$r=0$.
\end{proof}

\begin{definition}[Closed arithmetic string certificate]
\label{def:rh-closed-arithmetic-string-certificate}
A \emph{closed arithmetic string certificate} for an accepted
non-trivial completed zeta zero is a legal boundary-stable zero
certificate tower in the sense of
\autoref{def:rh-legal-boundary-stable-certificate-tower}, together with
the following mode interpretation:
$$
h_n=\log n,
\qquad
\log n=\sum_p v_p(n)\log p,
$$
finite certificate levels are finite normal-mode packets, the time row
is the unitary arithmetic string flow of
\autoref{def:rh-unitary-arithmetic-string-flow}, the boundary row is the
unit-channel condition of
\autoref{thm:rh-unitary-string-admissibility-excludes-radial-scale}, and
the far-end row is the no-absorption discipline required by
\autoref{def:rh-legal-boundary-stable-certificate-tower}. Thus a zero is
certified as a node of a closed conservative arithmetic spectral system,
not as a cancellation produced by unledgered radial amplitude change.
\end{definition}

\begin{principle}[Zeta closed arithmetic string target]
\label{prin:rh-zeta-closed-arithmetic-string-target}
The target
$$
\mathsf{ZetaClosedArithmeticString}
$$
asserts that every accepted non-trivial completed zeta zero admits a
closed arithmetic string certificate. Its source load has the following
rows:
$$
\mathsf{ZetaStringRealization},
$$
$$
\mathsf{StrongActiveCapture},
$$
$$
\mathsf{BoundaryStability},
$$
$$
\mathsf{NoFarEndAbsorption}_{\zeta},
$$
and
$$
\mathsf{ZeroPresentationCompleteness}.
$$
The first row identifies the logarithmic normal-mode carrier; the second
turns a zero into a canonical finite packet tower; the third supplies the
unitary boundary row; the fourth prevents the infinite tail from serving
as a radial scale reservoir; the fifth supplies coverage of all accepted
non-trivial zero readouts.
\end{principle}

\begin{corollary}[Closed arithmetic string coverage gives fixed-half readout]
\label{cor:rh-closed-arithmetic-string-coverage-fixed-half}
If $\mathsf{ZetaClosedArithmeticString}$ holds, then every accepted
non-trivial completed zeta zero has real part $1/2$.
\end{corollary}

\begin{proof}
Let $u$ be an accepted non-trivial completed zeta zero. The target
$\mathsf{ZetaClosedArithmeticString}$ supplies a closed arithmetic string
certificate for $u$, hence a legal boundary-stable certificate tower by
\autoref{def:rh-closed-arithmetic-string-certificate}. Applying
\autoref{cor:rh-legal-certificate-towers-fixed-half-limits} gives the
fixed-half readout.
\end{proof}

\begin{principle}[String language does not change the proof boundary]
\label{prin:rh-string-language-proof-boundary}
The arithmetic string language reorganizes the conservative oscillator
route; it does not remove any source obligation. Weak eta capture still
only supplies local analytic boxes. The closed string claim requires the
strong certificate rows of
\autoref{prin:rh-zeta-closed-arithmetic-string-target}. In particular,
unitary normal-mode language does not by itself prove that every zeta
zero is represented by a closed arithmetic string certificate, and it
does not by itself prove boundary stability for the canonical packets.
\end{principle}
