\chapter{All-Prime Object}
\label{ch:visions-rh-all-prime-object}
\origin{ai}

\autoref{ch:visions-rh-prime-generated-recursive-sieve} treats each
prime as a generator of local sieve rows.  The RH route also requires
the source containing all primes to be a BEDC object rather than an
unstructured ambient set.  The present chapter packages the
all-prime source as
$$
\mathcal P^\uparrow
$$
with a displayed time row, unit row, and ledger row.  This prevents a
radial residual from being hidden in an unaccounted prime tail.

\section{Prime Source Packet}
\label{sec:rh-all-prime-object-source-packet}

\begin{definition}[Finite prime source layer]
\label{def:rh-finite-prime-source-layer}
Let $p_1,p_2,p_3,\ldots$ be the increasing prime sequence and define
$$
\mathcal P_N:=\{p_1,\ldots,p_N\}.
$$
The transition
$$
\iota_{N,N+1}:\mathcal P_N\hookrightarrow\mathcal P_{N+1}
$$
is inclusion.
\end{definition}

\begin{definition}[AllPrimeObject]
\label{def:rh-all-prime-object}
The BEDC all-prime object is the packet
$$
\mathcal P^\uparrow
:=
\left(
(\mathcal P_N)_{N\ge1},
(\iota_{N,N+1})_{N\ge1},
\Lambda_{\mathcal P},
\operatorname{NameCert}_{\mathcal P},
\operatorname{TailPolicy}_{\mathcal P}
\right).
$$
It is a certified colimit-style source: finite prime layers supply the
visible stages, $\Lambda_{\mathcal P}$ records generation and sieve
history, $\operatorname{NameCert}_{\mathcal P}$ licenses the source as
a BEDC object, and $\operatorname{TailPolicy}_{\mathcal P}$ records
data not visible at a chosen finite layer.
\end{definition}

\begin{definition}[Prime object time row]
\label{def:rh-prime-object-time-row}
The time row
$$
\mathsf T_{\mathcal P}
$$
asserts that $\mathcal P^\uparrow$ is generated by the Euclidean
survivor tower of \autoref{def:rh-euclidean-prime-sieve-tower}.  With
$$
P_N:=\prod_{j=1}^{N}p_j,
\qquad
\mathcal N_N:=\{n\ge2:\gcd(n,P_N)=1\},
$$
the appended source row is
$$
p_{N+1}:=\min\mathcal N_N,
\qquad
\mathcal P_{N+1}=\mathcal P_N\cup\{p_{N+1}\},
$$
and the ledger update is
$$
\Lambda_{\mathcal P,N+1}
=
\Lambda_{\mathcal P,N}
+
(p_{N+1},\operatorname{minimalSurvivor}(P_N)).
$$
\end{definition}

\begin{theorem}[Prime object time row is append-only]
\label{thm:rh-prime-object-time-row-append-only}
The row $\mathsf T_{\mathcal P}$ is an append-only generation row.
\end{theorem}

\begin{proof}
By
\autoref{thm:rh-minimal-euclidean-survivor-new-prime}, the minimal
survivor of $\mathcal N_N$ is a prime distinct from
$p_1,\ldots,p_N$.  Hence adjoining it produces precisely the next
finite prime source layer.  The displayed update records that adjoining
event without deleting any prior prime row, so the generation history
is append-only.
\end{proof}

\section{Unit Rows}
\label{sec:rh-all-prime-object-unit-rows}

\begin{definition}[Prime circle layer]
\label{def:rh-prime-circle-layer}
For $N\ge1$, set $P=p_N$.  The $N$th prime circle layer is the embedded
finite set
$$
\mathbb P_N^\circ
:=
\{p_1,\ldots,p_{N-1},0\}
\subseteq
\mathbb F_P,
$$
obtained from
$$
\pi_P:\mathcal P_N\to\mathbb F_P,
\qquad
\pi_P(p_j)=p_j\bmod P.
$$
The element $0$ represents the terminal point $p_N=P$ of the finite
prime source layer.
\end{definition}

\begin{lemma}[Prime circle embedding]
\label{lem:rh-prime-circle-embedding}
The map $\pi_P:\mathcal P_N\to\mathbb F_P$ is injective.
\end{lemma}

\begin{proof}
For $j<N$, the integer $p_j$ satisfies $0<p_j<P$, while $p_N=P$ maps to
$0$.  The residues $p_1,\ldots,p_{N-1},0$ are therefore pairwise
distinct in $\mathbb F_P$.
\end{proof}

\begin{definition}[Prime object unit row]
\label{def:rh-prime-object-unit-row}
The unit row
$$
\mathsf U_{\mathcal P}
$$
contains two compatible unit readouts:
$$
\begin{gathered}
\psi_t(x):=\exp(2\pi i tx/P),
\qquad t,x\in\mathbb F_P,\\
\mathbb P_{\sigma}(v_p=k)=(1-p^{-\sigma})p^{-k\sigma},
\qquad \sigma>1.
\end{gathered}
$$
The first is the finite additive character system on the prime circle
layer.  The second is the Euler local probability row of
\autoref{def:rh-right-half-plane-euler-probability-law}.
\end{definition}

\begin{theorem}[Prime object unit rows are normalized]
\label{thm:rh-prime-object-unit-rows-normalized}
The finite character and Euler probability rows of
$\mathsf U_{\mathcal P}$ are unit rows.
\end{theorem}

\begin{proof}
For the finite character row,
$$
|\psi_t(x)|=1
$$
for all $t,x\in\mathbb F_P$.  For the Euler row,
$$
\sum_{k\ge0}(1-p^{-\sigma})p^{-k\sigma}=1
$$
at each prime $p$ when $\sigma>1$.  By
\autoref{thm:rh-euler-probability-factors-by-prime-valuations}, these
local rows factor the probability law by prime valuations.  By
\autoref{thm:rh-zeta-ratio-phase-expectation}, the normalized
right-half-plane zeta ratio is the unit phase expectation
$$
\mathbb E_{\mathbb P_\sigma}[n^{-it}]
=
\frac{\zeta(\sigma+it)}{\zeta(\sigma)},
\qquad
|n^{-it}|=1.
$$
Thus both finite and Euler layers expose unit readouts.
\end{proof}

\section{Ledger Rows}
\label{sec:rh-all-prime-object-ledger-rows}

\begin{definition}[Prime object ledger row]
\label{def:rh-prime-object-ledger-row}
The ledger row
$$
\mathsf L_{\mathcal P}
$$
records unique factorization and finite-layer tails.  For $n\ge1$,
$$
\operatorname{PrimeLedger}(n)
:=
(v_p(n))_{p\in\mathcal P}.
$$
At the finite layer $\mathcal P_N$ the ledger is
$$
\operatorname{PrimeLedger}_N(n)
:=
\left(
(v_{p_j}(n))_{j\le N},
n_{>N}
\right),
$$
where
$$
n_{>N}
:=
\prod_{j>N}p_j^{v_{p_j}(n)}.
$$
For the Euler product the corresponding finite-layer ledger is
$$
\operatorname{EulerLedger}_N(s)
:=
\left(
\prod_{j\le N}(1-p_j^{-s})^{-1},
\prod_{j>N}(1-p_j^{-s})^{-1}
\right)
$$
on the half-plane where the Euler product is licensed.
\end{definition}

\begin{theorem}[Prime object ledger records finite data and tail data]
\label{thm:rh-prime-object-ledger-records-tail-data}
The row $\mathsf L_{\mathcal P}$ prevents a finite-layer readout from
discarding information outside $\mathcal P_N$.
\end{theorem}

\begin{proof}
Unique factorization gives
$$
n=
\left(\prod_{j\le N}p_j^{v_{p_j}(n)}\right)n_{>N}.
$$
Thus the finite valuation row alone is not the full ledger unless the
tail $n_{>N}$ is also recorded.  The Euler product has the same
structure: the finite product over $j\le N$ is accompanied by the
displayed Euler tail over $j>N$.  Therefore every finite readout of the
all-prime object carries an explicit tail coordinate.
\end{proof}

\begin{theorem}[AllPrimeObject has an ontological triple]
\label{thm:rh-all-prime-object-has-ontological-triple}
The BEDC all-prime object has
$$
\operatorname{Ont}(\mathcal P^\uparrow)
=
(\mathsf T_{\mathcal P},\mathsf U_{\mathcal P},\mathsf L_{\mathcal P})
$$
in the sense of \autoref{def:rh-ontological-triple}.
\end{theorem}

\begin{proof}
The time row is
\autoref{def:rh-prime-object-time-row}, and it is append-only by
\autoref{thm:rh-prime-object-time-row-append-only}.  The unit row is
\autoref{def:rh-prime-object-unit-row}, normalized by
\autoref{thm:rh-prime-object-unit-rows-normalized}.  The ledger row is
\autoref{def:rh-prime-object-ledger-row}, with tail preservation given
by \autoref{thm:rh-prime-object-ledger-records-tail-data}.  These are
exactly the three coordinates required by
\autoref{def:rh-ontological-triple}.
\end{proof}

\section{Finite Detection}
\label{sec:rh-all-prime-object-finite-detection}

\begin{theorem}[Prime circle layer detects finite prime residuals]
\label{thm:rh-prime-circle-layer-detects-finite-prime-residuals}
Let
$$
R_N=\sum_{j=1}^{N}c_j[p_j]\ne0
$$
be a residual supported on $\mathcal P_N$, and set $P=p_N$.  Then
there exists $t\in\mathbb F_P$ such that
$$
\sum_{j=1}^{N}c_j\psi_t(p_j\bmod P)\ne0.
$$
\end{theorem}

\begin{proof}
Push $R_N$ forward along the embedding of
\autoref{lem:rh-prime-circle-embedding}:
$$
\widetilde R_N
:=
\sum_{j=1}^{N}c_j[\pi_P(p_j)]
\in
\mathbb C[\mathbb F_P].
$$
Since the points $\pi_P(p_j)$ are pairwise distinct, $R_N\ne0$ implies
$\widetilde R_N\ne0$.  The additive characters
$$
\psi_t(x)=\exp(2\pi i tx/P)
$$
form a Fourier basis for functions on the finite additive group
$\mathbb F_P$.  If every displayed character readout vanished on
$\widetilde R_N$, all Fourier coefficients of $\widetilde R_N$ would
vanish, forcing $\widetilde R_N=0$, a contradiction.
\end{proof}

\begin{definition}[Finite-or-tail prime residual]
\label{def:rh-finite-or-tail-prime-residual}
A residual over $\mathcal P^\uparrow$ is finite-or-tail visible when,
for every finite layer $\mathcal P_N$, it satisfies one of the following
ledger alternatives:
$$
\begin{gathered}
\text{a nonzero component is visible on some finite prime layer},\\
\text{the invisible component is recorded in }
\operatorname{TailPolicy}_{\mathcal P}.
\end{gathered}
$$
\end{definition}

\begin{theorem}[AllPrimeObject has no unledgered residual escape]
\label{thm:rh-all-prime-object-no-unledgered-residual-escape}
Assume a residual over $\mathcal P^\uparrow$ is finite-or-tail visible.
Then it cannot be hidden inside the all-prime source as an unledgered
ambient coordinate.
\end{theorem}

\begin{proof}
If the residual has a nonzero finite component on $\mathcal P_N$, then
\autoref{thm:rh-prime-circle-layer-detects-finite-prime-residuals}
detects it by a unit character row on the prime circle layer.  If no
finite component is visible at the chosen layer, the definition of
finite-or-tail visibility places the component in
$\operatorname{TailPolicy}_{\mathcal P}$.  Thus every residual is either
detected at a finite layer or recorded as tail data.  Neither case is an
unledgered source coordinate.
\end{proof}

\section{Zeta Source Interface}
\label{sec:rh-all-prime-object-zeta-source-interface}

\begin{definition}[Certified Euler prime limit]
\label{def:rh-certified-euler-prime-limit}
For $\Re(s)>1$, the certified Euler prime limit attached to
$\mathcal P^\uparrow$ is the packet
$$
\zeta_{\Re>1}^{\uparrow}
:=
\left(
(Z_N(s))_{N\ge1},
(\operatorname{EulerTail}_N(s))_{N\ge1},
\Lambda_{\mathcal P},
\operatorname{NameCert}_{\mathcal P}
\right),
$$
where
$$
Z_N(s):=\prod_{j\le N}(1-p_j^{-s})^{-1}
$$
and
$$
\operatorname{EulerTail}_N(s)
:=
\prod_{j>N}(1-p_j^{-s})^{-1}.
$$
\end{definition}

\begin{theorem}[AllPrimeObject supplies the right-half-plane zeta source]
\label{thm:rh-all-prime-object-supplies-right-half-plane-zeta-source}
On $\Re(s)>1$, the certified Euler prime limit is a BEDC-licensed
source for the normalized zeta phase readout.
\end{theorem}

\begin{proof}
The finite products $Z_N(s)$ are generated by appending Euler local
factors along the time row of $\mathcal P^\uparrow$.  The tails
$\operatorname{EulerTail}_N(s)$ are the Euler ledger tails of
\autoref{def:rh-prime-object-ledger-row}.  Absolute convergence on
$\Re(s)>1$ licenses the Euler product, and
\autoref{prin:rh-right-half-plane-zeta-prime-generated-phase-limit}
identifies the normalized zeta phase as the limit of prime-generated
local probability sieves.  The source is therefore certified by
finite products, tail rows, and the all-prime NameCert rather than by a
bare infinite set of primes.
\end{proof}

\begin{principle}[All-prime source audit boundary]
\label{prin:rh-all-prime-source-audit-boundary}
The all-prime object closes the source-side gap for the prime sieve
route.  It does not prove that an off-critical zero produces a
finite-or-tail prime residual.  It asserts that, once such a residual
is present over the prime source, the residual must be detected by a
finite prime circle layer or recorded in the all-prime tail ledger.
\end{principle}
