\chapter{Squarefree Prime Polarized Bridge}
\label{ch:visions-rh-squarefree-prime-polarized-bridge}
\origin{ai}

\autoref{ch:visions-rh-zeta-ray-lee-yang-container} isolates the
polarized golden Lee--Yang container as the stable object in the golden
route.  The present chapter records the arithmetic form of that
container after prime-height specialization.  The point is not to choose
a Lee--Yang container around a zero island, but to identify a
source-side prime ledger through which a zeta packet could be compared
with the polarized container.

\begin{definition}[Squarefree prime address set]
\label{def:rh-squarefree-prime-address-set}
Let $p_1,\ldots,p_D$ be distinct primes.  The squarefree prime address
set generated by this window is
$$
\mathcal N_D^{\mathrm{sq}}
:=
\left\{
\prod_{j\in S}p_j
:
S\subseteq \{1,\ldots,D\}
\right\}.
$$
For $n\in\mathcal N_D^{\mathrm{sq}}$, let
$$
\omega_D(n)
:=
\#\{j:p_j\mid n\}.
$$
\end{definition}

\begin{definition}[Squarefree prime packet]
\label{def:rh-squarefree-prime-packet}
A \emph{squarefree prime packet} on the prime window
$p_1,\ldots,p_D$ is a Dirichlet packet of the form
$$
S_D(s)
:=
\sum_{n\in\mathcal N_D^{\mathrm{sq}}}
b(n)n^{1/2-s}.
$$
The coefficients $b(n)$ are part of the source ledger.  They are not
chosen after a zero island is selected unless a separate capture row
records the choice and its boundary margin.
\end{definition}

\begin{definition}[Prime-height polarized readback]
\label{def:rh-prime-height-polarized-readback}
Let
$$
\mathcal Z^{\mathrm{pol}}(z_1,\ldots,z_D)
=
\sum_{k=0}^{D}
a_k\binom{D}{k}^{-1}
e_k(z_1,\ldots,z_D)
$$
be the polarization of a one-variable Lee--Yang seed.  Its
\emph{prime-height readback} is obtained by setting
$$
z_j(s)=e^{-(s-1/2)\log p_j}.
$$
Then
$$
\mathcal Z^{\mathrm{pol}}(z_1(s),\ldots,z_D(s))
=
\sum_{S\subseteq\{1,\ldots,D\}}
a_{|S|}\binom{D}{|S|}^{-1}
\left(\prod_{j\in S}p_j\right)^{1/2-s}.
$$
Thus the prime-height readback is a squarefree prime packet whose
coefficient depends only on the prime-support cardinality.
\end{definition}

\begin{principle}[Polarization enters the prime ledger]
\label{prin:rh-polarization-enters-prime-ledger}
The polarized golden container is not merely a formal multiaffine
replacement for a one-variable seed.  Under the specialization
$h_j=\log p_j$, every elementary symmetric term becomes a squarefree
prime monomial:
$$
e^{-(s-1/2)\sum_{j\in S}\log p_j}
=
\left(\prod_{j\in S}p_j\right)^{1/2-s}.
$$
Consequently, the bridge from the zeta packet to the polarized
Lee--Yang container factors through the arithmetic surface
$$
\text{zeta packet}
\quad\rightsquigarrow\quad
\text{squarefree prime packet}
\quad\rightsquigarrow\quad
\text{polarized Lee--Yang container}.
$$
The first arrow is analytic and zero-island dependent; the second arrow
is algebraic or boundary-small once the prime window and seed
coefficients have been fixed.
\end{principle}

\begin{definition}[Squarefree Rouch\'e dominance]
\label{def:rh-squarefree-rouche-dominance}
Let $P_m^{\zeta}(s)$ be a zero-relevant zeta golden packet on a zero
island $D_m$.  The packet satisfies
$$
\mathsf{SquarefreeRoucheDominance}
$$
if there exist a prime window $p_1,\ldots,p_D$ and a squarefree prime
packet $S_D(s)$ such that
$$
|P_m^{\zeta}(s)-S_D(s)|
<
|S_D(s)|
\qquad
s\in\partial D_m.
$$
Equivalently, $P_m^{\zeta}$ and $S_D$ have the same zero count in
$D_m$ by Rouch\'e's theorem.  The packet $S_D$ must be supplied by a
source ledger rather than by an unrecorded interpolation around the
candidate zero.
\end{definition}

\begin{definition}[Polarized coefficient matching]
\label{def:rh-polarized-coefficient-matching}
A squarefree prime packet $S_D$ satisfies
$$
\mathsf{PolarizedCoefficientMatching}
$$
if there is a one-variable Lee--Yang seed
$L(u)=\sum_{k=0}^D a_k u^k$ such that either
$$
b(n)=a_{\omega_D(n)}
\binom{D}{\omega_D(n)}^{-1}
\qquad
n\in\mathcal N_D^{\mathrm{sq}},
$$
or, for the corresponding polarized container
$\mathcal Z^{\mathrm{pol}}$, the boundary estimate
$$
|S_D(s)-\mathcal Z^{\mathrm{pol}}(z_1(s),\ldots,z_D(s))|
<
|\mathcal Z^{\mathrm{pol}}(z_1(s),\ldots,z_D(s))|
\qquad
s\in\partial D_m
$$
holds on the zero island under the prime-height specialization.
\end{definition}

\begin{theorem}[Squarefree bridge gives polarized transport]
\label{thm:rh-squarefree-bridge-gives-polarized-capture}
Let a zero-relevant zeta golden packet $P_m^{\zeta}$ on an island $D_m$
satisfy $\mathsf{SquarefreeRoucheDominance}$ through $S_D$.  If $S_D$
satisfies $\mathsf{PolarizedCoefficientMatching}$ through a polarized
golden Lee--Yang container, then $P_m^{\zeta}$ and the prime-height
readback of that container have the same zero-island winding on $D_m$.
\end{theorem}

\begin{proof}
The first row gives a strict Rouch\'e inequality between
$P_m^{\zeta}$ and $S_D$ on $\partial D_m$, hence the same zero count on
$D_m$.  The second row either identifies $S_D$ with the prime-height
readback of a polarized container, or supplies a second strict
Rouch\'e inequality between $S_D$ and that readback on the same
boundary.  Chaining the two boundary nonvanishing comparisons gives the
same zero-island winding for $P_m^{\zeta}$ and the polarized
Lee--Yang container.
\end{proof}

\begin{corollary}[Squarefree bridge gives fixed-half readout]
\label{cor:rh-squarefree-bridge-fixed-half-readout}
Under the hypotheses of
\autoref{thm:rh-squarefree-bridge-gives-polarized-capture}, every zero
captured by the island $D_m$ has readout with real part $1/2$.
\end{corollary}

\begin{proof}
By \autoref{thm:rh-squarefree-bridge-gives-polarized-capture}, the
zeta packet and the prime-height polarized container have the same
zero-island winding on $D_m$.  If a captured readout had real part
different from $1/2$, then all prime-height variables would lie
strictly inside the unit disk or strictly outside it.  The polarized
container has no zero on either region by
\autoref{prin:rh-polarization-supplies-container-row}.  Hence the
container has zero winding on that off-line island, and the same is true
of $P_m^{\zeta}$, contradicting capture.  Therefore the readout has real
part $1/2$.
\end{proof}

\begin{principle}[Analytic load of the squarefree bridge]
\label{prin:rh-analytic-load-squarefree-bridge}
The polarization and coefficient-matching rows are structural: after a
one-variable Lee--Yang seed and a prime window are fixed, their
prime-height readback is a squarefree prime packet by construction.
The analytic burden is the dominance row
$$
\mathsf{SquarefreeRoucheDominance}.
$$
It asks whether the zero-relevant zeta packet can be compared, with a
strict boundary gap, to a packet supported on squarefree products of a
finite prime window.  Prime powers and the non-squarefree tail must
therefore enter only through a controlled remainder, a reciprocal
Euler-product contour row, or an explicit-formula tail ledger.  Without
such a ledger, squarefree support is only a formal address surface and
does not supply the local Rouch\'e bridge.
\end{principle}
