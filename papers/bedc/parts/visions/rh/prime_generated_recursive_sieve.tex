\chapter{Prime-Generated Recursive Sieve}
\label{ch:visions-rh-prime-generated-recursive-sieve}
\origin{ai}

\autoref{ch:visions-rh-prime-sieve-tower-duality} treats primes as both
objects of classification and generators of local sieve surfaces.  The
present chapter isolates the more basic recursive package.  It starts
from the local residue classifier attached to a prime, iterates such
classifiers as a sieve tower, and reads the right-half-plane Euler
product as the phase expectation of the resulting prime-generated
probability system.

\section{Recursive Local Sieve}
\label{sec:rh-prime-generated-recursive-local-sieve}

\begin{definition}[Prime-generated recursive sieve step]
\label{def:rh-prime-generated-recursive-sieve-step}
Let $X_m$ be a displayed source set, let $p_m$ be a prime, and let
$$
\pi_{p_m}:X_m\to G_{p_m}
$$
be a finite classifier generated by $p_m$, such as a residue,
unit-residue, valuation, order, or character readout.  For an event
$A_m\subseteq G_{p_m}$, define
$$
C_m:=\{x\in X_m:\pi_{p_m}(x)\in A_m\},
$$
$$
X_{m+1}:=X_m\setminus C_m,
$$
and append the ledger row
$$
\Lambda_{m+1}
:=
\Lambda_m+
(p_m,A_m,\mu_{p_m}(A_m),C_m,\operatorname{reason}_m).
$$
\end{definition}

\begin{definition}[PrimeGeneratedRecursiveSieve]
\label{def:rh-prime-generated-recursive-sieve}
The row
$$
\mathsf{PrimeGeneratedRecursiveSieve}
$$
is an iterated system of
\autoref{def:rh-prime-generated-recursive-sieve-step} whose source
sets may be integers, primes, prime powers, finite residual supports,
or sieve-method parameter spaces.  Each step is generated by a prime
and must expose its classifier, event, probability mass, and ledger
append row.
\end{definition}

\begin{theorem}[Recursive prime-generated sieve preserves BEDC rows]
\label{thm:rh-recursive-prime-generated-sieve-preserves-bedc-rows}
Every finite stage of $\mathsf{PrimeGeneratedRecursiveSieve}$ preserves
the BEDC time, unit, and ledger rows.
\end{theorem}

\begin{proof}
The time row is the nested chain
$$
X_0\supseteq X_1\supseteq X_2\supseteq\cdots
$$
together with the append-only ledger equations in
\autoref{def:rh-prime-generated-recursive-sieve-step}.  The unit row is
the probability-amplitude split on each finite classifier:
$$
\psi_m=P_{A_m}\psi_m+P_{A_m^c}\psi_m,
\qquad
P_{A_m}\psi_m\perp P_{A_m^c}\psi_m,
$$
so
$$
\|\psi_m\|^2
=
\|P_{A_m}\psi_m\|^2+\|P_{A_m^c}\psi_m\|^2.
$$
This is the event projection law of
\autoref{thm:rh-event-projection-pythagorean-law}.  The ledger row is
the displayed $\Lambda_{m+1}$ update, which records the selected prime,
event, probability mass, excluded source, and reason.
\end{proof}

\section{Finite Residual Detection}
\label{sec:rh-prime-generated-recursive-finite-detection}

\begin{theorem}[Prime-generated sieve detects finite residuals]
\label{thm:rh-prime-generated-recursive-sieve-detects-finite-residuals}
Let
$$
R=\sum_{j=1}^{N}c_j[q_j]\ne0
$$
be a finite residual with distinct positive integers $q_j$.  Then there
are a prime $p$ and a character
$$
\chi:\mathbb F_p^\times\to S^1
$$
such that
$$
\sum_{j=1}^{N}c_j\chi(q_j\bmod p)\ne0.
$$
\end{theorem}

\begin{proof}
Choose a prime $p$ dividing none of the $q_j$ and none of the nonzero
differences $q_i-q_j$.  Then the residues $q_j\bmod p$ are pairwise
distinct elements of $\mathbb F_p^\times$.  If all characters of
$\mathbb F_p^\times$ had zero readout on $R$, character orthogonality
on the finite abelian group $\mathbb F_p^\times$ would force the
coefficient sum on each residue class to vanish.  Since the displayed
residues are pairwise distinct, every $c_j$ would be zero, contradicting
$R\ne0$.
\end{proof}

\begin{corollary}[Prime-generated recursive completeness]
\label{cor:rh-prime-generated-recursive-completeness}
Every nonzero finite residual is detected by a local sieve generated by
some prime.  Equivalently,
$$
R\ne0
\quad\Rightarrow\quad
\exists p,\exists \chi\in\widehat{\mathbb F_p^\times},
\quad
\langle R,\chi\circ\pi_p\rangle\ne0.
$$
\end{corollary}

\begin{proof}
Use
\autoref{thm:rh-prime-generated-recursive-sieve-detects-finite-residuals}
with $\pi_p(q)=q\bmod p$.
\end{proof}

\section{Euler Probability Readout}
\label{sec:rh-prime-generated-recursive-euler-probability}

\begin{definition}[Right-half-plane Euler probability law]
\label{def:rh-right-half-plane-euler-probability-law}
For $\sigma>1$, define a probability law on positive integers by
$$
\mathbb P_{\sigma}(n)
:=
\frac{n^{-\sigma}}{\zeta(\sigma)}.
$$
Under the prime factorization
$$
n=\prod_p p^{v_p(n)},
$$
the local valuation row at prime $p$ has distribution
$$
\mathbb P_{\sigma}(v_p=k)
=
(1-p^{-\sigma})p^{-k\sigma}.
$$
\end{definition}

\begin{theorem}[Euler probability factors by prime valuations]
\label{thm:rh-euler-probability-factors-by-prime-valuations}
For $\sigma>1$, the valuation rows $v_p$ are independent under
$\mathbb P_{\sigma}$, with the local distribution of
\autoref{def:rh-right-half-plane-euler-probability-law}.
\end{theorem}

\begin{proof}
For a finite prime window $S$ and exponents $k_p\ge0$,
$$
\mathbb P_{\sigma}(v_p=k_p\text{ for }p\in S)
=
\frac{1}{\zeta(\sigma)}
\left(\prod_{p\in S}p^{-k_p\sigma}\right)
\prod_{p\notin S}\left(\sum_{k\ge0}p^{-k\sigma}\right).
$$
Using the Euler product on $\sigma>1$, the terms outside $S$ cancel the
corresponding factors of $\zeta(\sigma)$, leaving
$$
\prod_{p\in S}(1-p^{-\sigma})p^{-k_p\sigma}.
$$
This is the product of the displayed local laws, hence the valuation
rows are independent.
\end{proof}

\begin{theorem}[Zeta ratio is the phase expectation]
\label{thm:rh-zeta-ratio-phase-expectation}
For $\sigma>1$ and real $t$,
$$
\mathbb E_{\mathbb P_{\sigma}}[n^{-it}]
=
\frac{\zeta(\sigma+it)}{\zeta(\sigma)}.
$$
\end{theorem}

\begin{proof}
By definition of $\mathbb P_{\sigma}$,
$$
\mathbb E_{\mathbb P_{\sigma}}[n^{-it}]
=
\sum_{n\ge1}\frac{n^{-\sigma}}{\zeta(\sigma)}n^{-it}
=
\frac{1}{\zeta(\sigma)}
\sum_{n\ge1}n^{-(\sigma+it)}
=
\frac{\zeta(\sigma+it)}{\zeta(\sigma)}.
$$
Absolute convergence is available because $\sigma>1$.
\end{proof}

\begin{definition}[Finite Euler probability phase packet]
\label{def:rh-finite-euler-probability-phase-packet}
For a finite prime window $S=\{p_1,\ldots,p_m\}$, define
$$
\mathcal Z_S(\sigma,t)
:=
\prod_{p\in S}
\frac{1-p^{-\sigma}}{1-p^{-\sigma-it}}.
$$
This is the phase expectation of the finite product of local geometric
valuation laws generated by primes in $S$.
\end{definition}

\begin{theorem}[Finite Euler packets converge to the zeta phase readout]
\label{thm:rh-finite-euler-packets-converge-zeta-phase-readout}
For $\sigma>1$ and cofinal finite prime windows $S$,
$$
\mathcal Z_S(\sigma,t)
\to
\frac{\zeta(\sigma+it)}{\zeta(\sigma)}.
$$
\end{theorem}

\begin{proof}
The finite product in
\autoref{def:rh-finite-euler-probability-phase-packet} is the quotient
of the finite Euler products for $\zeta(\sigma+it)$ and
$\zeta(\sigma)$ over $S$.  Both Euler products converge absolutely on
$\sigma>1$, so their quotient converges to the displayed zeta ratio.
\end{proof}

\begin{principle}[Right-half-plane zeta is a prime-generated phase limit]
\label{prin:rh-right-half-plane-zeta-prime-generated-phase-limit}
On $\Re(s)>1$, the normalized zeta phase
$\zeta(\sigma+it)/\zeta(\sigma)$ is already a legal inverse limit of
prime-generated local probability sieves.  This statement uses only
unique factorization, geometric valuation laws, and absolute
convergence of the Euler product.
\end{principle}

\section{Continuation Interface}
\label{sec:rh-prime-generated-recursive-continuation-interface}

\begin{definition}[AnalyticContinuationPreservesPrimeSieveLedger]
\label{def:rh-analytic-continuation-preserves-prime-sieve-ledger}
The row
$$
\mathsf{AnalyticContinuationPreservesPrimeSieveLedger}
$$
asserts that the analytic continuation and explicit-formula transport
from $\Re(s)>1$ to the critical strip preserves the prime-generated
sieve ledger: finite prime windows, local valuation rows, unit phase
readouts, residual projections, and excluded-mass ledger entries remain
visible as certified rows of the continued zeta object.
\end{definition}

\begin{theorem}[Prime-generated recursive sieve gives conditional fixed-half route]
\label{thm:rh-prime-generated-recursive-sieve-conditional-fixed-half-route}
Assume $\mathsf{AnalyticContinuationPreservesPrimeSieveLedger}$, every
off-critical zero supplies a nonzero prime-side residual visible at a
finite layer, and detected residuals cannot be hidden in legal closed
BEDC/PFZ zero packets.  Then every admitted nontrivial zero has
fixed-half readout.
\end{theorem}

\begin{proof}
An off-critical admitted zero supplies, by hypothesis, a nonzero
finite-layer prime-side residual.  By
\autoref{cor:rh-prime-generated-recursive-completeness}, some
prime-generated local sieve detects it.  The continuation-preservation
row keeps this detection inside the certified zeta ledger rather than
outside the continued object.  If the detected residual is recorded, the
zero packet carries an append ledger and is not closed.  If it is not
recorded, the ledger policy is violated.  Thus a legal closed admitted
zero cannot be off-critical, and
\autoref{thm:rh-li-coordinate-fixed-half-circle} gives fixed-half
readout.
\end{proof}

\begin{principle}[Recursive sieve audit boundary]
\label{prin:rh-prime-generated-recursive-sieve-audit-boundary}
The recursive prime-generated sieve packet proves the finite arithmetic
side: local prime sieves are BEDC-legal, detect finite residuals, and
construct the normalized Euler phase on $\Re(s)>1$.  The remaining
analytic burden is the preservation row
$\mathsf{AnalyticContinuationPreservesPrimeSieveLedger}$, together with
the explicit-formula row that turns an off-critical zero into a
prime-side residual.
\end{principle}
