\begin{definition}[TriadicGenerationInduction]
\label{def:rh-triadic-generation-induction}
The obligation
$$
\mathsf{TriadicGenerationInduction}
$$
is the finite structural induction principle for the
circle--triangle--register grammar. Its base rows are finite
distinctions, unit phase, and append-only histories. Its constructors are
unit-circle phase readout, orthogonal triangular decomposition, and
finite residual registration. Its conclusion is that every certified
object generated by this grammar carries the triadic invariant
$$
\mathcal I_{\triangle}
=
\mathcal I_{\mathsf P}
\wedge
\mathcal I_{\mathsf U}
\wedge
\mathcal I_{\mathsf T}.
$$
The grammar-specific induction theorem is
\autoref{thm:unified-reality-circle-triangle-register-generation-induction},
and its abstract foundation is
\autoref{thm:unified-reality-triadic-structural-induction}. The open
part is not the induction rule, but the coverage of zeta by the
constructor grammar.
\end{definition}

\begin{definition}[NoHiddenRadialRegister]

\begin{definition}[ZetaSourceConstruction]

\xi(s)
=

$$

\begin{theorem}[Direct sums preserve source Pick positivity]
\label{thm:rh-direct-sums-preserve-source-pick-positivity}
Let $K_X$ and $K_Y$ be positive kernels with source-built Hilbert
factorizations
$$
K_X(z,w)=\langle\Phi_X(z),\Phi_X(w)\rangle_{\mathcal H_X},
\qquad
K_Y(z,w)=\langle\Phi_Y(z),\Phi_Y(w)\rangle_{\mathcal H_Y}.
$$
Then
$$
K_X(z,w)+K_Y(z,w)
$$
has a source-built Hilbert factorization on
$\mathcal H_X\oplus\mathcal H_Y$.
\end{theorem}

\begin{proof}
Define
$$
\Phi(z):=\Phi_X(z)\oplus\Phi_Y(z).
$$
Then
$$
\langle\Phi(z),\Phi(w)\rangle_{\mathcal H_X\oplus\mathcal H_Y}
=
\langle\Phi_X(z),\Phi_X(w)\rangle_{\mathcal H_X}
+
\langle\Phi_Y(z),\Phi_Y(w)\rangle_{\mathcal H_Y}
=
K_X(z,w)+K_Y(z,w).
$$
The feature map is source-built from the two source-built feature maps by
the displayed direct-sum constructor.
\end{proof}

\begin{theorem}[Finite-window limits preserve positive kernels]
\label{thm:rh-finite-window-limits-preserve-positive-kernels}
Let $(K_m)$ be positive kernels on a domain $D$ and suppose
$$
K_m(z,w)\to K(z,w)
$$
for every displayed finite pair $z,w\in D$. Then $K$ is positive
definite.
\end{theorem}

\begin{proof}
Let $z_1,\ldots,z_N\in D$ and $c_1,\ldots,c_N\in\mathbb C$ be displayed.
For every $m$,
$$
\sum_{i,j=1}^{N}
c_i\overline{c_j}K_m(z_i,z_j)\ge0
$$
because $K_m$ is positive definite. Taking the finite sum through the
pointwise limit gives
$$
\sum_{i,j=1}^{N}
c_i\overline{c_j}K(z_i,z_j)\ge0.
$$
Since the finite window was arbitrary, $K$ is positive definite.
\end{proof}

\begin{principle}[Analytic continuation is the remaining constructor risk]
\label{prin:rh-analytic-continuation-remaining-constructor-risk}
The direct-sum and finite-window limit constructors above are
triadic-preserving once their source rows are displayed. The remaining
constructor risk in $\mathsf{ConstructorTriadicPreservation}_{\xi}$ is
the combined Euler-product, Gamma-completion, and analytic-continuation
step: it must preserve source-side Pick, Herglotz, or Hausdorff
positivity, not merely equality of meromorphic functions.
\end{principle}

\section{Single-Mode Source Bridge}
\label{sec:rh-radial-single-mode-source-bridge}

\begin{definition}[Prime exponent ledger for a zeta monomial]
\label{def:rh-prime-exponent-ledger-zeta-monomial}
For a positive integer $n$, the prime exponent ledger is the finite
support function
$$
\upsilon(n):=(v_p(n))_p,
\qquad
n=\prod_p p^{v_p(n)}.
$$
Its logarithmic height is
$$
h_n:=\log n=\sum_p v_p(n)\log p.
$$
\end{definition}

\begin{theorem}[A zeta monomial has a prime-ledger phase readout]
\label{thm:rh-zeta-monomial-prime-ledger-phase-readout}
For every displayed positive integer $n$ and real phase time $t$,
$$
n^{-it}
=
e^{-it\log n}
=
\exp\left(
-it\sum_p v_p(n)\log p
\right).
$$
\end{theorem}

\begin{proof}
The first equality is the definition of the exponential logarithmic
readout for the positive real $n$. The second equality substitutes the
prime exponent ledger identity
$$
\log n=\sum_p v_p(n)\log p
$$
from \autoref{def:rh-prime-exponent-ledger-zeta-monomial}. Since the
ledger has finite support, the displayed sum is finite.
\end{proof}

\begin{principle}[Status of the single-mode bridge]
\label{prin:rh-single-mode-bridge-status}
\autoref{thm:rh-zeta-monomial-prime-ledger-phase-readout} closes only the
single-mode bridge. It proves that one Dirichlet monomial has continuous
unit-phase readout and discrete prime-register provenance. It does not
prove the global
$\mathsf{ContinuousDiscreteBridge}$ obligation for completed zeta zero
packets, where functional-equation transport, analytic continuation,
explicit-formula rows, and cofinal tower coverage must share the same
source certificate.
\end{principle}

\section{Invariant Separation}
\label{sec:rh-radial-invariant-separation}

\begin{definition}[Expressible but not generated]
\label{def:rh-expressible-not-generated}
An object is \emph{expressible but not generated} relative to a BEDC
grammar when its external notation is well formed, but no derivation from
the declared initial packets and constructors reaches a certificate for
that object. The distinction is semantic for this chapter: a complex
coordinate, a formal zero packet, or a radial parameter may be written in
ambient notation without being admitted by the circle--triangle--register
construction grammar.
\end{definition}

\begin{definition}[Hidden scale energy]
\label{def:rh-hidden-scale-energy}
For a proposed pure phase closure packet, let
$$
\ell_1,\ldots,\ell_N
$$
be the displayed radial scale legs not recorded as scale-ledgered rows.
Its hidden scale energy is
$$
\mathcal E_{\mathrm{scale}}
:=
\sum_{i=1}^{N}\ell_i^2.
$$
For an iterated Li observation, the corresponding finite observation
energy at level $n$ is
$$
\mathcal E_{\mathrm{scale}}(n)
:=
\sum_{i=1}^{N}(e^{n\ell_i}-1)^2.
$$
A pure phase closure is scale-clean when
$$
\mathcal E_{\mathrm{scale}}=0.
$$
\end{definition}

\begin{principle}[Scale energy is not cancelled by mirror signs]
\label{prin:rh-scale-energy-not-cancelled-by-mirror-signs}
The obstruction is quadratic rather than additive. Opposite radial
parameters may have zero algebraic sum,
$$
\ell+(-\ell)=0,
$$
but their hidden scale energy is
$$
\ell^2+(-\ell)^2=2\ell^2.
$$
Thus a radial leg cannot be erased by pairing it with an opposite radial
leg inside a pure phase closure. It must either be absent, or be recorded
as explicit scale-ledger data.
\end{principle}

\begin{theorem}[Scale-clean invariant for generated pure closures]
\label{thm:rh-scale-clean-invariant-generated-pure-closures}
Every pure closure generated by the circle--triangle--register grammar
and satisfying the no-hidden-scale ledger rule has
$$
\mathcal E_{\mathrm{scale}}=0.
$$
\end{theorem}

\begin{proof}
Proceed by structural induction on the generating certificate. The
initial distinction split has only append-only history and orthogonal
residual rows. Any residual used by the packet is recorded in the
ledger, so it is not an unrecorded radial scale leg. A circle constructor
adds a unit phase $e^{i\theta}$ and hence has radial coordinate
$\log|e^{i\theta}|=0$. A triangle-readout constructor decomposes a
readout into an existing projection and an orthogonal residual; if that
residual is used, the register rule records it, and if it is not
recorded the packet is not legal. The register constructor converts a
displayed residual into ledger data and therefore does not create hidden
scale. Hence no generated pure closure contains an unrecorded nonzero
radial leg. The hidden scale energy is a sum of squares over an empty or
zero family, so it is $0$.
\end{proof}

\begin{theorem}[Off-line Li coordinates are not generated pure closures]
\label{thm:rh-off-line-li-coordinate-not-generated-pure-closure}
Let a completed zero candidate have Li coordinate
$$
z_{\rho}=1-\frac1{\rho}=e^{\ell+i\theta}.
$$
If $\ell\#0$, then the candidate cannot be a generated pure
circle--triangle--register closure satisfying the no-hidden-scale ledger
rule.
\end{theorem}

\begin{proof}
For any displayed positive observation level $n$, the Li defect
decomposition of
\autoref{thm:rh-li-observation-defect-scale-phase-split} contains the
radial square
$$
(e^{n\ell}-1)^2.
$$
When $\ell\#0$, this is a nonzero scale leg for some displayed
observation. If it is recorded, the packet is scale-ledgered and not a
pure phase closure by
\autoref{thm:rh-recorded-radial-scale-changes-closure-type}. If it is
not recorded, it contributes positive hidden scale energy, contradicting
\autoref{thm:rh-scale-clean-invariant-generated-pure-closures}. Hence
the candidate is expressible in ambient notation but not generated as a
legal pure closure by the stated grammar.
\end{proof}

\section{Compact Gauge Obstruction}
\label{sec:rh-radial-compact-gauge-obstruction}

\begin{definition}[Compact legal gauge]
\label{def:rh-compact-legal-gauge}
A legal gauge row $K$ is \emph{compact} when it is a compact topological
group and all gauge readouts used by a pure phase closure are continuous
group homomorphisms from $K$ to the displayed target group. Unit phase
containers and finite-prime solenoid fibres are read as compact legal
gauges at the scope where their compactness row is displayed.
\end{definition}

\begin{theorem}[Compact gauges have no nontrivial radial character]
\label{thm:rh-compact-gauge-no-nontrivial-radial-character}
Let $K$ be a compact legal gauge and let
$$
\phi:K\to\mathbb R_{>0}
$$
be a continuous group homomorphism, with $\mathbb R_{>0}$ under
multiplication. Then
$$
\phi(K)=\{1\}.
$$
\end{theorem}

\begin{proof}
The image $H:=\phi(K)$ is compact because $K$ is compact and $\phi$ is
continuous. It is a subgroup of $\mathbb R_{>0}$ because $\phi$ is a
group homomorphism. Apply the logarithm isomorphism
$$
\log:\mathbb R_{>0}\to\mathbb R.
$$
Then $\log H$ is a compact subgroup of the additive group
$\mathbb R$. If $x\in\log H$ and $x\#0$, then all integer multiples
$$
nx
\qquad
(n\in\mathbb Z)
$$
lie in $\log H$, because $\log H$ is a subgroup. This set is unbounded
in $\mathbb R$, contradicting compactness. Hence every element of
$\log H$ is $0$, so $\log H=\{0\}$ and therefore $H=\{1\}$.
\end{proof}

\begin{corollary}[Compact phase and prime gauges cannot hide scale]
\label{cor:rh-compact-phase-prime-gauges-cannot-hide-scale}
If the legal hidden row of a pure zeta phase closure is compact, then no
nontrivial radial factor $e^{\ell}$ with $\ell\#0$ can be supplied by
that hidden row.
\end{corollary}

\begin{proof}
A hidden compact gauge would supply the radial factor through a
continuous character
$$
\phi:K\to\mathbb R_{>0}.
$$
By \autoref{thm:rh-compact-gauge-no-nontrivial-radial-character}, the
image of $\phi$ is $\{1\}$. Thus every radial factor obtained from the
compact gauge is trivial.
\end{proof}

\section{Ledgered Scale Is Not Pure Closure}
\label{sec:rh-radial-ledgered-scale-not-pure-closure}

\begin{definition}[Scale-ledgered closure]
\label{def:rh-scale-ledgered-closure}
A zeta observation packet is \emph{scale-ledgered} when its ledger
contains a nontrivial radial row
$$
\ell\#0
$$
or the corresponding positive factor $e^{\ell}$. A scale-ledgered packet
may be a legal object with an explicit residual, but it is not a pure
unit-phase closure.
\end{definition}

\begin{theorem}[Recorded radial scale changes the closure type]
\label{thm:rh-recorded-radial-scale-changes-closure-type}
If a zeta observation packet records a radial row $\ell\#0$, then it is
not a pure unit-phase closure.
\end{theorem}

\begin{proof}
By \autoref{def:rh-scale-ledgered-closure}, recording $\ell\#0$ makes the
packet scale-ledgered. A pure unit-phase closure has only unit phase,
orthogonal unit readout, and legal compact or finite ledgers as hidden
rows; it has no nontrivial positive-real radial row. Therefore a packet
with recorded radial scale has changed closure type.
\end{proof}

\begin{principle}[Two ways a radial factor is refused]
\label{prin:rh-two-ways-radial-factor-refused}
The radial obstruction has two independent parts. If the radial factor is
sent through compact phase or compact prime gauge, it is trivial by
\autoref{thm:rh-compact-gauge-no-nontrivial-radial-character}. If it is
recorded as ledger data, it is no longer pure phase closure by
\autoref{thm:rh-recorded-radial-scale-changes-closure-type}. The
remaining open content of $\mathsf{NoHiddenRadialRegister}$ is the global
classification that every legal zeta gauge is either compact in this
sense or ledger-explicit in this sense.
\end{principle}

\section{Conditional Core}
\label{sec:rh-radial-register-conditional-core}

\begin{theorem}[Radial obstruction package is conditionally RH-sufficient]
\label{thm:rh-radial-obstruction-package-conditionally-sufficient}
Assume
$$
\mathsf{ZetaBEDCEmbedding},
\qquad
\mathsf{NoHiddenRadialRegister},
\qquad
\mathsf{CofinalTriangularTower},
$$
together with the support rows of
\autoref{def:rh-radial-projection-tower-support-obligations}. Then the
BEDC RH section target is inhabited.
\end{theorem}

\begin{proof}
The embedding row supplies, for each completed non-trivial zero packet
$\rho$, a circle--triangle--register observation with Li coordinate
$$
z_{\rho}=1-\frac1{\rho}=e^{\ell+i\theta}.
$$
If $\ell\#0$, the radial leg is nontrivial. The no-hidden-radial-register
row says that this leg is neither unit phase nor a legal hidden
prime-register gauge and cannot disappear as unledgered finite data. The
projection faithfulness support row exposes the same radial leg through
the Li, Weil, and Nyman readouts; the cofinal triangular tower prevents
it from being postponed to the infinite tail; diagonal safety prevents
posterior repair by a zero-dependent oracle. Thus a legal pure zeta
closure cannot have $\ell\#0$. The located readback gives $\ell=0$.
By \autoref{thm:rh-li-circle-criterion},
$$
\left|1-\frac1{\rho}\right|=1
\Longleftrightarrow
\Re(\rho)=\frac12.
$$
Since the zero packet was arbitrary, the RH section target is inhabited.
\end{proof}

\begin{principle}[Status of the radial obstruction frontier]
\label{prin:rh-radial-obstruction-frontier-status}
The compact-gauge obstruction and the single-mode prime-ledger bridge are
closed local rows. They do not close the zeta embedding, projection
faithfulness, or cofinal tower obligations. The remaining core is the
global theorem that every zero-relevant zeta gauge is compact or
ledger-explicit, and that the finite triangular tower covers the
completed zero ledger without an oracle-dependent repair step.
\end{principle}
