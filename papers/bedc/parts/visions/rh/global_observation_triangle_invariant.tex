\begin{definition}[Global zeta observation triangle]
\label{def:rh-global-zeta-observation-triangle}
The source row
$$
\mathsf{GlobalObservationTriangle}_{\xi}
$$
consists of four compatible pieces:
$$
\mathsf{Src}_{\xi},
\qquad
\mathsf{Pos}_{\xi},
\qquad
\mathsf{Leg}_{\xi},
\qquad
\mathsf{Exh}_{\xi}.
$$
The source piece $\mathsf{Src}_{\xi}$ constructs the kernels
$K_m$ and maps $\Phi_m$ from prime, gamma, normalization, solenoid,
golden, explicit-formula, or Stieltjes rows before the zero divisor is
read.  The positive piece $\mathsf{Pos}_{\xi}$ is the finite Hilbert
square identity
$$
K_m(z,w)=\langle\Phi_m(z),\Phi_m(w)\rangle_{\mathcal H_m}
$$
together with finite-matrix convergence to $K_{\xi}$.  The same-leg
piece $\mathsf{Leg}_{\xi}$ says that the Li, Weil, Nyman, and Pick
readouts use one radial scalar
$$
\ell(\rho)=\log\left|1-\frac1\rho\right|.
$$
The exhaustion piece $\mathsf{Exh}_{\xi}$ says that residual Hilbert
energy is either exhausted by the observation tower or displayed by a
separating functional.
\end{definition}

\begin{definition}[Legal global observation]
\label{def:rh-legal-global-observation}
A zero-relevant observation row $\mathcal O$ is legal for the global
triangle when it factors through the source row of
\autoref{def:rh-global-zeta-observation-triangle}.  Thus $\mathcal O$
is read either as a finite positive-kernel observation, as a limit of
such observations on a finite matrix family, or as one of the displayed
projection readouts of the same finite-to-limit system.  A row that
uses a completed zero divisor to define its Hilbert space, measure, or
positive kernel is not a source-side legal observation for this
purpose.
\end{definition}

\begin{definition}[Global triangle invariant]
\label{def:rh-global-triangle-invariant}
The global triangle invariant is the requirement that every legal
zero-relevant observation satisfies the following three rows.

First, the ledger row permits compact phase and compact prime-register
fibres, as in \autoref{def:rh-phase-channel-hidden-prime-fibre}, but
does not permit an unrecorded $\mathbb R_{>0}$ radial gauge.

Second, the positive row represents observation energy by Hilbert
squares:
$$
\sum_{i,j}c_i\overline{c_j}K_m(z_i,z_j)
=
\left\|\sum_i c_i\Phi_m(z_i)\right\|_{\mathcal H_m}^2
\ge0.
$$

Third, the refinement row preserves finite matrix readouts and
orthogonal residuals under passage to the limit.  If a residual remains
nonzero, it must be read by a displayed separator rather than hidden at
the end of the tower.
\end{definition}

\begin{principle}[Common radial leg]
\label{prin:rh-global-common-radial-leg}
The Li, Weil, Nyman, Pick, Lee--Yang, and endpoint rows may have
different finite presentations, but their off-fixed-half obstruction
must read the same radial scalar.  In Li time it appears as the radial
factor in $z(\rho)^n$.  In the Weil row it appears as a negative
direction for a reflected orbit.  In the Nyman row it appears as a
Mellin separator for an unexhausted residual.  In the Pick row it
appears as a finite matrix obstruction.  The global triangle invariant
requires these readings to be transports of one scale leg, not separate
defects that can be independently hidden.
\end{principle}

\begin{theorem}[Global triangle invariant gives Pick positivity]
\label{thm:rh-global-triangle-invariant-gives-pick-positivity}
Assume $\mathsf{GlobalObservationTriangle}_{\xi}$ and assume its finite
matrix convergence row holds for every displayed disk family.  Then
$$
\mathsf{TriadicPickPositivity}_{\xi}
$$
holds.
\end{theorem}

\begin{proof}
Fix a finite disk family $z_1,\ldots,z_N$ and coefficients
$c_1,\ldots,c_N$.  For every finite source level $m$, the positive
piece of \autoref{def:rh-global-zeta-observation-triangle} gives
$$
\sum_{i,j}c_i\overline{c_j}K_m(z_i,z_j)
=
\left\|\sum_i c_i\Phi_m(z_i)\right\|_{\mathcal H_m}^2
\ge0.
$$
The convergence row identifies the limit of the left side with
$$
\sum_{i,j}c_i\overline{c_j}K_{\xi}(z_i,z_j).
$$
Hence this latter finite quadratic form is nonnegative.  Since the
finite family was arbitrary, $K_{\xi}$ is positive definite.
\end{proof}

\begin{theorem}[Global observation triangle gives fixed-half readout]
\label{thm:rh-global-observation-triangle-fixed-half-readout}
Assume $\mathsf{GlobalObservationTriangle}_{\xi}$, the global triangle
invariant, common radial-leg transport, triangular residual exhaustion,
and the usual reflection consumption row for represented completed zero
ledgers.  Then every represented completed zero ledger legal for the
global triangle has fixed-half readout.
\end{theorem}

\begin{proof}
By
\autoref{thm:rh-global-triangle-invariant-gives-pick-positivity},
the global observation triangle supplies
$\mathsf{TriadicPickPositivity}_{\xi}$.  Let $\rho$ be a represented
completed zero ledger legal for the triangle.  Suppose
$$
\ell(\rho)\ne0.
$$
By common radial-leg transport, this nonzero scale leg must appear in
one of the displayed projection rows: as a finite Li-time witness, a
Weil negative direction, a Nyman residual separator, or a Pick finite
matrix obstruction.  The positive row forbids negative finite Hilbert
energy and Pick matrix obstructions.  The residual exhaustion row
forbids an unread Nyman residual; if such a residual remains, it is a
displayed separator and therefore not hidden by the tower.  The ledger
row forbids reclassifying the same radial scalar as a compact phase or
prime-register fibre.  Thus $\ell(\rho)\ne0$ is impossible.  Hence
$\ell(\rho)=0$, and
\autoref{thm:rh-radial-scale-fixed-half-surface} gives the fixed-half
readout.
\end{proof}

\begin{definition}[Zeta BEDC embedding row]
\label{def:rh-zeta-bedc-embedding-row}
The source row
$$
\mathsf{ZetaBEDCEmbedding}
$$
asserts that every represented critical-strip completed zero row
$\rho$ is presented by a BEDC legal observation certificate
$$
\mathcal O_\rho.
$$
The certificate carries a carrier coordinate, a classifier coordinate,
a ledger coordinate, the zero-row provenance, and the boundary rows
needed to compare its Li, Weil, Nyman, and Pick projections.  This row
is stronger than a statement-level encoding of the RH predicate: it
asserts that the analytic zero object has entered the BEDC observation
system with the source data required by
\autoref{def:rh-global-zeta-observation-triangle}.
\end{definition}

\begin{definition}[Zeta scale-leg coverage obstruction]
\label{def:rh-zeta-scale-leg-coverage-obstruction}
The source row
$$
\mathsf{ZetaScaleLegCoverageObstruction}
$$
asserts the following.  Let $\rho$ be a represented critical-strip
completed zero row with
$$
\ell(\rho)=\log\left|1-\frac1\rho\right|\ne0.
$$
Then the associated radial scale leg cannot be covered by any of the
three BEDC observation coordinates of
\autoref{def:rh-zeta-bedc-embedding-row}: it is not a legal carrier
coordinate, not a legal classifier coordinate, and not a legal ledger
endpoint coordinate.  It must therefore appear as an explicit defect
row in the global observation triangle, rather than as hidden source
structure.
\end{definition}

\begin{theorem}[Embedding plus scale obstruction gives fixed-half readout]
\label{thm:rh-embedding-scale-obstruction-fixed-half-readout}
Assume $\mathsf{ZetaBEDCEmbedding}$,
$$
\mathsf{ZetaScaleLegCoverageObstruction},
$$
and the global observation triangle hypotheses of
\autoref{thm:rh-global-observation-triangle-fixed-half-readout}.  Then
every represented critical-strip completed zero row in the embedding has
fixed-half readout.
\end{theorem}

\begin{proof}
Let $\rho$ be a represented critical-strip completed zero row.  The
embedding row supplies the BEDC legal observation certificate
$\mathcal O_\rho$ and its carrier, classifier, ledger, and boundary
coordinates.  If $\ell(\rho)\ne0$, the scale-leg obstruction row says
that this radial coordinate cannot be absorbed by any of the three BEDC
observation coordinates.  Hence it remains an explicit defect row of
the global observation triangle.  Applying
\autoref{thm:rh-global-observation-triangle-fixed-half-readout} to
$\mathcal O_\rho$ rules out that explicit defect.  Thus
$\ell(\rho)=0$, and
\autoref{thm:rh-radial-scale-fixed-half-surface} gives the fixed-half
readout.
\end{proof}

\begin{principle}[Boundary of the BEDC route]
\label{prin:rh-boundary-bedc-route}
The BEDC route does not consist merely of writing the RH statement in
BEDC syntax.  It requires the embedding row
$\mathsf{ZetaBEDCEmbedding}$ and the obstruction row
$\mathsf{ZetaScaleLegCoverageObstruction}$.  Only then can the abstract
kernel discipline of observation, distinction, time, carrier,
classifier, and ledger be consumed by the analytic zeta object.  The
non-circular source burden is to prove that a nonzero radial scale leg
cannot be one of the legal BEDC coordinates, rather than to assume
fixed-half readout in advance.
\end{principle}

\begin{definition}[BEDC formalization success for zeta]
\label{def:rh-bedc-formalization-success-zeta}
The source row
$$
\mathsf{BEDCFormalizationSuccess}_{\zeta}
$$
is the conjunction of four rows:
$$
\mathsf{BEDCSoundness},
\qquad
\mathsf{ZetaInterpretation},
\qquad
\mathsf{ZetaCoverage},
\qquad
\mathsf{GlobalTriadicPositivity}_{\xi}.
$$
Here $\mathsf{BEDCSoundness}$ says that BEDC legal objects satisfy the
carrier, classifier, ledger, unit-boundary, and generated-time
discipline.  The row $\mathsf{ZetaInterpretation}$ says that the
completed zeta carrier, its centered channels, and its zero-relevant
observations are interpreted inside that discipline.  The row
$\mathsf{ZetaCoverage}$ says that every represented non-trivial
critical-strip zero is covered by such a legal observation certificate.
The row $\mathsf{GlobalTriadicPositivity}_{\xi}$ says that the resulting
global observation kernels have a source-side Hilbert/Pick/Herglotz
positive representation.
\end{definition}

\begin{definition}[Strict sufficiency boundary]
\label{def:rh-strict-sufficiency-boundary}
The strict sufficiency boundary is the separation between two claims.
The weak claim is that BEDC supplies a formal language of observation,
distinction, generated time, and ledgered residuals.  The strong claim
is that $\mathsf{BEDCFormalizationSuccess}_{\zeta}$ holds.  Only the
strong claim can be consumed by the fixed-half route, because only it
asserts zeta interpretation, zero coverage, and global triadic
positivity in addition to the abstract BEDC discipline.
\end{definition}

\begin{theorem}[Formalization success supplies fixed-half readout]
\label{thm:rh-formalization-success-fixed-half-readout}
Assume $\mathsf{BEDCFormalizationSuccess}_{\zeta}$, the
scale-leg obstruction row
$$
\mathsf{ZetaScaleLegCoverageObstruction},
$$
and the usual reflection consumption row for represented completed zero
ledgers.  Then every represented non-trivial critical-strip zero has
fixed-half readout.
\end{theorem}

\begin{proof}
By $\mathsf{ZetaInterpretation}$ and $\mathsf{ZetaCoverage}$, each
represented non-trivial critical-strip zero is presented by a BEDC legal
observation certificate inside the global zeta observation triangle.  By
$\mathsf{GlobalTriadicPositivity}_{\xi}$, the triangle carries the
positive Hilbert/Pick/Herglotz row required by
\autoref{thm:rh-global-observation-triangle-fixed-half-readout}.  By
$\mathsf{BEDCSoundness}$, the carrier, classifier, ledger, unit-boundary,
and generated-time rows are legal BEDC rows.  The scale-leg obstruction
then prevents a nonzero radial scale from being absorbed by those legal
coordinates.  Applying
\autoref{thm:rh-embedding-scale-obstruction-fixed-half-readout} gives
fixed-half readout for the represented zero.
\end{proof}

\begin{principle}[Strong meaning of formalization success]
\label{prin:rh-strong-formalization-success-meaning}
The phrase ``formalization success'' is used in this route only in the
strong sense of
\autoref{def:rh-bedc-formalization-success-zeta}.  Formalizing the
abstract BEDC principles gives the invariant language, but it does not
by itself supply zeta interpretation, zero coverage, or the source-side
positive representation
$$
K_{\xi}(z,w)=\langle\Phi_{\xi}(z),\Phi_{\xi}(w)\rangle.
$$
The analytic core remains the construction of this positive global
observation system from source data rather than from a posterior
fixed-half zero divisor.
\end{principle}

\begin{principle}[Global observation source load]
\label{prin:rh-global-observation-source-load}
The mathematical load is concentrated in the construction of the source
piece $\mathsf{Src}_{\xi}$ and the convergence piece of
\autoref{def:rh-global-zeta-observation-triangle}.  The positive side
is then Hilbert-square formal, the time side is finite-matrix
witnessability, and the ledger side is the refusal of an unrecorded
real-positive scale gauge.  This is the global form of the triangle
route: every legal zeta observation is constrained by one positive
Hilbert--Pick--Herglotz system, and every nonzero radial scale must be
read rather than hidden.
\end{principle}
