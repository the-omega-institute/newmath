\chapter{Prime Sieve Tower Duality}
\label{ch:visions-rh-prime-sieve-tower-duality}
\origin{ai}

\autoref{ch:visions-rh-prime-sieve-duality-completeness} treats finite
type spaces and sieve readouts as dual coordinates.  The present
chapter records the generating side of that duality.  A prime is not
only an object to be classified by a sieve; it is also an atom that
creates further sieve surfaces.  Divisibility, residue class, finite
unit-group, order, and splitting readouts are all generated from prime
atoms, and their refinements preserve the BEDC time, unit, and ledger
rows.

\section{Prime-Generated Sieve Tower}
\label{sec:rh-prime-sieve-tower-generated}

\begin{definition}[Euclidean prime sieve tower]
\label{def:rh-euclidean-prime-sieve-tower}
Let $p_1,p_2,p_3,\ldots$ be the increasing sequence of primes and set
$$
P_m:=\prod_{j=1}^{m}p_j.
$$
The Euclidean prime sieve layer is
$$
\mathcal N_m
:=
\{n\ge2:\gcd(n,P_m)=1\}.
$$
Thus $\mathcal N_m$ is the set of integers not removed by the first $m$
prime divisibility tests.
\end{definition}

\begin{theorem}[Minimal survivor is a new prime]
\label{thm:rh-minimal-euclidean-survivor-new-prime}
If $\mathcal N_m$ is nonempty and $q=\min\mathcal N_m$, then $q$ is a
prime distinct from $p_1,\ldots,p_m$.
\end{theorem}

\begin{proof}
If $q$ were composite, write $q=ab$ with $1<a<q$ and $1<b<q$.  By
minimality of $q$ in $\mathcal N_m$, the integer $a$ is not in
$\mathcal N_m$.  Hence some $p_j$ with $j\le m$ divides $a$, and
therefore divides $q$.  This contradicts $\gcd(q,P_m)=1$.  Thus $q$ is
prime.  Since $q$ is coprime to $P_m$, it is not one of
$p_1,\ldots,p_m$.
\end{proof}

\begin{corollary}[Prime atoms extend the sieve tower]
\label{cor:rh-prime-atoms-extend-sieve-tower}
Adding the prime $p_{m+1}$ refines the survivor layer by
$$
\mathcal N_{m+1}
=
\mathcal N_m\cap\{n:p_{m+1}\nmid n\}.
$$
Hence each prime is both a survivor of previous sieve layers and a new
divisibility test for later layers.
\end{corollary}

\begin{proof}
The identity is the definition of coprimality with
$P_{m+1}=P_mp_{m+1}$.  The previous theorem identifies the next minimal
survivor as a new prime, and the displayed identity identifies its
corresponding divisibility test.
\end{proof}

\begin{principle}[Prime atoms are sieve generators]
\label{prin:rh-prime-atoms-are-sieve-generators}
A prime atom has a dual status in the sieve tower.  It is an arithmetic
object classified by earlier tests, and it generates later tests such
as divisibility by $p$, residue class modulo $p$, finite unit-group
readout in $\mathbb F_p^\times$, and order data in that unit group.  The
same atom therefore lives on both the source side and the method side of
the tower.
\end{principle}

\section{Local Prime Sieve Surfaces}
\label{sec:rh-prime-sieve-tower-local-surfaces}

\begin{definition}[Prime-generated local sieve]
\label{def:rh-prime-generated-local-sieve}
For a prime $p$, the prime-generated local sieve packet contains:
$$
\begin{gathered}
S_p^{\mathrm{div}}(n):=\mathbf 1_{p\mid n},\\
n\longmapsto n\bmod p,\qquad
\ell\longmapsto \ell\bmod p\quad(\ell\ne p),\\
\mathbb F_p^\times,\qquad
a\longmapsto \operatorname{ord}_p(a)\quad(p\nmid a).
\end{gathered}
$$
Each row is a finite classifier generated by the prime $p$.
\end{definition}

\begin{theorem}[Fermat unit row]
\label{thm:rh-fermat-unit-row}
If $p$ is prime and $p\nmid a$, then
$$
a^{p-1}\equiv1\pmod p.
$$
\end{theorem}

\begin{proof}
The nonzero residue classes modulo $p$ form the finite multiplicative
group $\mathbb F_p^\times$ of order $p-1$.  The order of the class of
$a$ divides the group order, giving the displayed congruence.
\end{proof}

\begin{corollary}[Fermat order sieve]
\label{cor:rh-fermat-order-sieve}
For fixed $a$ and $n$, the condition
$$
a^n\equiv1\pmod p,
\qquad
p\nmid a,
$$
is the same as
$$
\operatorname{ord}_p(a)\mid n.
$$
It is therefore a finite unit-group sieve on primes $p$.
\end{corollary}

\begin{proof}
In the finite group $\mathbb F_p^\times$, an element satisfies
$a^n=1$ exactly when its order divides $n$.
\end{proof}

\begin{principle}[Local prime sieves preserve unit readout]
\label{prin:rh-local-prime-sieves-preserve-unit-readout}
The local sieve generated by $p$ carries a unit row: characters have
values in $S^1$, nonzero residues live in $\mathbb F_p^\times$, and
Fermat order conditions express finite cyclic closure.  Passing from
divisibility to residue class or character readout changes the visible
coordinate system, not the ledgered arithmetic source.
\end{principle}

\begin{definition}[Zero fiber and quotient coordinate]
\label{def:rh-zero-fiber-quotient-coordinate}
For a prime $p$, the residue classifier
$$
\pi_p:\mathbb Z\to\mathbb Z/p\mathbb Z
$$
has zero fiber
$$
\pi_p^{-1}(0)=\{b:p\mid b\}.
$$
If $b=ap$, then $b$ lies in this zero fiber and $a=b/p$ is the first
quotient coordinate of the fiber.  More generally, writing
$$
b=p^v u,
\qquad
p\nmid u,
$$
exposes the valuation coordinate $v_p(b)=v$ and the unit residue
coordinate
$$
u\bmod p\in\mathbb F_p^\times.
$$
\end{definition}

\begin{principle}[Divisibility is a fiber readout]
\label{prin:rh-divisibility-fiber-readout}
The equation $b=ap$ is not only multiplication in $\mathbb Z$.  In the
local sieve generated by $p$, it says that $b$ is classified into the
zero fiber and that the quotient $a$ is internal fiber data.  Thus the
same prime-generated sieve records both a visible residue coordinate
and the higher valuation data that remains after entering the zero
fiber.
\end{principle}

\section{Quadratic Local Unit Sieve}
\label{sec:rh-prime-sieve-tower-quadratic-local-unit}

\begin{definition}[Gaussian norm local sieve]
\label{def:rh-gaussian-norm-local-sieve}
The Gaussian norm local sieve classifies rational primes by their
behavior in $\mathbb Z[i]$:
$$
\pi_{\mathbb Z[i]}(p)
=
\begin{cases}
\mathrm{ramified},&p=2,\\
\mathrm{split},&p\equiv1\pmod4,\\
\mathrm{inert},&p\equiv3\pmod4.
\end{cases}
$$
For $p\equiv1\pmod4$, the split row is represented by
$$
p=a^2+b^2=N(a+bi)=(a+bi)(a-bi).
$$
\end{definition}

\begin{theorem}[Split Gaussian primes carry a unit-circle quotient]
\label{thm:rh-split-gaussian-primes-carry-unit-circle-quotient}
If a rational prime $p$ has
$$
p=a^2+b^2
$$
with $\pi=a+bi$, then
$$
u:=\frac{\pi}{\overline\pi}
$$
has complex modulus $1$.
\end{theorem}

\begin{proof}
Since $N(\pi)=\pi\overline\pi=p$, one has
$$
|u|^2
=
\frac{\pi\overline\pi}{\overline\pi\pi}
=1.
$$
\end{proof}

\begin{theorem}[Modular two-square row detects the same split type]
\label{thm:rh-modular-two-square-row-detects-split-type}
For an odd prime $p$, the congruence
$$
a^2+b^2\equiv0\pmod p
$$
has a solution with $b\not\equiv0\pmod p$ if and only if $-1$ is a
square in $\mathbb F_p$, equivalently if and only if
$$
p\equiv1\pmod4.
$$
\end{theorem}

\begin{proof}
If $b\not\equiv0\pmod p$, then the displayed congruence is equivalent
to
$$
\left(\frac ab\right)^2\equiv-1\pmod p.
$$
Thus it holds exactly when $-1$ has a square root in $\mathbb F_p$.
For odd primes, Euler's criterion gives
$$
(-1)^{(p-1)/2}\equiv1\pmod p
$$
exactly when $p\equiv1\pmod4$.
\end{proof}

\begin{principle}[Two-square data is a local unit sieve]
\label{prin:rh-two-square-data-local-unit-sieve}
The integer row $p=a^2+b^2$ and the modular row
$a^2+b^2\equiv0\pmod p$ are two projections of the same local split
phenomenon.  The first reads $p$ through the Gaussian norm, while the
second reads the existence of $i^2=-1$ in $\mathbb F_p$.  Both rows
display the prime as carrying a quadratic unit-circle resource rather
than as a bare arithmetic atom.
\end{principle}

\section{Sieve Objects and Refinement}
\label{sec:rh-prime-sieve-tower-sieve-objects}

\begin{definition}[Sieve object]
\label{def:rh-sieve-object}
A sieve object is a packet
$$
\mathcal S=(X,G,\pi,\mu,E,\Lambda)
$$
where $X$ is the source being classified, $G$ is a finite type space,
$\pi:X\to G$ is the classifier, $\mu$ is a probability measure on $G$,
$E\subseteq G$ is the displayed event or pass region, and $\Lambda$ is
the ledger row recording the source, event, complement, probability
mass, and residual data.
\end{definition}

\begin{definition}[Sieve refinement]
\label{def:rh-sieve-refinement}
A sieve object
$$
\mathcal S_2=(X,G_2,\pi_2,\mu_2,E_2,\Lambda_2)
$$
refines
$$
\mathcal S_1=(X,G_1,\pi_1,\mu_1,E_1,\Lambda_1)
$$
when there is a map $F:G_2\to G_1$ such that
$$
\pi_1=F\circ\pi_2,
$$
$$
\mu_1(a)=\sum_{F(b)=a}\mu_2(b),
$$
and
$$
\Lambda_2=\Lambda_1+\operatorname{RefineLedger}_{2,1}.
$$
\end{definition}

\begin{theorem}[Sieve refinement preserves BEDC rows]
\label{thm:rh-sieve-refinement-preserves-bedc-rows}
Sieve refinement preserves the BEDC time, unit, and ledger rows.
\end{theorem}

\begin{proof}
The time row is the refinement history $G_2\to G_1$ together with
$\pi_1=F\circ\pi_2$.  The unit row is preserved by probability
pushforward:
$$
\sum_{a\in G_1}\mu_1(a)
=
\sum_{a\in G_1}\sum_{F(b)=a}\mu_2(b)
=
\sum_{b\in G_2}\mu_2(b)
=1.
$$
For an event $E$, the amplitude split
$$
\psi=P_E\psi+P_{E^c}\psi
$$
is orthogonal by
\autoref{thm:rh-event-projection-pythagorean-law}.  The ledger row is
preserved by the displayed equation
$\Lambda_2=\Lambda_1+\operatorname{RefineLedger}_{2,1}$, so refinement
does not discard previous source or residual data.
\end{proof}

\section{Composition and Prime Sieve Factors}
\label{sec:rh-prime-sieve-tower-prime-factors}

\begin{definition}[Composite and prime sieve factors]
\label{def:rh-composite-prime-sieve-factors}
A finite abelian sieve object with type group $G$ is decomposable when
there are nontrivial finite abelian groups $G_1,G_2$ such that
$$
G\simeq G_1\times G_2,
\qquad
\pi=(\pi_1,\pi_2),
\qquad
\mu=\mu_1\otimes\mu_2.
$$
It is a prime sieve factor when no such nontrivial decomposition is
part of its displayed ledger.
\end{definition}

\begin{theorem}[Finite abelian sieves decompose to prime-power factors]
\label{thm:rh-finite-abelian-sieves-decompose-prime-power}
Every finite abelian sieve type group decomposes into finite
prime-power type factors.
\end{theorem}

\begin{proof}
By the finite abelian group structure theorem,
$$
G\simeq\prod_j\mathbb Z/n_j\mathbb Z.
$$
Writing each $n_j$ as a product of prime powers and applying the
Chinese remainder theorem gives
$$
\mathbb Z/n_j\mathbb Z
\simeq
\prod_{\ell^a\parallel n_j}\mathbb Z/\ell^a\mathbb Z.
$$
Combining these decompositions expresses $G$ as a product of
prime-power type factors.
\end{proof}

\begin{principle}[Composite sieves carry factor ledgers]
\label{prin:rh-composite-sieves-carry-factor-ledgers}
A composite sieve is not a single opaque test.  Its factorization into
prime-power type factors is part of the ledger whenever factor-level
detection is used.  Thus a residual detected by a composite modulus,
Frobenius product, or character product can be localized to displayed
factor rows rather than hidden inside an unstructured global sieve.
\end{principle}

\section{Duality Energy and Ledger}
\label{sec:rh-prime-sieve-tower-energy-ledger}

\begin{theorem}[Finite Fourier change of sieve coordinates preserves energy]
\label{thm:rh-finite-fourier-sieve-coordinate-energy}
Let $G$ be a finite abelian group with uniform probability measure and
let $R\in L^2(G)$.  With Fourier coefficients
$$
\widehat R(\chi)
:=
\frac1{|G|}\sum_{a\in G}R(a)\overline{\chi(a)},
$$
one has
$$
\sum_{a\in G}|R(a)|^2
=
|G|\sum_{\chi\in\widehat G}|\widehat R(\chi)|^2.
$$
\end{theorem}

\begin{proof}
After normalization by $|G|^{-1/2}$, the character table is a unitary
matrix.  The displayed identity is Parseval's identity for this unitary
change of coordinates.
\end{proof}

\begin{theorem}[Sieve projection splits residual energy]
\label{thm:rh-sieve-projection-splits-residual-energy}
For a finite probability layer $(G,\mu)$ and event $E\subseteq G$,
every residual $R\in L^2(G,\mu)$ satisfies
$$
R=P_ER+P_{E^c}R,
\qquad
P_ER\perp P_{E^c}R,
$$
and hence
$$
\|R\|^2=\|P_ER\|^2+\|P_{E^c}R\|^2.
$$
\end{theorem}

\begin{proof}
The supports of $P_ER$ and $P_{E^c}R$ are disjoint, so their inner
product is zero.  The norm identity follows from orthogonal
decomposition in $L^2(G,\mu)$.
\end{proof}

\begin{corollary}[Sieve transformations preserve or ledger energy]
\label{cor:rh-sieve-transformations-preserve-or-ledger-energy}
A finite Fourier change from type coordinates to sieve coordinates
preserves total residual energy.  A projection sieve splits the energy
into the kept component and the complementary ledger component.
\end{corollary}

\begin{proof}
Use \autoref{thm:rh-finite-fourier-sieve-coordinate-energy} for a
coordinate change and
\autoref{thm:rh-sieve-projection-splits-residual-energy} for an event
projection.  The complementary projection is precisely the residual
entry required by \autoref{def:rh-sieve-ledger-row}.
\end{proof}

\section{Recursive Prime-Generated Completeness}
\label{sec:rh-prime-sieve-tower-recursive-completeness}

\begin{definition}[RecursivePrimeGeneratedSieveCompleteness]
\label{def:rh-recursive-prime-generated-sieve-completeness}
The row
$$
\mathsf{RecursivePrimeGeneratedSieveCompleteness}
$$
asserts that finite prime-generated residue and character sieves detect
every nonzero finite prime-power residual.  In concrete form, for every
nonzero finite residual
$$
R=\sum_{j=1}^{N}c_j[q_j]
$$
with distinct positive integers $q_j$, there are a prime $p$ avoiding
the support, an event $A\subseteq\mathbb F_p^\times$ or a character
$$
\chi:\mathbb F_p^\times\to S^1,
$$
and a nonzero readout of $R$ through the classifier
$$
\pi_p(q)=q\bmod p.
$$
\end{definition}

\begin{theorem}[Prime-generated local sieves detect finite residuals]
\label{thm:rh-prime-generated-local-sieves-detect-finite-residuals}
Every nonzero finite residual
$$
R=\sum_{j=1}^{N}c_j[q_j]
$$
with distinct positive integers $q_j$ has a detecting local sieve
generated by some prime $p$.
\end{theorem}

\begin{proof}
Choose a prime $p$ that divides none of the $q_j$ and none of the
nonzero differences $q_i-q_j$.  Then the residues
$$
q_j\bmod p
$$
are nonzero and pairwise distinct in $\mathbb F_p^\times$.  Pick
$j$ with $c_j\ne0$ and let
$$
A=\{q_j\bmod p\}.
$$
The event readout on $A$ is exactly $c_j$, hence nonzero.  Since
$\mathbb F_p^\times$ is a finite abelian group, the character readouts
also form a complete Fourier basis; equivalently, applying
\autoref{thm:rh-fourier-complete-finite-layers-detect-nonzero-residuals}
to this finite layer gives a character with nonzero readout whenever a
character witness is required.
\end{proof}

\begin{corollary}[Recursive generated sieves supply finite residual detection]
\label{cor:rh-recursive-generated-sieves-supply-finite-residual-detection}
The prime-generated local sieve family satisfies
$$
\mathsf{RecursivePrimeGeneratedSieveCompleteness}
$$
for finite residuals.
\end{corollary}

\begin{proof}
This is the content of
\autoref{thm:rh-prime-generated-local-sieves-detect-finite-residuals},
with event and character readouts supplied by the same prime-generated
finite unit group.
\end{proof}

\begin{principle}[Recursive generated sieves are BEDC-legal]
\label{prin:rh-recursive-generated-sieves-bedc-legal}
Recursive use of prime-generated sieves preserves the three BEDC rows.
Each step appends a prime-generated classifier to the time history,
reads through a finite unit or probability amplitude surface, and
records the excluded residue, valuation, or residual component in a
ledger.  Thus recursive filtering changes visibility but not the
accounted information or residual energy.
\end{principle}

\section{Tower Interface}
\label{sec:rh-prime-sieve-tower-interface}

\begin{definition}[PrimeSieveTowerDuality]
\label{def:rh-prime-sieve-tower-duality}
The row
$$
\mathsf{PrimeSieveTowerDuality}
$$
consists of the Euclidean prime-generated tower, the local prime sieve
packets, the zero-fiber and valuation rows, the Gaussian local unit
sieve, the sieve-object refinement category, the prime-power factor
ledger, the finite Fourier/event energy laws, and recursive
prime-generated finite residual detection.  It asserts that
prime-generated sieve transformations preserve the time source, unit or
energy readout, and ledger accountability of the prime classifier tower.
\end{definition}

\begin{theorem}[Prime sieve tower duality supplies duality completeness data]
\label{thm:rh-prime-sieve-tower-duality-supplies-completeness-data}
Assume $\mathsf{PrimeSieveTowerDuality}$ and that every zeta-facing
radial residual enters the generated profinite prime type tower.  Then
the finite type-sieve dual pairs required by
$$
\mathsf{PrimeSieveDualityCompleteness}
$$
are legal BEDC sieve objects with preserved time, unit, and ledger
rows.
\end{theorem}

\begin{proof}
The generated tower supplies the finite prime type spaces and their
refinement maps.  The local prime sieve packets supply legal event,
residue, zero-fiber, valuation, order, Gaussian norm, and character
readouts.  Sieve refinement preserves the time, unit, and ledger rows by
\autoref{thm:rh-sieve-refinement-preserves-bedc-rows}.  The Fourier and
event energy laws show that changing to the sieve-frequency side does
not erase residual energy, while projection sends complementary energy
to the ledger.  Finite residual detection is supplied by
\autoref{cor:rh-recursive-generated-sieves-supply-finite-residual-detection}.
These are exactly the legal data needed for the finite dual pairs of
\autoref{def:rh-prime-sieve-duality-completeness}.
\end{proof}

\begin{principle}[Prime sieve tower audit boundary]
\label{prin:rh-prime-sieve-tower-audit-boundary}
The tower duality packet is a structural certificate for sieve methods,
not an analytic source theorem for zeta zeros.  It says that once a
radial residual is represented in the prime-generated type tower, no
combination, factorization, Fourier change, or event projection of
sieves may erase it without a ledger row.  The analytic obligation is
therefore isolated to the bridge that places every off-critical
zero-side residual into this prime-generated tower.
\end{principle}
