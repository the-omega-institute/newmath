\chapter{Radial Scale Projection System}
\label{ch:visions-rh-radial-scale-projection-system}
\origin{ai}

\autoref{ch:visions-rh-triadic-pythagorean-obstruction} identifies the
analytic triangle as one scale defect, and
\autoref{ch:visions-rh-right-triangle-invariant-system} records its
finite Hilbert form.  The present chapter isolates the scalar readout
shared by the Li, Weil, and Nyman projections.  The scalar is the radial
coordinate of the Li map; every analytic line in the route reads the
same coordinate through a different finite or Hilbert witness.

\begin{definition}[Li radial scale coordinate]
\label{def:rh-li-radial-scale-coordinate}
For a nontrivial completed zeta zero row $\rho$, let
$$
z(\rho):=1-\frac{1}{\rho}
$$
as in \autoref{def:rh-li-time-coordinate}.  The Li radial scale
coordinate is
$$
\ell(\rho):=\log|z(\rho)|.
$$
When $z(\rho)=e^{\ell+i\theta}$, the coordinate $\ell$ is the scale
leg and $\theta$ is the phase angle.
\end{definition}

\begin{theorem}[Radial scale reads the fixed-half surface]
\label{thm:rh-radial-scale-fixed-half-surface}
For a nontrivial completed zero row $\rho$,
$$
\ell(\rho)=0
$$
if and only if the readout of $\rho$ lies on the fixed-half surface.
\end{theorem}

\begin{proof}
By definition, $\ell(\rho)=0$ exactly when $|z(\rho)|=1$.  By
\autoref{thm:rh-li-coordinate-fixed-half-circle}, this is equivalent to
the fixed-half readout.
\end{proof}

\begin{definition}[Critical half axis row]
\label{def:rh-critical-half-axis-row}
The critical half axis row is the recovered-carrier predicate
$$
\mathcal C_{1/2}:=\{s:\operatorname{Re}(s)=1/2\}
$$
read inside the completed zeta zero interface.  Its mirror source is
the involution of \autoref{def:rh-weil-mirror-involution},
$$
\iota(s)=1-\overline{s},
$$
its unit readout is the Li coordinate
$$
z(s)=1-\frac1s,
$$
and its ledger readout is the radial scale
$$
\ell(s)=\log|z(s)|.
$$
\end{definition}

\begin{theorem}[Critical half axis is the mirror fixed locus]
\label{thm:rh-critical-half-axis-mirror-fixed-locus}
For a completed zero coordinate $s=\sigma+i\gamma$,
$$
\iota(s)=s
$$
if and only if
$$
\sigma=\frac12.
$$
\end{theorem}

\begin{proof}
By \autoref{def:rh-weil-mirror-involution},
$$
\iota(s)=1-\overline{s}=1-\sigma+i\gamma.
$$
The equality $\iota(s)=s$ is therefore equivalent to
$$
1-\sigma=\sigma,
$$
which is equivalent to $\sigma=1/2$.
\end{proof}

\begin{definition}[Critical half axis triple]
\label{def:rh-critical-half-axis-triple}
The critical half axis carries the ontological triple
$$
\operatorname{Ont}(\mathcal C_{1/2})
=
(\mathsf T_{\mathcal C},\mathsf U_{\mathcal C},\mathsf L_{\mathcal C}).
$$
The time row $\mathsf T_{\mathcal C}$ is the fixed locus of the zeta
mirror involution.  The unit row $\mathsf U_{\mathcal C}$ is the unit
Li readout
$$
s\in\mathcal C_{1/2}
\Longleftrightarrow
|z(s)|=1.
$$
The ledger row $\mathsf L_{\mathcal C}$ is the zero radial account
$$
s\in\mathcal C_{1/2}
\Longleftrightarrow
\ell(s)=0.
$$
Thus the critical half axis is not an unlicensed host line: it is the
common fixed, unit, and zero-ledger row consumed by the closed zero
packet route.
\end{definition}

\begin{theorem}[Mirror reflection reverses the radial ledger]
\label{thm:rh-critical-half-axis-reverses-radial-ledger}
If $z(s)=e^{\ell+i\theta}$ and $\iota(s)$ is the mirror coordinate,
then
$$
z(\iota(s))=e^{-\ell+i\theta}.
$$
Consequently the critical half axis is exactly the zero ledger locus for
the mirror radial pair.
\end{theorem}

\begin{proof}
The identity is \autoref{lem:rh-li-radius-reverses-across-mirror}.  The
zero ledger locus is $\ell=0$, which is equivalent to fixed-half readout
by \autoref{thm:rh-radial-scale-fixed-half-surface}.
\end{proof}

\begin{theorem}[Closed radial packets lie on the critical half axis]
\label{thm:rh-closed-radial-packets-lie-critical-half-axis}
Let a licensed zeta zero packet carry
$$
\mathsf{ZetaZeroObjectsHaveOntTriple},
\qquad
\mathsf{GlobalOffLineZeroToFiniteNPWindow},
\qquad
\mathsf{LedgerClosureRefusal}.
$$
If the packet is closed, then its Li radial scale satisfies
$$
\ell(\rho)=0,
$$
and hence its readout lies on $\mathcal C_{1/2}$.
\end{theorem}

\begin{proof}
If $\ell(\rho)\ne0$, then
\autoref{thm:rh-radial-scale-fixed-half-surface} says that $\rho$ is
off the fixed-half surface.  The licensed triple route then gives the
closed-status refusal of
\autoref{thm:rh-offline-licensed-triple-cannot-be-closed}.  This
contradicts the closed-packet hypothesis.  Therefore $\ell(\rho)=0$,
and \autoref{def:rh-critical-half-axis-triple} identifies the readout
with the critical half axis.
\end{proof}

\begin{theorem}[Li radial Pythagorean identity]
\label{thm:rh-li-radial-pythagorean-identity}
Let
$$
z=e^{\ell+i\theta}.
$$
For every $n\ge1$,
$$
|1-z^n|^2
=
(e^{n\ell}-1)^2
+
2e^{n\ell}(1-\cos(n\theta)).
$$
\end{theorem}

\begin{proof}
This is \autoref{thm:rh-li-coordinate-pythagorean-decomposition},
rewritten with the notation of
\autoref{def:rh-li-radial-scale-coordinate}.
\end{proof}

\begin{definition}[Radial projection triple]
\label{def:rh-radial-projection-triple}
The radial projection triple attached to a completed zero ledger is
$$
\mathsf{RadialProjection}_{\zeta}
:=
(\mathsf T_{\ell},\mathsf W_{\ell},\mathsf N_{\ell}).
$$
The time projection $\mathsf T_{\ell}$ is the Li-time readout of
$z(\rho)^n$.  The energy projection $\mathsf W_{\ell}$ is the
Weil-quadratic readout of reflected zero pairs.  The container
projection $\mathsf N_{\ell}$ is the Nyman--Beurling readout of the
Mellin separator attached to a zero in the right open half of the
critical strip.
\end{definition}

\begin{principle}[Time projection of a nonzero radial scale]
\label{prin:rh-time-projection-nonzero-radial-scale}
If a reflected zero pair has Li coordinates
$$
z_+=e^{\ell+i\theta},
\qquad
z_-=e^{-\ell+i\theta},
$$
then Li time reads the radial coordinate through the factor
$$
\cosh(n\ell).
$$
For $\ell\ne0$, this factor is larger than $1$ for $n\ge1$.  The
finite Li--Weil witness row of
\autoref{def:rh-li-weil-finite-negative-witness} is the time-side
certificate that such radial amplification cannot coexist with the
global Li positivity row.
\end{principle}

\begin{principle}[Energy projection of a nonzero radial scale]
\label{prin:rh-energy-projection-nonzero-radial-scale}
A reflected off-fixed-half pair produces a Weil value block with a
negative direction, as in
\autoref{def:rh-weil-negative-direction-projection}.  The sign of
$\ell$ may be reversed by reflection, but the existence of a two-point
reflected orbit with $|\ell|>0$ is what gives the energy-side negative
direction.  Global Weil positivity forbids that projection when the
test row and tail ledgers are source-side controlled.
\end{principle}

\begin{principle}[Container projection of a nonzero radial scale]
\label{prin:rh-container-projection-nonzero-radial-scale}
If the reflected orbit has a representative
$$
s_0=\sigma+i\tau,
\qquad
\sigma>\frac12,
$$
then \autoref{lem:rh-off-half-zero-nyman-separator} supplies the
continuous Mellin functional
$$
L_{s_0}(f)=\int_0^1 f(x)x^{s_0-1}\,dx.
$$
It annihilates the displayed Nyman generators and does not annihilate
$\mathbf 1$.  Thus the same radial scale coordinate that Li time reads
as amplification and Weil energy reads as a negative direction is read
by the Nyman container as a residual separator.
\end{principle}

\begin{definition}[Zeta radial triangular closure]
\label{def:rh-zeta-radial-triangular-closure}
The source row
$$
\mathsf{ZetaRadialTriangularClosure}
$$
asserts that every accepted nontrivial completed zero ledger carries
the radial projection triple of
\autoref{def:rh-radial-projection-triple}, and that the three
projections are governed by one scalar
$$
\ell(\rho)=\log\left|1-\frac1\rho\right|.
$$
The row further asserts that a nonzero value of this scalar must be
detected by at least one of the three finite or Hilbert witnesses:
Li-time negativity, Weil negative direction, or Nyman residual
separation.
\end{definition}

\begin{theorem}[Radial triangular closure gives fixed-half readout]
\label{thm:rh-radial-triangular-closure-fixed-half-readout}
Assume $\mathsf{ZetaRadialTriangularClosure}$ and assume the three
projection rows
$$
\mathsf{GlobalLiWeilPositivity},
\qquad
\mathsf{GlobalWeilPositivity},
\qquad
\mathsf{NymanBeurlingExhaustion}
$$
are available with their usual reflection consumption rows.  Then every
accepted nontrivial completed zero has fixed-half readout.
\end{theorem}

\begin{proof}
Let $\rho$ be an accepted nontrivial completed zero.  If
$\ell(\rho)\ne0$, the closure row forces a detection in one of the
three projections.  A Li-time negative witness contradicts
$\mathsf{GlobalLiWeilPositivity}$.  A Weil negative direction
contradicts $\mathsf{GlobalWeilPositivity}$.  A Nyman residual separator
contradicts $\mathsf{NymanBeurlingExhaustion}$ by
\autoref{lem:rh-off-half-zero-nyman-separator}.  Hence
$\ell(\rho)=0$.  By
\autoref{thm:rh-radial-scale-fixed-half-surface}, $\rho$ has fixed-half
readout.
\end{proof}

\begin{principle}[Single radial burden]
\label{prin:rh-single-radial-burden}
The Li, Weil, and Nyman tasks should be read as three audits of one
radial coordinate, not as unrelated proof searches.  Li time asks
whether powers of $z(\rho)$ expose $\ell(\rho)$.  Weil energy asks
whether a test row can extract a negative direction from the reflected
orbit.  Nyman exhaustion asks whether the legal generator container
leaves a residual Mellin functional.  The source-side burden is to show
that the BEDC triadic carrier never admits a nonzero radial coordinate
without one of these witnesses.
\end{principle}

\begin{definition}[Cross-projection radial refutation packet]
\label{def:rh-cross-projection-radial-refutation-packet}
A cross-projection radial refutation packet for a completed zero ledger
is a source-side row that starts from a displayed assumption
$$
\ell(\rho)\ne0
$$
and exports this same scalar to the three projection ledgers
$$
\mathsf T_{\ell},
\qquad
\mathsf W_{\ell},
\qquad
\mathsf N_{\ell}
$$
of \autoref{def:rh-radial-projection-triple}.  The packet is accepted
only when the time ledger reads the reflected Li pair, the energy ledger
reads the corresponding Weil negative direction, and the container
ledger reads the Mellin separator in the open half selected by
reflection.  Thus the packet does not certify a contradiction inside a
single projection; it certifies that a nonzero radial coordinate has
the same readback in the other two projections as well.
\end{definition}

\begin{definition}[Triangular burden of a radial source row]
\label{def:rh-triangular-burden-radial-source-row}
Let $\mathcal Q$ be a RH-facing source row whose conclusion is consumed
by the fixed-half route.  Its triangular radial burden is the
requirement that any use of $\mathcal Q$ against a zero ledger with
$$
\ell(\rho)\ne0
$$
comes with a cross-projection radial refutation packet.  A row that
only records positivity, separation, or exhaustion in its native
projection is therefore a local analytic row.  It becomes a route row
only after the same radial scalar has been transported to the remaining
projection ledgers.
\end{definition}

\begin{theorem}[Two-sided projection audit closes the third side]
\label{thm:rh-two-sided-projection-audit-closes-third-side}
Assume a completed zero ledger carries a cross-projection radial
refutation packet.  Suppose two of the projection rows
$$
\mathsf{GlobalLiWeilPositivity},
\qquad
\mathsf{GlobalWeilPositivity},
\qquad
\mathsf{NymanBeurlingExhaustion}
$$
are source-side closed, and suppose the remaining projection is linked
to them by the triadic source transport of
\autoref{def:rh-triadic-source-transport}.  Then the zero ledger cannot
carry
$$
\ell(\rho)\ne0.
$$
\end{theorem}

\begin{proof}
Assume $\ell(\rho)\ne0$.  The cross-projection packet exports this
scalar to all three ledgers.  If the time ledger supplies a finite
negative Li witness, it contradicts
$\mathsf{GlobalLiWeilPositivity}$.  If the energy ledger supplies a
Weil negative direction, it contradicts
$\mathsf{GlobalWeilPositivity}$.  If the container ledger supplies a
Mellin separator, it contradicts $\mathsf{NymanBeurlingExhaustion}$ by
\autoref{lem:rh-off-half-zero-nyman-separator}.  The two closed sides
and the triadic source transport ensure that the remaining side cannot
hide the same scalar as an unledgered real-positive coordinate.  Hence
the assumption $\ell(\rho)\ne0$ is impossible.
\end{proof}

\begin{principle}[No single-projection consumption]
\label{prin:rh-no-single-projection-consumption}
The fixed-half route may consume a Li, Weil, or Nyman target only after
its triangular radial burden is displayed.  A Li square row must explain
where the corresponding radial scale would appear as Weil energy and as
Nyman residual separation.  A Weil factorization row must explain where
the same radial scale would appear in Li time and in the Nyman
container.  A Nyman exhaustion row must explain where its residual
separator would appear in Li time and Weil energy.  This is the
source-side form of the rule that every off-fixed-half assumption is
tested against the other two ledgers, not only against the ledger in
which it was first detected.
\end{principle}
