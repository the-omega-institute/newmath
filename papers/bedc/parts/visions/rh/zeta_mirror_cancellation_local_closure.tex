\chapter{Zeta Mirror Cancellation and Local Closure}
\label{ch:visions-rh-zeta-mirror-cancellation-local-closure}
\origin{ai}

\autoref{ch:visions-rh-zeta-virtual-machine-execution-trace} separates
ledger cancellation from scalar readout cancellation.  This chapter
records the corresponding mirror boundary for completed zeta zeros.
Functional-equation symmetry supplies mirror-pair balance, while the
fixed-half conclusion requires local zero-ledger closure at a single
zero packet.

\begin{definition}[Weak and strong mirror closure]
\label{def:rh-weak-strong-mirror-closure}
Let $J(s)=1-\overline{s}$ be the completed mirror row.  A zero packet
$\rho$ has weak mirror closure when the mirror zero packet $J(\rho)$ is
also present and the paired radial ledger balances across the orbit:
$$
\ell(\rho)+\ell(J(\rho))=0.
$$
It has strong mirror closure when
$$
\rho=J(\rho).
$$
By the fixed-surface calculation of \autoref{def:rh-mirror-map}, strong
mirror closure is the same as fixed-half readout.
\end{definition}

\begin{principle}[Global cancellation is not local closure]
\label{prin:rh-global-cancellation-not-local-closure}
Weak mirror closure is a global-orbit balance row.  It says that an
off-fixed residual at one zero can be paired with the inverse residual
at the mirror zero.  It does not close the ledger of the first zero as a
local object.  Local closure requires strong mirror closure, or another
displayed row proving that the residual of the single zero is already
zero at its own position.  This is the ledger formulation of
\autoref{cor:rh-mirror-closure-weaker-than-rh}.
\end{principle}

\begin{definition}[Half-density scaling ledger]
\label{def:rh-half-density-scaling-ledger}
For a pure zeta input $a\in M_{\mathbb P}$ of
\autoref{def:rh-positive-prime-ledger}, write
$$
L(a)=\sum_p a_p\log p.
$$
For $s=\sigma+it$, set
$$
\delta:=\sigma-\frac12
$$
and define the half-density scaling ledger
$$
\Lambda_s(a):=\delta L(a).
$$
The centered zeta mode then factors as
$$
e^{-sL(a)}
=
e^{-\frac12 L(a)}
e^{-\Lambda_s(a)}
e^{-itL(a)}.
$$
The final factor is the unit-phase rotation.  The middle factor is the
positive real scaling ledger.
\end{definition}

\begin{theorem}[Off-half scaling ledgers are same-signed and unbounded]
\label{thm:rh-offhalf-scaling-ledgers-samesigned-unbounded}
If $\sigma\ne1/2$, then every nonempty pure zeta input
$a\in M_{\mathbb P}$ has $\Lambda_s(a)\ne0$, all such ledgers have the
same sign, and $|\Lambda_s(a)|$ is unbounded on $M_{\mathbb P}$.
\end{theorem}

\begin{proof}
For a nonempty pure zeta input, the positivity of prime logarithms gives
$$
L(a)>0.
$$
Thus
$$
\Lambda_s(a)=(\sigma-\frac12)L(a)
$$
has the sign of $\sigma-1/2$ for every nonempty $a$, and it is nonzero
when $\sigma\ne1/2$.  To see unboundedness, take the powers along the
prime $2$ axis,
$$
a_m=m[2].
$$
Then
$$
|\Lambda_s(a_m)|
=
|\sigma-\frac12|\,m\log2,
$$
which is unbounded as $m$ ranges over $\mathbb N$.
\end{proof}

\begin{definition}[Zeta instruction-scale residual]
\label{def:rh-zeta-instruction-scale-residual}
For a centered zeta mode $s=1/2+r+it$ and a pure zeta input
$a\in M_{\mathbb P}$, the instruction-scale residual is
$$
\mathsf{Scal}_s(a)
:=
\log\left|e^{-rL(a)}\right|
=
-rL(a)
=
-\Lambda_s(a).
$$
The unit-phase instruction attached to the same input is
$$
U_t(a):=e^{-itL(a)},
\qquad
|U_t(a)|=1.
$$
\end{definition}

\begin{theorem}[Off-half scaling cancellation is not a zeta instruction]
\label{thm:rh-offhalf-scaling-cancellation-not-zeta-instruction}
On the pure zeta instruction surface generated by positive prime
registers and unit-phase rotations, no finite composite of phase
rotations, conjugate phase rotations, or same-window phase mirrors
cancels $\mathsf{Scal}_s(a)$ for a nonempty input $a$ when
$\Re(s)\ne1/2$.
\end{theorem}

\begin{proof}
Every admitted phase instruction has modulus $1$.  The same remains true
after taking finite products, inverses, conjugates, and mirror phase
reads, because these operations only change angular coordinates.  Hence
their log-modulus contribution is always
$$
0.
$$
For $s=1/2+r+it$, the non-phase part of the zeta monomial is
$$
e^{-rL(a)}.
$$
By \autoref{def:rh-zeta-instruction-scale-residual}, its log-modulus is
$-rL(a)$.  If $a$ is nonempty, then $L(a)>0$; if $r\ne0$, this
log-modulus is nonzero.  Therefore unit-phase instructions cannot cancel
the residual.  Closing it requires either $r=0$ or a displayed
positive-real scale row outside the pure zeta instruction surface.
\end{proof}

\begin{principle}[Pure zeta has no local scaling inverse]
\label{prin:rh-pure-zeta-no-local-scaling-inverse}
The pure zeta input surface contains nonnegative prime registers and
unit-phase rotations.  It does not contain an independent negative
length register, a positive-real scale counterweight, or a second
same-position value of $\delta$ with opposite sign.  Therefore an
off-half scaling ledger cannot be cancelled inside the same pure zeta
input.  The only way for all half-density scaling ledgers to vanish
locally is
$$
\sigma=\frac12.
$$
This is the zeta-specific form of the no-hidden-scale discipline of
\autoref{def:rh-no-hidden-scale-from-bedc-row} and the unit-phase
closure rule of
\autoref{prin:rh-unit-phase-closure-excludes-radial-scale}.
\end{principle}

\begin{theorem}[Pure zeta local closure is fixed-half closure]
\label{thm:rh-pure-zeta-local-closure-fixed-half}
For a completed zero packet whose source is required to close inside the
pure zeta input surface, local closure of the scaling ledger is
equivalent to fixed-half readout.
\end{theorem}

\begin{proof}
If the readout is fixed-half, then $\sigma=1/2$, so
\autoref{def:rh-half-density-scaling-ledger} gives
$$
\Lambda_s(a)=0
$$
for every pure zeta input $a$.

Conversely, assume the scaling ledger closes locally for every pure zeta
input.  Apply this to the one-prime input $[2]$.  Since
$$
L([2])=\log2>0,
$$
the equality $\Lambda_s([2])=0$ forces $\sigma-1/2=0$.  Hence
$\sigma=1/2$, which is fixed-half readout.
\end{proof}

\begin{corollary}[Scaling-cancellation zeros are not pure zeta closures]
\label{cor:rh-scaling-cancellation-zeros-not-pure-zeta-closures}
A zero packet whose cancellation requires an off-half scaling ledger is
not closed by the pure zeta input surface.
\end{corollary}

\begin{proof}
If the packet is off-half, then
\autoref{thm:rh-offhalf-scaling-ledgers-samesigned-unbounded} produces
same-signed nonzero scaling ledgers, already unbounded along the
$2$-axis.  By
\autoref{prin:rh-pure-zeta-no-local-scaling-inverse}, pure zeta supplies
no local scaling inverse for those ledgers.  Thus such a packet can
close only by adding an external scale counterweight or by using a
distinct mirror position, neither of which is a pure zeta local closure.
\end{proof}

\begin{principle}[Single-ledger zero principle]
\label{prin:rh-single-ledger-zero-principle}
A scalar zero read of the completed zeta carrier is accepted as a
zeta-local zero only when the same zero packet carries the prime source
ledger, phase readout, tail ledger, and scale-residual ledger needed to
close the cancellation.  A distinct mirror packet may witness global
orbit symmetry, but it is not a same-packet payment of the scale residual.
Thus an off-half scalar cancellation is not a pure zeta local closure
unless a separate theorem first proves that its scale residual is closed
inside the same zero ledger.
\end{principle}

\begin{theorem}[Single-ledger zeta closure excludes off-half scale]
\label{thm:rh-single-ledger-zeta-closure-excludes-offhalf-scale}
If a completed zero packet is closed by the single-ledger zero principle
on the pure zeta instruction surface, then its scale residual vanishes.
Consequently its readout is fixed-half.
\end{theorem}

\begin{proof}
Single-ledger closure forbids payment of the scale residual by a distinct
mirror position or by an undisplayed positive-real scale row.  By
\autoref{thm:rh-offhalf-scaling-cancellation-not-zeta-instruction},
unit-phase zeta instructions cannot cancel a nonzero instruction-scale
residual.  Hence the residual can close on the pure zeta instruction
surface only when
$$
\mathsf{Scal}_s(a)=0
$$
for every pure zeta input $a$.  Applying this to $[2]$ gives
$$
-\left(\Re(s)-\frac12\right)\log2=0,
$$
so $\Re(s)=1/2$.
\end{proof}

\begin{definition}[Mirror-cancellation certificate]
\label{def:rh-mirror-cancellation-certificate}
A mirror-cancellation certificate for the completed xi execution trace
is a displayed tuple
$$
\mathsf{MCC}_{\xi}
:=
(\mathcal A,\mathsf{Ledger},\mathcal R_{\xi},\Theta,\iota,
(W_n)_{n\ge0},(B_n)_{n\ge0},Q,\mathsf L_{\mathrm{tail}}).
$$
Here $\mathcal A$ is the ledger carrier, $\mathcal R_{\xi}$ is the
completed scalar readout, $\Theta$ is the completed mirror row
$$
s\mapsto 1-\overline{s},
$$
$\iota$ is the ledger inverse row, $(W_n)$ is a finite window tower,
$(B_n)$ is its tail-budget row, $Q$ is a positive quadratic or
Hilbert-square row, and $\mathsf L_{\mathrm{tail}}$ is the tail ledger.
The certificate requires:
$$
\iota^2=1,
\qquad
\delta+\iota(\delta)=0
$$
on the displayed cancellation class;
$$
\mathcal R_{\xi}
\text{ sends admitted mirror pairs to scalar cancellation;}
$$
the tail budgets are exhausted or closed by
$\mathsf L_{\mathrm{tail}}$; and $Q$ is nonnegative on the displayed
residual class.
\end{definition}

\begin{theorem}[Local zero-ledger closure gives fixed-half readout]
\label{thm:rh-local-zero-ledger-closure-fixed-half-readout}
Assume every completed non-trivial zero packet has local ledger closure:
it cannot use a distinct mirror position to close its own radial
residual.  Then every such zero has fixed-half readout.
\end{theorem}

\begin{proof}
Let $\rho$ be a completed non-trivial zero packet.  The functional
equation and conjugation rows supply the mirror packet $J(\rho)$ as in
\autoref{thm:rh-functional-equation-gives-mirror-closure}.  If
$\rho\ne J(\rho)$, the residual balance is only weak mirror closure in
the sense of \autoref{def:rh-weak-strong-mirror-closure}: the radial
ledger of $\rho$ is offset by the radial ledger of a distinct position.
By the local ledger-closure hypothesis, such cross-position balancing
does not close the zero packet.  A completed zero packet must therefore
satisfy $\rho=J(\rho)$.  The fixed-surface calculation in
\autoref{def:rh-mirror-map} gives $\Re(\rho)=1/2$, hence fixed-half
readout.
\end{proof}

\begin{principle}[Mirror supplies symmetry, positivity locks the line]
\label{prin:rh-mirror-symmetry-positivity-locks-line}
The mirror row by itself supplies paired structure: an off-fixed zero
candidate may be reflected to its mirror companion.  This symmetry is
not yet the fixed-half conclusion.  The fixed line is forced only when
mirror cancellation is combined with a positive or self-adjoint row and
with tail-ledger closure.  In the notation of
\autoref{def:rh-mirror-cancellation-certificate}, the mirror row
$\Theta$ explains the local form of cancellation, while $Q$ and
$\mathsf L_{\mathrm{tail}}$ refuse off-fixed residual escape.  This is
the same burden isolated by the global observation triangle of
\autoref{def:rh-global-zeta-observation-triangle}.
\end{principle}
