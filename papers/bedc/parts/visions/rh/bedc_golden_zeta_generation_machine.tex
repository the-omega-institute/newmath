\begin{definition}[BEDC--Golden--Zeta generation machine]
\label{def:rh-bedc-golden-zeta-generation-machine}
A \emph{BEDC--Golden--Zeta generation machine} is a finite-to-limit
source packet
$$
\mathfrak M_{\zeta}
$$
with the following displayed rows.

The kernel row is the raw mark and history spine
$$
\mathsf{b0},\mathsf{b1}
\longrightarrow
\mathsf{BHist}
$$
of \autoref{def:kernel-spine-raw-mark-carrier}, equipped with the
one-step time row of \autoref{def:omega-generation-one-step-time-row}.
Thus time is constructor-tail generation over histories rather than an
ambient parameter.

The zero row is an RH-free generated zero carrier
$$
Z=\mu F_Z
$$
with a realization
$$
\varepsilon:Z\to\mathcal Z_{\zeta}
$$
as in \autoref{def:rh-generated-zero-carrier}.  Its source
completeness row is the essential coverage component
$$
\prod_{u:\mathcal Z_{\zeta}}
\sum_{z:Z}\varepsilon(z)\sim u.
$$

The arithmetic row assigns active heights
$$
h_a>0
$$
from the logarithmic prime or integer ledger, so that centered zeta
channels have the form
$$
U_a(s):=e^{-(s-1/2)h_a}.
$$

The resolution row is a golden resolution ruler
$$
\delta_m=\varphi^{-m},
\qquad
N_m=F_{Lm},
$$
with Zeckendorf truncation ledgers, as in
\autoref{def:rh-golden-resolution-ruler} and
\autoref{def:rh-zeckendorf-resolution-ledger}.

The certificate row is a compatible resolution certificate tower in the
sense of \autoref{def:rh-resolution-certificate-tower}, carrying the
triadic invariant of \autoref{def:rh-certificate-tower-triadic-invariant}
at every displayed level.

The bridge row is a cofinal family of golden Lee--Yang bridge witnesses
in the sense of \autoref{def:rh-golden-lee-yang-bridge-witness}.  This
row connects the weak eta capture packet to a lifted Fibonacci path
packet with the same zero-island readout.

The defect row is the scale-energy extraction
$$
E_m(s)=r^2W_m+o(W_m),
\qquad
W_m\to\infty,
$$
for readouts
$$
s=\frac12+r+it.
$$
\end{definition}

\begin{definition}[Constructive layer witness]
\label{def:rh-constructive-machine-layer-witness}
A \emph{constructive layer witness} for a machine level is a quadruple
$$
(O,G,I,F).
$$
Here $O$ is the displayed object row, $G$ is the generator row producing
that object from the previous layer, $I$ is the invariant transported by
the generator, and $F$ is the projection-loss row recording what the
current layer forgets when read by the next visible observer.  The loss
row is a ledger, not a license to introduce hidden structure.
\end{definition}

\begin{definition}[Constructive BEDC--Golden--Zeta machine]
\label{def:rh-constructive-bedc-golden-zeta-machine}
A \emph{constructive BEDC--Golden--Zeta machine} is a
BEDC--Golden--Zeta generation machine together with constructive layer
witnesses for the source chain
$$
\mathsf{BMark}
\to
\mathsf{BHist}
\to
\mathsf{Time}
\to
\mathsf{PrimeLedger}
\to
\mathsf{Hamiltonian}
\to
\mathsf{GoldenResolution}
\to
\mathsf{EtaPacket}
\to
\mathsf{ZeckendorfPacket}
\to
\mathsf{SolenoidGauge}
\to
\mathsf{LeeYangBoundary}
\to
\mathsf{ZetaZero}.
$$
Each transition must be generated from the preceding object and the
triadic invariant rows.  A later layer may inspect the projection loss
of an earlier layer only through the displayed ledger $F$; it may not
repair that loss by importing an ungenerated coordinate.
\end{definition}

\begin{definition}[Machine legality]
\label{def:rh-bedc-golden-zeta-machine-legality}
The machine $\mathfrak M_{\zeta}$ is \emph{legal} when its hidden
coordinates are confined to the compact phase row and finite
prime-register fibres:
$$
\mathbb T,
\qquad
\prod_{p\in S}\mathbb Z_p,
$$
and when no constructor, bridge comparison, tail row, or limit socket
imports an independent real positive closure coordinate
$$
\mathbb R_{>0}.
$$
Equivalently, the only admissible readback of
$$
U_a(s)=e^{-r h_a}e^{-it h_a}
$$
inside the closed machine treats $e^{-it h_a}$ as phase data and
requires $e^{-r h_a}$ to be accounted for by the defect ledger rather
than absorbed as gauge.
\end{definition}

\begin{principle}[No downstream smuggling]
\label{prin:rh-no-downstream-smuggling}
In a constructive BEDC--Golden--Zeta machine, no layer after a projection
may supply data that the previous generator did not produce.  Thus a
Lee--Yang boundary row may stabilize an eta zero island only through a
displayed same-island bridge; it may not choose a stable packet after
the zero is known.  A solenoid row may hide only phase and
prime-register fibres; it may not supply a real positive scale
coordinate.  A limit row may read a compatible tower; it may not add an
external zero presentation.  This is the layerwise form of machine
legality.
\end{principle}

\begin{definition}[Internal golden ruler]
\label{def:rh-internal-golden-ruler}
An \emph{internal golden ruler} at level $m$ is the row
$$
\mathcal R_m:=
(\delta_m,N_m,Z_m)
$$
where
$$
\delta_m=\varphi^{-m},
\qquad
N_m=F_{Lm},
$$
and $Z_m$ is the Zeckendorf resolution ledger of
\autoref{def:rh-zeckendorf-resolution-ledger} for $N_m$.  It is internal
when the following three rows are generated inside the same BEDC packet.

The $\mathsf P$ row is Zeckendorf source uniqueness: the Fibonacci-block
address of $N_m$ is not replaceable by another unledgered truncation
source.

The $\mathsf U$ row is the no-adjacent path boundary: the digits of
$Z_m$ lie in the stable Zeckendorf window, hence in the path carrier used
by \autoref{def:rh-fibonacci-path-carrier}.  This supplies the
unit-boundary source for the later Lee--Yang lift; it does not by itself
prove zeta boundary stability.

The $\mathsf T$ row is Fibonacci refinement:
$$
\mathcal R_m\rightsquigarrow\mathcal R_{m+1},
$$
read as another level of the same generated ruler rather than as a
change of external measurement.
\end{definition}

\begin{definition}[Triadic ruler invariance]
\label{def:rh-triadic-ruler-invariance}
A family of internal golden rulers
$$
(\mathcal R_m)_{m\ge0}
$$
satisfies \emph{triadic ruler invariance} when every ruler level carries
the $\mathsf P$, $\mathsf U$, and $\mathsf T$ rows of
\autoref{def:rh-internal-golden-ruler}, and every refinement map
$$
\mathcal R_m\rightsquigarrow\mathcal R_{m+1}
$$
transports those rows.  The ruler is then part of the generated system:
it is not an outside observer, and it cannot supply an outside
$\mathbb R_{>0}$ scale ledger.
\end{definition}

\begin{principle}[Machine source order]
\label{prin:rh-bedc-golden-zeta-machine-source-order}
The source order of \autoref{def:rh-bedc-golden-zeta-generation-machine}
is strict.  Raw marks generate histories.  Histories provide the time
row.  The generated zero carrier supplies the zero-presentation surface.
The prime ledger supplies logarithmic heights.  The golden ruler supplies
finite resolution.  Weak eta capture supplies local zero islands.
The Lee--Yang bridge supplies boundary stability for the same islands.
The defect-energy row then turns any off-half radial coordinate into an
unbounded normal-energy readout.  No later row may be used as an
unledgered source for an earlier row.
\end{principle}

\begin{theorem}[Constructive layer discipline preserves legality]
\label{thm:rh-constructive-layer-discipline-preserves-legality}
Let a constructive BEDC--Golden--Zeta machine have legal first layer,
and assume every generator row transports the triadic invariant and
obeys \autoref{prin:rh-no-downstream-smuggling}.  Then every displayed
layer of the machine is legal in the sense needed by
\autoref{def:rh-bedc-golden-zeta-machine-legality}.
\end{theorem}

\begin{proof}
Proceed by induction along the displayed source chain of
\autoref{def:rh-constructive-bedc-golden-zeta-machine}.  The base layer
is legal by hypothesis.  For the induction step, assume the preceding
layer is legal.  The generator row transports the source, unit-boundary,
and time invariants to the next layer.  The projection-loss ledger
records exactly what becomes invisible at that transition, while
\autoref{prin:rh-no-downstream-smuggling} forbids the next layer from
replacing that loss by an ungenerated coordinate.  Hence the compact
phase and prime-register boundaries remain the only hidden gauges, and
no $\mathbb R_{>0}$ closure coordinate is introduced.  This is precisely
the legality condition for the next layer.
\end{proof}

\begin{principle}[Internal rulers remove the measurement escape]
\label{prin:rh-internal-rulers-remove-measurement-escape}
If the golden ruler is external, a failure of the route could be hidden
in the choice of resolution, truncation, or refinement schedule.  If the
ruler is internal and satisfies
\autoref{def:rh-triadic-ruler-invariance}, the measuring rows themselves
are subject to the same source, boundary, and time invariants as the zero
certificate tower.  Thus an off-half radial factor cannot be reclassified
as a feature of measurement; it must appear in the defect ledger of the
same closed machine.
\end{principle}

\begin{theorem}[Legal generation machines exclude radial limits]
\label{thm:rh-legal-generation-machines-exclude-radial-limits}
Let a legal BEDC--Golden--Zeta generation machine produce a completed
zero certificate tower with limit readout
$$
s_{\infty}=\frac12+r+it.
$$
If its bridge witnesses supply zero-compatible boundary bounds and its
defect row satisfies
$$
E_m(s_{\infty})=r^2W_m+o(W_m),
\qquad
W_m\to\infty,
$$
then $r=0$.
\end{theorem}

\begin{proof}
The certificate row of the machine is a resolution certificate tower.
The bridge row supplies the boundary-stability component by
\autoref{thm:rh-golden-bridge-witness-supplies-boundary-stability}; the
legality row supplies the refusal of hidden real-positive scale gauge;
the defect row supplies cofinal growth and scale-energy extraction.
Thus the tower is legal boundary-stable in the sense of
\autoref{def:rh-legal-boundary-stable-certificate-tower}.  Applying
\autoref{thm:rh-no-scale-closure-legal-certificate-towers} gives
$r=0$.
\end{proof}

\begin{theorem}[Internal ruler invariance excludes measurement exceptions]
\label{thm:rh-internal-ruler-invariance-excludes-measurement-exceptions}
Let a BEDC--Golden--Zeta generation machine be legal, and assume its
golden ruler family satisfies triadic ruler invariance.  If each zero
certificate level is generated from the corresponding internal ruler
level and the tower satisfies the defect extraction and boundary-bound
rows of \autoref{thm:rh-legal-generation-machines-exclude-radial-limits},
then its limit readout has $r=0$.
\end{theorem}

\begin{proof}
By triadic ruler invariance, every refinement level of the ruler carries
the $\mathsf P$, $\mathsf U$, and $\mathsf T$ rows and transports them to
the next level.  Since each certificate level is generated from the
corresponding ruler level, the certificate tower cannot appeal to an
external resolution row, external truncation row, or external scale row.
The only visible off-half component of
$$
e^{-(s-1/2)h_a}=e^{-rh_a}e^{-it h_a}
$$
is therefore the defect term $e^{-rh_a}$ recorded by the same machine.
The hypotheses then reduce exactly to the legal generation-machine
exclusion of \autoref{thm:rh-legal-generation-machines-exclude-radial-limits},
which gives $r=0$.
\end{proof}

\begin{definition}[BEDC--Golden--Zeta source row]
\label{def:rh-bedc-golden-zeta-source-row}
The source row
$$
\mathsf{BEDCGoldenZetaMachine}_{\zeta}
$$
asserts that every accepted non-trivial completed zeta zero is presented
by a legal BEDC--Golden--Zeta generation machine whose sufficiently fine
levels carry compatible golden Lee--Yang bridge witnesses.
\end{definition}

\begin{definition}[Constructive machine source row]
\label{def:rh-constructive-bedc-golden-zeta-source-row}
The source row

\mathsf{ConstructiveBEDCGoldenZetaMachine}_{\zeta}
$$
asserts that every accepted non-trivial completed zeta zero is presented
by a constructive BEDC--Golden--Zeta machine whose layer witnesses obey
\autoref{prin:rh-no-downstream-smuggling}, whose golden ruler satisfies
triadic ruler invariance, and whose sufficiently fine levels carry
compatible golden Lee--Yang bridge witnesses.
\end{definition}

\begin{definition}[Zeta constraint container]
\label{def:rh-zeta-constraint-container}
Let $\Lambda$ be the ledger of a compatible golden zero-certificate
tower.  A \emph{zeta constraint container} is the specialization of
\autoref{def:philosophy-constraint-container} to the RH-facing candidate
class
$$
\mathcal G_{\zeta,\varphi}
$$
whose visible candidates include the critical container
$$
\Gamma_{\mathrm{crit}}
:=
\{s:\Re(s)=1/2\}
$$
and the radial containers
$$
\Gamma_r
:=
\{s:\Re(s)=1/2+r\},
\qquad
r\ne0.
$$
Its probe row reads zero-island capture, Rouch\'e or winding data,
phase, prime-register fibres, Zeckendorf resolution ledgers, boundary
stability, and defect energy.  Its gauge row may contain only compact
phase, finite prime-register, and Zeckendorf-ledger freedoms already
displayed by the machine.  The container is therefore a stabilized
closure of ledger constraints, not a prior host line into which zeros
are placed.
\end{definition}

\begin{definition}[Zeta critical-container forcing]
\label{def:rh-zeta-critical-container-forcing}
For the limit ledger $\Lambda_{\infty}$ of a compatible golden
zero-certificate tower, write
$$
\Lambda_{\infty}
\Vdash_{\zeta,\varphi}
[\Gamma_{\mathrm{crit}}]
$$
when $\Gamma_{\mathrm{crit}}$ realizes the ledger and every candidate
$$
\Delta\in\mathcal G_{\zeta,\varphi}
$$
realizing the same ledger has the same probe readout as
$\Gamma_{\mathrm{crit}}$ across the probe row of
\autoref{def:rh-zeta-constraint-container}, modulo only the displayed
compact phase, prime-register, and Zeckendorf gauge rows.  Equivalence
at the completed-zero readout means that every presentation carried by
$\Delta$ has $r=0$ in
$$
s=\frac12+r+it.
$$
\end{definition}

\begin{theorem}[Constructive source rows imply machine source rows]
\label{thm:rh-constructive-source-row-implies-machine-source-row}
If
$$
\mathsf{ConstructiveBEDCGoldenZetaMachine}_{\zeta}
$$
holds, then
$$
\mathsf{BEDCGoldenZetaMachine}_{\zeta}
$$
holds.
\end{theorem}

\begin{proof}
The constructive source row supplies, for every accepted non-trivial
completed zeta zero, a constructive BEDC--Golden--Zeta machine.  By
\autoref{thm:rh-constructive-layer-discipline-preserves-legality}, its
layers are legal.  The same source row also supplies triadic ruler
invariance and compatible golden Lee--Yang bridge witnesses.  Forgetting
the displayed layer-witness quadruples leaves exactly the legal
BEDC--Golden--Zeta generation machine required by
\autoref{def:rh-bedc-golden-zeta-source-row}.
\end{proof}

\begin{corollary}[The machine source row gives fixed-half readout]
\label{cor:rh-bedc-golden-zeta-machine-fixed-half-readout}
If
$$
\mathsf{BEDCGoldenZetaMachine}_{\zeta}
$$
holds, then every accepted non-trivial completed zeta zero has fixed-half
readout.
\end{corollary}

\begin{proof}
Let an accepted non-trivial completed zeta zero be given.  The source
row supplies a legal generation machine presenting it.  By
\autoref{thm:rh-legal-generation-machines-exclude-radial-limits}, its
limit readout has $r=0$ in the decomposition
$$
s_{\infty}=\frac12+r+it.
$$
Therefore $\Re(s_{\infty})=1/2$.
\end{proof}

\begin{corollary}[Constructive machine source rows give fixed-half readout]
\label{cor:rh-constructive-machine-source-row-fixed-half-readout}
If
$$
\mathsf{ConstructiveBEDCGoldenZetaMachine}_{\zeta}
$$
holds, then every accepted non-trivial completed zeta zero has fixed-half
readout.
\end{corollary}

\begin{proof}
Apply
\autoref{thm:rh-constructive-source-row-implies-machine-source-row}, then
\autoref{cor:rh-bedc-golden-zeta-machine-fixed-half-readout}.
\end{proof}

\begin{theorem}[Constructive machines force the critical container]
\label{thm:rh-constructive-machines-force-critical-container}
If
$$
\mathsf{ConstructiveBEDCGoldenZetaMachine}_{\zeta}
$$
holds, then the limit ledger of every accepted non-trivial completed
zeta zero satisfies
$$
\Lambda_{\infty}
\Vdash_{\zeta,\varphi}
[\Gamma_{\mathrm{crit}}].
$$
\end{theorem}

\begin{proof}
Fix an accepted non-trivial completed zeta zero and its limit ledger.
The constructive source row supplies a constructive
BEDC--Golden--Zeta machine for this zero.  By
\autoref{thm:rh-constructive-layer-discipline-preserves-legality}, all
displayed layers are legal, and by
\autoref{thm:rh-constructive-source-row-implies-machine-source-row} the
underlying legal generation-machine source row is present.

Let $\Delta\in\mathcal G_{\zeta,\varphi}$ realize the same limit ledger.
If the readout of $\Delta$ has
$$
s=\frac12+r+it
$$
with $r\ne0$, then the defect row of the machine gives the radial
energy extraction
$$
E_m(s)=r^2W_m+o(W_m),
\qquad
W_m\to\infty,
$$
while the compatible golden Lee--Yang bridge witnesses supply the
zero-compatible boundary bound.  Since $\Delta$ realizes the same
ledger, it must realize these rows with the same legal gauge restrictions.
Thus the $\Delta$-readout satisfies the hypotheses of
\autoref{thm:rh-legal-generation-machines-exclude-radial-limits}, which
forces $r=0$, a contradiction.  Hence every surviving candidate has
fixed-half readout.  The remaining invisible rows are exactly the
displayed compact phase, prime-register, and Zeckendorf gauge rows, so
all surviving candidates have the same probe readout as
$\Gamma_{\mathrm{crit}}$ on the probes of
\autoref{def:rh-zeta-constraint-container}.  This is the forcing
judgment of \autoref{def:rh-zeta-critical-container-forcing}.
\end{proof}

\begin{principle}[Machine proof load]
\label{prin:rh-bedc-golden-zeta-machine-proof-load}
The machine formulation separates the proof load into source rows.  The
kernel and history rows are supplied by the BEDC spine.  The golden
resolution and weak eta capture rows are supplied by the local
Rouch\'e construction of
\autoref{thm:rh-golden-eta-weak-active-capture}.  The strong source
content has three coupled parts: constructive layer witnesses with no
downstream smuggling, triadic ruler invariance for the golden resolution
row, and compatible golden Lee--Yang bridge witnesses for the same zero
islands, including dominance, same-island transfer, radial purity, and
far-end energy control.  Those rows are the mathematical content of
$$
\mathsf{BEDCGoldenZetaMachine}_{\zeta}.
$$
Under those rows, the public geometric readout is the derived constraint
container judgment
$$
\Lambda_{\infty}
\Vdash_{\zeta,\varphi}
[\Gamma_{\mathrm{crit}}],
$$
not an initial assumption that zero certificates already live on the
critical line.
\end{principle}
