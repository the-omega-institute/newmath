\section{Unit Phase and Orthogonal Readout}
\label{sec:rh-phase-triangle-unit-readout}

\begin{definition}[Unit phase readout]
\label{def:rh-unit-phase-readout}
A unit phase row is a circle-valued coordinate
$$
u(\theta)=e^{i\theta}\in\mathbb T.
$$
Its orthogonal coordinate readout is
$$
\rho_{\perp}(u(\theta))
:=
(\cos\theta,\sin\theta),
$$
with invariant
$$
\cos^2\theta+\sin^2\theta=1.
$$
The coordinate pair is therefore a finite visible vector event on the
unit-circle surface, not a duplicate of every source row that produced
the phase.
\end{definition}

\begin{principle}[Triangle readout preserves norm and hides provenance]
\label{prin:rh-triangle-readout-preserves-norm-hides-provenance}
The passage
$$
e^{i\theta}\mapsto(\cos\theta,\sin\theta)
$$
preserves the quadratic norm row and exposes an orthogonal two-coordinate
event. It does not by itself expose winding history, cover data, branch
choice, prime-local provenance, or any hidden fiber attached to the phase
row. Those rows must be carried by separate ledgers if a later consumer
is allowed to recover them.
\end{principle}

\begin{definition}[Phase-triangle event]
\label{def:rh-phase-triangle-event}
A phase-triangle event is a readout packet
$$
T_{\theta}
:=
\bigl(x,y,h_{\theta}\bigr),
\qquad
x=\cos\theta,\quad y=\sin\theta,
$$
where
$$
h_{\theta}:x^2+y^2=1.
$$
It is the visible orthogonal slice of the phase row $e^{i\theta}$. In BEDC
terms, it is an output surface with a quadratic invariant, not the full
source carrier.
\end{definition}

\section{Prime Register Residue}
\label{sec:rh-prime-register-residue}

\begin{definition}[Solenoid phase register]
\label{def:rh-solenoid-phase-register}
In the notation of
\autoref{def:unified-reality-s-solenoid-over-unit-circle}, the solenoid
phase-register readout is the visible projection
$$
\Sigma_S
\to
\mathbb T
$$
with fiber $K_S=\ker(\pi_S)$. For finite $S$, this fiber is the
prime-register residue behind the circle phase: it records the
prime-adic source distinctions that are not visible in the circle
coordinate.
\end{definition}

\begin{principle}[Prime registers are not phase coordinates]
\label{prin:rh-prime-registers-not-phase-coordinates}
The projection $\Sigma_S\to\mathbb T$ licenses a visible phase readout,
but it does not erase the source ledger. Two source rows in $\Sigma_S$ may
share the same visible phase while differing by an element of
$K_S$. Thus a solenoid is a phase quotient with a profinite prime fiber,
not a bare circle and not an ambient real-scale container.
\end{principle}

\begin{definition}[Projection residue]
\label{def:rh-projection-residue}
For a displayed readout
$$
\pi:E\to V,
$$
the residue over a visible row $v:V$ is the fiber
$$
\operatorname{Res}_{\pi}(v)
:=
\sum_{e:E}\bigl(\pi(e)\sim_V v\bigr).
$$
A visible coordinate is lossless only when the packet also supplies a
certified section or classifier-faithful reconstruction row for this
residue. Otherwise the readout has exposed a surface while retaining the
unread distinctions as fiber or ledger data.
\end{definition}

\begin{principle}[Projection-residue uncertainty]
\label{prin:rh-projection-residue-uncertainty}
A single low-dimensional visible readout cannot, without additional
ledger rows, be simultaneously the full continuous phase, the full
prime-register residue, and an independent radial scale coordinate. A
readout may expose one surface and preserve a stated invariant, but every
unexposed source distinction must remain in a fiber, a transport ledger,
or an explicit obstruction row. This is the BEDC projection-residue form
of uncertainty.
\end{principle}

\section{Centered Zeta Modes}
\label{sec:rh-centered-zeta-modes-phase-scale}

\begin{definition}[Centered phase-scale mode]
\label{def:rh-centered-phase-scale-mode}
Let $h_a$ be a displayed positive frequency or prime-ledger height, and
write
$$
s=\frac12+r+it,
\qquad
r,t\in\RealUp.
$$
The centered zeta mode attached to $a$ is
$$
U_a(s)
:=
e^{-(s-\frac12)h_a}
=
e^{-rh_a}e^{-ith_a}.
$$
The factor $e^{-ith_a}$ is the unit-circle phase read. The factor
$e^{-rh_a}$ is a positive real radial-scale read. The source of $h_a$ is
the arithmetic or prime-ledger row that determines which frequency has
been inspected.
\end{definition}

\begin{principle}[Unit phase closure excludes radial scale]
\label{prin:rh-unit-phase-closure-excludes-radial-scale}
If a zero-relevant zeta packet is licensed as a pure unit phase closure,
then every centered mode must have no independent positive real scale
factor. In the notation of \autoref{def:rh-centered-phase-scale-mode},
that means the mode is read as
$$
U_a(s)=e^{-ith_a}\in\mathbb T
$$
rather than as
$$
e^{-rh_a}e^{-ith_a}
$$
with an extra radial register. This is the same proof role as the
no-independent-real-scale obligation in
\autoref{ch:visions-rh-unitary-scale-closure-route}.
\end{principle}

\begin{definition}[Orthogonal scale-phase energy readout]
\label{def:rh-orthogonal-scale-phase-energy-readout}
For a finite active packet $A_m$, define
$$
v_a:=(-rh_a,-th_a),
\qquad
W_m:=\sum_{a\in A_m}h_a^2.
$$
The orthogonal energy readout is
$$
|v_a|^2=r^2h_a^2+t^2h_a^2,
$$
and its scale component is
$$
E_m^{\mathrm{scale}}:=r^2W_m.
$$
\end{definition}

\begin{definition}[Scale-defect positivity packet]
\label{def:rh-scale-defect-positivity-packet}
A scale-defect positivity packet for the active family $(A_m)$ consists
of a displayed defect readout $D_m^{\mathrm{scale}}(s)$ and a bound row
$B$ such that
$$
D_m^{\mathrm{scale}}(s)
=
r^2W_m+\varepsilon_m,
\qquad
\varepsilon_m=o(W_m),
$$
and
$$
D_m^{\mathrm{scale}}(s)\le B
$$
for every displayed window $m$. The notation records an asymptotic
obligation inside the chosen analytic interface; it is not an automatic
consequence of the centered-mode definition.
\end{definition}

\begin{theorem}[Cofinal scale energy is incompatible with pure phase closure]
\label{thm:rh-cofinal-scale-energy-incompatible-pure-phase}
Assume a cofinal family of active packets satisfies $W_m\to\infty$. If
$r\#0$, then
$$
E_m^{\mathrm{scale}}\to\infty.
$$
Consequently any zero-relevant closure theorem that supplies pure unit
phase closure together with a bounded or absent radial-scale ledger must
force $r=0$.
\end{theorem}

\begin{proof}
By \autoref{def:rh-orthogonal-scale-phase-energy-readout},
$$
E_m^{\mathrm{scale}}=r^2W_m.
$$
If $r\#0$, then $r^2>0$, so the cofinal divergence of $W_m$ gives
$E_m^{\mathrm{scale}}\to\infty$. A pure unit phase closure has no
independent positive real scale row. Therefore a bounded or absent
scale-ledger condition is incompatible with $r\#0$, and the centered
coordinate must satisfy $r=0$.
\end{proof}

\begin{theorem}[Positive scale-defect criterion]
\label{thm:rh-positive-scale-defect-criterion}
Assume $W_m\to\infty$ and a scale-defect positivity packet in the sense of
\autoref{def:rh-scale-defect-positivity-packet}. Then $r$ is not apart
from $0$. In any located real readback that collapses absence of
apartness to equality at the stated resolution, the centered coordinate
satisfies $r=0$.
\end{theorem}

\begin{proof}
Assume $r\#0$. Then $r^2>0$. Since
$\varepsilon_m=o(W_m)$, there is a cofinal tail on which
$$
|\varepsilon_m|\le\frac{r^2}{2}W_m.
$$
Hence on that tail
$$
D_m^{\mathrm{scale}}(s)
\ge
\frac{r^2}{2}W_m.
$$
Because $W_m\to\infty$, the right side is unbounded. This contradicts the
displayed bound row $D_m^{\mathrm{scale}}(s)\le B$. Therefore the
apartness row $r\#0$ is impossible. A located real readback then gives the
displayed equality row $r=0$ at the resolution licensed by the packet.
\end{proof}
