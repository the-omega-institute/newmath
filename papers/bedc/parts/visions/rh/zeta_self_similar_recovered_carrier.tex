\begin{definition}[Prime-local geometric fiber]
\label{def:rh-prime-local-geometric-fiber}
For a displayed prime row $p\in\PrimeUp$ and a complex history $s$ in a
convergence window where the local series is licensed, the
prime-local geometric fiber is
$$
E_p(s):=\sum_{k\geq0}p^{-ks}.
$$
It carries the one-step recursion
$$
p^{-ks}\mapsto p^{-(k+1)s}
$$
with local ratio $p^{-s}$. Equivalently, on the Euler-product window,
$$
E_p(s)=(1-p^{-s})^{-1}.
$$
The fiber is local to the prime row and to the declared convergence and
branch ledgers.
\end{definition}

\end{definition}

\begin{principle}[Zeta self-similarity is carrier-level]
\label{prin:rh-zeta-self-similarity-carrier-level}
The expressions
$$
\sum_{n\geq1}n^{-s}
$$
and
$$
\prod_{p\in\PrimeUp}(1-p^{-s})^{-1}
$$
are projections of $\mathsf{ZETA}$ on the right half-plane, not
standalone definitions of the full carrier. The self-similar structure
belongs to the recovered carrier only after the Dirichlet row, Euler
row, analytic continuation row, archimedean completion row, functional
equation row, pole ledger, zero ledger, and branch transport ledger are
displayed together.
\end{principle}

\begin{proposition}[Euler gluing is multiplicative self-similarity]
\label{prop:rh-euler-gluing-multiplicative-self-similarity}
On the displayed Euler-product window, the zeta projection is obtained
by gluing the prime-local geometric fibers:
$$
\zeta_D(s)
=
\prod_{p\in\PrimeUp}E_p(s)
=
\prod_{p\in\PrimeUp}(1-p^{-s})^{-1}.
$$
This is a multiplicative self-similar projection indexed by ordinary
prime rows.
\end{proposition}

\begin{proof}
By \autoref{prop:rh-euler-product-prime-local-gluing}, the Dirichlet row
and Euler row agree on $\Re(s)>1$. The proof of that proposition groups
the Dirichlet terms by the unique factorization row
$$
n=\prod_p p^{k_p}
$$
and rewrites
$$
n^{-s}=\prod_p p^{-k_ps}.
$$
Thus the global right-half-plane projection is the product of the local
geometric fibers $E_p(s)$.
\end{proof}

\begin{definition}[Prime-occupancy partition readout]
\label{def:rh-prime-occupancy-partition-readout}
Let $T_{\mathbb P}$ be the prime-occupancy carrier of
\autoref{def:layered-projection-prime-occupancy-carrier}. On a displayed
right-half-plane convergence window, the prime-occupancy partition
readout is
$$
Z_{\mathbb P}(s)
:=
\sum_{k\in T_{\mathbb P}}
\exp\left(
-s\sum_{p\in\mathbb P}k_p\log p
\right).
$$
The sum ranges over finite-support prime occupancy rows. It is therefore
the analytic readout of the multiplicative arithmetic presentation, not
a separate unledgered source carrier.
\end{definition}

\begin{proposition}[Zeta is the prime-occupancy partition readout]
\label{prop:rh-zeta-prime-occupancy-partition-readout}
On the Euler-product window,
$$
Z_{\mathbb P}(s)=\zeta_D(s).
$$
Equivalently,
$$
Z_{\mathbb P}(s)
=
\prod_{p\in\PrimeUp}
\left(\sum_{k\geq0}p^{-ks}\right)
=
\prod_{p\in\PrimeUp}(1-p^{-s})^{-1}.
$$
\end{proposition}

\begin{proof}
For a finite-support occupancy row $k$, its readback integer is
$$
\Psi(k)=\prod_p p^{k_p}
$$
by \autoref{def:layered-projection-arithmetic-multiplication-presentation}.
Thus
$$
\log\Psi(k)=\sum_p k_p\log p
$$
and
$$
\exp\left(-s\sum_p k_p\log p\right)
=
\prod_p p^{-sk_p}.
$$
Summing over all finite-support occupancy rows is the same as summing
over positive integers by unique factorization, hence gives the
Dirichlet row. Splitting the independent prime coordinates gives the
Euler product by \autoref{prop:rh-euler-gluing-multiplicative-self-similarity}.
\end{proof}

\begin{principle}[Zero packets are recovered-carrier cancellations]
\label{prin:rh-zero-packets-recovered-carrier-cancellations}
In the prime-occupancy presentation, $\zeta_D$ is the analytic partition
readout of finite prime configurations on the right half-plane. A
non-trivial zero packet is not a bare row of $T_{\mathbb P}$ and not a
zero of one local geometric fiber. It belongs to the completed recovered
carrier assembled from Dirichlet, Euler, continuation, archimedean
completion, functional-equation, pole, zero, branch, and transport
ledgers.
\end{principle}

\begin{proposition}[Dirichlet projection is a discrete Laplace readout]
\label{prop:rh-zeta-self-similar-laplace-readout}
For the zeta coefficients $a_n=1$, the Dirichlet projection is the
discrete Laplace readout of the atomic source
$$
\mu_{\zeta}:=\sum_{n\geq1}\delta_{\log n}.
$$
That is, on any displayed convergence window,
$$
\sum_{n\geq1}n^{-s}
=
\mathcal L[\mu_{\zeta}](s).
$$
\end{proposition}

\begin{proof}
This is \autoref{prop:rh-layered-dirichlet-laplace-readout} with
$a_n=1$. The equality follows from
$$
e^{-s\log n}=n^{-s}
$$
on the declared branch and convergence window.
\end{proof}

\begin{proposition}[Euler phase channels are prime-axis readouts]
\label{prop:rh-euler-phase-channels-prime-axis-readouts}
On the displayed right-half-plane window, the exponential kernel sends a
finite-support prime occupancy row $k\in T_{\mathbb P}$ to the product of
its prime-axis phase-scale channels:
$$
\exp\left(-s\sum_{p}k_p\log p\right)
=
\prod_p\left(p^{-\sigma}e^{-i\tau\log p}\right)^{k_p}
$$
for $s=\sigma+i\tau$.
\end{proposition}

\begin{proof}
The prime-occupancy readback gives
$$
\log\Psi(k)=\sum_p k_p\log p.
$$
Substituting $s=\sigma+i\tau$ into the exponential kernel gives
$$
e^{-s\log p}=e^{-\sigma\log p}e^{-i\tau\log p}=p^{-\sigma}e^{-i\tau\log p}.
$$
Multiplying these local reads over the finite support of $k$ yields the
displayed product.
\end{proof}

\begin{definition}[Recovered-carrier self-dual mirror]
\label{def:rh-recovered-carrier-self-dual-mirror}
The recovered-carrier mirror is the functional-equation readback

$$
Writing

It preserves the prime phase clock $\tau\log p$ and reverses the
prime-amplitude bias $\delta\log p$ in the centered channels, as in
\autoref{prop:rh-mirror-prime-amplitude-bias}.
\end{definition}

\begin{theorem}[RH is fixed-half zero ledger for the self-similar carrier]
\label{thm:rh-self-similar-carrier-fixed-half-zero-ledger}
Under the recovered-carrier reading of
\autoref{def:rh-multiplicative-self-similar-zeta-packet}, the Riemann
Hypothesis is the assertion that every recognized non-trivial zero
packet lies in the fixed-half surface of the mirror:
$$
\mathsf{ZetaZero}(s)\wedge\mathsf{InCritStrip}(s)
\Longrightarrow
s=J(s).
$$
Equivalently, every recognized non-trivial zero packet has no
prime-visible mirror amplitude bias:
$$
\mathsf{ZetaZero}(s)\wedge\mathsf{InCritStrip}(s)
\Longrightarrow
\mathsf{PF}(s)=\mathsf{PF}(J(s)).
$$
\end{theorem}

\begin{proof}
The first displayed implication is the fixed-surface reading of
\autoref{thm:rh-layered-translation-synthesis}. The equivalence with
prime-fingerprint collapse is
\autoref{thm:rh-prime-fingerprint-normal-form}. The mirror and centered
prime-channel interpretation is supplied by
\autoref{def:rh-recovered-carrier-self-dual-mirror} and
\autoref{prop:rh-mirror-prime-amplitude-bias}.
\end{proof}

\begin{principle}[Self-similar comparison boundary]
\label{prin:rh-zeta-fibonacci-self-similar-comparison-boundary}
The zeta packet and the Fibonacci-Zeckendorf packet are both
self-similar only after their classifiers are named. The Fibonacci route
uses second-order recursive axis normalization
$$
e_k+e_{k+1}\rightsquigarrow e_{k+2}.
$$
The zeta route uses prime-indexed local geometric fibers and
multiplicative Euler gluing
$$
E_p(s)=1+p^{-s}+p^{-2s}+\cdots.
$$
The comparison is structural: it identifies classifier-relative
self-similar recovery patterns, not a common ordinary geometric object
and not a proof of the zero-ledger exclusion.
\end{principle}
