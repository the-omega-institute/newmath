\chapter{Finite Prime Circle Off-Line Contradiction}
\label{ch:visions-rh-finite-prime-circle-offline-contradiction}
\origin{ai}

\autoref{ch:visions-rh-global-offline-zero-finite-np-window} supplies the
source bridge from an off-line radial row to a finite prime-power
window or to the all-prime tail ledger.  The present chapter packages
the conditional consequence for legal zeta zero packets: if completed
zeta and its non-trivial zeros are admitted as generated BEDC objects
with ontological triples and closure ledgers, then an off-line zero
cannot also be closed.

\section{Legal Zeta Zero Packets}
\label{sec:rh-prime-circle-legal-zero-packets}

\begin{definition}[Zeta legal zero packet]
\label{def:rh-zeta-legal-zero-packet}
A zeta legal zero packet is a completed non-trivial zero packet $\rho$
carried by the generated zeta closure row
\autoref{def:rh-zeta-bedc-closure}, equipped with:
$$
\begin{gathered}
\operatorname{Ont}(\rho)=(\mathsf T_\rho,\mathsf U_\rho,\mathsf L_\rho),\\
\mathsf{GlobalOffLineZeroToFiniteNPWindow},\\
\mathsf{LedgerClosureRefusal}.
\end{gathered}
$$
The first row is the ontological triple of
\autoref{def:rh-ontological-triple}.  The second row is the finite or
tail source bridge of
\autoref{def:rh-global-offline-zero-to-finite-np-window}.  The third
row is the closed-packet refusal of \autoref{def:rh-ledger-closure-refusal}.
\end{definition}

\begin{definition}[Closed zeta zero packet]
\label{def:rh-closed-zeta-zero-packet}
A zeta legal zero packet is closed when no nonzero residual row remains
as an unprocessed append row.  A detected residual that is not recorded
violates $\mathsf L_\rho$; a detected residual that is recorded is a
continuing ledger row rather than closed zero-packet status.
\end{definition}

\begin{principle}[Zeta-shaped statements are theorem targets]
\label{prin:rh-zeta-shaped-statements-theorem-targets}
The rows $\mathsf{ZetaBEDCClosure}$,
$\mathsf{GlobalOffLineZeroToFiniteNPWindow}$, and the resulting
fixed-half readout are theorem-facing rows.  They are not seed names for
zeta or for $\mathsf{RH}$.  Each must be supplied by the Dirichlet,
Euler product, analytic-continuation, explicit-formula, finite-window,
and ledger certificates on which it depends.
\end{principle}

\section{Off-Line Residual Refusal}
\label{sec:rh-prime-circle-offline-residual-refusal}

\begin{theorem}[Off-line legal zero packets expose residuals]
\label{thm:rh-offline-legal-zero-packets-expose-residuals}
If a zeta legal zero packet is off the critical line, then it supplies a
nonzero radial residual with finite-window or tail status.
\end{theorem}

\begin{proof}
An off-line zero has a nonzero radial row by
\autoref{def:rh-offline-radial-distribution-row} and
\autoref{thm:rh-offline-zero-supplies-compact-radial-witness}.  The
global bridge row in \autoref{def:rh-zeta-legal-zero-packet} applies to
that residual.  By
\autoref{thm:rh-global-bridge-gives-finite-or-tail-residual-status},
the residual is represented in a legal finite $(N,P)$ window or is
recorded in $\operatorname{TailPolicy}_{\mathcal P}$.
\end{proof}

\begin{theorem}[Finite prime circle off-line contradiction]
\label{thm:rh-finite-prime-circle-offline-contradiction}
A zeta legal zero packet cannot be both off the critical line and
closed.
\end{theorem}

\begin{proof}
Suppose that a zeta legal zero packet $\rho$ is off the critical line.
By
\autoref{thm:rh-offline-legal-zero-packets-expose-residuals}, its
radial residual has finite-window or tail status.

If the residual is finite-window represented, then
\autoref{thm:rh-global-bridge-feeds-finite-address-refusal} applies:
the internal prime-power detector, finite prime-circle detector, gap
ledger, or tail ledger prevents the packet from being closed and missed.
If the residual is tail-recorded, then
\autoref{thm:rh-tail-address-cannot-close-silently} says that the tail
row is an append ledger row; erasing it would violate $\mathsf L_\rho$.
Both branches are precisely the alternatives refused by
\autoref{def:rh-ledger-closure-refusal}.  Therefore $\rho$ cannot be a
closed packet.
\end{proof}

\begin{corollary}[Legal closed zeta zeros have fixed-half readout]
\label{cor:rh-legal-closed-zeta-zeros-fixed-half-readout}
Every closed zeta legal zero packet has
$$
\operatorname{Re}(\rho)=\frac12.
$$
\end{corollary}

\begin{proof}
If $\operatorname{Re}(\rho)\ne1/2$, then $\rho$ is off the critical line.
By
\autoref{thm:rh-finite-prime-circle-offline-contradiction}, it cannot be
closed.  Hence every closed zeta legal zero packet has no off-line
radial displacement.  Equivalently,
\autoref{thm:rh-li-coordinate-fixed-half-circle} gives
$\operatorname{Re}(\rho)=1/2$ from the unit Li readout.
\end{proof}

\section{Conditional RH Interface}
\label{sec:rh-prime-circle-conditional-rh-interface}

\begin{definition}[PrimeCircleBEDCRoute]
\label{def:rh-prime-circle-bedc-route}
The route
$$
\mathsf{PrimeCircleBEDCRoute}
$$
consists of the following rows:
$$
\begin{gathered}
\mathsf{ZetaBEDCClosure},\\
\text{every admitted non-trivial zero is a zeta legal zero packet},\\
\text{every admitted non-trivial zero packet is closed},\\
\mathsf{GlobalOffLineZeroToFiniteNPWindow}.
\end{gathered}
$$
It is a conditional route: each row is an explicit certificate or
theorem target, not a primitive assertion about zeta.
\end{definition}

\begin{theorem}[PrimeCircleBEDCRoute gives the BEDC RH predicate]
\label{thm:rh-prime-circle-bedc-route-gives-rh-predicate}
Assume $\mathsf{PrimeCircleBEDCRoute}$.  Then every admitted
non-trivial zero satisfies the fixed-half predicate of
\autoref{def:rh-predicate}.
\end{theorem}

\begin{proof}
Let $\rho$ be an admitted non-trivial zero packet.  By the route, it is
a zeta legal zero packet and it is closed.  Apply
\autoref{cor:rh-legal-closed-zeta-zeros-fixed-half-readout}.  The
fixed-half readout is exactly the $\mathsf{OnCritLine}$ conclusion in
\autoref{def:rh-predicate}.
\end{proof}

\begin{corollary}[PrimeCircle conditional RH]
\label{cor:rh-prime-circle-conditional-rh}
Under $\mathsf{PrimeCircleBEDCRoute}$, the BEDC RH statement is
inhabited as a conditional theorem target.
\end{corollary}

\begin{proof}
The predicate $\mathsf{RH}$ is the $\Pi$-statement over admitted
non-trivial zero packets recorded in \autoref{def:rh-predicate}.  The
preceding theorem supplies the fixed-half output for each such packet.
Thus the conditional route inhabits the BEDC RH statement.
\end{proof}

\begin{principle}[Prime circle contradiction audit boundary]
\label{prin:rh-prime-circle-contradiction-audit-boundary}
The contradiction is conditional on the legal-object rows.  It proves
that the simultaneous package
$$
\text{off-line zero}
+\text{ontological triple}
+\text{closed zero packet}
+\mathsf{GlobalOffLineZeroToFiniteNPWindow}
$$
is impossible.  It does not assert that $\mathsf{ZetaBEDCClosure}$,
zero-packet closure, or the global window bridge are primitive facts;
those remain the explicit certificates required before the conditional
route can be read as a BEDC proof of $\mathsf{RH}$.
\end{principle}
