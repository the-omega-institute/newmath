\chapter{RH Proof Obligation Package}
\label{ch:visions-rh-proof-obligation-package}
\origin{ai}

\autoref{ch:visions-rh-zeta-zero-type-factorization} isolates the
type-theoretic zero fiber, and
\autoref{ch:visions-rh-bulk-boundary-prime-shadow-interface} isolates the
prime-shadow rigidity interface. The present chapter records the proof
obligation package obtained by composing those two interfaces. It is not a
proof of $\mathsf{RH}$. It states which rows are definitional, which rows
are already structural consequences, and which rows carry the remaining
analytic content.

\section{Constructive Target}
\label{sec:rh-proof-obligation-constructive-target}

\begin{definition}[RH witness function]
\label{def:rh-proof-obligation-witness-function}
The BEDC-RH witness function is the dependent function
$$
\Phi:
\mathsf{NTZeroType}\to\mathsf{FixedHalfZeroType}
$$
whose projection back to $\mathsf{NTZeroType}$ is classifier-equal to the
identity. Equivalently, for every source row $s$ with
$$
\mathsf{ZetaZero}(s)\times\mathsf{InCritStrip}(s),
$$
it returns a displayed $\mathsf{OnCritLine}(s)$ witness. This is the
constructive target of \autoref{def:rh-predicate} and the section form of
\autoref{thm:rh-zeta-zero-type-rh-fixed-half-coverage}.
\end{definition}

\begin{definition}[Quantitative off-line barrier]
\label{def:rh-proof-obligation-quantitative-barrier}
A quantitative off-line barrier is a function
$$
\beta:\NatUp\times\NatUp\to\RatUp_{>0}
$$
such that for every critical-strip row $s$,
$$
|\Re(s)-1/2|>2^{-n}
\wedge
|\Im(s)|\le T
\to
|\zeta(s)|\ge \beta(n,T).
$$
It is the positive-apartness strengthening on a displayed compact height
window. A height-uniform function of $n$ alone is a separate global analytic
barrier assumption and is not part of the ordinary fixed-half witness.
\end{definition}

\begin{principle}[Refutation data must be typed]
\label{prin:rh-proof-obligation-refutation-data-typed}
A contradiction route contributes to the BEDC-RH witness target only when
it is typed as a same-source conflict between a zero ledger and an
apart-zero ledger. A classical refutation without the transport rows needed
to build $\Phi$ does not inhabit \autoref{def:rh-proof-obligation-witness-function}.
\end{principle}

\section{Structural Rows}
\label{sec:rh-proof-obligation-structural-rows}

\begin{definition}[Zero-fiber rows]
\label{def:rh-proof-obligation-zero-fiber-rows}
The zero-fiber part of the package consists of the following rows:
$$
\mathsf{ZetaZeroType}\simeq\operatorname{Fib}_{0}(\zeta),
\qquad
\mathsf{NTZeroType}
\simeq
\sum_s
\bigl(
\mathsf{ZetaZero}(s)\times\mathsf{InCritStrip}(s)
\bigr),
$$
and the section equivalence
$$
\mathsf{RH}
\Longleftrightarrow
\mathsf{NTZeroType}\to\mathsf{FixedHalfZeroType}.
$$
These are structural descriptions of the target, not analytic exclusions.
\end{definition}

\begin{definition}[Zero-generator rows]
\label{def:rh-proof-obligation-zero-generator-rows}
The zero-generator part of the package consists of the factorization row
$$
G:A\to\mathsf{ZetaDom}^{\uparrow},
\quad
\theta_G(a):
\sum_{z:\ComplexUp}
\bigl(
\mathsf{ZetaVal}(G(a),z)
\times
z\sim_{\ComplexUp}e_{\ComplexUp}
\bigr)
\quad
\Longrightarrow
\quad
\overline G:A\to\mathsf{ZetaZeroType},
$$
and the canonical-presentation row asserting that two canonical
presentations of the same zero type are equivalent licensed
presentations. Hence incompatible canonical generators cannot be
presentations of one zero type.
\end{definition}

\begin{definition}[Omega representation row]
\label{def:rh-proof-obligation-omega-representation-row}
The $\Omega$ representation row is the assertion that every legal
non-trivial zeta-zero packet is represented by some row of the Omega
non-trivial zero carrier
$$
T_{\Omega}^{nt0}
:=
\sum_{x:T_{\Omega}}
\left(
\mathsf{ZetaZero}(\pi_{\zeta}(x))
\times
\mathsf{InCritStrip}(\pi_{\zeta}(x))
\right).
$$
This row is the transport bridge from the zero-type statement to the
bulk-boundary tower statement.
\end{definition}

\section{Rigidity Rows}
\label{sec:rh-proof-obligation-rigidity-rows}

\begin{definition}[Prime normal defect]
\label{def:rh-proof-obligation-prime-normal-defect}
The prime normal defect predicate is the tower-visible defect
$$
\mathsf{PrimeNormalDefect}_{\Omega}(x)
:=
\sum_{i\in I}\mathsf{PrimeDefect}_i(x),
$$
where each $\mathsf{PrimeDefect}_i(x)$ is a nonzero prime-shadow readout
at boundary layer $i$. Its negation is
$$
\mathsf{NoPrimeNormalDefect}_{\Omega}(x)
:=
\neg\mathsf{PrimeNormalDefect}_{\Omega}(x).
$$
This notation packages the same defect surface as
\autoref{def:rh-prime-shadow-defect-interface}; it emphasizes the normal
direction read by finite prime-window and bulk-boundary shadows.
\end{definition}

\begin{definition}[Zero atom prime-normal rigidity]
\label{def:rh-proof-obligation-zero-atom-prime-normal-rigidity}
Zero atom prime-normal rigidity is the statement
$$
\mathsf{ZeroAtomNoPrimeNormalDefect}_{\Omega}
:=
\prod_{x:T_{\Omega}}
\left(
\mathsf{NTZeroAtom}_{\Omega}(x)
\to
\mathsf{NoPrimeNormalDefect}_{\Omega}(x)
\right).
$$
It says that a non-trivial zero atom may carry phase data, but cannot carry
a nonzero prime-normal defect.
\end{definition}

\begin{definition}[Prime-normal critical-line row]
\label{def:rh-proof-obligation-prime-normal-critical-line}
The prime-normal critical-line row is
$$
\mathsf{PrimeNormalImpliesCriticalLine}_{\Omega}
:=
\prod_{x:T_{\Omega}}
\left(
\mathsf{NTZeroAtom}_{\Omega}(x)
\times
\mathsf{NoPrimeNormalDefect}_{\Omega}(x)
\to
\mathsf{OnCritLine}(\pi_{\zeta}(x))
\right).
$$
This is the fixed-half interpretation of the absence of prime-normal
defect.
\end{definition}

\begin{definition}[Apart-zero defect row]
\label{def:rh-proof-obligation-apart-zero-defect-row}
The apart-zero defect row is
$$
\mathsf{PrimeDefectApartZero}_{\Omega}
:=
\prod_{x:T_{\Omega}}
\left(
\mathsf{InCritStrip}(\pi_{\zeta}(x))
\times
\mathsf{PrimeNormalDefect}_{\Omega}(x)
\to
\mathsf{Apart}_{0}(\zeta(\pi_{\zeta}(x)))
\right).
$$
Together with zero-ledger soundness, it turns prime-normal defect into a
same-source zero and apart-zero conflict.
\end{definition}

\begin{theorem}[Apart-zero defect gives zero atom prime-normal rigidity]
\label{thm:rh-proof-obligation-apart-zero-gives-zero-atom-rigidity}
Assume $\mathsf{PrimeDefectApartZero}_{\Omega}$ and zero-ledger soundness,
namely
$$
\mathsf{ZeroAtom}_{\Omega}(x)
\to
\zeta(\pi_{\zeta}(x))=0.
$$
Then $\mathsf{ZeroAtomNoPrimeNormalDefect}_{\Omega}$ holds.
\end{theorem}

\begin{proof}
Let $x:T_{\Omega}$ carry $\mathsf{NTZeroAtom}_{\Omega}(x)$. Suppose, for
contradiction, that $\mathsf{PrimeNormalDefect}_{\Omega}(x)$ holds. The
non-trivial zero atom supplies both $\mathsf{InCritStrip}(\pi_{\zeta}(x))$
and $\mathsf{ZeroAtom}_{\Omega}(x)$. The apart-zero defect row gives
$$
\mathsf{Apart}_{0}(\zeta(\pi_{\zeta}(x))).
$$
Zero-ledger soundness gives $\zeta(\pi_{\zeta}(x))=0$. This is the
same-source conflict of
\autoref{thm:rh-zeta-zero-apart-conflict-impossible}. Hence the assumed
prime-normal defect is impossible.
\end{proof}

\begin{theorem}[Proof obligation package implies RH]
\label{thm:rh-proof-obligation-package-implies-rh}
Assume the following rows:
$$
\mathsf{PrimeDefectApartZero}_{\Omega},
\qquad
\text{zero-ledger soundness},
\qquad
\mathsf{PrimeNormalImpliesCriticalLine}_{\Omega},
$$
and representation of every legal non-trivial zeta-zero packet in
$T_{\Omega}^{nt0}$. Then the BEDC-RH predicate holds.
\end{theorem}

\begin{proof}
By \autoref{thm:rh-proof-obligation-apart-zero-gives-zero-atom-rigidity},
the apart-zero defect row and zero-ledger soundness give
$\mathsf{ZeroAtomNoPrimeNormalDefect}_{\Omega}$. For any represented
non-trivial zero row $x:T_{\Omega}^{nt0}$, this yields
$\mathsf{NoPrimeNormalDefect}_{\Omega}(x)$. Applying
\autoref{def:rh-proof-obligation-prime-normal-critical-line} gives
$\mathsf{OnCritLine}(\pi_{\zeta}(x))$. Representation transports this
critical-line witness to every $u:\mathsf{NTZeroType}$, and
\autoref{thm:rh-zeta-zero-type-rh-fixed-half-coverage} gives the BEDC-RH
predicate.
\end{proof}

\begin{principle}[Remaining mathematical content]
\label{prin:rh-proof-obligation-remaining-content}
After the zero-fiber, generator-factorization, and representation rows are
in place, the remaining mathematical content is concentrated in two
rigidity rows:
$$
\mathsf{PrimeDefectApartZero}_{\Omega},
\qquad
\mathsf{PrimeNormalImpliesCriticalLine}_{\Omega}.
$$
The first says that a positive prime-normal defect forces apartness from
the zero value on the same source row. The second says that absence of such
defect is read in the complex projection as the fixed-half condition.
\end{principle}

\section{Proof Route Ordering}
\label{sec:rh-proof-obligation-route-ordering}

\begin{definition}[RH route state]
\label{def:rh-proof-obligation-route-state}
An RH route state is a packet
$$
\mathsf{RHRouteState}
:=
\bigl(
\mathsf{AnalyticBase},
\mathsf{ZeroType},
T_{\Omega},
\mathsf{BulkBoundaryExact},
\mathsf{PrimeSieve},
\mathsf{ZeroRigidity},
\mathsf{CriticalLineReadout},
\mathsf{WitnessExport}
\bigr).
$$
The components are interpreted as follows. $\mathsf{AnalyticBase}$ records
the closed arithmetic, real, complex, holomorphic, Dirichlet, and zeta
interfaces consumed by \autoref{def:rh-predicate}. $\mathsf{ZeroType}$ is
the zero-fiber package of \autoref{def:rh-proof-obligation-zero-fiber-rows}.
$T_{\Omega}$ is the compatible tower body. $\mathsf{BulkBoundaryExact}$ is
the fullness, faithfulness, and ledger-exactness row of
\autoref{def:rh-bulk-boundary-exactness}. $\mathsf{PrimeSieve}$ is the
prime-invariant sieve of \autoref{def:rh-prime-sieve-interface}.
$\mathsf{ZeroRigidity}$ abbreviates
\autoref{def:rh-proof-obligation-zero-atom-prime-normal-rigidity}.
$\mathsf{CriticalLineReadout}$ abbreviates
\autoref{def:rh-proof-obligation-prime-normal-critical-line}. Finally,
$\mathsf{WitnessExport}$ records extraction of $\Phi$ and the associated
modulus and barrier witnesses.
\end{definition}

\begin{principle}[Route dependency order]
\label{prin:rh-proof-obligation-route-dependency-order}
The route state is ordered by dependency. The analytic base licenses
zeta-value and zero predicates. The zero-type package turns those
predicates into the source carrier $\mathsf{NTZeroType}$. The
$\Omega$ body and bulk-boundary exactness provide a tower representation
whose boundary shadows and ledgers are visible. The prime sieve defines the
defect surface. The zero-rigidity row rules out prime-normal defect on zero
atoms. The critical-line readout converts absence of prime-normal defect
into $\mathsf{OnCritLine}$. Only after those rows are present can
$\mathsf{WitnessExport}$ produce the BEDC-RH witness function.
\end{principle}

\begin{theorem}[Route state closes the witness target]
\label{thm:rh-proof-obligation-route-state-closes-witness-target}
If an RH route state carries inhabited rows for
$$
\mathsf{ZeroRigidity},
\qquad
\mathsf{CriticalLineReadout},
\qquad
\mathsf{WitnessExport},
$$
and the representation row of
\autoref{def:rh-proof-obligation-omega-representation-row}, then it
supplies the witness function
$$
\Phi:\mathsf{NTZeroType}\to\mathsf{FixedHalfZeroType}.
$$
\end{theorem}

\begin{proof}
Let $u:\mathsf{NTZeroType}$. The representation row supplies
$x:T_{\Omega}^{nt0}$ whose zeta projection is classified with the source
of $u$. The zero-rigidity row yields
$\mathsf{NoPrimeNormalDefect}_{\Omega}(x)$. The critical-line readout row
then yields $\mathsf{OnCritLine}(\pi_{\zeta}(x))$. Transporting this
witness along the source classifier gives $\mathsf{OnCritLine}(\pi_s(u))$.
Pairing $u$ with that witness gives an element of
$\mathsf{FixedHalfZeroType}$ over $u$, which is $\Phi(u)$.
\end{proof}

\section{Witness Export}
\label{sec:rh-proof-obligation-witness-export}

\begin{definition}[Witness export packet]
\label{def:rh-proof-obligation-witness-export-packet}
The witness export packet is
$$
\mathsf{WitnessExport}
:=
\bigl(
\Phi,\mu,\beta_{\mathrm{loc}},
\mathsf{CriticalLineRestriction}
\bigr).
$$
Here $\Phi$ is the fixed-half section of
\autoref{def:rh-proof-obligation-witness-function}. For a zero-on-strip
row $s$, the modulus
$$
\mu_s:\NatUp\to\NatUp
$$
extracts rational approximants witnessing $\Re(s)\sim 1/2$. The barrier
$$
\beta_{\mathrm{loc}}:\NatUp\times\NatUp\to\RatUp_{>0}
$$
is the height-local off-line apartness function of
\autoref{def:rh-proof-obligation-quantitative-barrier}. The remaining field
is the restriction equivalence between non-trivial zero packets and
critical-line height rows described in \autoref{sec:rh-witness-types}. The
Riemann-von Mangoldt count and the trivial-zero classifier are support ledgers
for the full zeta formalization; they are not fields exported by the RH section.
\end{definition}

\begin{theorem}[Fixed-half section exports the modulus witness]
\label{thm:rh-proof-obligation-fixed-half-exports-modulus}
An inhabitant of
$$
\Phi:\mathsf{NTZeroType}\to\mathsf{FixedHalfZeroType}
$$
exports, for each $u:\mathsf{NTZeroType}$, a modulus row
$$
\mu_{\pi_s(u)}:\NatUp\to\NatUp
$$
of the kind stated in \autoref{thm:rh-witness-modulus}.
\end{theorem}

\begin{proof}
The value $\Phi(u)$ contains the same zero packet $u$ and a witness of
$\mathsf{OnCritLine}(\pi_s(u))$. By
\autoref{thm:rh-witness-modulus}, the constructive content of that
$\mathsf{OnCritLine}$ witness is a modulus selecting rational
approximants whose distance from $1/2$ is bounded by the requested
dyadic tolerance.
\end{proof}

\begin{principle}[Barrier export is analytic content]
\label{prin:rh-proof-obligation-barrier-export-analytic-content}
The function $\beta_{\mathrm{loc}}$ is not obtained from the section form
alone. It is exported only from the apart-zero defect row when that row is
strengthened to a quantitative lower-bound statement on a displayed height
window:
$$
|\Re(s)-1/2|>2^{-n}
\wedge
|\Im(s)|\le T
\to
|\zeta(s)|\ge \beta(n,T).
$$
Thus $\Phi$ records fixed-half membership, while $\beta_{\mathrm{loc}}$
records the compact-window lower-bound form of off-line exclusion. A global
height-uniform barrier remains a distinct route assumption.
\end{principle}

\begin{principle}[Witness export boundary]
\label{prin:rh-proof-obligation-witness-export-boundary}
The route ordering and witness export packet do not reduce the analytic
difficulty of $\mathsf{RH}$. They specify the shape of a closed BEDC proof:
first a same-source zero/apart-zero exclusion via
$\mathsf{PrimeDefectApartZero}_{\Omega}$, then a fixed-half readout via
$\mathsf{PrimeNormalImpliesCriticalLine}_{\Omega}$, and only then the
export of $\Phi$, $\mu$, height-local barrier data when supplied by the
analytic route, and critical-line restriction witnesses.
\end{principle}
