\chapter{Internal Ruler Constraint Container}
\label{ch:visions-rh-internal-ruler-constraint-container}

\autoref{ch:visions-rh-bedc-golden-zeta-generation-machine} packages the
BEDC--Golden--Zeta machine, while
\autoref{def:philosophy-constraint-container} supplies the general
constraint-container interface.  The present chapter records the common
source row in which the golden ruler is itself part of the generated
container.  The point is not to place zero certificates inside a prior
critical-line host; the point is to make the internal ruler, the
zero-certificate tower, and the surviving candidate container use the
same source, boundary, and time ledgers.

\begin{definition}[Container-mature internal ruler]
\label{def:rh-container-mature-internal-ruler}
Let
$$
(\mathcal R_m)_{m\ge0}
$$
be an internal golden ruler family in the sense of
\autoref{def:rh-internal-golden-ruler}.  It is
\emph{container-mature for a zero-certificate tower} when the following
rows are present.

The source row is triadic ruler invariance:
$$
\mathcal R_m\rightsquigarrow\mathcal R_{m+1}
$$
transports the $\mathsf P$, $\mathsf U$, and $\mathsf T$ rows of
\autoref{def:rh-triadic-ruler-invariance}.

The weak capture row is a cofinal family of golden eta packets
$$
\mathcal G_m(\rho)=(D_m,P_m,R_m,\Gamma_m)
$$
as in \autoref{def:rh-golden-eta-certificate-packet}, with the Rouch\'e
boundary gap supplied by \autoref{thm:rh-golden-eta-weak-active-capture}
where its hypotheses apply.

The boundary row is a cofinal family of golden Lee--Yang bridge
witnesses in the sense of
\autoref{def:rh-golden-lee-yang-bridge-witness}.  Thus the same
zero-island ledger has a lifted path packet, a Lee--Yang unit-boundary
certificate, same-island transfer, radial purity, and dominance.

The audit row records projection loss through the displayed Zeckendorf,
prime-register, phase, tail, boundary, and defect ledgers.  It does not
allow the ruler to supply an independent real-positive scale coordinate.
\end{definition}

\begin{definition}[Internal-ruler constraint container package]
\label{def:rh-internal-ruler-constraint-container-package}
An \emph{internal-ruler constraint container package} for a completed
zeta zero readout consists of:

\begin{itemize}
\item a constructive BEDC--Golden--Zeta machine in the sense of
\autoref{def:rh-constructive-bedc-golden-zeta-machine};
\item a container-mature internal ruler family
$(\mathcal R_m)_{m\ge0}$;
\item a compatible zero-certificate tower generated levelwise from
$\mathcal R_m$;
\item the zeta constraint container of
\autoref{def:rh-zeta-constraint-container} for the limit ledger
$\Lambda_{\infty}$;
\item the defect extraction and cofinal active-height rows
$$
E_m(s)=r^2W_m+o(W_m),
\qquad
W_m\to\infty,
$$
for readouts
$$
s=\frac12+r+it.
$$
\end{itemize}
\end{definition}

\begin{principle}[The ruler cannot be the scale reservoir]
\label{prin:rh-ruler-cannot-be-scale-reservoir}
In an internal-ruler constraint container package, the ruler is part of
the audited container rather than an outside measurement device.  Hence
an off-half factor
$$
e^{-r h_a}\in\mathbb R_{>0}
$$
cannot be reclassified as a choice of resolution, truncation schedule,
Zeckendorf address, or path-model lift.  The only admissible hidden
rows are the displayed compact phase row, the finite prime-register
fibres, and the Zeckendorf ledger rows.  Any real-positive radial
factor must therefore appear in the defect-energy ledger.
\end{principle}

\begin{theorem}[Internal-ruler containers exclude radial readouts]
\label{thm:rh-internal-ruler-containers-exclude-radial-readouts}
Let an internal-ruler constraint container package have limit readout
$$
s_{\infty}=\frac12+r+it.
$$
Then $r=0$.
\end{theorem}

\begin{proof}
The package contains a constructive BEDC--Golden--Zeta machine and a
container-mature internal ruler family.  By
\autoref{thm:rh-constructive-layer-discipline-preserves-legality}, the
displayed layers are legal.  By
\autoref{def:rh-container-mature-internal-ruler}, the ruler transports
the triadic rows and does not supply an independent
$\mathbb R_{>0}$ gauge.  Thus
\autoref{prin:rh-ruler-cannot-be-scale-reservoir} forces any off-half
factor to be read through the defect ledger.

The same package supplies the golden Lee--Yang bridge witnesses, hence
the zero-compatible boundary bound, and it also supplies
$$
E_m(s_{\infty})=r^2W_m+o(W_m),
\qquad
W_m\to\infty.
$$
These are exactly the hypotheses of
\autoref{thm:rh-legal-generation-machines-exclude-radial-limits}.
Therefore $r=0$.
\end{proof}

\begin{principle}[Proof load of the internal-ruler container]
\label{prin:rh-internal-ruler-container-proof-load}
The internal-ruler container formulation separates four rows.  The
general constraint-container interface is supplied by
\autoref{def:philosophy-constraint-container}.  The golden ruler and
Zeckendorf address rows are supplied by
\autoref{def:rh-golden-resolution-ruler} and
\autoref{def:rh-zeckendorf-resolution-ledger}.  The weak analytic
capture row is supplied by
\autoref{thm:rh-golden-eta-weak-active-capture}.  The strong source row
is the same-island Lee--Yang bridge of
\autoref{def:rh-zeta-golden-lee-yang-bridge-source-row}.  The container
package closes only when these rows are displayed together with radial
purity, dominance, cofinal active-height growth, and no external
scale-gauge row.
\end{principle}

\begin{principle}[Bridge reduction for the internal golden certificate]
\label{prin:rh-internal-golden-certificate-bridge-reduction}
The internal golden certificate row does not add a second analytic
route beyond the golden bridge.  The weak eta capture row supplies
finite zero-island capture; the internal ruler row supplies source,
refinement, and Zeckendorf address control; the constraint-container row
records the surviving candidate class and admissible gauge.  The only
strong zeta-specific row is the same-island Lee--Yang bridge, namely a
cofinal supply of the witnesses of
\autoref{def:rh-golden-lee-yang-bridge-witness}.  Thus failure of that
bridge leaves the package at weak capture, while success of that bridge
upgrades the package to the internal-ruler container source row.
\autoref{def:rh-golden-bridge-analytic-subrows} records the analytic
subrows of this bridge: spectral expansion and spectral gap for the
selected path branch, and stable Rouch\'e homotopy to the analytic packet
on the same boundary.
\end{principle}
