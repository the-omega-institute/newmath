\section{Open Projection Obligations}
\label{sec:rh-unresolved-projection-obligations}

\begin{definition}[Zeta BEDC embedding obligation]
\label{def:rh-zeta-bedc-embedding-obligation}
The obligation
$$
\mathsf{ZetaBEDCEmbedding}
$$
asks for a single BEDC observation certificate for each completed
non-trivial zeta zero packet. The certificate must package the carrier,
classifier, ledger, boundary, and stability rows for completed zeta,
zero readout, functional-equation transport, explicit-formula transport,
Li readout, Weil pairing, Nyman container rows, and the prime-register
source ledger. The prime-register phase source is already visible; the
open part is the unified observation certificate.
\end{definition}

\begin{definition}[No hidden scale obligation]
\label{def:rh-no-hidden-scale-obligation}
The obligation
$$
\mathsf{NoHiddenScaleFromBEDC}
$$
asks for an internal BEDC theorem that a nonzero radial scale leg cannot
be used as a hidden component of a legal pure zeta closure. The theorem
must distinguish unit phase, prime-register data, finite encodable
residue, and noncompact positive-real scale. Its target is a ledger
statement, not a physical analogy.
\end{definition}

\begin{definition}[Projection faithfulness obligation]
\label{def:rh-tri-projection-faithfulness-obligation}
The obligation
$$
\mathsf{TriProjectionFaithfulness}
$$
asks for a commutative projection certificate showing that Li time
readout, Weil energy readout, and Nyman container readout detect the same
radial scale leg when it exists. The open content is not the individual
criterion statements; it is the shared source witness and the transport
between the three readouts.
\end{definition}

\begin{definition}[Dominant observation strand obligation]
\label{def:rh-dominant-observation-strand-obligation}
The obligation
$$
\mathsf{DominantObservationStrand}
$$
asks for a source-side certificate selecting the zero-relevant
observation strand. A visible branch, local model, or finite carrier is
not enough; the selected strand must be certified as controlling the
zero-relevant boundary through a dominance, Rouch\'e, energy, or ledger
comparison row.
\end{definition}

\begin{definition}[Unit-boundary transport obligation]
\label{def:rh-unit-boundary-transport-obligation}
The obligation
$$
\mathsf{UnitBoundaryTransport}
$$
asks for a stable transport from the dominant zeta observation strand to
a unit-boundary container such as a Lee--Yang, unitary, compact, or Pick
positive model. The transport must expose its boundary nonvanishing and
ledger compatibility rows; it may not pass through an unrecorded radial
scale.
\end{definition}

\begin{definition}[Cofinal triangular tower obligation]
\label{def:rh-cofinal-triangular-tower-obligation}
The obligation
$$
\mathsf{CofinalTriangularTower}
$$
asks for a tower of finite triangular observation spaces whose residuals
are exhausted in the limit. Finite orthogonal increments and finite Pick
positivity are already reduction rows; the open content is the cofinal
exhaustion statement that prevents a radial residual from being hidden at
the far end.
\end{definition}

\begin{definition}[Diagonal safety obligation]
\label{def:rh-diagonal-safety-obligation}
The obligation
$$
\mathsf{DiagonalSafety}
$$
asks that the proof system be source-generated and append-only. It
forbids a local oracle strategy that first detects a bad zero-side
direction and then modifies the certificate to cover that direction.
Negative Li witnesses, Weil probes, and Nyman separating functionals must
be excluded by the source-side construction itself.
\end{definition}

\begin{definition}[No hidden radial register obligation]
\label{def:rh-no-hidden-radial-register-obligation}
The obligation
$$
\mathsf{NoHiddenRadialRegister}
$$
is the strengthened gauge form of
\autoref{def:rh-no-hidden-scale-obligation}. It asks for a theorem that a
nonzero radial coordinate is not unit phase, not a prime-register gauge,
not freely erasable finite data, and not an admissible tower tail. A
recorded radial coordinate changes the object into a scale-ledgered row;
an unrecorded radial coordinate violates ledger completeness.
\end{definition}

\begin{definition}[Triangular projection equivalence obligation]
\label{def:rh-triangular-projection-equivalence-obligation}
The obligation
$$
\mathsf{TriangularProjectionEquivalence}
$$
asks for the converse direction of the projection dictionary: any Li,
Weil, or Nyman defect relevant to the completed zeta zero packet must
come from the same radial scale witness, and the three projection
defects must share a common source certificate. This is the diagrammatic
equivalence row tying time, unit, and ledger projections together.
\end{definition}

\section{Conditional Package}
\label{sec:rh-unresolved-conditional-package}

\begin{theorem}[Projection obligation package is conditionally sufficient]
\label{thm:rh-projection-obligation-package-conditionally-sufficient}
Assume the obligations
$$
\mathsf{ZetaBEDCEmbedding},
\quad
\mathsf{NoHiddenScaleFromBEDC},
\quad
\mathsf{TriProjectionFaithfulness},
\quad
\mathsf{DominantObservationStrand},
\quad
\mathsf{UnitBoundaryTransport},
\quad
\mathsf{CofinalTriangularTower},
\quad
\mathsf{DiagonalSafety},
\quad
\mathsf{NoHiddenRadialRegister},
\quad
\mathsf{TriangularProjectionEquivalence}.
$$
Then the BEDC RH section target is inhabited.
\end{theorem}

\begin{proof}
The embedding obligation places each completed non-trivial zero packet in
the BEDC observation system. If a radial scale witness were present, the
faithfulness and equivalence obligations would expose it through the Li,
Weil, and Nyman projections as the same source-side defect. Dominance
ensures that the defect belongs to the zero-relevant strand; unit-boundary
transport prevents it from passing through a legal unit container;
cofinality prevents it from escaping to the tower tail; diagonal safety
prevents posterior repair. The hidden-scale and hidden-radial-register
obligations then rule out the radial witness inside a legal closure. The
zero packet is therefore forced onto the unit Li boundary, and the
earlier Li criterion transports that row to the critical-line section.
\end{proof}

\section{Compressed Core}
\label{sec:rh-unresolved-compressed-core}

\begin{principle}[Minimal open core]
\label{prin:rh-unresolved-minimal-open-core}
The preceding package compresses to three core proof tasks:
$$
\mathsf{ZetaBEDCEmbedding},
\qquad
\mathsf{NoHiddenRadialRegister},
\qquad
\mathsf{CofinalTriangularTower}.
$$
The first supplies the legal observation object, the second forbids a
nonzero radial scale as a hidden gauge or ledger-free residual, and the
third prevents a residual from being postponed to the infinite tail.
Tri-projection faithfulness, dominant-strand selection, unit-boundary
transport, diagonal safety, and projection equivalence are support rows
that make these three tasks usable in the completed zeta carrier.
\end{principle}

\begin{principle}[Status of the unresolved projection ledger]
\label{prin:rh-unresolved-projection-ledger-status}
This ledger lists open obligations only. The already displayed algebraic
identities, coordinate criteria, finite factorization rows, and moment
reductions are not counted as remaining proof load here. The route to RH
through this ledger is conditional on constructing the listed source-side
certificates.
\end{principle}
