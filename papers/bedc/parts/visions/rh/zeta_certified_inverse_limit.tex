\section{Ontological Rows}
\label{sec:rh-zeta-certified-inverse-limit-ontological-rows}

\begin{definition}[Ontological triple]
\label{def:rh-ontological-triple}
For a proof object, tool, or carrier $X$, its ontological triple is
$$
\operatorname{Ont}(X):=(\mathsf T_X,\mathsf U_X,\mathsf L_X).
$$
Here $\mathsf T_X$ is the time-source row, recording the lower-level
objects from which $X$ is generated and whether the history is
append-only.  The row $\mathsf U_X$ is the unit or triangle row,
recording unit normalization, orthogonal splitting, positivity, or
Pythagorean energy conservation.  The row $\mathsf L_X$ is the ledger
row, recording hidden gap, tail, phase, classifier, or probability-mass
data.
\end{definition}

\begin{principle}[All zeta-route tools require ontological triples]
\label{prin:rh-zeta-route-tools-require-ontological-triples}
Every tool used by the RH route must expose its ontological triple.  A
finite classifier exposes prime-source time, unit character or amplitude
readout, and type-probability ledger.  A fixed-window analytic tool
exposes finite construction time, unit or positive boundary data, and a
tail ledger.  A limiting or completion tool exposes the lower-layer
inverse system, the preserved unit or positivity invariant, and a
limit-ledger policy.  A row without these three coordinates is not
available as a BEDC proof object for the route.
\end{principle}

\begin{definition}[Certified zeta layer]
\label{def:rh-certified-zeta-layer}
A certified zeta layer is a finite packet
$$
\mathcal Z_m
:=
\left(
G_m,\mu_m,\Psi_{m,\varphi},
\operatorname{SieveLedger}_m,
\operatorname{Gap}_m,
\operatorname{DomCert}_m,
\operatorname{NameCert}_m
\right),
$$
where $(G_m,\mu_m)$ is a prime probability classifier, $\Psi_{m,\varphi}$
is a normalized filtered prime-distribution curve, the sieve ledger
records allowed and excluded probability mass, the gap row records the
unexcluded candidate region, the dominance certificate records the
fixed-window analytic row available at that layer, and the naming
certificate licenses the layer as a BEDC object.
\end{definition}

\begin{definition}[Certified zeta inverse system]
\label{def:rh-certified-zeta-inverse-system}
A certified zeta inverse system is a projective family
$$
\mathcal Z_0
\leftarrow
\mathcal Z_1
\leftarrow
\mathcal Z_2
\leftarrow\cdots
$$
whose transition maps preserve the classifier projection, probability
pushforward, sieve ledger, gap inclusion, dominance certificate
projection, and NameCert provenance rows.  The certified inverse-limit
object is
$$
\mathsf{ZetaCertifiedLimit}
:=
\left(
\varprojlim_m\mathcal Z_m,
\operatorname{NameCert}_\infty,
\operatorname{LedgerPolicy}_\infty,
\operatorname{StabilityCert}_\infty
\right).
$$
\end{definition}

\begin{principle}[Zeta is not a seed assertion]
\label{prin:rh-zeta-not-seed-assertion}
The certified inverse-limit object is a route for admitting completed
zeta after the finite layers, transports, continuation rows, explicit
formula rows, and limit ledgers have been exposed.  It does not assert
analytic continuation or RH-shaped statements as naming seeds.  Those
rows remain theorem targets or operation-certificate targets until the
required finite and limiting ledgers are supplied.
\end{principle}

\section{Coverage}
\label{sec:rh-zeta-certified-inverse-limit-coverage}

\begin{definition}[Candidate gap and coverage]
\label{def:rh-candidate-gap-coverage}
At layer $m$, let $\operatorname{GAP}_m$ be the candidate region not yet
excluded by the layer's classifiers, probability sieves, fixed-window
certificates, and ledgers.  Let $\mathbb P_m$ be the candidate
probability measure visible at that layer.  The proof coverage of the
layer is
$$
\operatorname{Cov}_m
:=
1-\mathbb P_m(\operatorname{GAP}_m).
$$
\end{definition}

\begin{principle}[Coverage is not theorem probability]
\label{prin:rh-coverage-not-theorem-probability}
The number $\operatorname{Cov}_m$ is an audited coverage ratio for the
candidate space visible at layer $m$.  It is not a probability that
$\mathsf{RH}$ is true.  The theorem is a fixed mathematical statement;
the layer number measures how much of the candidate gap has been
excluded, classified, or forced into a ledger by the proof system.
\end{principle}

\begin{theorem}[Coverage is monotone under append-only constraints]
\label{thm:rh-coverage-monotone-append-only}
Assume the transition from layer $m$ to layer $m+1$ only adds legal
constraints and never revokes a previous ledger exclusion.  Then
$$
\operatorname{GAP}_{m+1}\subseteq\operatorname{GAP}_m,
$$
and therefore
$$
\operatorname{Cov}_{m+1}\ge\operatorname{Cov}_m.
$$
\end{theorem}

\begin{proof}
Every candidate surviving layer $m+1$ satisfies all constraints from
layer $m$ together with the additional constraints of layer $m+1$.
Hence it also survives layer $m$, giving the gap inclusion.  Taking
candidate measures gives
$$
\mathbb P_{m+1}(\operatorname{GAP}_{m+1})
\le
\mathbb P_m(\operatorname{GAP}_m)
$$
after transport along the displayed refinement map.  The displayed
coverage inequality follows by subtracting from $1$.
\end{proof}

\begin{definition}[FiniteWitnessCompleteness]
\label{def:rh-finite-witness-completeness}
The row
$$
\mathsf{FiniteWitnessCompleteness}
$$
asserts that every off-critical zero candidate, if present, generates a
finite-layer witness in the certified zeta inverse system.  Equivalently,
an off-critical zero cannot survive only as a nonlocated point of the
limit; it must expose a finite classifier, probability-sieve,
fixed-window, or radial-residual row at some layer.
\end{definition}

\begin{theorem}[Coverage limit excludes certified off-critical witnesses]
\label{thm:rh-coverage-limit-excludes-certified-offcritical-witnesses}
Assume
$$
\lim_m\operatorname{Cov}_m=1
$$
and $\mathsf{FiniteWitnessCompleteness}$.  Then no off-critical zero has
a legal BEDC witness in $\mathsf{ZetaCertifiedLimit}$.
\end{theorem}

\begin{proof}
An off-critical zero with a legal BEDC witness supplies, by
\autoref{def:rh-finite-witness-completeness}, a finite-layer witness in
some candidate gap row.  Since the coverage tends to $1$, the limiting
candidate gap has measure zero.  A legal BEDC witness cannot enter the
object layer merely as an unlocated measure-zero survivor; it must be
carried by a finite NameCert or ledger row.  The finite witness is
therefore eventually excluded or ledgered by the certified inverse
system, contradicting legal off-critical survival.
\end{proof}

\section{RH Interface}
\label{sec:rh-zeta-certified-inverse-limit-rh-interface}

\begin{definition}[CertifiedInverseLimitRHRoute]
\label{def:rh-certified-inverse-limit-rh-route}
The row
$$
\mathsf{CertifiedInverseLimitRHRoute}
$$
consists of:
$$
\begin{gathered}
\mathsf{ZetaCertifiedLimit},\quad
\mathsf{FiniteWitnessCompleteness},\quad
\mathsf{LayeredProbabilitySieveCompleteness},\\
\mathsf{CofinalSieveDissipation},\quad
\mathsf{CriticalProbabilityStability},\quad
\lim_m\operatorname{Cov}_m=1.
\end{gathered}
$$
It is a proof-engineering package: each row must be supplied by finite
or limiting certificate data before the package can be used.
\end{definition}

\begin{theorem}[Certified inverse-limit route gives conditional fixed-half readout]
\label{thm:rh-certified-inverse-limit-route-conditional-fixed-half}
Assume $\mathsf{CertifiedInverseLimitRHRoute}$ and that every completed
nontrivial zero packet admitted by the route is a legal BEDC closure
object.  Then every admitted nontrivial zero has fixed-half readout.
\end{theorem}

\begin{proof}
If an admitted nontrivial zero were off the critical line, it would give
an off-critical candidate in the certified zeta inverse system.  By
\autoref{thm:rh-coverage-limit-excludes-certified-offcritical-witnesses},
no such candidate has a legal BEDC witness.  Equivalently, if its
radial residual is exposed on a probability-filtered curve, then
\autoref{thm:rh-cofinal-probability-sieve-conditional-fixed-half-readout}
excludes it by cofinal dissipation and critical probability stability.
Thus the radial coordinate is zero, and
\autoref{thm:rh-li-coordinate-fixed-half-circle} gives fixed-half
readout.
\end{proof}

\begin{principle}[Certified inverse-limit audit boundary]
\label{prin:rh-certified-inverse-limit-audit-boundary}
The inverse-limit route closes only when every layer is licensed by
source, unit, and ledger rows, the coverage numbers are interpreted as
gap-coverage measures, and every off-critical candidate is finite-witness
visible.  Without finite witness completeness, measure-one coverage
would be insufficient for a constructive BEDC proof.  With finite
witness completeness and critical probability stability, a surviving
off-critical zero has no legal NameCert route into the completed zeta
object.
\end{principle}
