\section{Closure Strength}
\label{sec:rh-zeta-bedc-closure-strength}

\begin{definition}[Zeta encodability row]
\label{def:rh-zeta-encodability-row}
The zeta encodability row is the statement that every Dirichlet monomial
can be written through prime and PZG data:
$$
n=\prod_p p^{v_p(n)}
=
\prod_{p,k}(p^{F_k})^{z_{p,k}(n)}
$$
and therefore
$$
n^{-s}
=
\exp\left(
-s\sum_{p,k}z_{p,k}(n)F_k\log p
\right).
$$
This row is a syntax and readout statement. It does not by itself supply
analytic continuation, zero packets, positive kernels, limit ledgers, or
closure under the circle--triangle--register invariant.
\end{definition}

\begin{definition}[Zeta BEDC closure]
\label{def:rh-zeta-bedc-closure}
The obligation
$$
\mathsf{ZetaBEDCClosure}
$$
asserts that the completed zeta route enters BEDC as a generated closure
object, not merely as an encodable formula. It consists of the following
certificate rows:
$$
\begin{gathered}
\zeta^{\uparrow},\quad
\xi^{\uparrow},\quad
\mathsf{NTZeroType},\quad
\mathsf{ZetaZero},\quad
\mathsf{InCritStrip},\\
\text{functional-equation transport},\quad
\text{analytic-continuation transport},\quad
\text{explicit-formula transport},\\
\text{Li coordinate readout},\quad
\text{zero-observation readout},\quad
\text{analytic prime-shift constraint},\quad
\text{prime-shift analytic compatibility},\quad
\text{limit and tail ledgers}.
\end{gathered}
$$
Each displayed row must be admitted through a naming or operation
certificate with source specification, pattern specification, classifier
specification, stability certificate, and ledger policy. In particular,
analytic continuation and limiting interfaces must carry the ledger rows
that record branch, pole, gap, and tail data.
\end{definition}

\begin{definition}[Triadic invariant preservation for zeta]
\label{def:rh-triadic-invariant-preservation-zeta}
Triadic invariant preservation for zeta is the assertion that every
constructor used by \autoref{def:rh-zeta-bedc-closure} preserves the
circle--triangle--register invariant
$$
\mathcal I_{\triangle}
=
\mathcal I_{\mathsf P}
\wedge
\mathcal I_{\mathsf U}
\wedge
\mathcal I_{\mathsf T}.
$$
Here $\mathsf P$ records prime, Gamma, rational-boundary, PZG, and
source-ledger rows; $\mathsf U$ records unit-boundary, positive-kernel,
or positive-measure rows with no hidden $\RealUp_{>0}$ gauge; and
$\mathsf T$ records append-only histories, finite witnesses, analytic
transport, and cofinal tower ledgers. A constructor may expose a residual
only by recording it in one of these rows.
\end{definition}

\begin{principle}[Analytic prime shifts are part of zeta closure]
\label{prin:rh-analytic-prime-shifts-part-of-zeta-closure}
The prime-index motion inside \autoref{def:rh-zeta-bedc-closure} must
satisfy $\mathsf{SmoothPrimeShiftConstraint}$ from
\autoref{def:rh-smooth-prime-shift-constraint}. Thus the PZG registers
may supply discrete source uniqueness, but the completed zeta object is
licensed only when those discrete shifts are compatible with holomorphic
charts, compact-uniform height control, branch ledgers, Rouch\'e
no-wall-crossing, and cofinal tower coverage.
\end{principle}

\begin{principle}[Zeta continuation is prime-shift compatible]
\label{prin:rh-zeta-continuation-prime-shift-compatible}
The analytic-continuation part of \autoref{def:rh-zeta-bedc-closure} is
the stronger obligation
$\mathsf{ZetaAnalyticContinuationIsPrimeShiftCompatible}$ from
\autoref{def:rh-zeta-analytic-continuation-prime-shift-compatible}. It
requires nested cell compatibility, cocycle compatibility,
character-flow compatibility, normal convergence, and no radial anomaly
for the continuation rows used by completed zeta.
\end{principle}

\begin{definition}[Legal zero closure object]
\label{def:rh-legal-zero-closure-object}
A legal zero closure object is a completed non-trivial zero packet
$\rho$ admitted by \autoref{def:rh-zeta-bedc-closure} whose zero
observation, Li coordinate, and closure transport are internal rows of
the same generated object. Thus
$$
z_{\rho}:=1-\frac1{\rho}
$$
is not an external diagnostic attached after the zero has been selected;
it is a displayed readout of the zero packet. Writing
$$
z_{\rho}=e^{\ell+i\theta},
$$
the radial row $\ell$ must be either absent, ledgered as a scale row, or
refused by the legal closure classifier. It cannot be hidden as unit
phase, compact prime-register fibre, or ledger-free tail.
\end{definition}

\section{Conditional Consequence}
\label{sec:rh-zeta-bedc-closure-conditional-consequence}

\begin{theorem}[Strong zeta closure reduces RH to the radial obstruction package]
\label{thm:rh-strong-zeta-closure-reduces-rh-radial-package}
Assume
$$
\mathsf{ZetaBEDCClosure},
\qquad
\mathsf{ZetaAnalyticContinuationIsPrimeShiftCompatible},
\qquad
\mathsf{SmoothPrimeShiftConstraint},
\qquad
\text{\autoref{def:rh-triadic-invariant-preservation-zeta}},
$$
and assume that every completed non-trivial zero packet is a legal zero
closure object in the sense of
\autoref{def:rh-legal-zero-closure-object}. Then the embedding row
needed by
\autoref{thm:rh-radial-obstruction-package-conditionally-sufficient}
is supplied. Consequently, together with
$$
\mathsf{NoHiddenRadialRegister},
\qquad
\mathsf{CofinalTriangularTower},
$$
and the support rows of
\autoref{def:rh-radial-projection-tower-support-obligations}, the BEDC
RH section target is inhabited.
\end{theorem}

\begin{proof}
The closure row supplies the completed zeta carrier, completed zero
packets, analytic transports, Li coordinate, and limit ledgers as one
generated BEDC object. The prime-shift compatibility row supplies the
nested cell, cocycle, character-flow, normal-convergence, and
no-radial-anomaly data for the analytic transports. By
\autoref{thm:rh-prime-shift-analytic-compatibility-refines-smooth-shift},
the smooth prime-shift row follows as the corresponding tower-level
constraint: the prime-index motion is not an arbitrary discrete address
stream, but an analytic tower with chart compatibility, branch ledger,
height control, and no-wall-crossing rows. The triadic preservation row
prevents the constructors from changing the circle--triangle--register
invariant while passing through Euler products, completion,
continuation, zero readout, and positive-kernel or moment readout. The
legal zero closure row supplies, for each completed non-trivial zero
packet, the internal Li coordinate
$$
z_{\rho}=1-\frac1{\rho}=e^{\ell+i\theta}.
$$
Thus the required $\mathsf{ZetaBEDCEmbedding}$ row of
\autoref{thm:rh-radial-obstruction-package-conditionally-sufficient} is
present. Applying that theorem with
$\mathsf{NoHiddenRadialRegister}$, $\mathsf{CofinalTriangularTower}$, and
the stated support rows gives the BEDC RH section target.
\end{proof}

\begin{principle}[Encodability is not closure]
\label{prin:rh-encodability-is-not-closure}
The implication used in this route is not
$$
\text{zeta is encodable}
\Longrightarrow
\mathsf{RH}.
$$
The needed implication is the stronger closure statement
$$
\mathsf{ZetaBEDCClosure}
\wedge
\mathsf{ZetaAnalyticContinuationIsPrimeShiftCompatible}
\wedge
\mathsf{SmoothPrimeShiftConstraint}
\wedge
\text{\autoref{def:rh-triadic-invariant-preservation-zeta}}
\wedge
\text{\autoref{def:rh-legal-zero-closure-object}}
\Longrightarrow
\mathsf{ZetaBEDCEmbedding}.
$$
Only after this gate has been crossed can
$\mathsf{NoHiddenRadialRegister}$ and $\mathsf{CofinalTriangularTower}$
force the Li radial coordinate to vanish. The remaining analytic work is
therefore not another notation for zeta, but the construction of the
legal generated closure object.
\end{principle}
