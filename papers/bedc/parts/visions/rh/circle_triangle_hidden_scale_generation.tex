\chapter{Circle-Triangle Generation and Hidden Scale Refusal}
\label{ch:visions-rh-circle-triangle-hidden-scale-generation}
\origin{ai}

\autoref{ch:visions-rh-phase-triangle-prime-register-readout} separates
unit phase, orthogonal readout, and prime-register residue. The present
chapter records the corresponding generation discipline at the level of
zero-relevant observations. A legal phase object is not a bare complex
number. It is a certificate-bearing packet whose visible component is a
unit-circle phase or an orthogonal readout of that phase, and whose
unread distinctions are retained by an explicit ledger. The radial scale
row is therefore not free background structure. If it participates in a
closure, it must be displayed as a ledgered residual; if it is not
displayed, it is not available to the object.

\section{Generated Time and Orthogonal Readout}
\label{sec:rh-circle-triangle-generated-time}

\begin{definition}[Circle-triangle generation packet]
\label{def:rh-circle-triangle-generation-packet}
A circle-triangle generation packet is a certificate row
$$
\mathcal G
:=
\bigl(
B,\Hist_{\mathcal G},
\Cont_{\mathcal G},
\mathbb T_{\mathcal G},
\rho_{\perp,\mathcal G},
\Lambda_{\mathcal G}
\bigr)
$$
with the following data. The distinction row contains two apart marks
$$
B_0\#B_1.
$$
The history row $\Hist_{\mathcal G}$ is generated by finite append of
those marks, and $\Cont_{\mathcal G}$ records continuation of such
histories. The phase row is a unit-circle readout
$$
u(\theta)=e^{i\theta}\in\mathbb T_{\mathcal G},
$$
and the orthogonal observation row is
$$
\rho_{\perp,\mathcal G}(u(\theta))
=
(\cos\theta,\sin\theta).
$$
The ledger $\Lambda_{\mathcal G}$ records distinctions not visible in the
circle or triangle readout.
\end{definition}

\begin{principle}[Observation time is append order]
\label{prin:rh-circle-triangle-observation-time-append-order}
The time row in a circle-triangle generation packet is the order in which
finite marks are appended and continued. A later observation is a longer
or transported history row, not an independent real-time coordinate
inserted before the finite kernel. Thus every residual that must be kept
for later recovery has a ledger position in the append-and-continuation
surface.
\end{principle}

\begin{definition}[Orthogonal observation event]
\label{def:rh-circle-triangle-orthogonal-observation-event}
For a phase row $u(\theta)$ in $\mathcal G$, the associated orthogonal
observation event is the finite vector packet
$$
O_{\theta}
:=
\bigl(\cos\theta,\sin\theta,h_{\theta}\bigr),
$$
where
$$
h_{\theta}:\cos^2\theta+\sin^2\theta=1.
$$
It exposes the quadratic invariant of the unit phase. It does not expose
the complete source history, winding route, prime provenance, or solenoid
fiber unless these are separately present in $\Lambda_{\mathcal G}$.
\end{definition}

\begin{principle}[Circle and triangle are readout surfaces]
\label{prin:rh-circle-triangle-readout-surfaces}
The circle row is the continuous unit-phase surface. The triangle row is
one orthogonal coordinate read of that phase. Neither row is licensed to
erase source distinctions. Every unread distinction must be retained as a
fiber, a ledger entry, a transport witness, or an obstruction row.
\end{principle}

\section{Prime Registers and Addressed Residue}
\label{sec:rh-circle-triangle-prime-register}

\begin{definition}[Zeta prime-register readout]
\label{def:rh-zeta-prime-register-readout}
For a positive integer $n$, the prime-register readout is the finite
support vector
$$
\upsilon(n):=(v_p(n))_p,
\qquad
n=\prod_p p^{v_p(n)}.
$$
The corresponding frequency row is
$$
\log n
=
\sum_p v_p(n)\log p.
$$
Thus the oscillatory zeta mode
$$
e^{-it\log n}
$$
has visible phase behavior, while the finite exponent vector
$\upsilon(n)$ records the irreducible arithmetic source ledger behind
that behavior.
\end{definition}

\begin{principle}[Finite addressability is not free hiding]
\label{prin:rh-finite-addressability-not-free-hiding}
\autoref{def:rh-zeckendorf-prime-ledger-interface} supplies an internal
route from a finite binary history to a Zeckendorf address and then to a
prime-exponent ledger. This addressability licenses finite residual data
only when the relevant address row is displayed. It does not license a
consumer to use residual scale, branch, phase-source, or prime-fiber
information while omitting the ledger that carries it.
\end{principle}

\begin{definition}[No hidden scale row]
\label{def:rh-no-hidden-scale-from-bedc-row}
For a circle-triangle generation packet $\mathcal G$, the row
$$
\mathsf{NoHiddenScaleFromBEDC}(\mathcal G)
$$
states the following refusal. If a radial factor
$$
e^{\ell},
\qquad
\ell\#0,
$$
participates in a proposed closure, then either the factor is displayed
in $\Lambda_{\mathcal G}$ as a residual scale ledger, or the closure is
not admitted as a BEDC object. If the factor is displayed, the closure is
not a pure unit-phase closure. If the factor is omitted, the closure has
an unledgered residual and is not legal.
\end{definition}

\begin{principle}[Pure phase closure has no radial leg]
\label{prin:rh-pure-phase-closure-no-radial-leg}
In a generation packet satisfying
$\mathsf{NoHiddenScaleFromBEDC}(\mathcal G)$, a pure phase closure cannot
use a nonzero radial scale leg. Such a leg is either a recorded residual,
which changes the object from pure phase closure to scale-ledgered
closure, or an unrecorded residual, which violates ledger completeness.
\end{principle}

\section{Li Coordinates as Circle-Triangle Reads}
\label{sec:rh-circle-triangle-li-coordinate}

\begin{definition}[Li zero coordinate]
\label{def:rh-li-zero-coordinate}
For a zero packet with complex coordinate $\rho\ne0$, the Li coordinate
is
$$
z_{\rho}
:=
1-\frac1{\rho}.
$$
When $z_{\rho}\ne0$, a located scale-phase read is a representation
$$
z_{\rho}=e^{\ell+i\theta},
\qquad
\ell,\theta\in\RealUp.
$$
The row $\ell=0$ is exactly the unit-circle readout
$z_{\rho}\in\mathbb T$.
\end{definition}

\begin{theorem}[Li circle criterion]
\label{thm:rh-li-circle-criterion}
Let
$$
\rho=\sigma+i\gamma,
\qquad
\rho\ne0.
$$
Then
$$
\left|1-\frac1{\rho}\right|=1
\quad\Longleftrightarrow\quad
\sigma=\frac12.
$$
\end{theorem}

\begin{proof}
Write
$$
z_{\rho}=1-\frac1{\rho}=\frac{\rho-1}{\rho}.
$$
The equality $|z_{\rho}|=1$ is equivalent to
$$
|\rho-1|=|\rho|.
$$
With $\rho=\sigma+i\gamma$, this is
$$
(\sigma-1)^2+\gamma^2
=
\sigma^2+\gamma^2.
$$
Cancelling the common term gives
$$
-2\sigma+1=0,
$$
hence $\sigma=1/2$. The same calculation read backwards gives the
converse.
\end{proof}

\begin{theorem}[Li observation defect splits into scale and phase legs]
\label{thm:rh-li-observation-defect-scale-phase-split}
If
$$
z_{\rho}=e^{\ell+i\theta},
$$
then for every displayed $n\in\mathbb N_{>0}$,
$$
|1-z_{\rho}^n|^2
=
(e^{n\ell}-1)^2
+
2e^{n\ell}(1-\cos(n\theta)).
$$
\end{theorem}

\begin{proof}
Since
$$
z_{\rho}^n=e^{n\ell}e^{in\theta},
$$
we have
$$
|1-z_{\rho}^n|^2
=
1+e^{2n\ell}-2e^{n\ell}\cos(n\theta).
$$
The right side rewrites as
$$
e^{2n\ell}-2e^{n\ell}+1
+
2e^{n\ell}-2e^{n\ell}\cos(n\theta),
$$
which is the displayed decomposition.
\end{proof}

\begin{principle}[Scale leg is the radial obstruction]
\label{prin:rh-li-scale-leg-radial-obstruction}
In the decomposition of
\autoref{thm:rh-li-observation-defect-scale-phase-split}, the term
$$
(e^{n\ell}-1)^2
$$
is the radial scale leg, while
$$
2e^{n\ell}(1-\cos(n\theta))
$$
is the phase-observation leg. A pure unit-phase closure can use the
second leg as an orthogonal readout of phase, but it cannot absorb a
nonzero first leg without either recording radial scale or leaving a
residual unrecorded.
\end{principle}

\section{Conditional RH Route}
\label{sec:rh-circle-triangle-conditional-route}

\begin{definition}[Zeta BEDC circle-triangle embedding]
\label{def:rh-zeta-bedc-circle-triangle-embedding}
For a completed non-trivial zeta zero packet $\rho$, the statement
$$
\mathsf{ZetaBEDCEmbedding}^{\circ\triangle}(\rho)
$$
names a certificate-bearing existence obligation in the sense of
\autoref{def:ask-sig-spine-certified-existence}. It requires a
circle-triangle generation packet $\mathcal G$, the prime-register
readout of \autoref{def:rh-zeta-prime-register-readout}, the Li
coordinate row of \autoref{def:rh-li-zero-coordinate}, the scale-phase
defect row of
\autoref{thm:rh-li-observation-defect-scale-phase-split}, zero-ledger
transport, and a
$\mathsf{NoHiddenScaleFromBEDC}(\mathcal G)$ row. It is a proof
obligation, not an additional axiom and not a definition of
$\mathsf{RH}$.
\end{definition}

\begin{theorem}[Circle-triangle embedding is conditionally RH-sufficient]
\label{thm:rh-circle-triangle-embedding-conditionally-sufficient}
If every completed non-trivial zeta zero packet $\rho$ carries a
$\mathsf{ZetaBEDCEmbedding}^{\circ\triangle}(\rho)$ certificate, and if
the associated located scale readback converts absence of apartness from
zero into equality for the radial coordinate, then the BEDC RH section
target is inhabited.
\end{theorem}

\begin{proof}
Let $\rho$ be a completed non-trivial zero packet and let
$$
z_{\rho}=e^{\ell+i\theta}
$$
be the scale-phase read supplied by its embedding certificate. The
certificate carries
$\mathsf{NoHiddenScaleFromBEDC}(\mathcal G)$ and classifies the zero
packet as a pure phase-prime-triangle closure. If $\ell\#0$, then for
any displayed positive observation $n$ the radial leg
$$
(e^{n\ell}-1)^2
$$
in \autoref{thm:rh-li-observation-defect-scale-phase-split} is nonzero.
By \autoref{def:rh-no-hidden-scale-from-bedc-row}, that leg must either
be displayed as a residual scale ledger or remain unavailable. The first
case is not a pure phase closure, and the second case is not a legal
BEDC object. Both contradict the embedding certificate. Hence $\ell$ is
not apart from $0$. The located readback in the theorem hypothesis gives
$$
\ell=0.
$$
Therefore
$$
|z_{\rho}|=1.
$$
By \autoref{thm:rh-li-circle-criterion}, the coordinate of the zero
packet satisfies
$$
\Re(\rho)=\frac12.
$$
Since $\rho$ was arbitrary, the section target is inhabited.
\end{proof}

\begin{principle}[Cofinal mechanism rather than pointwise repair]
\label{prin:rh-circle-triangle-cofinal-mechanism}
The embedding obligation is not a zero-by-zero repair rule. It asks for a
single source mechanism whose finite observation windows grow cofinally:
generated histories, unit phase reads, orthogonal observation events,
prime-register ledgers, Li-coordinate scale reads, and no-hidden-scale
refusal must be supplied by the same certificate family. The finite
Pick-tower route of
\autoref{ch:visions-rh-triadic-solenoid-pick-tower} is one analytic
candidate for producing the required positive source mechanism.
\end{principle}

\begin{principle}[Status of the circle-triangle generation route]
\label{prin:rh-circle-triangle-generation-status}
The circle-triangle generation route reduces the geometric part of the
RH-facing argument to two explicit obligations:
$$
\mathsf{ZetaBEDCEmbedding}^{\circ\triangle}
\qquad\text{and}\qquad
\mathsf{NoHiddenScaleFromBEDC}.
$$
The algebraic identities for the Li coordinate and for the
scale-phase defect are displayed above. The remaining mathematical
content is the construction showing that completed zeta zero packets are
indeed admitted through this generation mechanism and that radial scale
cannot be silently supplied outside the ledger.
\end{principle}
