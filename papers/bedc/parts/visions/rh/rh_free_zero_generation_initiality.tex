\chapter{RH-Free Zero Generation by Initiality}
\label{ch:visions-rh-free-zero-generation-initiality}
\origin{ai}

\autoref{ch:visions-rh-zeta-self-similar-recovered-carrier} displays the
recovered zeta carrier as a diagram of prime, Dirichlet, Euler, Laplace,
continuation, completion, mirror, zero-ledger, window, code, certificate,
and provenance rows. \autoref{ch:visions-rh-layered-translation} identifies
the Riemann Hypothesis with a section into the fixed-half fiber over the
critical-strip zero-packet carrier. The present chapter isolates the
initial-algebra pattern that would turn a zero-generation theorem into that
section. It supplies no analytic continuation, no zero source, and no
fixed-half witness by itself; it only separates the exact obligations.

\begin{definition}[RH-free zero signature]
\label{def:rh-free-zero-signature}
\leandef{BEDC.Derived.RHRoute.ZeroGenerationInitiality.RHFreeZeroSignature}
\leanchecked{BEDC.Derived.RHRoute.ZeroGenerationInitiality.rhFreeZeroSignatureKinds\_no\_rh\_circ}
An RH-free zero signature is a displayed endofunctor $F_Z$ on the zeta
zero-presentation stratum together with a circularity audit
$$
\mathsf{NoRHCirc}(F_Z).
$$
The audit asserts that no constructor of $F_Z$ consumes an
$\mathsf{OnCritLine}$ witness, a row asserting $\Re(s)=1/2$, a section of
the fixed-half inclusion, the proposition $\mathsf{RH}$, or an arbitrary
unfactored seed
$$
\mathcal Z_{\zeta}\to F_Z(X).
$$
The admissible inputs are pre-RH rows such as prime-local spectral atoms,
functional-equation transport, conjugation transport, classifier transport,
analytic gluing, compatible limit seals, and ledger reconstruction. The
word ``pre-RH'' means that the row is present in the neutral zeta carrier
before the fixed-half consequence is imposed.
\end{definition}

\begin{definition}[Generated zero carrier]
\label{def:rh-generated-zero-carrier}
\leandef{BEDC.Derived.RHRoute.ZeroGenerationInitiality.GeneratedZero}
\leanchecked{BEDC.Derived.RHRoute.ZeroGenerationInitiality.generatedZero\_initiality}
For an RH-free zero signature $F_Z$, a generated zero carrier is an initial
algebra
$$
(Z,\mathsf{in}_{Z}) := \mu F_Z.
$$
Its intended realization into the source carrier of
\autoref{def:rh-layered-source-carrier} is a displayed map
$$
\varepsilon:Z\to\mathcal Z_{\zeta}.
$$
Zero-presentation exactness is the conjunction of soundness, faithfulness,
and essential coverage:
$$
\prod_{z:Z}\varepsilon(z):\mathcal Z_{\zeta},
$$
$$
\varepsilon(z)\sim\varepsilon(z')\to z\sim_Z z',
$$
and
$$
\prod_{u:\mathcal Z_{\zeta}}
\sum_{z:Z}\varepsilon(z)\sim u.
$$
When these rows are displayed, $\varepsilon$ is written
$Z\simeq\mathcal Z_{\zeta}$. This equivalence is not supplied by
initiality; it is the completeness obligation saying that the displayed
generators account for every accepted critical-strip zero packet.
\end{definition}

\begin{definition}[Generated fixed-half subcarrier]
\label{def:rh-generated-fixed-half-subcarrier}
\leandef{BEDC.Derived.RHRoute.ZeroGenerationInitiality.FixedHalfSubcarrier}
Given $\varepsilon:Z\to\mathcal Z_{\zeta}$, define the generated
fixed-half predicate
$$
P_{1/2}(z)
:=
\mathsf{OnCritLine}\bigl(\pi_s(\varepsilon(z))\bigr).
$$
The fixed-half subcarrier of $Z$ is
$$
H_Z
:=
\sum_{z:Z}P_{1/2}(z),
$$
with projection $m:H_Z\to Z$.
\end{definition}

\begin{definition}[Fixed-half closed zero signature]
\label{def:rh-fixed-half-closed-zero-signature}
\leandef{BEDC.Derived.RHRoute.ZeroGenerationInitiality.FixedHalfClosedZeroSignature}
The RH-free zero signature $F_Z$ is fixed-half closed relative to
$\varepsilon$ when $H_Z$ carries an $F_Z$-algebra
$$
h:F_Z(H_Z)\to H_Z
$$
whose projection agrees with the original zero algebra:
$$
m\circ h
\sim
\mathsf{in}_{Z}\circ F_Z(m).
$$
Thus every displayed zero constructor, when fed fixed-half inputs, emits a
fixed-half output. This is a constructor-by-constructor obligation, not a
pointwise enumeration of zeros.
\end{definition}

\begin{theorem}[Conditional initiality route to the RH section]
\label{thm:rh-conditional-initiality-route-section}
\leanchecked{BEDC.Derived.RHRoute.ZeroGenerationInitiality.rh\_conditional\_initiality\_route\_section}
Assume $F_Z$ is RH-free, $(Z,\mathsf{in}_{Z})=\mu F_Z$ is displayed,
$\varepsilon:Z\simeq\mathcal Z_{\zeta}$ is zero-presentation exact, and
$F_Z$ is fixed-half closed relative to $\varepsilon$. Then the section form
$\mathsf{Sec}_{\mathsf{RH}}$ of \autoref{def:rh-layered-section-form} is
inhabited.
\end{theorem}

\begin{proof}
By \autoref{def:rh-fixed-half-closed-zero-signature}, $H_Z$ is an
$F_Z$-algebra. Initiality of $Z$ gives the unique algebra morphism
$$
r:Z\to H_Z.
$$
The composite $m\circ r:Z\to Z$ is an $F_Z$-algebra endomorphism of the
initial algebra. The identity map on $Z$ is also such an endomorphism, so
initiality gives
$$
m\circ r\sim\mathrm{id}_{Z}.
$$
For each $z:Z$, the second component of $r(z):H_Z$ is a witness of
$P_{1/2}(z)$. Transporting this witness along
$\varepsilon:Z\simeq\mathcal Z_{\zeta}$ yields a map from every source
zero packet $u:\mathcal Z_{\zeta}$ to the fixed-half fiber over $u$. This
is exactly a section of the projection
$\operatorname{Fix}_{\mathbb R}\to\mathcal Z_{\zeta}$, hence an inhabitant
of $\mathsf{Sec}_{\mathsf{RH}}$.
\end{proof}

\begin{corollary}[Conditional extraction of the BEDC RH witness]
\label{cor:rh-conditional-initiality-extracts-witness}
Under the hypotheses of
\autoref{thm:rh-conditional-initiality-route-section}, the constructive
target of \autoref{prop:rh-constructive-form} is inhabited.
\end{corollary}

\begin{proof}
\autoref{thm:rh-conditional-initiality-route-section} supplies
$\mathsf{Sec}_{\mathsf{RH}}$. By
\autoref{prop:rh-layered-rh-fixed-half-section}, this is the scoped
fixed-half section readback used by the BEDC statement $\mathsf{RH}$ of
\autoref{def:rh-predicate}, whose proof content is the function described in
\autoref{prop:rh-constructive-form}.
\end{proof}

\begin{definition}[Weak eta capture packet]
\label{def:rh-weak-eta-capture-packet}
Let $\rho$ be a non-trivial zeta zero in the open critical strip. A
\emph{weak eta capture packet} for $\rho$ is a tuple
$$
(D,N,F_N,T_N,\gamma_D)
$$
where $D$ is a closed disk with $\rho$ in its interior and
$\overline D\subset\{0<\Re(s)<1\}$, the functions
$$
F_N(s):=\sum_{n=1}^{N}(-1)^{n-1}n^{-s},
\qquad
T_N(s):=\eta(s)-F_N(s)
$$
satisfy
$$
\eta(s)=F_N(s)+T_N(s),
$$
and the boundary gap row is
$$
\gamma_D:
\prod_{s\in\partial D}
\bigl(|T_N(s)|<|F_N(s)|\bigr).
$$
The packet captures the local zero count of $\eta$ by finite eta data and
a tail ledger. It is not, by itself, a fixed-half or unit-boundary packet.
\end{definition}

\begin{theorem}[Weak eta capture gives local zero-count capture]
\label{thm:rh-weak-eta-capture-local-zero-count}
For every non-trivial zeta zero $\rho$ in the open critical strip there
exists a weak eta capture packet whose finite part $F_N$ has the same
number of zeros as $\eta$ inside the disk $D$, counted with multiplicity.
\end{theorem}

\begin{proof}
In the open critical strip,
$$
\eta(s)=(1-2^{1-s})\zeta(s),
$$
and the factor $1-2^{1-s}$ has no zero there. Hence $\rho$ is a zero of
$\eta$. Choose $D$ so that its boundary has no $\eta$-zero and
$\overline D$ stays in the open strip. Then
$$
M:=\min_{s\in\partial D}|\eta(s)|
$$
is positive. The eta expansion converges uniformly on $\overline D$, so
for some $N$ the tail satisfies
$$
\sup_{s\in\partial D}|T_N(s)|<M/2.
$$
For $s\in\partial D$,
$$
|F_N(s)|
\ge
|\eta(s)|-|T_N(s)|
>
M/2,
$$
and therefore $|T_N(s)|<|F_N(s)|$ on $\partial D$. Rouch\'e's theorem
gives the same zero count for $F_N$ and $\eta$ inside $D$.
\end{proof}

\begin{principle}[Weak capture is not zero-presentation exactness]
\label{prin:rh-weak-capture-not-zero-presentation-exactness}
The weak eta capture packet is a local analytic capture row. It supplies
finite eta data, a tail ledger, a disk, and a Rouch\'e gap around a
given zero. It does not supply the essential coverage row of
\autoref{def:rh-generated-zero-carrier}. In particular, it does not build
a canonical generated carrier $Z$, a global realization
$\varepsilon:Z\to\mathcal Z_{\zeta}$, or the equivalence
$Z\simeq\mathcal Z_{\zeta}$. Thus weak capture may be used as evidence
for a local presentation mechanism, but it is not
zero-presentation exactness.
\end{principle}

\begin{principle}[Boundary stability is the strong packet gap]
\label{prin:rh-boundary-stability-strong-packet-gap}
The fixed-half closure hypothesis of
\autoref{def:rh-fixed-half-closed-zero-signature} requires more than
local zero-count capture. A strong zero packet must carry enough
boundary data to show that the generated channel has no nontrivial real
positive scale defect, equivalently that the active centered channels
are unit-boundary rows:
$$
|e^{-(s-1/2)h_a}|=1
$$
for every active height $h_a>0$ in the packet. Weak eta capture records
a Rouch\'e gap for a finite analytic approximation, but it does not
prove this unit-boundary stability. The missing row is therefore a
boundary-stability theorem, not a consequence of Rouch\'e capture alone.
\end{principle}

\begin{principle}[Conservative oscillator reading of the gap]
\label{prin:rh-conservative-oscillator-reading-gap}
The eta packet admits a physical reading as a many-oscillator phase sum.
For
$$
s=\frac12+r+it,
$$
the $n$-th centered eta term is
$$
(-1)^{n-1}n^{-1/2}e^{-r\log n}e^{-it\log n}.
$$
The factor $e^{-it\log n}$ is a conservative phase rotation, while
$e^{-r\log n}$ is a damping or gain factor. Thus a vanishing eta readout
is not automatically a conservative phase closure. It may be a zero of a
nonunitary amplitude-adjusted packet unless the packet carries
unit-boundary stability.
\end{principle}

\begin{definition}[Conservative normal-mode packet]
\label{def:rh-conservative-normal-mode-packet}
A \emph{conservative normal-mode packet} for a zeta zero is a strong
generated packet whose finite oscillator rows display:
$$
\text{canonical dominance or active capture},
$$
$$
\text{unitary phase evolution},
$$
$$
\text{no nontrivial real damping or gain factor},
$$
$$
\text{far-end compatibility without hidden scale absorption},
$$
and
$$
\text{transport to the accepted zeta-zero source row}.
$$
This is the physical reading of the strong packet required by
\autoref{prin:rh-boundary-stability-strong-packet-gap}: the zero is a
normal phase cancellation of a closed conservative system, not a
cancellation produced by adjusting amplitudes.
\end{definition}

\begin{definition}[Galperin-style inertial unfolding]
\label{def:rh-galperin-style-inertial-unfolding}
A \emph{Galperin-style inertial unfolding} for a generated zeta-zero
packet is a displayed carrier
$$
(\mathcal M,H,\Phi_t,\Pi)
$$
with the following rows. The flow row is a single conservative phase flow
$$
\Phi_t=e^{-itH},
\qquad
\Phi_t^*\Phi_t=I.
$$
The projection row $\Pi$ reads the eta or zeta polygonal packet as a
folded shadow of this unfolded orbit. The reflection row says that every
folding, alternating-sign turn, functional-equation mirror, or
prime-routing identification changes only phase direction inside the
unfolded carrier. The unit row forbids any reflected step from changing
radial scale. The crossing row reads a zero as a normalized cancellation
of the projected orbit. The far-end row is a compatibility socket, not a
reservoir for radial defect, and the coverage row transports the unfolded
packet to the accepted zero source carrier.
\end{definition}

\begin{principle}[Unique inertial time excludes radial time]
\label{prin:rh-unique-inertial-time-excludes-radial-time}
In a Galperin-style inertial unfolding, the active centered channels are
generated by one time parameter $t$ through the unitary flow
$$
e^{-it h_a}.
$$
For
$$
s=\frac12+r+it,
$$
the corresponding analytic channel is
$$
e^{-(s-1/2)h_a}=e^{-r h_a}e^{-it h_a}.
$$
The second factor belongs to the inertial phase flow. The first factor is
a real positive scale flow. If $r\ne0$ and some active height satisfies
$h_a>0$, this supplies a second nonunitary time direction, not a
reflection of the unitary unfolding. Thus the unique-inertial-time row can
hold only when the radial parameter vanishes on every active channel.
\end{principle}

\mathsf{ZetaGalperinInertialUnfolding}

\begin{definition}[Zeta cosmos closure row]
\label{def:rh-zeta-cosmos-closure-row}
The \emph{zeta cosmos closure row} is the source-internal form of the

one BEDC triadic carrier:
$$
\mathsf{ZetaCosmosClosure}_{\zeta}.
$$
This row has four visible components. First, every accepted non-trivial
zero is covered by a generated zero:
$$
\prod_{u:\mathcal Z_{\zeta}}
\sum_{z:Z}\varepsilon(z)\sim u.
$$
Second, every zero constructor preserves the triadic invariants
$$
\mathsf P,\qquad \mathsf U,\qquad \mathsf T,
$$
read respectively as prime-ledger distinction, unit-boundary legality,
and recursive time generation. Third, hidden coordinates are confined to
unit phase or prime-register fibres:
$$
\mathbb T
\quad\text{and}\quad
\prod_{p\in S}\mathbb Z_p.
$$
Fourth, no generated row may import an independent real positive scale
ledger
$$
\mathbb R_{>0}
$$
as a closure coordinate. Thus the zeta sector is not treated as a
projection embedded in a larger dissipative system; the row asks for the
completed zero presentation itself to be source-internal.
\end{definition}

Write the closure target as

\mathsf{ZetaHistFlowClosure}_{\zeta}.

For zeta, this row says that every accepted zero packet receives its

\begin{principle}[Hist-flow closure rejects scale time]
\label{prin:rh-hist-flow-closure-rejects-scale-time}
Assume a generated zero packet carries

e^{-(s-1/2)h_a}

The internal Hist-flow supplies constructor extension and its phase

\begin{principle}[Cosmos closure absorbs the strong route obligations]
\label{prin:rh-cosmos-closure-absorbs-strong-route-obligations}
The row $\mathsf{ZetaCosmosClosure}_{\zeta}$ packages the same analytic
load as the conservative-packet route, but with the hidden-scale boundary
made explicit. Its coverage component is zero-presentation exactness for
\autoref{def:rh-generated-zero-carrier}. Its invariant-preservation

\autoref{def:rh-fixed-half-closed-zero-signature}. Its phase and

\autoref{prin:rh-zeta-galperin-inertial-unfolding-target}. Its refusal of
an $\mathbb R_{>0}$ closure coordinate is the no-radial-time row of
\autoref{prin:rh-unique-inertial-time-excludes-radial-time}.
\end{principle}

\begin{corollary}[Cosmos closure is conditionally sufficient]
\label{cor:rh-cosmos-closure-conditionally-sufficient}
If $\mathsf{ZetaCosmosClosure}_{\zeta}$ holds for the RH-free zero
signature, then the hypotheses of
\autoref{thm:rh-strong-capture-boundary-stability-initiality-route} hold.
\end{corollary}

\begin{proof}
The coverage component supplies the essential coverage row
$Z\simeq\mathcal Z_{\zeta}$. The triadic invariant-preservation component
supplies constructor closure for the fixed-half subcarrier. The absence
of an independent $\mathbb R_{>0}$ closure coordinate rules out a
nontrivial factor $e^{-r h_a}$ on every active unit-boundary channel, so
boundary stability is available for the strong packet. These are exactly
the two rows used by
\autoref{thm:rh-strong-capture-boundary-stability-initiality-route}.
\end{proof}

\begin{principle}[Physical gap inventory]
\label{prin:rh-physical-gap-inventory}
In the conservative-oscillator reading, the initiality route leaves five
separate mathematical obligations. First, each zero must have a canonical
dominant normal-mode packet, not merely an arbitrary finite truncation.
Second, that packet must strongly capture the zeta zero, upgrading weak
eta/Rouch\'e capture to a source-compatible generated packet. Third, the
packet must satisfy unitary boundary stability, excluding the damping or
gain factor $e^{-r h}$. Fourth, the infinite tail must satisfy the
no-absorption discipline of
\autoref{def:unified-reality-no-far-end-absorption}, so the far end does
not act as a hidden reservoir for scale defect. Fifth, all accepted
non-trivial zeta zeros must be covered by these generated packets, which
is the zero-presentation exactness row of
\autoref{def:rh-generated-zero-carrier}.
\end{principle}

\begin{corollary}[Physical form of the conditional route]
\label{cor:rh-physical-form-conditional-route}
If every accepted non-trivial zeta zero is represented by a conservative
normal-mode packet, then the only admissible centered-channel closure is
pure phase closure. Hence the hypotheses of
\autoref{thm:rh-strong-capture-boundary-stability-initiality-route} are
met, and the conditional initiality route supplies the RH section.
\end{corollary}

\begin{proof}
Representation by conservative normal-mode packets supplies strong
capture and zero-presentation coverage. The unitary phase and no
damping-or-gain rows supply boundary stability. The far-end compatibility
row prevents the infinite tail from absorbing a displayed real scale
defect. These are exactly the two inputs isolated in
\autoref{thm:rh-strong-capture-boundary-stability-initiality-route}:
zero-presentation exactness and fixed-half closure. Applying that theorem
gives the RH section.
\end{proof}

\begin{theorem}[Strong capture and boundary stability imply the initiality route]
\label{thm:rh-strong-capture-boundary-stability-initiality-route}
Assume an RH-free zero signature $F_Z$ with initial algebra $Z$ and a
realization $\varepsilon:Z\to\mathcal Z_{\zeta}$. If every accepted
non-trivial zeta-zero packet is represented by a strong generated packet
and every constructor of that strong packet preserves the unit-boundary
row of
\autoref{prin:rh-boundary-stability-strong-packet-gap}, then the
hypotheses of
\autoref{thm:rh-conditional-initiality-route-section} are exactly the
zero-presentation exactness and fixed-half closure supplied by those two
rows.
\end{theorem}

\begin{proof}
Strong representation of every accepted non-trivial zero supplies the
essential coverage component of
\autoref{def:rh-generated-zero-carrier}; together with the displayed
soundness and faithfulness rows, this is
$\varepsilon:Z\simeq\mathcal Z_{\zeta}$. Boundary stability states that
each constructor preserves the unit-boundary readout, hence preserves
the predicate
$$
P_{1/2}(z)=
\mathsf{OnCritLine}\bigl(\pi_s(\varepsilon(z))\bigr)
$$
appearing in \autoref{def:rh-generated-fixed-half-subcarrier}. This is
precisely the algebra structure on $H_Z$ required by
\autoref{def:rh-fixed-half-closed-zero-signature}. Applying
\autoref{thm:rh-conditional-initiality-route-section} then yields the
RH section.
\end{proof}

\begin{principle}[Initiality does not give zero completeness]
\label{prin:rh-initiality-not-zero-completeness}
The initiality of $Z=\mu F_Z$ proves induction over the zeros generated by
$F_Z$. It does not prove that $F_Z$ has generated every accepted
critical-strip zero packet. The essential coverage component of
\autoref{def:rh-generated-zero-carrier} is therefore one of the two central
mathematical obligations of the route. Without it, an off-half zero outside
the generated carrier would not be seen by initial-algebra induction.
\end{principle}

\begin{principle}[Constructor closure is the fixed-half hinge]
\label{prin:rh-constructor-closure-fixed-half-hinge}
The other central obligation is fixed-half closure of the constructors.
Administrative constructors, such as classifier transport and ledger
reconstruction, should preserve $P_{1/2}$ by stability. Symmetry
constructors, such as conjugation and functional-equation transport, must
preserve the mirror fixed surface. Limit constructors must preserve
$P_{1/2}$ through real-part continuity and uniqueness of the completed
limit. Primitive zero-source constructors are the analytic hinge: they must
produce fixed-half zero packets without smuggling an
$\mathsf{OnCritLine}$ witness into their input data.
\end{principle}

\begin{principle}[Window induction and limit windows]
\label{prin:rh-window-induction-limit-windows}
The same distinction applies to inspection windows. Finite generated
windows are handled by structural induction over their window signature.
Limit windows are handled as compatible inverse systems
$$
W_0\leftarrow W_1\leftarrow W_2\leftarrow\cdots,
$$
with a limit-window predicate read as
$$
P_{\infty}(w):=\prod_n P_n(\pi_n w).
$$
This removes the need to inspect an untyped totality of windows while still
recording the exact compatibility data required for a completed zero
carrier.
\end{principle}

\begin{principle}[Failure readings]
\label{prin:rh-initiality-route-failure-readings}
The initiality route has three mathematically distinct failure readings.
If $\mathsf{NoRHCirc}(F_Z)$ fails, the proposed generator has assumed the
fixed-half conclusion. If zero-presentation exactness fails, the generator
does not cover the accepted non-trivial zero carrier. If fixed-half closure
fails for a constructor, the failing constructor identifies the analytic
mechanism by which an off-half packet could enter the recovered carrier.
None of these failures is a route-closure decider or an independence
theorem for the fixed proposition $\mathsf{RH}$.
\end{principle}

\begin{theorem}[Completeness bridge to constructive RH]
\label{thm:rh-generation-completeness-bridge}
\leanchecked{BEDC.Derived.RHRoute.GenerativeZeroCompletion.constructiveRH\_of\_completion}
\leanvariant{BEDC.Derived.RHRoute.GenerativeZeroCompletion.symmetryReadbackRH\_of\_completion}
Let $\varepsilon:Z\to\mathcal Z_{\zeta}$ be a generated realization whose
fixed-half closure is displayed, as in \autoref{def:rh-generated-zero-carrier}.
Package the completeness datum as the record whose rows are $\varepsilon$, the
fixed-half closure witness, and the coverage row
$$
\prod_{u:\mathcal Z_{\zeta}}\mathsf{NontrivZero}(u)\to
\sum_{z:Z}\varepsilon(z)\sim u.
$$
From this record, together with the located readback row
$$
\mathsf{OnCritLineReadback}(u)\to\mathsf{OnCritLine}(u),
$$
the fixed-half closure supplies $\mathsf{CritLine}(\varepsilon(z))$ for the
covering generator, and critical-line transport along $\sim$ yields
$\mathsf{OnCritLine}(u)$. This is the constructive Riemann hypothesis of
\autoref{ch:visions-rh-layered-translation}, read off the generation side. The
theorem is conditional: it consumes the coverage record and the readback row as
hypotheses and constructs neither. The coverage row is the completeness
obligation of \autoref{def:rh-generated-zero-carrier}; it is not discharged
here, and no inhabitant of the completeness record is exhibited.
\end{theorem}

\begin{principle}[Zero-free forward dynamics and the obligation firewall]
\label{prin:rh-generation-forward-firewall}
\leandef{BEDC.Derived.RHRoute.LocatedGenerationTower.ZetaBridgeObligations}
\leandef{BEDC.Derived.RHRoute.LocatedGenerationTower.generated\_true\_zero\_twoWay}
The generation side is presented as a forward dynamics that never inspects a
zeta value. A tick carries a radial log-scale row, an angular rotation row, and
a fresh orthogonal residual, gated by a no-adjacent Zeckendorf certificate. A
state stores only its generated syntax; its located point is the fold of that
syntax, never a stored coordinate, so no tick can fit a zero backwards. The
analytic content is quarantined behind an evaluator interface whose binding,
completeness, and soundness rows are the only bridge to the true zero set, and
the forward dynamics constructs none of them. Under the packaged bridge the
generated orbit and the true zero set stand in a displayed two-way
correspondence.
\end{principle}

\section{Spectral reformulation and the Hilbert--P\'olya wall}
\label{sec:rh-generation-spectral-reformulation}

The completeness obligation of \autoref{def:rh-generated-zero-carrier} can be
carried by a single spectral statement rather than a pointwise zeta binding. The
generation side supplies, $\zeta$-free, a formal phase register, a self-adjoint height
operator, and the coupling that removes every internal bookkeeping obligation; the
only surviving analytic content is that the generated spectrum is the true zero
spectrum.

\begin{principle}[Prime-phase character factorization]
\label{prin:rh-generation-prime-phase}
\leandef{BEDC.Derived.RHRoute.OCLSDPhaseAlgebra.readPZGPhase}
\leanchecked{BEDC.Derived.RHRoute.OCLSDPhaseAlgebra.readPZGPhase\_tick}
The prime register reads back into a formal phase algebra by multiplying one atom
phase per activated Zeckendorf bit,
$$
\chi_t(n)=\prod_{(p,k)}\mathsf{primePhase}(p,k),
$$
multiplicatively over register concatenation. The atom phase depends on the prime
and Fibonacci index alone, so it is prime-locked: no free angle enters, and the
phase cannot be tuned to a target height.
\end{principle}

\begin{definition}[Spectral zero identity]
\label{def:rh-generation-spectral-identity}
\leandef{BEDC.Derived.RHRoute.OCLSDSpectral.SpectralZeroIdentity}
\leandef{BEDC.Derived.RHRoute.OCLSDSpectral.zetaBridge\_from\_spectral}
Let $H_{\mathrm{OCLSD}}$ be the diagonal operator whose orthonormal basis is the
forward orbits and whose eigenvalue on an orbit is its cumulative angular height.
The spectral zero identity is the pair of readings
$$
\gamma\in\operatorname{Spec}(H_{\mathrm{OCLSD}})\ \Longleftrightarrow\ \zeta(\tfrac12+i\gamma)=0.
$$
It is strictly weaker than pointwise evaluator binding, and it suffices: with a
binding stub and the located readback it assembles the full zero-bridge record.
\end{definition}

\begin{principle}[Self-adjoint generator, prime-locked spectrum]
\label{prin:rh-generation-self-adjoint}
\leandef{BEDC.Derived.RHRoute.OCLSDHamiltonian.hamiltonian\_selfAdjoint}
\leanchecked{BEDC.Derived.RHRoute.OCLSDPrimeLock.primeLocked\_angle\_independent}
Because the orbit heights are located-real, $H_{\mathrm{OCLSD}}$ is self-adjoint and
generates a unitary phase flow. Because the per-tick height factors through the
prime register only, equal prime data forces equal height: the spectrum is
determined by prime arithmetic, not by the free angle row.
\end{principle}

\begin{principle}[Critical coupling closes the internal obligations]
\label{prin:rh-generation-critical-coupling}
\leandef{BEDC.Derived.RHRoute.OCLSDCriticalCoupling.orbitCriticalReadback\_from\_stateHeight}
\leandef{BEDC.Derived.RHRoute.OCLSDCriticalCoupling.zetaBridge\_from\_spectral\_coupled}
The orbit-to-critical-line readback is not an analytic wall but a coupling gap
between the syntactic point readback and the accumulated height. A state-level
height that commutes with both discharges it by induction over the orbit recursor.
After this the entire zero-bridge record follows from the spectral zero identity and
a binding stub alone. The one remaining obligation is the spectral zero identity
itself --- that the prime-generated spectrum is the true zeta zero spectrum. That is
the Hilbert--P\'olya content; it is not discharged here, and nothing in this chapter
constructs it.
\end{principle}
