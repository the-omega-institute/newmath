\chapter{EventFlow Certificate Flow for Primes, Zeta, and RH}
\label{ch:visions-rh-eventflow-certificate-flow}
\origin{ai}

\autoref{ch:visions-rh-zero-skew-eventflow-target} places the Riemann
Hypothesis at the compiler-facing layer as a transformer from recognized
zero-event packets to recognized critical-line packets.  The present chapter
records the certificate-flow substrate below that transformer.  It shows how
prime recognition, zeta recovery, and the RH fixed-point target are separated
into finite EventFlow rows.  This separation does not prove $\mathsf{RH}$.
It turns the conjecture into a fixed-output certificate target whose missing
row is visible.

\section{Boundary before semantics}
\label{sec:rh-eventflow-boundary-before-semantics}

\begin{definition}[Channel fixed boundary]
\label{def:rh-channel-fixed-boundary}
For an event flow $S=(w_0,\ldots,w_{n-1})$, the channel layer records a
source event-flow row $E$, a legal channel-code row $C$, a decoder row $D$,
and a round-trip witness row $R$:
$$
E\to C\to D\to R.
$$
The round-trip row asserts only
$$
\mathsf{Decode}(\mathsf{Compile}(S))\hsame S.
$$
It recovers the event boundaries and the event sequence.  It does not recover
primehood, zeta values, zerohood, or proof rows.
\end{definition}

\begin{proposition}[Channel reconstruction is a boundary fixed point]
\label{prop:rh-channel-reconstruction-boundary-fixed-point}
Every accepted legal-channel read makes the source event flow a fixed point
of the compile-decode composite:
$$
(\mathsf{Decode}\circ\mathsf{Compile})(S)\hsame S.
$$
\end{proposition}

\begin{proof}
The accepted read must pass through the displayed rows $E,C,D,R$.  The row
$C$ stores the legal channel code of the source flow named by $E$, the row $D$
stores the decoder read, and $R$ identifies the decoded output with the
source event flow up to $\hsame$.  Hence the compile-decode composite returns
the same event sequence and no semantic classifier has yet been applied.
\end{proof}

\begin{corollary}[Terminator rows are not zeta-zero rows]
\label{cor:rh-eventflow-terminator-not-zero-row}
If the channel representation uses $11$ as event terminator, each occurrence
of $11$ records only the end of one raw event.  A zeta-zero reading can appear
only after a generated recognizer accepts a whole decoded packet as a
$\mathsf{ZeroCode}_{\mathsf{RH}}$ or $\mathsf{ZeroEventFlow}_{\mathsf{RH}}$
surface.
\end{corollary}

\begin{proof}
This is the channel-level specialization of
\autoref{prop:rh-terminator-not-zero-record}.  The terminator belongs to the
legal-code row, while a zero row belongs to a later recognizer and certificate
gate.
\end{proof}

\section{Prime flow as factorization fixed point}
\label{sec:rh-prime-flow-factorization-fixed-point}

\begin{definition}[Arithmetic event packets]
\label{def:rh-arithmetic-event-packets}
A natural-number event packet $\mathsf{NatFlow}(n)$ is a recognized finite
EventFlow packet exposing
$$
n,\quad \NatUp,\quad \NameCert_{\NatUp},\quad
\mathsf{ContRoute},\quad \mathsf{Pkg}.
$$
A split event packet $\mathsf{SplitFlow}(n;a,b)$ exposes natural-number rows
for $a,b,n$ together with a multiplication row
$$
a\cdot b\hsame n.
$$
The split is non-trivial when both displayed factors are separated from the
unit row.
\end{definition}

\begin{definition}[Prime event flow]
\label{def:rh-prime-event-flow}
A prime event flow $\mathsf{PrimeFlow}(p)$ is a natural-number event packet
for $p$ together with a strict row $p>1$ and a refusal row excluding every
non-trivial split packet $\mathsf{SplitFlow}(p;a,b)$.
\end{definition}

\begin{definition}[Factorization normalizer]
\label{def:rh-factorization-normalizer}
The factorization normalizer $\mathsf{FactNorm}$ sends a recognized
natural-number packet to the finite multiset of prime packets generated by
its displayed factorization route.  A singleton output is written $[p]$.
\end{definition}

\begin{proposition}[Prime flow is the atomic factorization fixed point]
\label{prop:rh-prime-flow-atomic-fixed-point}
For a recognized natural-number packet $p$ with $p>1$,
$$
\mathsf{PrimeFlow}(p)
\quad\Longleftrightarrow\quad
\mathsf{FactNorm}(p)\hsame [p].
$$
\end{proposition}

\begin{proof}
If $\mathsf{PrimeFlow}(p)$ holds, the refusal row excludes all non-trivial
split packets.  The factorization normalizer therefore cannot expose a
proper product decomposition and returns the singleton packet $[p]$.  Conversely,
if $\mathsf{FactNorm}(p)\hsame[p]$ while a non-trivial split
$p\hsame a\cdot b$ existed with both factors separated from the unit row, the
normalizer would have to continue through the factorization routes of $a$ and
$b$, yielding a non-singleton prime packet list.  This contradicts the
displayed singleton row.  Hence no non-trivial split exists, and the packet is
prime.
\end{proof}

\begin{remark}[No arithmetic self-assertion]
\label{rem:rh-prime-flow-no-self-assertion}
The fixed-point reading of \autoref{prop:rh-prime-flow-atomic-fixed-point} is
not a bare self-assertion that $p$ is prime.  It is a factorization readback:
the object is atomic because the recognized multiplication-split flow returns
the same prime packet and no proper split packet.
\end{remark}

\section{Zeta as socket recovery flow}
\label{sec:rh-zeta-socket-recovery-flow}

\begin{definition}[Prime-local and Dirichlet event projections]
\label{def:rh-prime-local-dirichlet-event-projections}
For a recognized prime packet $\mathsf{PrimeFlow}(p)$ and a complex history
$s$, the prime-local channel event is
$$
\mathsf{PrimeChan}_p(s)=p^{-s}.
$$
On the right half-plane $\Re(s)>1$, the Euler event projection and Dirichlet
event projection are
$$
\mathsf{EulerFlow}(s)=
\prod_{p\in\mathsf{PrimeFlow}}(1-p^{-s})^{-1},
\qquad
\mathsf{DirichletFlow}(s)=\sum_{n\geq 1}n^{-s}.
$$
\end{definition}

\begin{definition}[Zeta event-flow socket]
\label{def:rh-zeta-eventflow-socket}
A zeta event-flow socket is a finite packet
$$
Z=(B,E,A,P,F,T,G,H,C,N)
$$
whose rows are read as follows:
$$
\begin{gathered}
B:\ZetaBasicUp\text{ source on }\Re(s)>1,\quad
E:\eta\text{-half-plane handoff},\\
A:\AnalyticContUp\text{ handoff},\quad
P:\text{pole ledger at }s=1,\\
F:\text{functional-equation factor ledger},\quad
T:\text{trivial-zero ledger},\\
G:\GammaUp\text{ boundary row},\quad
H:\hsame\text{ transports},\\
C:\text{finite }\Cont\text{ routes},\quad
N:\NameCert\text{ row}.
\end{gathered}
$$
This is the EventFlow reading of the $\ZetaContinuationSocketUp$ packet.
\end{definition}

\begin{proposition}[Zeta recovery from prime-local flow and socket rows]
\label{prop:rh-zeta-recovery-from-eventflow-socket}
On $\Re(s)>1$,
$$
\mathsf{DirichletFlow}(s)\hsame\mathsf{EulerFlow}(s).
$$
Together with the rows of \autoref{def:rh-zeta-eventflow-socket}, this
right-half-plane agreement recovers the global meromorphic carrier
$\mathsf{ZETA}$ as a compatible projection packet rather than as a single
series expression.
\end{proposition}

\begin{proof}
Absolute convergence on $\Re(s)>1$ permits grouping by the displayed
factorization routes.  Each natural-number index decomposes uniquely as a
finite product of prime packets:
$$
n=\prod_p p^{k_p}.
$$
Therefore
$$
n^{-s}=\prod_p p^{-k_ps},
$$
and the sum over positive indices becomes the product of local geometric
series
$$
\sum_{n\geq 1}n^{-s}
=
\prod_p\sum_{k\geq 0}p^{-ks}
=
\prod_p(1-p^{-s})^{-1}.
$$
This supplies the prime-local projection row.  The zeta socket then adds the
eta handoff, analytic-continuation handoff, pole ledger, functional-equation
ledger, Gamma-boundary row, transports, and finite continuation routes.  These
rows form the compatible recovery surface for $\mathsf{ZETA}$.
\end{proof}

\begin{corollary}[Zeta is not exhausted by its series projection]
\label{cor:rh-zeta-not-exhausted-by-series-flow}
The event-flow reading of $\mathsf{ZETA}$ is not the bare row
$\sum_{n\geq 1}n^{-s}$.  It is the recovered carrier obtained from the
Dirichlet row, Euler prime row, eta handoff, analytic-continuation handoff,
pole row, functional-equation row, Gamma boundary, transport rows, and local
NameCert row.
\end{corollary}

\begin{proof}
The Dirichlet row is only one projection on $\Re(s)>1$.  The remaining rows of
\autoref{def:rh-zeta-eventflow-socket} are necessary to read the same carrier
outside that window and to preserve its pole, mirror, trivial-zero, and
transport ledgers.
\end{proof}

\section{RH as fixed-output target}
\label{sec:rh-as-fixed-output-target}

\begin{definition}[Prime-unitary balance event flow]
\label{def:rh-prime-unitary-balance-event-flow}
For a complex history $s$, a prime-unitary balance event flow is a recognized
packet exposing, for every prime packet $\mathsf{PrimeFlow}(p)$, the centered
channel row
$$
U_p(s)=p^{1/2}p^{-s}=p^{1/2-s}
$$
and the unit-amplitude witness
$$
\left|U_p(s)\right|\hsame 1.
$$
\end{definition}

\begin{definition}[Prime-balanced RH compiler target]
\label{def:rh-prime-balanced-compiler-target}
The prime-balanced RH compiler target is a certified compiler or proof flow
$C_{\mathsf{RH}}^{\mathrm{prime}}$ with type
$$
C_{\mathsf{RH}}^{\mathrm{prime}}:
\mathsf{ZeroEventFlow}_{\mathsf{RH}}(s)
\to
\mathsf{PrimeUnitaryBalanceFlow}_{\mathsf{RH}}(s).
$$
Expanded through the recognized rows, it is the target
$$
\bigl(
\mathsf{ZETA}(s)\hsame 0_{\mathbb C}
\wedge
\mathsf{InCritStrip}(s)
\bigr)
\to
\forall p\in\mathsf{PrimeFlow},\,
\left|p^{1/2-s}\right|\hsame 1.
$$
\end{definition}

\begin{proposition}[Prime-balanced compiler target is an RH witness]
\label{prop:rh-prime-balanced-compiler-target-witness}
An accepted prime-balanced RH compiler target supplies the same mathematical
content as the zero--skew transformer of \autoref{prop:rh-zero-skew-form}.
\end{proposition}

\begin{proof}
\autoref{thm:rh-prime-unitary-balance-critical-line} identifies the
prime-unitary output packet with the critical-line row $\Re(s)\hsame 1/2$.
Thus a compiler flow from a recognized zero event packet to a recognized
prime-unitary balance packet yields the skew-zero output row required by
\autoref{prop:rh-zero-skew-form}.  Conversely, a zero--skew transformer gives
the prime-unitary output by reading $\Re(s)\hsame 1/2$ through the normalized
prime-channel amplitude formula.
\end{proof}

\begin{definition}[Zero-orbit collapse event flow]
\label{def:rh-zero-orbit-collapse-event-flow}
Let $J(s)=1-\overline{s}$.  A zero-orbit collapse event flow for $s$ is a
finite packet containing
$$
\mathsf{ZeroEventFlow}_{\mathsf{RH}}(s),\quad
\mathsf{ZeroEventFlow}_{\mathsf{RH}}(J(s)),\quad
\hsame(s,J(s)),
$$
together with the displayed transport, route, provenance, and local naming
rows.
\end{definition}

\begin{proposition}[RH is zero-orbit collapse]
\label{prop:rh-is-zero-orbit-collapse}
The BEDC statement $\mathsf{RH}$ is equivalent to the existence of a total
target
$$
\mathsf{ZeroEventFlow}_{\mathsf{RH}}(s)
\to
\mathsf{ZeroOrbitCollapseFlow}_{\mathsf{RH}}(s).
$$
\end{proposition}

\begin{proof}
If $\mathsf{RH}$ holds, a recognized non-trivial zero packet has
$\Re(s)\hsame 1/2$.  The mirror calculation of \autoref{def:rh-mirror-map}
then gives $\hsame(s,J(s))$, and the functional-equation mirror rows supply
the companion zero packet for $J(s)$ when the zeta socket rows are displayed.
This is the zero-orbit collapse packet.  Conversely, if every zero event flow
has an accepted collapse packet, its $\hsame(s,J(s))$ row implies
$\Re(s)\hsame 1/2$ by the same mirror calculation, hence supplies
$\mathsf{OnCritLine}(s)$ for every critical-strip zero packet.
\end{proof}

\section{Fixed-point stratification}
\label{sec:rh-fixed-point-stratification}

\begin{principle}[Certified fixed points only]
\label{prin:rh-certified-fixed-points-only}
The EventFlow reading admits fixed points only as displayed certificate rows:
channel round-trip rows, factorization singleton rows, projection-recovery
rows, mirror-collapse rows, and fixed-output witness rows.  It does not admit
bare self-reference, host-level fixed-point principles, or self-application
oracles.
\end{principle}

\begin{theorem}[Three fixed-point layers]
\label{thm:rh-three-fixed-point-layers}
The EventFlow reconstruction of primes, zeta, and RH separates three
fixed-point layers:
$$
\mathsf{PrimeFlow}(p)
\Longleftrightarrow
\mathsf{FactNorm}(p)\hsame[p],
$$
$$
P_i\bigl(\operatorname{Rec}((D_i,f_i)_i)\bigr)\hsame f_i
\quad\text{for each displayed zeta projection row},
$$
and
$$
\mathsf{RH}
\Longleftrightarrow
\forall s,\,
\mathsf{ZeroEventFlow}_{\mathsf{RH}}(s)
\to
\mathsf{ZeroOrbitCollapseFlow}_{\mathsf{RH}}(s).
$$
\end{theorem}

\begin{proof}
The first equivalence is \autoref{prop:rh-prime-flow-atomic-fixed-point}.  The
second is the projection-recovery discipline of
\autoref{prop:rh-zeta-recovery-from-eventflow-socket}: recovering the carrier
from compatible local rows and projecting back to any displayed window returns
that window's row.  The third is \autoref{prop:rh-is-zero-orbit-collapse}.
Each layer is a certified readback or fixed-output row, not a self-application
principle.
\end{proof}

\begin{remark}[Generator fixed-point boundary]
\label{rem:rh-generator-fixed-point-boundary}
The same discipline is enforced by the $\GeneratorFixedPointUp$ packet.  A
fixed-output witness is accepted only through the displayed rows
$G,L,Q,W,O,H,C,P,N$; any diagonal or self-application attempt outside those
rows remains a $\mathsf{GAP}$ boundary.  The RH target therefore asks for a
finite output witness, not for a host theorem that a proof generator must have
a fixed point.
\end{remark}
