\chapter{Source-Side Triadic Audit Criterion}
\label{ch:visions-rh-source-side-triadic-audit-criterion}
\origin{ai}

\autoref{ch:visions-rh-right-triangle-invariant-system} records the
finite scale-leg invariant, while
\autoref{def:rh-certificate-tower-triadic-invariant} records the
certificate-tower rows $\mathsf P$, $\mathsf U$, and $\mathsf T$.  The
present chapter states the audit rule that every RH-facing source row
must satisfy before it can be consumed by the fixed-half route.  A row
that occupies only one side of the triangle is a local tool.  A row
that is used in the route must display its source ledger, unit-positive
boundary, and refinement or finite-witness order.

\begin{definition}[Source-side triadic audit of a route row]
\label{def:rh-source-side-triadic-audit-route-row}
Let $\mathcal Q$ be a source row in the RH route.  A source-side triadic
audit of $\mathcal Q$ is a displayed factorization
$$
\mathcal Q
\leadsto
(\mathcal Q_{\mathsf P},\mathcal Q_{\mathsf U},\mathcal Q_{\mathsf T})
$$
where $\mathcal Q_{\mathsf P}$ is the source-ledger and legal-hidden
fibre row, $\mathcal Q_{\mathsf U}$ is the unit-boundary or positive
energy row, and $\mathcal Q_{\mathsf T}$ is the refinement, cofinality,
or finite-witness row.  The audit is closed when the three rows read the
same scale defect and no side introduces an unrecorded real-positive
coordinate.
\end{definition}

\begin{definition}[Triadic source transport]
\label{def:rh-triadic-source-transport}
For an audited route row $\mathcal Q$, triadic source transport is the
requirement that the following three transport rows are source-side
valid:
$$
\mathcal Q_{\mathsf P}+\mathcal Q_{\mathsf T}\to\mathcal Q_{\mathsf U},
\qquad
\mathcal Q_{\mathsf P}+\mathcal Q_{\mathsf U}\to\mathcal Q_{\mathsf T},
\qquad
\mathcal Q_{\mathsf U}+\mathcal Q_{\mathsf T}\to\mathcal Q_{\mathsf P}.
$$
The first row says that a legal ledger and an internal time readout
force the absence of an independent radial gauge.  The second says that
a legal ledger and a unit boundary leave only the phase-time parameter
as a free readout.  The third says that a unit boundary and a refinement
law select the zero-relevant modes that must be ledgered.
\end{definition}

\begin{definition}[Triadic no-scale packet]
\label{def:rh-triadic-no-scale-packet}
Let an active packet $A_m$ carry heights $h_a>0$ and centered channel
$$
U_a(s)=e^{-(s-1/2)h_a}=e^{-rh_a}e^{-it h_a}.
$$
The packet is triadic no-scale when its $\mathsf P$ row accepts compact
phase and prime-register fibres only, its $\mathsf U$ row supplies a
source-side bound for the scale energy, and its $\mathsf T$ row supplies
cofinal active-weight growth or a finite negative witness whenever
$r\ne0$.  In symbols the scale row is
$$
d_a(s):=\log|U_a(s)|=-r h_a,
\qquad
E_m^{\mathrm{scale}}(s)=\sum_{a\in A_m}d_a(s)^2=r^2W_m,
$$
where
$$
W_m:=\sum_{a\in A_m}h_a^2.
$$
\end{definition}

\begin{principle}[Stable rows already fit the audit]
\label{prin:rh-stable-rows-fit-triadic-audit}
The BEDC history generator supplies the $\mathsf T$ side: finite
records extend by append-only constructor rows.  The phase-channel
solenoid of \autoref{def:rh-phase-channel-hidden-prime-fibre} supplies
the $\mathsf P$ side: hidden state may live in compact prime-register
fibres, not in an external real-positive scale line.  Weak eta capture
of \autoref{def:rh-weak-eta-capture-packet} supplies local analytic
capture and part of the time side, but it does not supply the unit side.
The Fibonacci path and Lee--Yang lift rows of
\autoref{def:rh-fibonacci-path-carrier} and
\autoref{def:rh-lee-yang-lift-certificate} supply a finite internal
ruler and a unit-boundary seed, but not by themselves a prime-compatible
zeta container.  The quartic endpoint model supplies the methodological
load: a branch must be shown dominant by source certificates before it
can carry a zero law.
\end{principle}

\begin{theorem}[Triadic no-scale criterion]
\label{thm:rh-triadic-no-scale-criterion}
Let every zero-relevant accepted certificate tower be represented by
triadic no-scale packets.  Assume that each such packet satisfies
triadic source transport, has defect extraction
$$
E_m^{\mathrm{scale}}(s_\infty)=r^2W_m+o(W_m),
$$
has either $W_m\to\infty$ or a finite negative witness when $r\ne0$,
and carries a source-side unit-positive row excluding unbounded scale
energy and negative witnesses.  Then every represented readout
$$
s_\infty=\frac12+r+it
$$
has $r=0$.
\end{theorem}

\begin{proof}
Assume $r\ne0$.  If the time row supplies cofinal active-weight growth,
then the defect extraction gives a cofinal tail on which
$$
E_m^{\mathrm{scale}}(s_\infty)\ge\frac{r^2}{2}W_m.
$$
The energies are unbounded along that tail, contradicting the
source-side unit-positive row.  If instead the time row supplies a
finite negative witness, that witness contradicts the same
unit-positive row in its finite test form.  In both cases the assumption
$r\ne0$ is impossible.  Hence $r=0$.
\end{proof}

\begin{principle}[Compressed source target]
\label{prin:rh-compressed-source-target}
The compressed source target is to realize one positive-spectrum row
whose triadic audit is source-side.  The Stieltjes log-derivative target
of \autoref{def:rh-xi-stieltjes-log-derivative-row} is the most compact
form of this requirement: the measure must be generated before reading a
fixed-half divisor, positivity supplies the unit side, and the
representation supplies a half-plane time row.  When such a source row
is transported through the prime and certificate ledgers, the
right-triangle scale coordinate has no legal carrier.
\end{principle}
