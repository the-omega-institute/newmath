\chapter{Golden Bridge Three-Level Closure}
\label{ch:visions-rh-golden-bridge-three-level-closure}
\origin{ai}

\autoref{ch:visions-rh-golden-lee-yang-bridge-obligations} isolates
the analytic bridge rows that remain between weak eta capture and a
Lee--Yang stable golden packet.  The present chapter records the same
load in three strata: an abstract finite theorem, a fixed quartic finite
model, and the zeta-facing golden source package.  The strata are kept
separate so that a verified finite pattern is not read as a global zeta
theorem.

\begin{definition}[Three-level golden bridge package]
\label{def:rh-three-level-golden-bridge-package}
The \emph{three-level golden bridge package} consists of the following
rows.

The abstract row is the finite dominant-branch control theorem
\autoref{thm:rh-dominant-gap-gives-rouche-control}.  It requires a
displayed spectral expansion, a dominant pair, and the boundary
inequality
$$
|P_m^{\mathrm{sub}}(z)|<|P_m^{\mathrm{dom}}(z)|
\qquad
z\in\partial\Omega_m.
$$

The finite-model row is the quartic endpoint carrier of
\autoref{prin:rh-quartic-endpoint-finite-model}.  It shows that the
dominant-branch, Puiseux, two-branch interference, and Rouch\'e
isolation rows can be realized inside one finite spectral carrier.

The zeta-facing row is the source package of
\autoref{def:rh-golden-bridge-analytic-source-rows}, namely
$$
\mathsf{GoldenSpectralExpansion},
\qquad
\mathsf{GoldenSpectralGap},
\qquad
\mathsf{GoldenStableRoucheHomotopy}.
$$
This row is the only stratum that can feed the golden zeta bridge.
\end{definition}

\begin{theorem}[Abstract dominant control is finite]
\label{thm:rh-abstract-dominant-control-is-finite}
The abstract row of
\autoref{def:rh-three-level-golden-bridge-package} is a finite analytic
theorem: it transports zero count from $P_m^{\mathrm{dom}}$ to $P_m$
under the displayed strict boundary inequality.
\end{theorem}

\begin{proof}
This is exactly \autoref{thm:rh-dominant-gap-gives-rouche-control}.
The theorem uses only the decomposition
$$
P_m=P_m^{\mathrm{dom}}+P_m^{\mathrm{sub}}
$$
and the strict boundary estimate.  No zeta-specific coverage row,
Lee--Yang bridge row, or infinite tower row is consumed.
\end{proof}

\begin{principle}[Quartic endpoint as finite witness]
\label{prin:rh-quartic-endpoint-as-finite-witness}
The quartic endpoint carrier witnesses the shape of the argument, not
its zeta coverage.  It contains a fixed spectral polynomial, a
normalization ledger, a real projection ledger, a dominance ledger, and
an endpoint-zero ledger.  Its scope boundary refuses a growing-window
endpoint-zero theorem and refuses any zeta conclusion.  Thus it licenses
the methodology of finite dominance followed by local Rouch\'e
isolation, while leaving the zeta golden spectral expansion and gap as
separate source obligations.
\end{principle}

\begin{definition}[Zeta-facing strong golden bridge package]
\label{def:rh-zeta-facing-strong-golden-bridge-package}
The source row
$$
\mathsf{ZetaFacingStrongGoldenBridge}_{\zeta,\varphi}
$$
asserts that every accepted non-trivial completed zeta zero has a golden
certificate tower on which the following compatible rows are present:
golden weak eta capture, the Zeckendorf--Mellin exact projection of
\autoref{thm:rh-zeckendorf-mellin-exact-projection},
the analytic source rows of
\autoref{def:rh-golden-bridge-analytic-source-rows}, cofinal
active-height growth, and no unledgered radial scale gauge.  The package
does not assert the analytic rows by definition; it records the source
surface that must be supplied before the fixed-half readout can be
exported.
\end{definition}

\begin{theorem}[Strong golden bridge gives the zeta Lee--Yang bridge]
\label{thm:rh-strong-golden-bridge-gives-zeta-lee-yang-bridge}
If
$$
\mathsf{ZetaFacingStrongGoldenBridge}_{\zeta,\varphi}
$$
holds, then
$$
\mathsf{ZetaGoldenLeeYangBridge}
$$
holds.
\end{theorem}

\begin{proof}
Fix an accepted non-trivial completed zeta zero.  The source row of
\autoref{def:rh-zeta-facing-strong-golden-bridge-package} supplies a
golden certificate tower for that zero and supplies the analytic source
rows of \autoref{def:rh-golden-bridge-analytic-source-rows}.  These rows
provide the dominant branch selection, the stable Rouch\'e transport,
and the Lee--Yang stable zero-free regions on the same zero island.
Together with the weak eta capture and Zeckendorf address ledgers, they
are precisely the golden Lee--Yang bridge witnesses required by
\autoref{def:rh-zeta-golden-lee-yang-bridge-source-row}.
\end{proof}

\begin{theorem}[Strong golden bridge gives fixed-half readout]
\label{thm:rh-strong-golden-bridge-fixed-half-readout}
If
$$
\mathsf{ZetaFacingStrongGoldenBridge}_{\zeta,\varphi}
$$
holds, then every accepted non-trivial completed zeta zero has real part
$1/2$.
\end{theorem}

\begin{proof}
By
\autoref{thm:rh-strong-golden-bridge-gives-zeta-lee-yang-bridge}, the
source row $\mathsf{ZetaGoldenLeeYangBridge}$ holds.  Apply
\autoref{cor:rh-golden-lee-yang-bridge-fixed-half-readout}.
\end{proof}

\begin{principle}[Three-level proof boundary]
\label{prin:rh-three-level-proof-boundary}
The abstract row and the quartic endpoint row are evidence about proof
shape.  They do not cover zeta zeros.  The zeta-facing row must still
provide the analytic bridge package
$$
\mathsf{GoldenSpectralExpansion}
\quad
\mathsf{GoldenSpectralGap}
\quad
\mathsf{GoldenStableRoucheHomotopy},
$$
or, through the Mellin factorization of
\autoref{prin:rh-mellin-factorization-bridge-load}, the two lower rows
$$
\mathsf{ZeckendorfTransferDominatedSplitting}
\qquad
\mathsf{SelfDualMellinLeeYangOperator}.
$$
Only after those rows are attached to the same golden zero island does
the no-scale argument apply to the readout
$$
s=\frac12+r+it.
$$
\end{principle}

\begin{definition}[Renormalized Zeckendorf endpoint transfer]
\label{def:rh-renormalized-zeckendorf-endpoint-transfer}
The endpoint scale for the Zeckendorf--Mellin fibre is not a fixed
$u>0$.  It is the normalized scale
$$
u=\frac{\alpha}{F_{m+2}}.
$$
At this scale the Mellin fibre variables become
$$
\widetilde x_{m,j}(\alpha)
:=
(-1)^{F_{j+1}}
\exp\left(-\alpha\frac{F_{j+1}}{F_{m+2}}\right).
$$
The associated non-autonomous path transfer matrices are
$$
T_{m,j}(\alpha)
:=
\begin{pmatrix}
1&\widetilde x_{m,j}(\alpha)\\
1&0
\end{pmatrix},
$$
with total endpoint transfer
$$
\mathcal T_m(\alpha)
:=
T_{m,m}(\alpha)T_{m,m-1}(\alpha)\cdots T_{m,1}(\alpha).
$$
The source row
$$
\mathsf{RenormalizedZeckendorfEndpointTransfer}
$$
asserts that the zero-relevant Mellin window is read through this
normalized transfer product, not through a fixed-$u$ fibre where the
large Fibonacci weights freeze the tail variables.
\end{definition}

\begin{principle}[Endpoint scaling is part of dominance]
\label{prin:rh-endpoint-scaling-part-of-dominance}
For fixed $u>0$, the variables
$$
x_j(u)=(-1)^{F_{j+1}}e^{-uF_{j+1}}
$$
decay rapidly with $j$, so the last variables of a long Zeckendorf
window no longer participate in a balanced Fibonacci competition.  The
renormalized scale of
\autoref{def:rh-renormalized-zeckendorf-endpoint-transfer} keeps the
terminal ratios
$$
\frac{F_{j+1}}{F_{m+2}}
$$
visible.  Hence the dominated-splitting row in
\autoref{def:rh-zeckendorf-transfer-dominated-splitting} must be proved
for the normalized endpoint transfer product $\mathcal T_m(\alpha)$.
\end{principle}

\begin{definition}[Mellin stability attack surfaces]
\label{def:rh-mellin-stability-attack-surfaces}
The source row
$$
\mathsf{SelfDualMellinLeeYangOperator}
$$
may be attacked through any of the following stability-preservation
surfaces.

The total-positivity surface proves that the completed Mellin kernel
produces a P\'olya-frequency or Laguerre--P\'olya type transform
compatible with the Lee--Yang zero-free regions.

The Hermite--Biehler surface rewrites the completed Mellin packet as a
function
$$
E(z)=A(z)-iB(z)
$$
whose half-plane or unit-circle stability is controlled by a
de Branges-type ledger.

The Rouch\'e-stable discretization surface approximates the completed
Mellin integral by finite sums
$$
\sum_{\ell}w_{\ell}(s)L_{m,u_{\ell}},
$$
with a stability-preserving choice of weights and a boundary error
ledger strong enough to pass to the integral without crossing zero.
\end{definition}

\begin{principle}[Mellin stability as no-demon finite form]
\label{prin:rh-mellin-stability-no-demon-finite-form}
The self-dual Mellin stability row is the finite packet analogue of the
no-demon constraint.  An off-half readout
$$
s=\frac12+r+it
$$
introduces the real scale factor
$$
e^{-rh_a}
$$
and therefore tilts any active-mode distribution by real-positive
weights.  If the completed self-dual Mellin readout preserves the
Lee--Yang inside and outside zero-free regions, that tilt cannot be
hidden as a phase or prime-register gauge.  The only remaining legal
readout is the unit boundary readout, where
$$
|e^{-(s-1/2)h_a}|=1.
$$
\end{principle}

\begin{principle}[Mellin projection obstruction]
\label{prin:rh-mellin-projection-obstruction}
Fibrewise Lee--Yang stability is not, by itself, a stability theorem for
Dirichlet-polynomial readout.  The Zeckendorf fibre identity of
\autoref{thm:rh-zeckendorf-mellin-exact-projection} places the finite
geometric packet
$$
K_M(u)=\sum_{N=0}^{M-1}(-1)^N e^{-uN}
$$
on the unit boundary in the variable $z=e^{-u}$.  The Mellin-projected
Dirichlet packet, however, reads the same finite data through powers
$n^{-s}$.  The smallest non-trivial Dirichlet packet
$$
P_2(s)=1-2^{-s}
$$
has zero set
$$
s=\frac{2\pi i k}{\log 2},
\qquad
k\in\mathbb Z,
$$
and therefore lies on the line $\Re(s)=0$.  Thus the unrestricted
implication from unit-boundary fibre zeros to fixed-half Mellin zeros is
false.  Any zeta-facing use of the Mellin row must carry the additional
ledger that transports the additive Zeckendorf height into the
multiplicative prime height without creating a real-positive scale
gauge.
\end{principle}

\begin{definition}[Self-dual Zeckendorf completed packet]
\label{def:rh-self-dual-zeckendorf-completed-packet}
Let
$$
K_m(u):=I_m(\mathbf x(u))-1
=
\sum_{n=1}^{F_{m+2}-1}(-1)^n e^{-nu}
$$
be the one-sided Zeckendorf--Mellin kernel.  A
\emph{self-dual Zeckendorf completed packet} is a completed kernel in
the logarithmic variable $y=\log u$ of the form
$$
\Psi_m(y)
:=
e^{y/2}K_m(e^y)+e^{-y/2}K_m(e^{-y}),
$$
or a ledger-equivalent Poisson-completed kernel with the same symmetry
row
$$
\Psi_m(y)=\Psi_m(-y).
$$
Its completed Mellin readout is
$$
\mathcal X_m\left(\frac12+z\right)
:=
\int_{-\infty}^{\infty}\Psi_m(y)e^{zy}\,dy,
$$
whenever the displayed integral is replaced, if necessary, by the
corresponding regularized finite-packet ledger.  The point of the row is
not the particular displayed completion, but the presence of a
functional-equation ledger pairing the $u$ and $u^{-1}$ directions before
Lee--Yang stability is read out.
\end{definition}

\begin{definition}[Self-dual Zeckendorf Hermite--Biehler target]
\label{def:rh-self-dual-zeckendorf-hb-target}
The source row
$$
\mathsf{SelfDualZeckendorfHB}
$$
asserts that the completed transform of
\autoref{def:rh-self-dual-zeckendorf-completed-packet} belongs to a
Hermite--Biehler, de Branges, or Laguerre--P\'olya stability class strong
enough to force every zero of
$$
\mathcal X_m(s)
$$
in the zero-relevant island to satisfy
$$
\Re(s)=\frac12.
$$
Equivalently, after writing $s=1/2+z$, the completed transform has no
zero with $\Re(z)\ne0$ on the island covered by the certificate.
\end{definition}

\begin{theorem}[Self-dual Hermite--Biehler target gives fixed-half packet]
\label{thm:rh-self-dual-hb-gives-fixed-half-packet}
If a zero-relevant golden packet is captured by a self-dual Zeckendorf
completed packet satisfying $\mathsf{SelfDualZeckendorfHB}$, and if the
capture homotopy preserves winding on the boundary of the same island,
then all zeros captured on that island have real part $1/2$.
\end{theorem}

\begin{proof}
Write $s=1/2+z$.  The self-dual completed packet supplies a transform
$$
\mathcal X_m\left(\frac12+z\right)
=
\int_{-\infty}^{\infty}\Psi_m(y)e^{zy}\,dy
$$
with the symmetry ledger $\Psi_m(y)=\Psi_m(-y)$.  The
$\mathsf{SelfDualZeckendorfHB}$ row excludes zeros with $\Re(z)\ne0$ on
the zero-relevant island.  Thus any zero of the completed packet in that
island has $z=it$ and hence real part $1/2$.  Boundary winding
preservation transfers the same zero count from the captured golden
packet to the completed packet on the island, so the captured packet
zeros have the same fixed-half readout.
\end{proof}

\begin{principle}[One-sided Mellin is a source identity, not a stability row]
\label{prin:rh-one-sided-mellin-source-not-stability-row}
\autoref{thm:rh-zeckendorf-mellin-exact-projection} remains the exact
source identity connecting the eta golden truncation to the
Zeckendorf path partition function.  Its role is to expose the finite
path ledger and the cyclotomic unit-boundary fibre.  It does not license
unit-boundary stability after Mellin projection.  The stability row must
instead pass through the completed self-dual packet of
\autoref{def:rh-self-dual-zeckendorf-completed-packet} and the
Hermite--Biehler target of
\autoref{def:rh-self-dual-zeckendorf-hb-target}.
\end{principle}

\begin{definition}[Zeckendorf--G\"odel prime code]
\label{def:rh-zeckendorf-godel-prime-code}
Let
$$
N=\sum_{j} z_jF_j,
\qquad
z_j\in\{0,1\},
\qquad
z_jz_{j+1}=0,
$$
be a Zeckendorf address, and let $p_j$ be the $j$-th prime.  The
\emph{Zeckendorf--G\"odel prime code} of this address is
$$
g(N):=\prod_j p_j^{z_j}.
$$
Its prime height is
$$
H_P(N):=\log g(N)=\sum_j z_j\log p_j.
$$
This converts the no-adjacent Zeckendorf ledger into a prime-register
ledger while keeping the address bits finite and certificate-readable.
\end{definition}

\begin{definition}[Prime--Zeckendorf stable transport]
\label{def:rh-prime-zeckendorf-stable-transport}
A \emph{prime--Zeckendorf stable transport} is a family of finite packet
transports
$$
\mathcal T_\tau,
\qquad
0\le\tau\le1,
$$
from the additive Zeckendorf height
$$
H_Z(N)=N
$$
to a prime height such as $H_P(N)=\log g(N)$, or to a zeta-facing height
$H_\zeta(n)=\log n$, satisfying the following certificate rows on the
same zero island:
ledger uniqueness, the no-adjacent address constraint, unit phase
boundary, absence of an unledgered $\mathbb R_{>0}$ scale anomaly,
Rouch\'e winding invariance, and preservation of the active dominant
branch.  The row records precisely the structure missing from a bare
Mellin projection.
\end{definition}

\begin{definition}[Prime-compatible golden tower]
\label{def:rh-prime-compatible-golden-tower}
A \emph{prime-compatible golden tower} is a tower of finite packets
$\mathcal P_m$ such that each layer is generated by a
path-independent-set or Zeckendorf address ledger, each active address
bit is read as a prime-register channel, the finite packet satisfies a
multivariate Lee--Yang container ledger, refinement is cofinal in the
zero-relevant packet family, Rouch\'e capture is available for the
accepted zeta zero packet under discussion, and the transport contains
no unledgered real-positive scale gauge.
\end{definition}

\begin{theorem}[Prime-compatible tower gives fixed-half readout]
\label{thm:rh-prime-compatible-golden-tower-fixed-half-readout}
If every accepted non-trivial completed zeta zero is captured by a
prime-compatible golden tower, then every such zero has real part
$1/2$.
\end{theorem}

\begin{proof}
Let
$$
\rho=\frac12+r+it
$$
be an accepted non-trivial completed zeta zero, and choose a
prime-compatible golden tower capturing its zero packet.  The tower
supplies a Lee--Yang unit-boundary ledger for each active channel, hence
for each active height $h_a>0$ the centered readout satisfies
$$
\left|e^{-(\rho-1/2)h_a}\right|=1.
$$
Since
$$
e^{-(\rho-1/2)h_a}=e^{-rh_a}e^{-ith_a},
$$
the modulus condition is
$$
e^{-rh_a}=1.
$$
Because $h_a>0$, this gives $r=0$.  Therefore
$\Re(\rho)=1/2$.
\end{proof}

\begin{principle}[Bridge factorization through prime registers]
\label{prin:rh-bridge-factorization-prime-registers}
The zeta-facing golden bridge is not licensed by a single unrestricted
Mellin stability assertion.  Its certificate surface factors through
two source rows:
$$
\mathsf{PrimeZeckendorfLift}
\qquad
\mathsf{PrimePacketToZetaHomotopy}.
$$
The first row transports Fibonacci or path-independent-set
Lee--Yang packets into Zeckendorf--G\"odel prime packets.  The second
row transports those prime packets into the eta or zeta golden packet
without changing boundary winding and without introducing a radial
scale ledger.  This factorization is the prime-ledger refinement of
$$
\mathsf{ZetaGoldenLeeYangBridge}.
$$
\end{principle}

\begin{definition}[Multivariate Lee--Yang container]
\label{def:rh-multivariate-lee-yang-container}
Let $A_m$ be a finite active-mode set with heights $h_a>0$.  For a
readout $s$ set
$$
z_a(s):=e^{-(s-1/2)h_a},
\qquad
\mathbf z_m(s):=(z_a(s))_{a\in A_m}.
$$
A \emph{multivariate Lee--Yang container} is a finite polynomial
$$
\mathcal Z_m(\mathbf z)
$$
equipped with a Lee--Yang nonvanishing ledger
$$
\mathcal Z_m(\mathbf z)\neq0
\qquad
\text{if } |z_a|<1\text{ for all }a,
$$
and
$$
\mathcal Z_m(\mathbf z)\neq0
\qquad
\text{if } |z_a|>1\text{ for all }a.
$$
The container may be supplied by a finite ferromagnetic Ising packet, an
Asano-contraction packet, a matching-stability packet, or a
Rouch\'e-stable specialization of such a packet.
\end{definition}

\begin{theorem}[Multivariate container gives no-scale readout]
\label{thm:rh-multivariate-container-no-scale-readout}
Let a zero-relevant golden packet be Rouch\'e equivalent on a zero
island to a multivariate Lee--Yang container evaluated at
$\mathbf z_m(s)$.  If all active heights satisfy $h_a>0$, then every
zero captured on that island has real part $1/2$.
\end{theorem}

\begin{proof}
Let a captured zero have readout
$$
s=\frac12+r+it.
$$
If $r>0$, then
$$
|z_a(s)|=e^{-rh_a}<1
$$
for every $a\in A_m$.  If $r<0$, then
$$
|z_a(s)|=e^{-rh_a}>1
$$
for every $a\in A_m$.  Both cases are excluded by the multivariate
Lee--Yang nonvanishing ledger of
\autoref{def:rh-multivariate-lee-yang-container}.  Since the Rouch\'e
equivalence preserves the zero count on the same island, the captured
zero cannot have $r\ne0$.  Therefore $r=0$.
\end{proof}

$$
asserts that the zero-relevant zeta or eta golden packet is Rouch\'e
equivalent, on the same zero island, to this lifted path container after

\end{definition}

\begin{principle}[Path hard-core data is a skeleton]
\label{prin:rh-path-hard-core-skeleton}
The Zeckendorf path independent-set packet is the finite combinatorial
skeleton of the golden bridge, not the final Lee--Yang container.  Raw
hard-core specialization can produce Dirichlet packets such as
$1-p^{-s}$, whose zeros need not have fixed-half readout.  The
Joukowsky, Asano, Ising, or matching lift is the row that turns the path
skeleton into a multivariate unit-boundary container.  Thus the
zeta-facing target is not a direct homotopy to a single variable
Fibonacci polynomial, but a Rouch\'e-stable realization inside a
multivariate Lee--Yang container.
\end{principle}
