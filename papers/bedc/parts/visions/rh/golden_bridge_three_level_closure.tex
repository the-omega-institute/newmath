\chapter{Golden Bridge Three-Level Closure}
\label{ch:visions-rh-golden-bridge-three-level-closure}
\origin{ai}

\autoref{ch:visions-rh-golden-lee-yang-bridge-obligations} isolates
the analytic bridge rows that remain between weak eta capture and a
Lee--Yang stable golden packet.  The present chapter records the same
load in three strata: an abstract finite theorem, a fixed quartic finite
model, and the zeta-facing golden source package.  The strata are kept
separate so that a verified finite pattern is not read as a global zeta
theorem.

\begin{definition}[Three-level golden bridge package]
\label{def:rh-three-level-golden-bridge-package}
The \emph{three-level golden bridge package} consists of the following
rows.

The abstract row is the finite dominant-branch control theorem
\autoref{thm:rh-dominant-gap-gives-rouche-control}.  It requires a
displayed spectral expansion, a dominant pair, and the boundary
inequality
$$
|P_m^{\mathrm{sub}}(z)|<|P_m^{\mathrm{dom}}(z)|
\qquad
z\in\partial\Omega_m.
$$

The finite-model row is the quartic endpoint carrier of
\autoref{prin:rh-quartic-endpoint-finite-model}.  It shows that the
dominant-branch, Puiseux, two-branch interference, and Rouch\'e
isolation rows can be realized inside one finite spectral carrier.

The zeta-facing row is the source package of
\autoref{def:rh-golden-bridge-analytic-source-rows}, namely
$$
\mathsf{GoldenSpectralExpansion},
\qquad
\mathsf{GoldenSpectralGap},
\qquad
\mathsf{GoldenStableRoucheHomotopy}.
$$
This row is the only stratum that can feed the golden zeta bridge.
\end{definition}

\begin{theorem}[Abstract dominant control is finite]
\label{thm:rh-abstract-dominant-control-is-finite}
The abstract row of
\autoref{def:rh-three-level-golden-bridge-package} is a finite analytic
theorem: it transports zero count from $P_m^{\mathrm{dom}}$ to $P_m$
under the displayed strict boundary inequality.
\end{theorem}

\begin{proof}
This is exactly \autoref{thm:rh-dominant-gap-gives-rouche-control}.
The theorem uses only the decomposition
$$
P_m=P_m^{\mathrm{dom}}+P_m^{\mathrm{sub}}
$$
and the strict boundary estimate.  No zeta-specific coverage row,
Lee--Yang bridge row, or infinite tower row is consumed.
\end{proof}

\begin{principle}[Quartic endpoint as finite witness]
\label{prin:rh-quartic-endpoint-as-finite-witness}
The quartic endpoint carrier witnesses the shape of the argument, not
its zeta coverage.  It contains a fixed spectral polynomial, a
normalization ledger, a real projection ledger, a dominance ledger, and
an endpoint-zero ledger.  Its scope boundary refuses a growing-window
endpoint-zero theorem and refuses any zeta conclusion.  Thus it licenses
the methodology of finite dominance followed by local Rouch\'e
isolation, while leaving the zeta golden spectral expansion and gap as
separate source obligations.
\end{principle}

\begin{definition}[Zeta-facing strong golden bridge package]
\label{def:rh-zeta-facing-strong-golden-bridge-package}
The source row
$$
\mathsf{ZetaFacingStrongGoldenBridge}_{\zeta,\varphi}
$$
asserts that every accepted non-trivial completed zeta zero has a golden
certificate tower on which the following compatible rows are present:
golden weak eta capture, the Zeckendorf--Mellin exact projection of
\autoref{thm:rh-zeckendorf-mellin-exact-projection},
the analytic source rows of
\autoref{def:rh-golden-bridge-analytic-source-rows}, cofinal
active-height growth, and no unledgered radial scale gauge.  The package
does not assert the analytic rows by definition; it records the source
surface that must be supplied before the fixed-half readout can be
exported.
\end{definition}

\begin{theorem}[Strong golden bridge gives the zeta Lee--Yang bridge]
\label{thm:rh-strong-golden-bridge-gives-zeta-lee-yang-bridge}
If
$$
\mathsf{ZetaFacingStrongGoldenBridge}_{\zeta,\varphi}
$$
holds, then
$$
\mathsf{ZetaGoldenLeeYangBridge}
$$
holds.
\end{theorem}

\begin{proof}
Fix an accepted non-trivial completed zeta zero.  The source row of
\autoref{def:rh-zeta-facing-strong-golden-bridge-package} supplies a
golden certificate tower for that zero and supplies the analytic source
rows of \autoref{def:rh-golden-bridge-analytic-source-rows}.  These rows
provide the dominant branch selection, the stable Rouch\'e transport,
and the Lee--Yang stable zero-free regions on the same zero island.
Together with the weak eta capture and Zeckendorf address ledgers, they
are precisely the golden Lee--Yang bridge witnesses required by
\autoref{def:rh-zeta-golden-lee-yang-bridge-source-row}.
\end{proof}

\begin{theorem}[Strong golden bridge gives fixed-half readout]
\label{thm:rh-strong-golden-bridge-fixed-half-readout}
If
$$
\mathsf{ZetaFacingStrongGoldenBridge}_{\zeta,\varphi}
$$
holds, then every accepted non-trivial completed zeta zero has real part
$1/2$.
\end{theorem}

\begin{proof}
By
\autoref{thm:rh-strong-golden-bridge-gives-zeta-lee-yang-bridge}, the
source row $\mathsf{ZetaGoldenLeeYangBridge}$ holds.  Apply
\autoref{cor:rh-golden-lee-yang-bridge-fixed-half-readout}.
\end{proof}

\begin{principle}[Three-level proof boundary]
\label{prin:rh-three-level-proof-boundary}
The abstract row and the quartic endpoint row are evidence about proof
shape.  They do not cover zeta zeros.  The zeta-facing row must still
provide the analytic bridge package
$$
\mathsf{GoldenSpectralExpansion}
\quad
\mathsf{GoldenSpectralGap}
\quad
\mathsf{GoldenStableRoucheHomotopy},
$$
or, through the Mellin factorization of
\autoref{prin:rh-mellin-factorization-bridge-load}, the two lower rows
$$
\mathsf{ZeckendorfTransferDominatedSplitting}
\qquad
\mathsf{MellinLeeYangStabilityOperator}.
$$
Only after those rows are attached to the same golden zero island does
the no-scale argument apply to the readout
$$
s=\frac12+r+it.
$$
\end{principle}

\begin{definition}[Renormalized Zeckendorf endpoint transfer]
\label{def:rh-renormalized-zeckendorf-endpoint-transfer}
The endpoint scale for the Zeckendorf--Mellin fibre is not a fixed
$u>0$.  It is the normalized scale
$$
u=\frac{\alpha}{F_{m+2}}.
$$
At this scale the Mellin fibre variables become
$$
\widetilde x_{m,j}(\alpha)
:=
(-1)^{F_{j+1}}
\exp\left(-\alpha\frac{F_{j+1}}{F_{m+2}}\right).
$$
The associated non-autonomous path transfer matrices are
$$
T_{m,j}(\alpha)
:=
\begin{pmatrix}
1&\widetilde x_{m,j}(\alpha)\\
1&0
\end{pmatrix},
$$
with total endpoint transfer
$$
\mathcal T_m(\alpha)
:=
T_{m,m}(\alpha)T_{m,m-1}(\alpha)\cdots T_{m,1}(\alpha).
$$
The source row
$$
\mathsf{RenormalizedZeckendorfEndpointTransfer}
$$
asserts that the zero-relevant Mellin window is read through this
normalized transfer product, not through a fixed-$u$ fibre where the
large Fibonacci weights freeze the tail variables.
\end{definition}

\begin{principle}[Endpoint scaling is part of dominance]
\label{prin:rh-endpoint-scaling-part-of-dominance}
For fixed $u>0$, the variables
$$
x_j(u)=(-1)^{F_{j+1}}e^{-uF_{j+1}}
$$
decay rapidly with $j$, so the last variables of a long Zeckendorf
window no longer participate in a balanced Fibonacci competition.  The
renormalized scale of
\autoref{def:rh-renormalized-zeckendorf-endpoint-transfer} keeps the
terminal ratios
$$
\frac{F_{j+1}}{F_{m+2}}
$$
visible.  Hence the dominated-splitting row in
\autoref{def:rh-zeckendorf-transfer-dominated-splitting} must be proved
for the normalized endpoint transfer product $\mathcal T_m(\alpha)$.
\end{principle}

\begin{definition}[Mellin stability attack surfaces]
\label{def:rh-mellin-stability-attack-surfaces}
The source row
$$
\mathsf{MellinLeeYangStabilityOperator}
$$
may be attacked through any of the following stability-preservation
surfaces.

The total-positivity surface proves that the Mellin kernel produces a
P\'olya-frequency or Laguerre--P\'olya type transform compatible with
the Lee--Yang zero-free regions.

The Hermite--Biehler surface rewrites the Mellin-projected packet as a
function
$$
E(z)=A(z)-iB(z)
$$
whose half-plane or unit-circle stability is controlled by a
de Branges-type ledger.

The Rouch\'e-stable discretization surface approximates the Mellin
integral by finite sums
$$
\sum_{\ell}w_{\ell}(s)L_{m,u_{\ell}},
$$
with a stability-preserving choice of weights and a boundary error
ledger strong enough to pass to the integral without crossing zero.
\end{definition}

\begin{principle}[Mellin stability as no-demon finite form]
\label{prin:rh-mellin-stability-no-demon-finite-form}
The Mellin stability row is the finite packet analogue of the no-demon
constraint.  An off-half readout
$$
s=\frac12+r+it
$$
introduces the real scale factor
$$
e^{-rh_a}
$$
and therefore tilts any active-mode distribution by real-positive
weights.  If Mellin projection preserves the Lee--Yang inside and
outside zero-free regions, that tilt cannot be hidden as a phase or
prime-register gauge.  The only remaining legal readout is the unit
boundary readout, where
$$
|e^{-(s-1/2)h_a}|=1.
$$
\end{principle}
