\chapter{Zeta Virtual Machine Execution Trace}
\label{ch:visions-rh-zeta-virtual-machine-execution-trace}
\origin{ai}

\autoref{ch:visions-rh-zeta-self-similar-recovered-carrier} reads zeta
as a recovered carrier assembled from Dirichlet, Euler, continuation,
completion, mirror, and zero-ledger rows.
\autoref{ch:visions-rh-pzg-phase-readback-log-energy} supplies the PZG
length and phase readback behind that carrier, while
\autoref{ch:visions-rh-pzg-finite-window-self-reference-recurrence}
separates finite zero reads from completed zero objects.  The present
chapter records the program semantics joining these rows: the ontology
kernel is consumed as an instruction surface, zeta is the all-prime
execution trace of that surface, and zero search is a separate certified
readout process over the completed trace.

\begin{definition}[Ontology virtual machine row]
\label{def:rh-ontology-virtual-machine-row}
An ontology virtual machine row is a BEDC-facing instruction surface
$$
\mathsf{VM}_{\mathcal K}
:=
\left(
\begin{gathered}
\mathsf{Gen},\quad
\mathsf{Enc},\quad
\mathsf{Norm},\quad
\mathsf{Decode},\quad
\mathsf{Length},\quad
\mathsf{Phase},\\
\mathsf{Read},\quad
\mathsf{Complete},\quad
\mathsf{Reflect},\quad
\mathsf{Ledger},\quad
\mathsf{Refine},\quad
\mathsf{Certify}
\end{gathered}
\right).
$$
The instruction names are schematic abbreviations for already displayed
BEDC rows.  Generation extends histories; encoding assigns finite syntax
to PZG addresses; normalization gives canonical PZG form; decoding reads
the positive-integer source; length and phase read the log-energy and
unit-phase rows; finite reading consumes a displayed window; completion
and reflection build the analytic and mirror rows; ledgering records
residuals; refinement enlarges the visible window; certification turns a
finite read together with the required tail and boundary rows into an
auditable theorem packet.
\end{definition}

\begin{principle}[The kernel is executable semantics]
\label{prin:rh-kernel-executable-semantics}
The BEDC ontology kernel is not an external list of mathematical
objects.  In the RH-facing route it is used as the executable semantics
of \autoref{def:rh-ontology-virtual-machine-row}: each accepted object
is generated by instructions whose source histories, normal forms,
public readouts, and residual ledgers are visible.  This is the same
discipline as the dynamic ontology state row of
\autoref{def:ask-sig-spine-dynamic-ontology-state-row} and the
BEDC--PZG dynamic tuple of
\autoref{def:ask-sig-spine-bedc-pzg-dynamic-tuple}.
\end{principle}

\begin{definition}[PZG length heat trace]
\label{def:rh-pzg-length-heat-trace}
Let $\mathsf Z_{\mathsf{PZG}}$ be the finite-support PZG register
carrier, and let
$$
L(z):=\sum_{p,k}z_{p,k}F_k\log p
$$
be the log-energy length row.  The right-half-plane PZG length heat
trace is
$$
\mathsf{ZetaRead}(s)
:=
\sum_{z\in\mathsf Z_{\mathsf{PZG}}}e^{-sL(z)}
$$
on every displayed convergence window where the sum is licensed.
\end{definition}

\begin{theorem}[PZG heat trace is the Dirichlet zeta readout]
\label{thm:rh-pzg-heat-trace-dirichlet-zeta-readout}
On the right-half-plane convergence window, the PZG length heat trace is
the Dirichlet zeta readout:
$$
\mathsf{ZetaRead}(s)=\sum_{n\geq1}n^{-s}.
$$
\end{theorem}

\begin{proof}
By \autoref{thm:rh-prime-zeckendorf-goedel-uniqueness}, each positive
integer $n$ has a unique PZG register $Z_{\mathsf{PZG}}(n)$, and every
finite-support legal register is the register of its decoded positive
integer.  By \autoref{thm:rh-log-energy-reads-logarithmic-height},
$$
L(Z_{\mathsf{PZG}}(n))=\log n.
$$
Hence the summand indexed by the register of $n$ is
$$
e^{-sL(Z_{\mathsf{PZG}}(n))}
=
e^{-s\log n}
=
n^{-s}.
$$
Reindexing the register sum by the decoded positive integers gives the
displayed Dirichlet row.
\end{proof}

\begin{principle}[Zeta is a readout object, not a search program]
\label{prin:rh-zeta-readout-not-search-program}
The symbol $\zeta$ names the completed recovered-carrier readout whose
right-half-plane projection is
\autoref{def:rh-pzg-length-heat-trace}; it is not itself the procedure
that searches for zeros.  The search procedure is a separate packet over
the completed $\xi$ readout.  Thus the source separation is
$$
\mathsf{VM}_{\mathcal K}
\rightsquigarrow
\mathsf{ZetaRead}
\rightsquigarrow
\mathsf{XiRead}
\rightsquigarrow
\mathsf{ZeroSearch}_{\xi},
$$
where $\mathsf{XiRead}$ adds the analytic completion, archimedean factor,
reflection, pole, branch, and zero-ledger rows.
\end{principle}

\begin{definition}[Completed xi execution trace]
\label{def:rh-completed-xi-execution-trace}
A completed xi execution trace is a PZG analytic completion certificate
of \autoref{def:rh-pzg-analytic-completion-certificate} together with
the completed readout
$$
\mathsf{XiRead}(s)
:=
\frac12 s(s-1)\pi^{-s/2}\Gamma(s/2)\mathsf{ZetaRead}(s)
$$
and the mirror row
$$
s\mapsto 1-s.
$$
The trace is closed only to the extent that its continuation, completion,
branch, pole, zero, tail, and semantic self-reference ledgers are closed.
\end{definition}

\begin{definition}[Scalar readout kernel event]
\label{def:rh-scalar-readout-kernel-event}
Let $\mathcal A$ be a displayed ledger carrier and let
$$
R:\mathcal A\to\mathbb C
$$
be a scalar readout row.  A scalar readout kernel event is a ledger
state $a\in\mathcal A$ such that
$$
R(a)=0.
$$
This event asserts vanishing in the scalar readout.  It does not assert
$$
a=0
$$
unless a separate faithfulness row for $R$ is displayed.
\end{definition}

\begin{principle}[Zeros are readout-kernel events]
\label{prin:rh-zeros-readout-kernel-events}
A completed zeta zero is a scalar readout kernel event for the completed
xi execution trace.  The assertion
$$
\mathsf{XiRead}(\rho)=0
$$
means that the completed analytic readout sends the represented ledger
state at $\rho$ to zero.  It does not by itself say that the underlying
source ledger, hidden fibre, prime-register residue, branch residue, or
tail residue has vanished.  Those rows are closed only by faithfulness,
tail control, mirror cancellation, positivity, or explicit ledger
closure data.
\end{principle}

\begin{definition}[Prime exponent ledger group]
\label{def:rh-prime-exponent-ledger-group}
The prime exponent ledger group is the finite-support additive group
$$
G_{\mathbb P}:=\bigoplus_{p\in\PrimeUp}\mathbb Z[p].
$$
Its positive cone is the free commutative monoid
$$
G_{\mathbb P}^{+}:=\bigoplus_{p\in\PrimeUp}\mathbb N[p],
$$
which is the exponent ledger behind positive integers.  Passing from
$G_{\mathbb P}^{+}$ to $G_{\mathbb P}$ is the group-completion row that
allows inverse ledger entries.  The inverse of
$$
\delta=(\delta_p)_p
$$
is
$$
-\delta=(-\delta_p)_p.
$$
\end{definition}

\begin{definition}[Positive prime ledger]
\label{def:rh-positive-prime-ledger}
The positive prime ledger is the free commutative monoid
$$
M_{\mathbb P}:=\bigoplus_{p\in\PrimeUp}\mathbb N[p].
$$
Its element
$$
a=\sum_p a_p[p]
$$
reads the positive integer
$$
n(a)=\prod_p p^{a_p}.
$$
The PZG register is the canonical normal-form code of this ledger: each
coordinate $a_p$ is written in Zeckendorf normal form, producing the
finite-support array $(z_{p,k})_{p,k}$ of
\autoref{def:rh-prime-zeckendorf-goedel-register}.
\end{definition}

\begin{definition}[Tagged PZG zeta family]
\label{def:rh-tagged-pzg-zeta-family}
For a parameter $s$ on a licensed convergence or completion window, the
tagged PZG zeta family is the labelled coefficient family
$$
\mathbf Z_{\mathsf{tag}}(s)
:=
\bigl(a,e^{-sL(a)}\bigr)_{a\in M_{\mathbb P}}.
$$
The tag $a$ is part of the datum.  Before a scalar readout is applied,
two different prime ledgers remain different summands even when their
complex coefficients have cancelling directions.
\end{definition}

\begin{theorem}[Tagged zeta family is not the zero family]
\label{thm:rh-tagged-zeta-family-not-zero}
For every licensed parameter $s$, the tagged PZG zeta family is not the
zero family.
\end{theorem}

\begin{proof}
The empty prime ledger $0\in M_{\mathbb P}$ satisfies
$$
L(0)=0.
$$
Hence its tagged coefficient is
$$
e^{-sL(0)}=1.
$$
A zero tagged family would have zero coefficient at every tag.  The
coefficient at the empty ledger is not zero, so the tagged family is not
the zero family.
\end{proof}

\begin{definition}[Forgetful zeta scalar readout]
\label{def:rh-forgetful-zeta-scalar-readout}
The forgetful zeta scalar readout is the scalar row that drops the prime
ledger tag and sums the displayed coefficients:
$$
\Pi_{\zeta}\bigl(\mathbf Z_{\mathsf{tag}}(s)\bigr)
:=
\sum_{a\in M_{\mathbb P}}e^{-sL(a)}
$$
whenever the convergence, continuation, completion, and tail rows license
that sum.  On the right-half-plane convergence window, this row is
$\mathsf{ZetaRead}(s)$.
\end{definition}

\begin{theorem}[Zeta zeros are forgetful-kernel events]
\label{thm:rh-zeta-zeros-forgetful-kernel-events}
On any licensed zero-read window, a zeta zero is a kernel event for the
forgetful scalar readout, not a vanishing of the tagged PZG family.
\end{theorem}

\begin{proof}
By \autoref{def:rh-forgetful-zeta-scalar-readout}, the scalar zeta value
is obtained only after the prime ledger tags have been forgotten and the
coefficients have been summed in the completed scalar target.  Therefore
a zero read has the form
$$
\Pi_{\zeta}\bigl(\mathbf Z_{\mathsf{tag}}(\rho)\bigr)=0.
$$
This is a scalar readout kernel event in the sense of
\autoref{def:rh-scalar-readout-kernel-event}.  By
\autoref{thm:rh-tagged-zeta-family-not-zero}, the tagged PZG family
itself is not zero.
\end{proof}

\begin{principle}[Projection cancellation is not source-ledger closure]
\label{prin:rh-projection-cancellation-not-source-ledger-closure}
Complex-plane cancellation after $\Pi_{\zeta}$ may be a polygonal
cancellation among many tagged coefficients.  Such cancellation does not
identify their prime ledger tags and does not by itself pay source,
branch, tail, or scale residuals.  The fixed-half conclusion requires a
separate closure row, such as the single-ledger condition of
\autoref{prin:rh-single-ledger-zero-principle}, proving that the zero
packet closes inside its own zeta ledger rather than only after forgetting
labels.
\end{principle}

\begin{theorem}[Positive prime ledgers have no nonzero self-cancellation]
\label{thm:rh-positive-prime-ledgers-no-nonzero-self-cancellation}
Let $a,b\in M_{\mathbb P}$.  If
$$
a+b=0,
$$
then
$$
a=0
\qquad\text{and}\qquad
b=0.
$$
\end{theorem}

\begin{proof}
The equality is coordinatewise.  For each prime $p$,
$$
a_p+b_p=0
$$
with $a_p,b_p\in\mathbb N$.  Hence $a_p=b_p=0$ for every $p$, so both
finite-support ledgers are zero.
\end{proof}

\begin{theorem}[Prime-ledger local cancellation is inverse cancellation]
\label{thm:rh-prime-ledger-local-cancellation-inverse}
Let $\delta,\eta\in G_{\mathbb P}$.  If
$$
\delta+\eta=0,
$$
then
$$
\eta=-\delta.
$$
\end{theorem}

\begin{proof}
The group $G_{\mathbb P}$ is the direct sum of the integer coordinate
groups indexed by primes.  Therefore the equation
$$
\delta+\eta=0
$$
is equivalent to the coordinate equations
$$
\delta_p+\eta_p=0
\qquad(p\in\PrimeUp).
$$
Thus $\eta_p=-\delta_p$ for every prime coordinate, hence
$$
\eta=-\delta.
$$
\end{proof}

\begin{corollary}[Nonzero prime-ledger cancellation uses group completion]
\label{cor:rh-nonzero-prime-ledger-cancellation-uses-group-completion}
A nonzero positive prime ledger cannot be completely cancelled inside
$M_{\mathbb P}$.  Any complete cancellation of a nonzero prime-ledger
residual requires the group-completion row
$$
M_{\mathbb P}\to G_{\mathbb P}
$$
and the inverse element supplied by
\autoref{thm:rh-prime-ledger-local-cancellation-inverse}.
\end{corollary}

\begin{proof}
The no-cancellation statement inside $M_{\mathbb P}$ is
\autoref{thm:rh-positive-prime-ledgers-no-nonzero-self-cancellation}.
Inside $G_{\mathbb P}$, any element cancelling $\delta$ must be
$-\delta$ by
\autoref{thm:rh-prime-ledger-local-cancellation-inverse}.
\end{proof}

\begin{principle}[Readout cancellation and ledger cancellation differ]
\label{prin:rh-readout-cancellation-ledger-cancellation-differ}
Ledger cancellation is the structural equation
$$
\delta+(-\delta)=0
$$
inside a displayed ledger group such as $G_{\mathbb P}$.  Scalar readout
cancellation is the equation
$$
\sum_i R_s(a_i)=0
$$
inside the target field of a readout.  The latter says that the readout
does not see the total contribution at the parameter $s$; it does not
close the ledger unless the readout is faithful on the relevant
residual class or every hidden component is closed by a ledger row.
\end{principle}

\begin{definition}[Certified zero-search process]
\label{def:rh-certified-zero-search-process}
A certified zero-search process over
\autoref{def:rh-completed-xi-execution-trace} is a cofinal finite-window
process
$$
W_0\subset W_1\subset W_2\subset\cdots
$$
whose stage $j$ displays:
$$
\left(
W_j,\quad
\mathsf{FiniteApprox}_{\xi,W_j},\quad
\mathsf{Cand}_{W_j},\quad
\mathsf{TryCert}_{W_j},\quad
\mathsf{ZeroLedger}_{W_j}
\right).
$$
Here $\mathsf{FiniteApprox}_{\xi,W_j}$ is the finite analytic readout
visible in the window, $\mathsf{Cand}_{W_j}$ is the finite candidate
region list, $\mathsf{TryCert}_{W_j}$ is the attempted winding,
Rouch\'e, argument-principle, or tail-bound certificate, and
$\mathsf{ZeroLedger}_{W_j}$ records every unclosed boundary, branch,
source, precision, and tail residue.
\end{definition}

\begin{principle}[Infinite execution is finite reading plus tail control]
\label{prin:rh-infinite-execution-finite-read-tail-control}
The virtual machine does not process an infinite object by expanding all
of its coordinates in one step.  It processes infinity through cofinal
finite reads, compatibility under refinement, tail bounds, and ledger
closure.  A finite read may become a global conclusion only through a
certificate whose unobserved part is controlled by a displayed tail row.
Without that row, the output remains a finite read or a residual ledger
entry.
\end{principle}

\begin{definition}[Tail-budget ledger tower]
\label{def:rh-tail-budget-ledger-tower}
A tail-budget ledger tower for an infinite readout is a cofinal finite
window tower
$$
W_0\subset W_1\subset W_2\subset\cdots
$$
together with stage data
$$
(\mathsf L_n,\mathsf T_n,B_n)_{n\ge0}.
$$
Here $\mathsf L_n$ is the ledger visible at $W_n$,
$$
\mathsf T_n:=W_{\infty}\setminus W_n
$$
is the tail outside the displayed window, and $B_n$ is the budget row
for the unread tail.  A transition
$$
(\mathsf L_n,\mathsf T_n,B_n)
\longrightarrow
(\mathsf L_{n+1},\mathsf T_{n+1},B_{n+1})
$$
is legal only when it transports the visible ledger, reallocates or
reduces the tail budget, and records every residual not closed at the
later window.
\end{definition}

\begin{definition}[Budgeted infinite object]
\label{def:rh-budgeted-infinite-object}
A budgeted infinite object is a triple
$$
X_{\infty}:=
\left((X_B)_{B\in\mathcal B},\ (\rho_{B'B})_{B\subseteq B'},\
\mathcal L_{\infty}\right).
$$
Here $\mathcal B$ is a cofinal family of finite budgets, each $X_B$ is
the finite readout available under budget $B$, each
$$
\rho_{B'B}:X_{B'}\to X_B
$$
is the readback map from a larger budget to a smaller one, and
$\mathcal L_{\infty}$ is the ledger of residuals not closed by the
displayed finite budgets.
\end{definition}

\begin{principle}[Infinity is budgeted tail-ledger execution]
\label{prin:rh-infinity-budgeted-tail-ledger-execution}
An infinite analytic object is not consumed by a single total read.  In
the virtual-machine semantics of \autoref{def:rh-ontology-virtual-machine-row},
its processing consists of tail-budget transitions, compatibility under
refinement, and ledger closure.  The formal reading of
$$
\infty
$$
in such a route is therefore a budgeted tail ledger: what has not been
read at $W_n$ is not absent, but assigned to $\mathsf T_n$ with budget
$B_n$ and a required transport row into later stages.
\end{principle}

\begin{theorem}[Infinite processing factors through budgets and ledgers]
\label{thm:rh-infinite-processing-factors-through-budgets-ledgers}
Any auditable process that claims to handle a budgeted infinite object
must display finite budget reads, compatibility readbacks, and a future
ledger rule.
\end{theorem}

\begin{proof}
An auditable process uses finite input data, finite precision, finite
rules, and finite proof records at each stage.  Thus its immediate
output is a finite read $X_B$ for some displayed budget $B$.  If the
process is to speak about the infinite object rather than only that
stage, it must also show how a larger budget $B'$ reads back to $B$ by
$\rho_{B'B}$ and how every residual not resolved at $B$ is carried by
$\mathcal L_{\infty}$.  These are exactly the rows of
\autoref{def:rh-budgeted-infinite-object}.
\end{proof}

\begin{theorem}[Global closure requires budget exhaustion or ledger closure]
\label{thm:rh-global-closure-requires-budget-exhaustion-ledger-closure}
Let a tail-budget ledger tower be used to promote finite reads to a
global conclusion.  The promotion is closed only if every tail component
is either exhausted by the budgets or carried by a closed ledger row at
the limit.
\end{theorem}

\begin{proof}
At each stage, \autoref{def:rh-tail-budget-ledger-tower} assigns the
unread part to the tail $\mathsf T_n$ and its budget $B_n$.  A global
conclusion consumes all stages of the tower.  Any tail component that is
neither exhausted by the budget row nor closed by the limit ledger is
still an unread residual.  Such a residual is not part of the certified
finite read and has not been removed by the transition rules.  Therefore
the promoted conclusion is closed only when no such component remains.
\end{proof}

\begin{theorem}[Zero search factors through finite reads]
\label{thm:rh-zero-search-factors-through-finite-reads}
Every auditable stage of a certified zero-search process factors through
a finite zero read in the sense of
\autoref{def:rh-pzg-finite-zero-read}.
\end{theorem}

\begin{proof}
At stage $j$, the zero-search process displays a finite window, a finite
analytic approximation, a finite candidate-region list, a finite
certificate attempt, and a zero ledger.  These supply exactly the region,
precision or tolerance, finite count or zero-type readout, finite
protocol, and error-boundary-branch-source-tail ledger fields of
\autoref{def:rh-pzg-finite-zero-read}.  Equivalently, this is an instance
of the finite-read theorem
\autoref{thm:rh-pzg-zeroread-finite-read}.
\end{proof}

\begin{theorem}[Certified zeros require closure beyond search]
\label{thm:rh-certified-zeros-require-closure-beyond-search}
A certified zero-search process may enumerate and certify finite zero
regions.  It presents a completed zero object only when the corresponding
finite zero reads assemble into a compatible zero-certificate tower with
closed boundary, branch, source, precision, and tail ledgers.
\end{theorem}

\begin{proof}
By \autoref{thm:rh-zero-search-factors-through-finite-reads}, every
auditable search stage supplies finite zero reads.  A completed zero is
not identical to one such read; by
\autoref{prin:rh-pzg-completed-zeros-compatible-readout-closures}, it is
the closed readout selected by a compatible system of finite zero reads.
The required compatible system is the zero-certificate tower of
\autoref{def:rh-zero-certificate-tower}.  Any field not closed by that
tower remains in the zero ledger rather than becoming a completed zero
object.
\end{proof}

\begin{principle}[RH proof is not zero enumeration]
\label{prin:rh-proof-not-zero-enumeration}
Zero search concerns located zero regions of the completed trace.  The RH
route requires a stronger global section certificate: every admitted
non-trivial zero packet must factor through the fixed-half readout.  Thus
enumerating more zero certificates and proving the fixed-half section are
different program rows over the same completed xi trace.  The latter
requires global positivity, self-adjoint, no-scale, or no-escape rows of
the kind isolated by the ambient RH route, not merely a longer search
ledger.
\end{principle}

\begin{principle}[Program code is an internal object]
\label{prin:rh-zero-search-code-internal-object}
Any finite description of
$\mathsf{ZeroSearch}_{\xi}$ is itself a finite syntactic packet and hence
has a PZG address by
\autoref{thm:rh-pzg-finite-descriptions-unique-addresses}.  Therefore the
search process, its finite stages, and its ledgers are internal objects
of the same ontology virtual machine.  Internality does not make the
search complete: it only says that the code, run history, finite reads,
certificates, and residual ledgers are generated and auditable by the
same BEDC instruction surface.
\end{principle}
