\chapter{Off-Line Zero to Prime Sieve Contradiction}
\label{ch:visions-rh-offline-zero-prime-sieve-contradiction}
\origin{ai}

\autoref{ch:visions-rh-fixed-window-dominance-certificate} isolates the
spectral packet that an off-critical mirror pair must expose, while
\autoref{ch:visions-rh-explicit-formula-scale-residual-bridge} records
the compact-test bridge from spectral scale to a finite prime-power
row.  \autoref{ch:visions-rh-prime-generated-recursive-sieve} proves
that finite prime residuals are detected by local sieves generated by
primes.  The present chapter packages these interfaces into a single
BEDC proof object:
$$
\mathsf{OffLineZeroToPrimeSieveContradiction}^{\uparrow}.
$$
Its role is not to assert the Riemann hypothesis as a closed theorem.
It identifies the exact certificate chain under which an off-line zero
cannot be admitted as a closed BEDC/PFZ zero packet.

\section{Contradiction Packet}
\label{sec:rh-offline-zero-prime-sieve-contradiction-packet}

\begin{definition}[OffLineZeroToPrimeSieveContradiction]
\label{def:rh-offline-zero-prime-sieve-contradiction}
The packet
$$
\mathsf{OffLineZeroToPrimeSieveContradiction}
$$
consists of the following rows:
$$
\begin{gathered}
\mathsf{AnalyticContinuationPreservesOntTriple},\\
\mathsf{AnalyticContinuationPreservesPrimeSieveLedger},\\
\mathsf{ZetaFixedWindowDominance},\\
\mathsf{PrimeProjectionCert},\\
\mathsf{PrimeGeneratedRecursiveSieve},\\
\mathsf{LedgerClosureRefusal}.
\end{gathered}
$$
The first two rows license the continued zeta object as a
source-preserving, unit-aware, ledger-visible object.  The fixed-window
row supplies the off-line spectral packet.  The prime-projection row
turns that spectral packet into a nonzero finite prime-power residual.
The recursive sieve row detects every such finite residual by a local
prime-generated classifier.  The ledger-refusal row says that a closed
zero packet cannot carry either an unrecorded detected residual or an
append row that has not closed.
\end{definition}

\begin{definition}[PrimeProjectionCert]
\label{def:rh-prime-projection-cert}
For an off-critical mirror pair
$$
\rho_+=\frac12+r+i\gamma,
\qquad
\rho_-=\frac12-r+i\gamma,
\qquad
r\ne0,
$$
the certificate
$$
\mathsf{PrimeProjectionCert}(\rho_+,\rho_-)
$$
asserts that the fixed-window radial readout supplied by
$$
\mathsf{ZetaFixedWindowDominance}
$$
has an admissible compact explicit-formula test row $g$ for which the
explicit-formula ledger
$$
\mathcal Z(g)=\mathcal P(g)+\mathcal A(g)
$$
produces a nonzero finite prime-power residual
$$
R_S(g)=\sum_{q\in S}c_q(g)[q]\ne0.
$$
Equivalently, the archimedean, trivial-zero, and pole correction rows
are displayed and do not absorb the fixed radial readout.
\end{definition}

\begin{definition}[LedgerClosureRefusal]
\label{def:rh-ledger-closure-refusal}
The row
$$
\mathsf{LedgerClosureRefusal}
$$
asserts that a legal closed BEDC/PFZ zero packet cannot satisfy either
of the following conditions:
$$
\begin{gathered}
\text{a detected prime-sieve residual is absent from the ledger},\\
\text{a detected prime-sieve residual is present as an unclosed append row}.
\end{gathered}
$$
Thus a detected residual is fatal to closed zero-packet admission: if it
is not ledgered, the ledger policy fails; if it is ledgered, the packet
has not closed.
\end{definition}

\section{From Spectral Packet to Prime Residual}
\label{sec:rh-offline-zero-prime-sieve-spectral-to-prime}

\begin{theorem}[Fixed packet gives a finite prime residual]
\label{thm:rh-fixed-packet-gives-finite-prime-residual}
Assume $\mathsf{ZetaFixedWindowDominance}$ and
$$
\mathsf{PrimeProjectionCert}(\rho_+,\rho_-)
$$
for an off-critical mirror pair.  Then the pair supplies a nonzero
finite prime-power residual
$$
R_S(g)=\sum_{q\in S}c_q(g)[q].
$$
\end{theorem}

\begin{proof}
The fixed-window dominance row unpacks through
\autoref{def:rh-zeta-fixed-window-dominance}.  For the displayed mirror
pair it supplies a finite spectral window, a mirror-odd differential
classifier, a compact observation interval, and a positive dominance
margin.  By
\autoref{thm:rh-dominance-certificate-audited-radial-readout}, the
corresponding classifier readout is nonzero and contains a nonunit
radial row.

The prime-projection certificate supplies an admissible compact
explicit-formula test row whose correction rows are ledgered and do not
absorb that radial readout.  Therefore the explicit-formula identity
$$
\mathcal Z(g)=\mathcal P(g)+\mathcal A(g)
$$
has a nonzero correction-subtracted prime side.  This is the
prime-effective realization of
\autoref{def:rh-prime-effective-dominance-realization}.  Applying
\autoref{thm:rh-prime-effective-dominance-finite-prime-residual} gives
the displayed nonzero finite residual.
\end{proof}

\begin{corollary}[Explicit-formula isolation is a projection certificate]
\label{cor:rh-explicit-formula-isolation-prime-projection-cert}
The hypotheses of
\autoref{thm:rh-explicit-formula-scale-residual-from-arch-null-isolation}
provide an instance of
$$
\mathsf{PrimeProjectionCert}(\rho_+,\rho_-)
$$
for the corresponding off-critical mirror pair.
\end{corollary}

\begin{proof}
\autoref{thm:rh-explicit-formula-scale-residual-from-arch-null-isolation}
starts from the abstract explicit-formula ledger and
$\mathsf{ArchNullZeroIsolation}$.  It produces a nonzero finite
prime-power residual from an off-critical zero.  This is precisely the
prime-projection certificate: the compact test row is admissible, the
correction row is null in the stated ledger, and the prime side is
finite and nonzero.
\end{proof}

\section{Prime-Generated Detection}
\label{sec:rh-offline-zero-prime-sieve-detection}

\begin{theorem}[Finite residual enters a prime-generated sieve]
\label{thm:rh-finite-residual-enters-prime-generated-sieve}
Assume $\mathsf{PrimeGeneratedRecursiveSieve}$.  If
$$
R_S=\sum_{j=1}^{N}c_j[q_j]\ne0
$$
is a finite prime-power residual with distinct support points, then
there are a prime $p$ and a character
$$
\chi:\mathbb F_p^\times\to S^1
$$
such that
$$
\sum_{j=1}^{N}c_j\chi(q_j\bmod p)\ne0.
$$
The readout is a legal finite row of the prime-generated recursive
sieve ledger.
\end{theorem}

\begin{proof}
The existence of the detecting prime and character is exactly
\autoref{cor:rh-prime-generated-recursive-completeness}.  The
legality of the row follows from
\autoref{def:rh-prime-generated-recursive-sieve}: a local residue or
character readout generated by a prime is one of the permitted finite
classifier rows, and the recursive sieve step records the selected
prime, event or character, readout, source residual, and reason.
\end{proof}

\begin{corollary}[Signed residuals also enter the recursive sieve]
\label{cor:rh-signed-residuals-enter-recursive-sieve}
Under $\mathsf{PrimeGeneratedRecursiveSieve}$, every finite prime
negative residual of
\autoref{def:rh-finite-prime-negative-residual} has a prime-generated
classifier readout.
\end{corollary}

\begin{proof}
By
\autoref{thm:rh-prime-negative-residual-gives-classifier-obstruction},
a finite prime negative residual supplies a nonzero finite
prime-classifier residual.  Apply
\autoref{thm:rh-finite-residual-enters-prime-generated-sieve}.
\end{proof}

\section{Contradiction Route}
\label{sec:rh-offline-zero-prime-sieve-contradiction-route}

\begin{theorem}[Off-line zero to prime-sieve contradiction]
\label{thm:rh-offline-zero-to-prime-sieve-contradiction}
Assume
$$
\mathsf{OffLineZeroToPrimeSieveContradiction}.
$$
If every off-critical mirror pair in the continued zeta ledger satisfies
$$
\mathsf{PrimeProjectionCert}(\rho_+,\rho_-),
$$
then no off-critical zero is admitted as a legal closed BEDC/PFZ zero
packet.
\end{theorem}

\begin{proof}
Suppose that a legal closed packet admits an off-critical zero.  Its
mirror pair has the form
$$
\rho_+=\frac12+r+i\gamma,
\qquad
\rho_-=\frac12-r+i\gamma,
\qquad
r\ne0.
$$
The continuation rows in
\autoref{def:rh-offline-zero-prime-sieve-contradiction} keep this pair
inside the certified zeta ledger and prevent analytic continuation
from hiding a radial residual, as expressed by
\autoref{thm:rh-ontological-continuation-preserves-prime-sieve-ledger}
and \autoref{thm:rh-continuation-cannot-hide-radial-residual}.

By $\mathsf{ZetaFixedWindowDominance}$ and the prime-projection
certificate for the mirror pair,
\autoref{thm:rh-fixed-packet-gives-finite-prime-residual} gives a
nonzero finite prime-power residual $R_S$.  By
\autoref{thm:rh-finite-residual-enters-prime-generated-sieve}, a local
prime-generated classifier detects $R_S$ with nonzero readout.

That detected readout has only two possible ledger statuses.  If it is
not recorded, the prime-sieve ledger policy fails.  If it is recorded,
the zero packet carries an append residual row and is not closed.  Both
alternatives contradict $\mathsf{LedgerClosureRefusal}$.  Therefore a
legal closed BEDC/PFZ zero packet cannot admit an off-critical zero.
\end{proof}

\begin{corollary}[Conditional fixed-half readout]
\label{cor:rh-offline-zero-prime-sieve-conditional-fixed-half-readout}
Under the hypotheses of
\autoref{thm:rh-offline-zero-to-prime-sieve-contradiction}, every
admitted closed nontrivial zero has fixed-half readout:
$$
\operatorname{Re}(\rho)=\frac12.
$$
\end{corollary}

\begin{proof}
The theorem excludes every closed admitted off-critical zero.  Hence an
admitted closed nontrivial zero must have no radial displacement from
the half line.  Equivalently, by
\autoref{thm:rh-li-coordinate-fixed-half-circle}, its Li coordinate lies
on the unit circle precisely when
$$
\operatorname{Re}(\rho)=\frac12.
$$
\end{proof}

\section{Audit Boundary}
\label{sec:rh-offline-zero-prime-sieve-audit-boundary}

\begin{principle}[Finite sieve side is closed under the packet]
\label{prin:rh-offline-zero-prime-sieve-finite-side-closed}
Inside
$$
\mathsf{OffLineZeroToPrimeSieveContradiction},
$$
the finite arithmetic part is not the source of the obstruction.  Once
a nonzero finite prime-power residual is displayed, the
prime-generated recursive sieve detects it by
\autoref{cor:rh-prime-generated-recursive-completeness}; the detected
row is then governed by the ledger dichotomy of
\autoref{def:rh-ledger-closure-refusal}.
\end{principle}

\begin{principle}[Analytic certificate boundary]
\label{prin:rh-offline-zero-prime-sieve-analytic-boundary}
The analytic burden of the packet is concentrated in the two certificate
rows
$$
\mathsf{ZetaFixedWindowDominance}
\qquad\text{and}\qquad
\mathsf{PrimeProjectionCert}.
$$
The first row is the fixed spectral-window dominance statement of
\autoref{def:rh-zeta-fixed-window-dominance}.  The second row is the
compact explicit-formula projection statement of
\autoref{def:rh-prime-projection-cert}.  Once those rows are supplied,
the contradiction route is finite, prime-generated, and ledgered.
\end{principle}
