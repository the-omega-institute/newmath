\chapter{Triadic Pythagorean Obstruction}
\label{ch:visions-rh-triadic-pythagorean-obstruction}
\origin{ai}

\autoref{def:rh-certificate-tower-triadic-invariant} records the three
certificate rows $\mathsf P$, $\mathsf U$, and $\mathsf T$ level by
level.  \autoref{ch:visions-rh-li-weil-herglotz-target} records the
unit-positive analytic face, while
\autoref{ch:visions-rh-thermodynamic-no-scale-route} records the
no-scale energy face.  This chapter packages these rows as one
obstruction rather than as independent routes.  The obstruction says that
an off-fixed-half readout must appear simultaneously as an illegal
prime-ledger scale, a finite-time witness, and a violation of the
positive spectral row.

\begin{definition}[Triadic obstruction coordinates]
\label{def:rh-triadic-obstruction-coordinates}
A triadic obstruction coordinate system for an active zero packet
consists of three source rows:
$$
\mathsf P,\qquad \mathsf U,\qquad \mathsf T.
$$
The row $\mathsf P$ is the prime-ledger row: hidden coordinates may live
in compact phase and prime-register fibres, as in
\autoref{def:rh-phase-channel-hidden-prime-fibre}, but no independent
real-positive scale ledger is accepted as a gauge.  The row $\mathsf U$
is the unit-positive row: it may be presented as Lee--Yang boundary
stability, global Weil positivity, Li CND, Herglotz positivity, or the
Stieltjes source row of
\autoref{def:rh-xi-stieltjes-log-derivative-row}.  The row $\mathsf T$
is the finite-time row: any off-fixed-half defect must project to a
finite Li--Weil witness of the kind isolated in
\autoref{def:rh-li-weil-finite-negative-witness}, rather than to an
open-ended search through certificate levels.
\end{definition}

\begin{definition}[Triadic closure relation]
\label{def:rh-triadic-closure-relation}
For a zero-relevant active packet with heights $h_a>0$, write the
centered readout as
$$
s=\frac12+r+it.
$$
The active channel at mode $a$ is
$$
U_a(s):=e^{-(s-1/2)h_a}
=
e^{-r h_a}e^{-it h_a}.
$$
The triadic closure relation
$$
\mathcal C(\mathsf P,\mathsf U,\mathsf T;s)=0
$$
means that the same active packet satisfies the prime-ledger gauge row,
the unit-positive boundary row, and the finite-time witness row without
introducing an unledgered $\mathbb R_{>0}$ coordinate.
\end{definition}

\begin{definition}[Triadic scale defect energy]
\label{def:rh-triadic-scale-defect-energy}
For a finite active packet $A_m$, define the scale defect at mode $a$ by
$$
d_a(s):=\log|U_a(s)|=-r h_a.
$$
The triadic defect vector and its quadratic energy are
$$
\Omega_m(s):=(d_a(s))_{a\in A_m},
$$
and
$$
\mathcal E_m(s)
:=
\sum_{a\in A_m}d_a(s)^2.
$$
With
$$
W_m:=\sum_{a\in A_m}h_a^2,
$$
one has the exact finite identity
$$
\mathcal E_m(s)=r^2W_m.
$$
\end{definition}

\begin{principle}[Pythagorean reading of the defect]
\label{prin:rh-pythagorean-reading-of-defect}
The equation of \autoref{def:rh-triadic-scale-defect-energy} is the
quadratic obstruction behind the triadic route.  Linear phase
cancellation may make a projected sum vanish, and reflection may exchange
$r$ with $-r$, but the squared defect
$$
r^2W_m
$$
has no sign that can be cancelled by phase or by mirror exchange.  Thus
an off-fixed-half readout can be hidden only by supplying an external
scale ledger, by losing the finite witness row, or by breaking the
unit-positive row.
\end{principle}

\begin{theorem}[Triadic Pythagorean obstruction]
\label{thm:rh-triadic-pythagorean-obstruction}
Let a zero-relevant certificate tower have limit readout
$$
s_{\infty}=\frac12+r+it.
$$
Assume that it satisfies the triadic closure relation of
\autoref{def:rh-triadic-closure-relation}, has cofinal active-weight
growth
$$
W_m\to\infty,
$$
has defect extraction
$$
\mathcal E_m(s_{\infty})=r^2W_m+o(W_m),
$$
and carries a unit-positive stability bound
$$
\mathcal E_m(s_{\infty})\le B
$$
with $B$ independent of $m$.  Then
$$
r=0.
$$
\end{theorem}

\begin{proof}
Assume $r\ne0$.  Then $r^2>0$.  From the defect extraction row there is
a cofinal tail on which
$$
|o(W_m)|\le\frac{r^2}{2}W_m.
$$
Hence
$$
\mathcal E_m(s_{\infty})
\ge
\frac{r^2}{2}W_m
$$
on that tail.  Since $W_m\to\infty$, the energies
$\mathcal E_m(s_{\infty})$ are unbounded along the tail.  This
contradicts the stability bound $\mathcal E_m(s_{\infty})\le B$.
Therefore $r=0$.
\end{proof}

\begin{corollary}[Triadic obstruction gives fixed-half readout]
\label{cor:rh-triadic-obstruction-fixed-half-readout}
If every accepted non-trivial completed zeta zero is represented by a
zero-relevant certificate tower satisfying the hypotheses of
\autoref{thm:rh-triadic-pythagorean-obstruction}, then every such
accepted zero has fixed-half readout.
\end{corollary}

\begin{proof}
Let $u$ be an accepted non-trivial completed zeta zero and let
$$
s_{\infty}=\frac12+r+it
$$
be its represented readout.  The tower hypotheses give $r=0$ by
\autoref{thm:rh-triadic-pythagorean-obstruction}.  Thus
$\operatorname{Re}(s_{\infty})=1/2$.
\end{proof}

\begin{definition}[Triadic positive-spectrum closure]
\label{def:rh-triadic-positive-spectrum-closure}
The source row
$$
\mathsf{TriadicPositiveSpectrumClosure}_{\zeta}
$$
asserts that the unit row $\mathsf U$ of
\autoref{def:rh-triadic-obstruction-coordinates} is carried by one of
the positive spectral presentations already named in the RH route:
$$
\mathsf{XiStieltjesLogDerivative},
\qquad
\mathsf{XiLogDerivativeHerglotz},
\qquad
\mathsf{LiSecondDifferencePositiveDefinite},
\qquad
\mathsf{GlobalLiWeilPositivity}.
$$
The row is source-side: it supplies positivity before the zero divisor is
read, and it may not be manufactured by first assuming fixed-half
readout and then placing a posterior spectral measure.
\end{definition}

\begin{theorem}[Positive spectrum supplies the unit side]
\label{thm:rh-positive-spectrum-supplies-unit-side}
If $\mathsf{TriadicPositiveSpectrumClosure}_{\zeta}$ is realized through
$\mathsf{XiStieltjesLogDerivative}$, then the unit row $\mathsf U$ of
the triadic obstruction is available.
\end{theorem}

\begin{proof}
\autoref{thm:rh-xi-stieltjes-gives-xi-herglotz} turns
$\mathsf{XiStieltjesLogDerivative}$ into
$\mathsf{XiLogDerivativeHerglotz}$.  The latter is the positive-real
unit row of
\autoref{def:rh-xi-log-derivative-herglotz-row}; equivalently, by the
Li--Weil comparison of
\autoref{thm:rh-li-beta-readout-xi-log-derivative}, it is the analytic
face of the Li beta positivity row.  Hence the unit-positive component
of the triadic closure relation is supplied.
\end{proof}

\begin{principle}[Three rows as one obstruction]
\label{prin:rh-three-rows-one-obstruction}
The triadic route should be consumed as a single obstruction.  The
prime-ledger row says that $e^{-rh_a}$ cannot be hidden in the compact
phase or prime-register fibre.  The finite-time row says that any such
scale defect must have a finite Li--Weil witness.  The unit-positive row
says that the relevant witness functional is nonnegative through
Weil, Li, Herglotz, or Stieltjes positivity.  Therefore a nonzero real
offset cannot be discharged by moving among the three rows; it is the
same defect seen in three projections.
\end{principle}

\begin{definition}[Analytic triadic closure]
\label{def:rh-analytic-triadic-closure}
The analytic specialization of the triadic obstruction has three rows:
$$
\mathsf T,\qquad \mathsf W,\qquad \mathsf N.
$$
The time row $\mathsf T$ is $\mathsf{GlobalLiWeilPositivity}$, or any
source-side stronger row such as Li CND or a Stieltjes log-derivative
representation that implies it.  The energy row $\mathsf W$ is
$\mathsf{GlobalWeilPositivity}$ for the Weil quadratic form of
\autoref{def:rh-weil-quadratic-form-row}.  The container row $\mathsf N$
is $\mathsf{NymanBeurlingExhaustion}$ of
\autoref{def:rh-nyman-exhaustion-row}.  The analytic triadic closure row
$$
\mathsf{AnalyticTriadicClosure}_{\zeta}
$$
asserts that these three rows are supplied by one source carrier rather
than by three posterior arguments over an already-read zero divisor.
\end{definition}

\begin{theorem}[Li coordinate Pythagorean decomposition]
\label{thm:rh-li-coordinate-pythagorean-decomposition}
Let
$$
z=e^{\ell+i\theta}
$$
with $\ell,\theta\in\mathbb R$.  For every $n\ge1$,
$$
|1-z^n|^2
=
(e^{n\ell}-1)^2
+
2e^{n\ell}(1-\cos(n\theta)).
$$
\end{theorem}

\begin{proof}
Since
$$
z^n=e^{n\ell}(\cos(n\theta)+i\sin(n\theta)),
$$
one has
$$
|1-z^n|^2
=
(1-e^{n\ell}\cos(n\theta))^2
+
(e^{n\ell}\sin(n\theta))^2.
$$
Expanding and using
$$
\cos^2(n\theta)+\sin^2(n\theta)=1
$$
gives
$$
|1-z^n|^2
=
1-2e^{n\ell}\cos(n\theta)+e^{2n\ell}.
$$
The right-hand side also equals
$$
(e^{n\ell}-1)^2
+
2e^{n\ell}(1-\cos(n\theta)),
$$
which proves the identity.
\end{proof}

\begin{principle}[Radial Li defect is the time projection]
\label{prin:rh-radial-li-defect-time-projection}
For a zero row $\rho$, the Li coordinate of
\autoref{def:rh-li-time-coordinate} has polar form
$$
z(\rho)=e^{\ell+i\theta}.
$$
By \autoref{thm:rh-li-coordinate-fixed-half-circle}, the fixed-half
readout is exactly $\ell=0$.  If $\ell\ne0$, the radial factor
$e^{n\ell}$ is amplified by Li time.  The real reflected-pair
contribution is the form already isolated in
\autoref{prin:rh-radial-li-time-amplification}: along suitable finite
times, the radial term produces a finite Li--Weil negative witness
unless the source carrier supplies a positive row that forbids it.
\end{principle}

\begin{definition}[Weil negative direction projection]
\label{def:rh-weil-negative-direction-projection}
Let $\rho_+$ and $\rho_-$ be a reflected off-fixed-half pair in the
completed zero ledger.  A Weil negative direction projection is a
test-row packet $f$ whose Weil transforms separate the reflected pair
with opposite signs, while the remaining displayed zero rows are
controlled by the interpolation and tail ledgers of the Weil packet.  It
is the zero-side obstruction described by
\autoref{prin:rh-offline-mirror-pairs-create-negative-weil-directions}: an
off-fixed-half pair creates a negative direction unless
$\mathsf{GlobalWeilPositivity}$ forbids such a packet.
\end{definition}

\begin{definition}[Nyman residual projection]
\label{def:rh-nyman-residual-projection}
Let $s_0$ be a zero with $\operatorname{Re}(s_0)>1/2$.  The Nyman
residual projection is the continuous Mellin functional
$$
L_{s_0}(f):=\int_0^1 f(x)x^{s_0-1}\,dx
$$
of \autoref{lem:rh-off-half-zero-nyman-separator}.  It annihilates every
Nyman generator $\rho_{1/n}$ and satisfies
$$
L_{s_0}(\mathbf 1)=\frac{1}{s_0}\ne0.
$$
Thus it separates the constant row from the generated Nyman closure
whenever the Nyman exhaustion row is absent.
\end{definition}

\begin{theorem}[Nyman separator gives a quantitative residual]
\label{thm:rh-nyman-separator-quantitative-residual}
Assume the Nyman Mellin bridge row.  If $s_0=\sigma+i\tau$ is a
nontrivial zeta zero with $\sigma>1/2$, then
$$
\operatorname{dist}_{L^2}
\left(
\mathbf 1,
\overline{\operatorname{span}\{\rho_{1/n}:n\ge1\}}^{\,L^2(0,1)}
\right)
\ge
\frac{\sqrt{2\sigma-1}}{|s_0|}.
$$
\end{theorem}

\begin{proof}
By \autoref{lem:rh-off-half-zero-nyman-separator}, the functional
$L_{s_0}$ vanishes on every generator $\rho_{1/n}$ and
$L_{s_0}(\mathbf 1)=1/s_0$.  Its representing vector is
$x^{\overline{s_0}-1}$, hence
$$
\|L_{s_0}\|
=
\left(\int_0^1 x^{2\sigma-2}\,dx\right)^{1/2}
=
\frac{1}{\sqrt{2\sigma-1}}.
$$
For any $v$ in the closed Nyman span,
$$
|L_{s_0}(\mathbf 1-v)|
=
|L_{s_0}(\mathbf 1)|
=
\frac{1}{|s_0|}.
$$
Therefore
$$
\|\mathbf 1-v\|_2
\ge
\frac{|L_{s_0}(\mathbf 1-v)|}{\|L_{s_0}\|}
=
\frac{\sqrt{2\sigma-1}}{|s_0|}.
$$
Taking the infimum over $v$ gives the claim.
\end{proof}

\begin{principle}[Analytic triangle as one scale defect]
\label{prin:rh-analytic-triangle-one-scale-defect}
The rows $\mathsf T$, $\mathsf W$, and $\mathsf N$ are not three
unrelated proof attempts.  They are three projections of the same
nonunit Li coordinate.  In the time projection, the radial parameter
$\ell=\log|z(\rho)|$ is amplified by $z(\rho)^n$ and appears as a
finite Li--Weil negative witness.  In the energy projection, the
reflected off-fixed-half pair supplies a negative Weil direction unless
global Weil positivity forbids it.  In the container projection, the
Mellin functional $L_{s_0}$ separates $\mathbf 1$ from the Nyman closure
unless Nyman exhaustion forbids the residual.  Thus a nonzero scale
coordinate is not a local anomaly that can be moved from one row to
another; it is the same obstruction read by time, energy, and container
coordinates.
\end{principle}

\begin{theorem}[Analytic triadic closure gives fixed-half readout]
\label{thm:rh-analytic-triadic-closure-fixed-half-readout}
Assume $\mathsf{AnalyticTriadicClosure}_{\zeta}$ and the usual
reflection consumption rows for the completed zeta zero ledger.  Then
every accepted non-trivial completed zeta zero has fixed-half readout.
\end{theorem}

\begin{proof}
The row $\mathsf{AnalyticTriadicClosure}_{\zeta}$ includes
$\mathsf{NymanBeurlingExhaustion}$.  By
\autoref{thm:rh-nyman-exhaustion-fixed-half-readout}, the Nyman row and
the reflection consumption row exclude zero readouts off the fixed-half
surface.  The Li and Weil rows give the same exclusion through their
finite-witness and energy forms; they are recorded here to identify that
the excluded residual is the same scale defect as in
\autoref{prin:rh-analytic-triangle-one-scale-defect}.  Hence every
accepted non-trivial completed zero has fixed-half readout.
\end{proof}

\begin{principle}[Ontic closure target for the analytic triangle]
\label{prin:rh-ontic-closure-target-analytic-triangle}
The remaining source-side target is not to prove the Li, Weil, and
Nyman rows by unrelated searches.  It is to show that BEDC triadic
closure supplies them as one source carrier:
$$
\mathsf{OnticTriadicClosure}_{\zeta}
\Longrightarrow
\mathsf{GlobalLiWeilPositivity}
\wedge
\mathsf{GlobalWeilPositivity}
\wedge
\mathsf{NymanBeurlingExhaustion}.
$$
Once this carrier is present, an off-fixed-half readout has no place to
hide: Li time detects it finitely, Weil energy detects it as a negative
direction, and the Nyman container detects it as a positive residual
distance.
\end{principle}
