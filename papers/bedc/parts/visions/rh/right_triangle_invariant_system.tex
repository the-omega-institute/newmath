\chapter{Right-Triangle Invariant System}
\label{ch:visions-rh-right-triangle-invariant-system}
\origin{ai}

\autoref{ch:visions-rh-triadic-pythagorean-obstruction} isolates the
scale component of a centered active packet as a squared residual, and
\autoref{ch:visions-rh-xi-log-potential-no-scale} records the global
log-potential field that should forbid such residuals at the source.
This chapter names the finite geometric system behind those rows.  Its
content is not an additional analytic object; it is the invariant
discipline that makes observation order, orthogonal residuals, and
positive energy one constraint.

\begin{definition}[Right-triangle observation system]
\label{def:rh-right-triangle-observation-system}
Let $A_m$ be a finite active packet with heights $h_a>0$.  For a
centered readout
$$
s=\frac12+r+it,
$$
define the logarithmic observation row
$$
L_m(s):=\bigl((-r h_a)_{a\in A_m},\,(-t h_a)_{a\in A_m}\bigr)
\in \mathbb R^{A_m}\oplus\mathbb R^{A_m}.
$$
The first factor is the scale leg, and the second factor is the phase
leg.  The weight of the packet is
$$
W_m:=\sum_{a\in A_m}h_a^2.
$$
A \emph{right-triangle observation system} is such a packet together
with the source ledger that records $A_m$, the heights $h_a$, the
orthogonal sum
$$
\mathbb R^{A_m}\oplus\mathbb R^{A_m},
$$
and the assertion that hidden coordinates may refine compact phase or
prime-register fibres but may not add an unrecorded
$\mathbb R_{>0}$ scale fibre.
\end{definition}

\begin{theorem}[Right-triangle energy identity]
\label{thm:rh-right-triangle-energy-identity}
For the observation row of
\autoref{def:rh-right-triangle-observation-system},
$$
\|L_m(s)\|^2
=
r^2W_m+t^2W_m.
$$
Its scale leg has energy
$$
E_{m,\mathrm{scale}}(s)
:=
\|(-r h_a)_{a\in A_m}\|^2
=
r^2W_m.
$$
\end{theorem}

\begin{proof}
The two displayed factors of
$\mathbb R^{A_m}\oplus\mathbb R^{A_m}$ are orthogonal.  Hence
$$
\|L_m(s)\|^2
=
\sum_{a\in A_m}r^2h_a^2+
\sum_{a\in A_m}t^2h_a^2
=
r^2W_m+t^2W_m.
$$
The scale formula is the first summand.
\end{proof}

\begin{corollary}[Scale leg lower bound]
\label{cor:rh-scale-leg-lower-bound}
For every finite active packet,
$$
\|L_m(s)\|^2\ge E_{m,\mathrm{scale}}(s).
$$
Consequently, phase notation cannot cancel a nonzero scale leg inside
the squared observation energy.
\end{corollary}

\begin{proof}
By \autoref{thm:rh-right-triangle-energy-identity},
$$
\|L_m(s)\|^2-E_{m,\mathrm{scale}}(s)=t^2W_m\ge0.
$$
\end{proof}

\begin{principle}[Projection residual as coordinate creation]
\label{prin:rh-projection-residual-coordinate-creation}
Let $\mathcal H_m$ be the Hilbert subspace generated by the displayed
ledger rows before adding a further observation $v$.  The projection
decomposition
$$
v=P_{\mathcal H_m}v+(I-P_{\mathcal H_m})v
$$
separates the explained part from the residual.  The residual is
orthogonal to $\mathcal H_m$ and therefore contributes the nonnegative
energy
$$
\|(I-P_{\mathcal H_m})v\|^2.
$$
Thus an unexplained scale component must either appear as a new
orthogonal coordinate or be ruled out by a source-side positivity row;
it cannot be erased by changing the phase coordinate.
\end{principle}

\begin{definition}[Finite invariant ledger]
\label{def:rh-finite-invariant-ledger}
A finite invariant ledger for the right-triangle system consists of the
following rows for every displayed observation family
$v_1,\ldots,v_N$:
$$
G_N:=(\langle v_i,v_j\rangle)_{1\le i,j\le N}\succeq0,
$$
the Bessel row
$$
\sum_{i=1}^N|\langle v,e_i\rangle|^2\le\|v\|^2
$$
for every displayed orthonormal subfamily $e_1,\ldots,e_N$, and the
Schur residual row
$$
\Delta_N:=a_N-b_N^\ast G_N^+b_N\ge0
$$
whenever
$$
G_{N+1}=
\begin{pmatrix}
G_N & b_N\\
b_N^\ast & a_N
\end{pmatrix}
$$
is the next displayed Gram extension with the required range row
$b_N\in\operatorname{Range}(G_N)$.
\end{definition}

\begin{principle}[Invariant ledger meaning]
\label{prin:rh-invariant-ledger-meaning}
The rows of \autoref{def:rh-finite-invariant-ledger} are the finite
Hilbert expression of BEDC ledger discipline.  Positive semidefinite
Gram matrices say that displayed correlations cannot exceed displayed
energies.  Bessel rows say that recorded coordinates cannot contain more
energy than the observation.  Schur rows say that a further observation
has nonnegative conditional energy after projection to the previous
ledger.  In the RH route these are the common finite form of the
Li--Weil time row, the Weil energy row, and the Nyman container row.
\end{principle}

\begin{theorem}[Schur row is residual energy]
\label{thm:rh-schur-row-residual-energy}
Assume that $G_N$ is the Gram matrix of vectors
$v_1,\ldots,v_N$ in a Hilbert space, that
$v_{N+1}$ is another vector, and that
$$
a_N=\langle v_{N+1},v_{N+1}\rangle,
\qquad
b_N=(\langle v_1,v_{N+1}\rangle,\ldots,
\langle v_N,v_{N+1}\rangle)^T.
$$
Then the Schur residual $\Delta_N$ of
\autoref{def:rh-finite-invariant-ledger} is the squared norm of the
orthogonal residual of $v_{N+1}$ modulo
$\operatorname{span}\{v_1,\ldots,v_N\}$.
\end{theorem}

\begin{proof}
Let $P_N$ be the orthogonal projection onto
$\operatorname{span}\{v_1,\ldots,v_N\}$.  The best-approximation theorem
gives
$$
\|v_{N+1}-P_Nv_{N+1}\|^2
=
\|v_{N+1}\|^2-\|P_Nv_{N+1}\|^2.
$$
Writing the projection coefficients through the Gram system gives
$$
\|P_Nv_{N+1}\|^2=b_N^\ast G_N^+b_N
$$
on the range row.  Hence
$$
\|v_{N+1}-P_Nv_{N+1}\|^2
=
a_N-b_N^\ast G_N^+b_N
=
\Delta_N.
$$
\end{proof}

\begin{principle}[Mirror midpoint obstruction]
\label{prin:rh-mirror-midpoint-obstruction}
Let
$$
z_+=e^{\ell+i\theta},
\qquad
z_-=e^{-\ell+i\theta}.
$$
The midpoint after Li time $n$ is
$$
M_n:=\frac{z_+^n+z_-^n}{2}
=
\cosh(n\ell)e^{in\theta}.
$$
Thus
$$
|M_n|=\cosh(n\ell).
$$
If $\ell\ne0$, the midpoint lies outside the unit circle.  A reflected
pair therefore does not remove the scale coordinate; it moves the
center of the pair away from the legal phase boundary.
\end{principle}

\begin{principle}[Parallelogram obstruction for reflected scale]
\label{prin:rh-parallelogram-reflected-scale-obstruction}
For
$$
a=z_+^n,
\qquad
b=z_-^n,
$$
the parallelogram identity gives
$$
|a+b|^2+|a-b|^2=2|a|^2+2|b|^2.
$$
Since
$$
|a|=e^{n\ell},
\qquad
|b|=e^{-n\ell},
$$
the right-hand side is
$$
2e^{2n\ell}+2e^{-2n\ell}.
$$
For $\ell\ne0$ this is strictly larger than $4$.  Mirror exchange can
change the sign of the scale parameter, but the square-energy ledger
records both sides as positive length.
\end{principle}

\begin{definition}[Triangular invariant audit]
\label{def:rh-triangular-invariant-audit}
Let $\mathsf R$ be a zero-relevant source row in the RH route.  A
\emph{triangular invariant audit} of $\mathsf R$ is a triple
$$
\mathsf{Audit}(\mathsf R)
:=
(\mathsf T_{\mathsf R},\mathsf U_{\mathsf R},\mathsf P_{\mathsf R})
$$
together with a displayed defect readout.  The row
$\mathsf T_{\mathsf R}$ records the observation order or refinement
order consumed by $\mathsf R$.  The row $\mathsf U_{\mathsf R}$ records
the unit, positive, or no-negative-energy side of $\mathsf R$.  The row
$\mathsf P_{\mathsf R}$ records the ledger side: which generators,
test rows, certificates, or containers are allowed to carry the
observation.  The defect readout must factor through the same
right-triangle scale leg as
\autoref{def:rh-right-triangle-observation-system}.
\end{definition}

\begin{definition}[Audited RH source family]
\label{def:rh-audited-source-family}
The audited source family is the list of RH-route rows
$$
\mathcal A_{\triangle}
:=
\{
\mathsf{LiTimePositiveTower},
\mathsf{GlobalWeilPositivity},
\mathsf{NymanBeurlingExhaustion},
\mathsf{LeeYangBoundaryStability},
\mathsf{GoldenDominantBranch},
\mathsf{ActiveCapture},

\}.
$$
Each member of $\mathcal A_{\triangle}$ is to be read through
\autoref{def:rh-triangular-invariant-audit}, rather than as an isolated
route.
\end{definition}

\begin{principle}[Li audit row]
\label{prin:rh-li-audit-row}
The audit of $\mathsf{LiTimePositiveTower}$ has
$\mathsf T$-side given by the discrete Li time $n$ and the iteration
$$
z(\rho)^n.
$$
Its $\mathsf U$-side is the nonnegative phase contribution
$$
2-2\cos(n\theta)\ge0
$$
when $z(\rho)=e^{i\theta}$.  Its $\mathsf P$-side is the reflected
zero ledger used in the Li--Weil coefficient row.  A radial pair
$e^{\ell+i\theta}$ and $e^{-\ell+i\theta}$ is audited by the factor
$$
\cosh(n\ell),
$$
which exposes the same scale coordinate as the midpoint obstruction of
\autoref{prin:rh-mirror-midpoint-obstruction}.
\end{principle}

\begin{principle}[Weil audit row]
\label{prin:rh-weil-audit-row}
The audit of $\mathsf{GlobalWeilPositivity}$ has $\mathsf T$-side given
by the cofinal selection of test rows, $\mathsf U$-side given by the
nonnegativity of the Weil quadratic form, and $\mathsf P$-side given by
the source test space and regularization ledger of
\autoref{def:rh-weil-quadratic-form-row}.  A reflected off-fixed-half
pair is audited by the finite value block
$$
\begin{pmatrix}
0 & 1\\
1 & 0
\end{pmatrix},
$$
whose negative direction is the energy-side projection of the same
scale leg that appears in
\autoref{thm:rh-right-triangle-energy-identity}.
\end{principle}

\begin{principle}[Nyman audit row]
\label{prin:rh-nyman-audit-row}
The audit of $\mathsf{NymanBeurlingExhaustion}$ has $\mathsf T$-side
given by the increasing Hilbert subspaces generated by the Nyman rows,
$\mathsf U$-side given by the projection identity
$$
\|1-P_{N+1}1\|^2
=
\|1-P_N1\|^2-\|P_{N+1}1-P_N1\|^2,
$$
and $\mathsf P$-side given by the displayed generator ledger
$\rho_{1/n}$.  An off-fixed-half zero supplies a continuous Mellin
functional that annihilates the displayed generators while separating
the constant row; this is the container-side readout of a nonzero
residual coordinate.
\end{principle}

\begin{principle}[Boundary and dominance audit rows]
\label{prin:rh-boundary-dominance-audit-rows}
The audit of $\mathsf{LeeYangBoundaryStability}$ has $\mathsf T$-side
given by finite-volume refinement, $\mathsf U$-side given by
unit-boundary nonzero regions, and $\mathsf P$-side given by stable
operations such as polarization, Asano contraction, or a representation
theoretic compactness ledger.  A golden or Zeckendorf ruler can be used
only as a displayed refinement scale; it is not by itself the unit
container.

The audit of $\mathsf{GoldenDominantBranch}$ has $\mathsf T$-side given
by the finite recurrence or endpoint parameter, $\mathsf U$-side given
by the maximal-modulus branch condition, and $\mathsf P$-side given by
the exact algebraic certificates that distinguish the branch.  Thus a
branch may be used in a zero law only after dominance and source
identity are both ledgered.
\end{principle}

\begin{principle}[Capture, Nyman--Mobius, and Schur audit rows]
\label{prin:rh-capture-nyman-schur-audit-rows}
The audit of $\mathsf{ActiveCapture}$ has $\mathsf T$-side given by
certificate refinement, $\mathsf U$-side given by boundary stability or
transport into a positive container, and $\mathsf P$-side given by the
source-side construction of the container before any zero island is
selected.

The audit of $\mathsf{GoldenNymanGramExhaustion}$ has $\mathsf T$-side
given by the Fibonacci refinement order, $\mathsf U$-side given by
nonnegative Hilbert residual gains, and $\mathsf P$-side given by the
Möbius and Nyman generator ledgers that must exhaust the constant row.

The audit of $\mathsf{SchurResidualSquareRepresentation}$ has
$\mathsf T$-side given by appending a test direction, $\mathsf U$-side
given by
$$
\Delta_N\ge0,
$$
and $\mathsf P$-side given by a source-side square representation of
$\Delta_N$, not by a posterior factorization of an already positive
matrix.
\end{principle}

\begin{definition}[Triangular audit coherence]
\label{def:rh-triangular-audit-coherence}
The source row
$$
\mathsf{TriangularAuditCoherence}_{\zeta}
$$
asserts that every member of $\mathcal A_{\triangle}$ carries a
triangular invariant audit and that the audits are compatible by the
following two-out-of-three transport rows:
$$
\mathsf T+\mathsf U\to\mathsf P,
\qquad
\mathsf U+\mathsf P\to\mathsf T,
\qquad
\mathsf P+\mathsf T\to\mathsf U.
$$
The arrows are not logical shortcuts between unrelated theorems.  They
state that if two sides of the same finite right-triangle ledger are
source-side certified, then the third side must read the same scale
defect and cannot introduce a different hidden coordinate.
\end{definition}

\begin{principle}[Common audit invariant]
\label{prin:rh-common-audit-invariant}
The common invariant behind every audited row is the Li-coordinate
decomposition
$$
|1-z^n|^2
=
(e^{n\ell}-1)^2+
2e^{n\ell}(1-\cos(n\theta)).
$$
The first summand is the scale residual and the second summand is the
phase residual.  The audit discipline requires every named RH-route row
to say where this scale residual is seen in time, where it is forbidden
by positivity or unit boundary, and where it is recorded in the ledger.
\end{principle}

\begin{definition}[Ontic Pythagorean closure row]
\label{def:rh-ontic-pythagorean-closure-row}
The source row
$$
\mathsf{OnticPythagoreanClosure}_{\zeta}
$$
asserts that every zero-relevant certificate tower admits finite
right-triangle observation systems whose scale legs are controlled by a
source-side unit-positive row:
$$
E_{m,\mathrm{scale}}(s_\infty)\le B
$$
for a tower bound $B$ independent of the refinement level, while the
active weights are cofinal:
$$
W_m\to\infty.
$$
The row also requires that any hidden fibre accepted by the tower is a
compact phase or prime-register fibre, not an unrecorded real-positive
scale fibre.
\end{definition}

\begin{theorem}[Ontic Pythagorean closure gives fixed-half readout]
\label{thm:rh-ontic-pythagorean-closure-fixed-half-readout}
Let a zero-relevant certificate tower have limit readout
$$
s_\infty=\frac12+r+it.
$$
If the tower satisfies
$\mathsf{OnticPythagoreanClosure}_{\zeta}$ and its finite
right-triangle systems extract the scale energy
$$
E_{m,\mathrm{scale}}(s_\infty)=r^2W_m+o(W_m),
$$
then $r=0$.
\end{theorem}

\begin{proof}
Assume $r\ne0$.  Then $r^2>0$.  From the extraction row there is a
cofinal tail on which
$$
|o(W_m)|\le\frac{r^2}{2}W_m.
$$
Therefore
$$
E_{m,\mathrm{scale}}(s_\infty)\ge\frac{r^2}{2}W_m
$$
on that tail.  Since $W_m\to\infty$, the scale energies are unbounded
along the tail, contradicting the tower bound in
\autoref{def:rh-ontic-pythagorean-closure-row}.  Hence $r=0$.
\end{proof}

\begin{principle}[Invariant target of the RH route]
\label{prin:rh-invariant-target-route}
The analytic burden of the route is to supply the source-side
unit-positive row used in
\autoref{def:rh-ontic-pythagorean-closure-row}.  The Stieltjes and
Herglotz rows of \autoref{ch:visions-rh-li-weil-herglotz-target}, the
log-potential no-scale field of
\autoref{ch:visions-rh-xi-log-potential-no-scale}, the Gram and Schur
rows of \autoref{ch:visions-rh-golden-weil-gram-induction}, and the
Nyman square tower of \autoref{ch:visions-rh-golden-nyman-square-tower}
are different finite readouts of that single positivity demand.  The
right-triangle system specifies what these readouts must forbid: a
nonzero scale leg carried by $e^{-rh_a}$.
\autoref{def:rh-triangular-audit-coherence} records the corresponding
bookkeeping demand: each named route must display its time side, its
unit-positive side, and its ledger side as projections of that same
scale-leg obstruction.
\end{principle}
