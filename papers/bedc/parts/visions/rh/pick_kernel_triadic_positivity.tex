\chapter{Pick Kernel Triadic Positivity}
\label{ch:visions-rh-pick-kernel-triadic-positivity}
\origin{ai}

\autoref{ch:visions-rh-li-weil-herglotz-target} records the positive-real
log-derivative row, and
\autoref{ch:visions-rh-source-side-triadic-audit-criterion} records the
$\mathsf P$, $\mathsf U$, and $\mathsf T$ audit discipline for
RH-facing source rows.  This chapter rewrites the Herglotz target as a
Pick-kernel target.  The purpose is to make the unit side finite-matrix
checkable: a failure of positivity is witnessed by a finite Pick matrix,
while a source-side Hilbert factorization supplies the right-triangle
norm form.

\begin{definition}[Cayley right-half-plane readout]
\label{def:rh-cayley-right-half-plane-readout}
Let
$$
s(z):=\frac{1}{1-z}.
$$
Then
$$
s(z)-\frac12
=
\frac{1+z}{2(1-z)}.
$$
Thus the unit disk is the Cayley coordinate for the open half-plane to
the right of the fixed line.  Define the completed-zeta logarithmic
Pick readout by
$$
f_{\xi}(z)
:=
2\frac{\xi'(s(z))}{\xi(s(z))}
$$
on the disk locus where this expression is holomorphic.
\end{definition}

\begin{definition}[Xi Pick kernel]
\label{def:rh-xi-pick-kernel}
The Xi Pick kernel attached to
\autoref{def:rh-cayley-right-half-plane-readout} is
$$
K_{\xi}(z,w)
:=
\frac{f_{\xi}(z)+\overline{f_{\xi}(w)}}{1-z\overline w}.
$$
The source row
$$
\mathsf{TriadicPickPositivity}_{\xi}
$$
asserts that $f_{\xi}$ is holomorphic on the unit disk and that
$K_{\xi}$ is positive definite: for every finite family
$z_1,\ldots,z_N\in\mathbb D$ and $c_1,\ldots,c_N\in\mathbb C$,
$$
\sum_{i,j=1}^{N}
c_i\overline{c_j}K_{\xi}(z_i,z_j)\ge0.
$$
\end{definition}

\begin{theorem}[Pick positivity gives the Herglotz row]
\label{thm:rh-pick-positivity-gives-herglotz-row}
If $\mathsf{TriadicPickPositivity}_{\xi}$ holds, then
$$
\mathsf{XiLogDerivativeHerglotz}
$$
holds.
\end{theorem}

\begin{proof}
Positive definiteness of $K_{\xi}$ applied to the one-point family
$z$ gives
$$
K_{\xi}(z,z)
=
\frac{2\operatorname{Re}f_{\xi}(z)}{1-|z|^2}
\ge0.
$$
Since $1-|z|^2>0$ in the disk,
$$
\operatorname{Re}f_{\xi}(z)\ge0.
$$
Under the coordinate $s=s(z)$ this is precisely the positive-real row
for
$$
2\frac{\xi'(s)}{\xi(s)}
$$
on $\operatorname{Re}(s)>1/2$, which is
$\mathsf{XiLogDerivativeHerglotz}$.
\end{proof}

\begin{theorem}[Herglotz row gives Pick positivity]
\label{thm:rh-herglotz-row-gives-pick-positivity}
Assume $f_{\xi}$ is holomorphic on the unit disk and has a Herglotz
representation
$$
f_{\xi}(z)
=
ib+
\int_{\mathbb T}\frac{\zeta+z}{\zeta-z}\,d\mu(\zeta),
$$
where $b\in\mathbb R$ and $\mu$ is a positive finite measure.  Then
$K_{\xi}$ is positive definite.
\end{theorem}

\begin{proof}
The imaginary constant cancels in
$f_{\xi}(z)+\overline{f_{\xi}(w)}$.  Substituting the integral
representation gives
$$
K_{\xi}(z,w)
=
2\int_{\mathbb T}
\frac{1}{\zeta-z}
\frac{1}{\overline{\zeta-w}}
\,d\mu(\zeta).
$$
For a finite family $z_i$ and coefficients $c_i$,
$$
\sum_{i,j}c_i\overline{c_j}K_{\xi}(z_i,z_j)
=
2\int_{\mathbb T}
\left|
\sum_i\frac{c_i}{\zeta-z_i}
\right|^2
d\mu(\zeta)
\ge0.
$$
Thus the kernel is positive definite.
\end{proof}

\begin{corollary}[Pick positivity gives fixed-half readout]
\label{cor:rh-pick-positivity-fixed-half-readout}
Assume $\mathsf{TriadicPickPositivity}_{\xi}$ and the usual reflection
consumption row for the completed zeta zero ledger.  Then the completed
zeta zero ledger has fixed-half readout.
\end{corollary}

\begin{proof}
By \autoref{thm:rh-pick-positivity-gives-herglotz-row},
$\mathsf{XiLogDerivativeHerglotz}$ holds.  Apply
\autoref{thm:rh-xi-herglotz-row-fixed-half-readout}.
\end{proof}

\begin{definition}[Finite Pick witness]
\label{def:rh-finite-pick-witness}
A finite Pick witness against the Xi Pick row is a tuple
$$
(z_1,\ldots,z_N;c_1,\ldots,c_N)
$$
with $z_i\in\mathbb D$ and $c_i\in\mathbb C$ such that
$$
\sum_{i,j=1}^{N}
c_i\overline{c_j}K_{\xi}(z_i,z_j)<0.
$$
It is the finite-matrix form of a negative energy witness.
\end{definition}

\begin{principle}[Pick witness as the time side]
\label{prin:rh-pick-witness-time-side}
The Pick formulation supplies the $\mathsf T$ side of the audit in a
finite form.  If the kernel row fails, the failure appears at some
finite matrix size with finitely many disk points and coefficients.
Thus the obstruction is not postponed to an unbounded search through
all certificate levels; it has a displayed finite witness.
\end{principle}

\begin{definition}[Source Pick Hilbert factorization]
\label{def:rh-source-pick-hilbert-factorization}
The source row
$$
\mathsf{SourcePickHilbertFactorization}_{\xi}
$$
asserts that there are a Hilbert space $\mathcal H$ and a source-side
map
$$
\Phi_{\xi}:\mathbb D\to\mathcal H
$$
such that
$$
K_{\xi}(z,w)=\langle\Phi_{\xi}(z),\Phi_{\xi}(w)\rangle_{\mathcal H}.
$$
The map $\Phi_{\xi}$ must be constructed from prime, gamma,
normalization, golden, Stieltjes, Herglotz, or explicit-formula source
rows before the completed zero divisor is read.
\end{definition}

\begin{theorem}[Hilbert factorization supplies Pick positivity]
\label{thm:rh-hilbert-factorization-supplies-pick-positivity}
If $\mathsf{SourcePickHilbertFactorization}_{\xi}$ holds, then
$$
\mathsf{TriadicPickPositivity}_{\xi}
$$
holds.
\end{theorem}

\begin{proof}
For any finite family,
$$
\sum_{i,j}c_i\overline{c_j}K_{\xi}(z_i,z_j)
=
\left\|
\sum_i c_i\Phi_{\xi}(z_i)
\right\|_{\mathcal H}^2
\ge0.
$$
\end{proof}

\begin{definition}[Finite source Pick approximation]
\label{def:rh-finite-source-pick-approximation}
A finite source Pick approximation is a sequence of positive kernels
$$
K_m(z,w)=\langle\Phi_m(z),\Phi_m(w)\rangle_{\mathcal H_m}
$$
whose construction is source-side and whose finite matrix entries
converge to the Xi Pick entries:
$$
K_m(z_i,z_j)\to K_{\xi}(z_i,z_j)
$$
for every finite disk family $z_1,\ldots,z_N$.
\end{definition}

\begin{theorem}[Finite Pick approximations close the Pick row]
\label{thm:rh-finite-pick-approximations-close-pick-row}
If a finite source Pick approximation exists, then
$$
\mathsf{TriadicPickPositivity}_{\xi}
$$
holds.
\end{theorem}

\begin{proof}
For each $m$ and every finite coefficient family,
$$
\sum_{i,j}c_i\overline{c_j}K_m(z_i,z_j)\ge0.
$$
Passing to the displayed finite-matrix limit gives
$$
\sum_{i,j}c_i\overline{c_j}K_{\xi}(z_i,z_j)\ge0.
$$
This is positive definiteness of $K_{\xi}$.
\end{proof}

\begin{principle}[Triadic meaning of the Pick row]
\label{prin:rh-triadic-meaning-pick-row}
The Pick row is the compact Hilbert-kernel form of the triadic audit.
Its $\mathsf P$ side is the requirement that $K_{\xi}$ or its
approximants are built from prime, gamma, and normalization ledgers.
Its $\mathsf U$ side is positive definiteness, equivalently the norm
square in \autoref{def:rh-source-pick-hilbert-factorization}.  Its
$\mathsf T$ side is the finite Pick witness row of
\autoref{def:rh-finite-pick-witness}.  The same obstruction that appears
as a scale leg in
\autoref{def:rh-right-triangle-observation-system} appears here as a
failure of a finite Gram matrix.
\end{principle}
