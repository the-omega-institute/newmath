\chapter{Global Weil Positive Container Tower}
\label{ch:visions-rh-global-weil-positive-container-tower}
\origin{ai}

\autoref{ch:visions-rh-squarefree-prime-polarized-bridge} records the
squarefree prime surface for polarized Lee--Yang containers, and
\autoref{ch:visions-rh-xi-jensen-internal-container} records the

the explicit formula.  This chapter records that target as the
non-local form of the no-scale route.

\begin{definition}[Multiplicative test row]
\label{def:rh-multiplicative-test-row}
Let $\mathcal A$ be a source test space on
$\mathbb R_{+}^{\times}$ stable under the operations required by the
explicit formula.  For $f\in\mathcal A$, its Mellin readout is
$$
\widehat f(s)
:=
\int_0^\infty f(x)x^s\frac{dx}{x},
$$
whenever this integral is read directly, and otherwise by the
regularized test-row convention of the explicit formula.  In logarithmic
coordinates $x=e^u$, with $\phi(u)=f(e^u)$, this is the Laplace--Fourier
readout
$$
\widehat f(s)
=
\int_{-\infty}^{\infty}\phi(u)e^{su}\,du.
$$
\end{definition}

\begin{definition}[Weil quadratic form row]
\label{def:rh-weil-quadratic-form-row}
The Weil quadratic form attached to the completed zeta zero ledger is
the regularized pairing
$$
\mathcal W(f)
:=
\sum_{\rho}
\widehat f(\rho)\,
\overline{\widehat f(1-\overline{\rho})},
$$
where $\rho$ ranges over the completed zeta zero ledger with the
standard symmetry and multiplicity conventions of the explicit formula.
The displayed formula records the zero-side contribution; the
regularization and archimedean rows are part of the Weil-form source
ledger.
\end{definition}

\begin{definition}[Global Weil positivity]
\label{def:rh-global-weil-positivity}
The source row
$$
\mathsf{GlobalWeilPositivity}
$$
asserts that
$$
\mathcal W(f)\ge0
\qquad
(f\in\mathcal A).
$$
It is the global test-function version of the no-scale condition: no
displayed test row may extract negative energy from the zero ledger.
\end{definition}

\begin{principle}[Weil positivity as the no-scale global target]
\label{prin:rh-weil-positivity-no-scale-global-target}
The local no-scale route reads an off-half parameter as a radial defect
$$
e^{-(s-1/2)h}=e^{-rh}e^{-ith}.
$$
Finite packets express this as a scale-energy row.  The global Weil
route expresses the same obstruction as a test-function positivity row:
an off-half mirror pair would allow a test readout that separates the two
mirror points and contributes a negative term to
\autoref{def:rh-weil-quadratic-form-row}.  Thus the thermodynamic
no-demon condition and the explicit-formula positivity condition have a
common target, namely $\mathsf{GlobalWeilPositivity}$.
\end{principle}

\begin{definition}[Finite Weil-positive container]
\label{def:rh-finite-weil-positive-container}
A \emph{finite Weil-positive container} is a pair
$$
(\mathcal A_N,\mathcal W_N)
$$
where $\mathcal A_N\subseteq\mathcal A$ is a finite displayed
test space and
$$
\mathcal W_N:\mathcal A_N\to\mathbb R
$$
is a positive semidefinite quadratic row:
$$
\mathcal W_N(f)\ge0
\qquad
(f\in\mathcal A_N).
$$
The container is source-side when $\mathcal A_N$ and $\mathcal W_N$ are
generated before any zero island is selected and carry no unledgered
$\mathbb R_{>0}$ scale coordinate.
\end{definition}

\begin{definition}[Cofinal Weil-positive container tower]
\label{def:rh-cofinal-weil-positive-container-tower}
The source row
$$
\mathsf{CofinalWeilPositiveContainerTower}
$$
asserts the existence of finite Weil-positive containers
$$
(\mathcal A_N,\mathcal W_N)_{N\ge1}
$$
with the following rows.

The nesting row is
$$
\mathcal A_N\subseteq\mathcal A_{N+1}.
$$
The cofinality row says that
$$
\bigcup_N\mathcal A_N
$$
is dense in the test space used by
\autoref{def:rh-weil-quadratic-form-row}.  The compatibility row says
that the restrictions of later quadratic forms to earlier test spaces
agree with earlier forms up to a displayed error ledger tending to zero.
The limit row says that for every test function $f$ and every admissible
cofinal approximation $f_N\in\mathcal A_N$ with $f_N\to f$,
$$
\mathcal W_N(f_N)\to\mathcal W(f).
$$
\end{definition}

\begin{theorem}[Cofinal Weil-positive tower gives global positivity]
\label{thm:rh-cofinal-weil-positive-tower-gives-global-positivity}
If $\mathsf{CofinalWeilPositiveContainerTower}$ holds, then
$\mathsf{GlobalWeilPositivity}$ holds.
\end{theorem}

\begin{proof}
Let $f\in\mathcal A$.  By cofinality choose admissible
$f_N\in\mathcal A_N$ with $f_N\to f$.  Finite positivity gives
$$
\mathcal W_N(f_N)\ge0
$$
for every $N$.  The limit row gives
$$
\mathcal W_N(f_N)\to\mathcal W(f).
$$
The limit of nonnegative real numbers is nonnegative, so
$$
\mathcal W(f)\ge0.
$$
Since $f$ was arbitrary, $\mathsf{GlobalWeilPositivity}$ holds.
\end{proof}

\begin{definition}[Weil criterion consumption row]
\label{def:rh-weil-criterion-consumption-row}
The source row
$$
\mathsf{WeilCriterionConsumesGlobalPositivity}
$$
is the classical criterion that consumes
$\mathsf{GlobalWeilPositivity}$ and returns the fixed-half zero ledger
for the completed zeta function.  In this vision it is treated as a
named bridge row, separate from the construction of the finite
positive tower.
\end{definition}

\begin{theorem}[Finite Weil-positive tower gives fixed-half readout]
\label{thm:rh-finite-weil-positive-tower-fixed-half-readout}
Assume $\mathsf{CofinalWeilPositiveContainerTower}$ and
$\mathsf{WeilCriterionConsumesGlobalPositivity}$.  Then the completed
zeta zero ledger has fixed-half readout.
\end{theorem}

\begin{proof}
By
\autoref{thm:rh-cofinal-weil-positive-tower-gives-global-positivity},
the tower gives $\mathsf{GlobalWeilPositivity}$.  The consumption row
then returns the fixed-half zero ledger.
\end{proof}

\begin{principle}[Finite container routes as Weil-tower candidates]
\label{prin:rh-finite-container-routes-as-weil-tower-candidates}
The Lee--Yang, Jensen, golden-ruler, squarefree-prime, and quartic
endpoint routes are best read as candidate constructions of the finite
rows in
\autoref{def:rh-cofinal-weil-positive-container-tower}.  A Lee--Yang
packet supplies finite nonvanishing and stability; a Jensen section
supplies finite hyperbolicity; a golden ruler supplies source-side
cofinal refinement; a squarefree prime packet supplies arithmetic
ledgering; and a quartic endpoint model supplies a finite dominance and
Rouch\'e laboratory.  None of these finite rows is by itself the global
argument.  The global argument is the cofinal passage from finite
positivity to $\mathsf{GlobalWeilPositivity}$.
\end{principle}

\begin{definition}[Scale-closure to Weil positivity row]
\label{def:rh-scale-closure-to-weil-positivity-row}
The source row

\mathsf{ZetaUnitaryScaleClosureToWeilPositivity}
$$

rows of the zeta unitary scale route assemble into finite
Weil-positive containers compatible with
\autoref{def:rh-weil-quadratic-form-row}.  It is the bridge from local
radial-defect exclusion to the global positivity row.
\end{definition}

\begin{principle}[Final non-local proof obligation]
\label{prin:rh-final-non-local-proof-obligation}
The non-local proof obligation is not to build a stable packet around a
candidate zero.  It is to construct a source-side cofinal tower of
finite positive quadratic rows.  In symbols, the active target is
$$
\mathsf{CofinalWeilPositiveContainerTower}
\Longrightarrow
\mathsf{GlobalWeilPositivity}
\Longrightarrow
\text{fixed-half zero readout}.
$$
The first implication is the finite-container reduction proved in this
chapter; the second is delegated to the Weil criterion consumption row.
\end{principle}
