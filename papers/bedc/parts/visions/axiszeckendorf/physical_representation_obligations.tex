\begin{definition}[Low-energy electromagnetic representation certificate]
\label{def:zeckendorf-low-energy-electromagnetic-representation-certificate}

$$

\begin{principle}[Hidden boundary parity]
\label{prin:zeckendorf-hidden-boundary-parity}

smooth correction in ordinary continuous low-energy probes. If it has a

integral, mod-\(2\), boundary-condition, topology-sensitive, or coherent

constrains the parity row to remain hidden from continuous probes; it
does not by itself identify or eliminate the discrete row.
\end{principle}

\begin{theorem}[Independent denominator is required for predictivity]
\label{thm:zeckendorf-denominator-predictivity-requirement}
Let
$$
  R(D)
  :=
  2\pi\varphi^5
  \frac{
    F_8+F_9C(D)
  }{
    F_8\varphi^{-1}+F_9C(D)\varphi^{-2}
  },
$$
where
$$
  C(D)
  :=
  \frac12
  +
  \frac14\cos^2\left(\frac{\pi}{\varphi}\right)
  +
  \frac{1}{D\varphi^5}.
$$
For positive \(D\), the readout \(R(D)\) is strictly decreasing in
\(D\). Consequently, if \(D\) is obtained by inverting an observed
low-energy value of \(\alpha_{\mathrm{EM}}^{-1}\), then the expression
has no predictive force as a Window-six claim. A predictive use requires
a denominator rule
$$
  D_6=\Gamma(X_6,\mathsf{Fold}_6,\text{boundary parity},
  \text{tail ledger})
$$
that does not use the measured electromagnetic coupling.
\end{theorem}
\begin{proof}
The derivative with respect to \(C\) is
$$
  \frac{\partial R}{\partial C}
  =
  2\pi\varphi^5
  \frac{
    F_8F_9(\varphi^{-1}-\varphi^{-2})
  }{
    (F_8\varphi^{-1}+F_9C\varphi^{-2})^2
  }.
$$
Since \(\varphi^{-1}>\varphi^{-2}\), this derivative is positive. Also
$$
  \frac{dC}{dD}
  =
  -\frac{1}{D^2\varphi^5}
  <
  0.
$$
Thus \(dR/dD<0\). In any interval where the inverse problem is posed,
the measured scalar can therefore determine a denominator by inversion
rather than by Window-six structure. Such an inverted denominator is a
fit, not an independent prediction. Predictivity requires that the
denominator be supplied by the Window-six ledger before comparison with
the physical scalar.
\end{proof}

\begin{theorem}[Physical refutation surface for the Window-six readout]
\label{thm:zeckendorf-window-six-physical-refutation-surface}
The physical interpretation of \(R_{Z6}^\star\) is refutable at the
derived-interface level by any of the following failures:
no admissible low-energy projection gate selects the \(F_{10}\) branch;
the coefficient \(8/9\) remains a fitted rational rather than a
boundary-parity over hidden-residual derivation; independent low-energy
routes converge to incompatible values after systematic corrections; a
high-energy comparison reuses \(R_{Z6}^\star\) without a scale-indexed
ledger or window-refinement flow; or later residual corrections require
terms outside the gated ladder
$$
  \varphi^{-7-10k}\qquad(k\geq 0).
$$
\end{theorem}
\begin{proof}
The first failure violates the protocol-gate requirement in
\autoref{def:zeckendorf-low-energy-alpha-projection-gate}. The second
violates the residual closure principle in
\autoref{prin:zeckendorf-window-six-residual-closure}, the
protocol-gated closure principle in
\autoref{prin:zeckendorf-protocol-gated-window-six-minimal-residual-closure},
and the interpretation of \(8/9\) in
\autoref{rem:zeckendorf-eight-nine-closure-coefficient}. The third
violates the route-convergence requirement of
\autoref{prin:zeckendorf-independent-route-convergence}. The fourth
violates the infrared boundary of
\autoref{prin:zeckendorf-running-coupling-window-refinement}. The fifth
violates the displayed Window-six residual denominator, whose correction
terms occur at exponents \(7\), \(17\), and \(27\), and hence on the
ladder \(7+10k\). Each failure breaks an explicit interface condition
without needing to deny the finite Zeckendorf fold construction itself.
\end{proof}
