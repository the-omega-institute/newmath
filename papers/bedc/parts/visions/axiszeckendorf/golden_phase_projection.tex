\section{Window observers and residual fibers}

\begin{definition}[Stable Zeckendorf window]
\label{def:zeckendorf-stable-window}
For \(m\geq 0\), let
$$
  \Omega_m:=\{0,1\}^{m}.
$$
The stable Zeckendorf window is the subcarrier
$$
  X_m
  :=
  \{x\in\Omega_m:\ x_kx_{k+1}=0\ \text{for all admissible }k\}.
$$
Equivalently, a visible word in \(X_m\) contains no adjacent \(11\)
pattern.
\end{definition}

\begin{definition}[Fibonacci value and fold]
\label{def:zeckendorf-window-fold}
Let \(F_0=0\), \(F_1=1\), and \(F_{n+1}=F_n+F_{n-1}\) for \(n\geq 1\).
For \(\omega\in\Omega_m\), define its Fibonacci value by
$$
  N_m(\omega)
  :=
  \sum_{k=1}^{m}\omega_kF_{k+1}.
$$
Let \(Z(N_m(\omega))\) be the Zeckendorf normal digit sequence of this
value. The window fold is the visible truncation
$$
  \mathsf{Fold}_m:\Omega_m\to X_m,
  \qquad
  \mathsf{Fold}_m(\omega):=\pi_m Z(N_m(\omega)).
$$
The map is a projection from raw binary words to stable visible words;
it is not an identification of the raw histories.
\end{definition}

\begin{definition}[Zeckendorf window observer]
\label{def:zeckendorf-window-observer}
A Zeckendorf window observer at width \(m\) consumes only the visible
word
$$
  x=\mathsf{Fold}_m(\omega)\in X_m.
$$
The hidden residual fiber over \(x\) is
$$
  \mathcal{F}_m(x)
  :=
  \mathsf{Fold}_m^{-1}(x),
  \qquad
  d_m(x):=|\mathcal{F}_m(x)|.
$$
Thus the observer sees a stable Zeckendorf word while the ledger retains
the raw microstates compressed by the fold.
\end{definition}

\begin{principle}[Fold readout does not erase the fiber]
\label{prin:zeckendorf-fold-readout-fiber}
Any consumer of \(\mathsf{Fold}_m(\omega)=x\) may use \(x\) as a visible
normal-form token only together with the residual-fiber row
\(\mathcal{F}_m(x)\) or an explicit exactness certificate. In
particular, equality of visible windows is not equality of raw histories.
\end{principle}

\begin{theorem}[Golden pair has half trace center]
\label{thm:zeckendorf-golden-pair-half-trace-center}

\qquad

\qquad
B_1=1-B_0,
\qquad
B_0=1-B_1,
$$
and hence
$$
c_{\varphi}=\frac12.
$$
\end{theorem}

\begin{proof}
Substituting the displayed algebraic readings gives
$$
B_0+B_1
=
\frac{1+\sqrt5}{2}+\frac{1-\sqrt5}{2}
=1
$$

$$

=

=
\frac{1-5}{4}
=-1.
$$
The two equations $B_1=1-B_0$ and $B_0=1-B_1$ are immediate from the
sum equation. Therefore
$$
c_{\varphi}
=
\frac{B_0+B_1}{2}
=
\frac12.
$$
\end{proof}

\begin{theorem}[Golden phase is a derived readout]
\label{thm:zeckendorf-golden-phase-derived-readout}
The golden phase projection is a derived interface over the
Axis-Zeckendorf fold ledger. It does not identify raw histories, does not
replace \(\mathsf{ZCarry}\) by \(\HSame\), does not export field
operations of \(\RealUp\) or a circle carrier, and does not assert a
physical law.
\end{theorem}
\begin{proof}
By Definition~\ref{def:zeckendorf-golden-phase-projection}, the phase
token \(\Theta_m(x)\) is assigned only after the visible word \(x\) has
been obtained from \(\mathsf{Fold}_m\). By
Principle~\ref{prin:zeckendorf-fold-readout-fiber}, use of the visible
word must retain the residual-fiber row or an exactness certificate.
Therefore the phase token cannot collapse raw histories. The carry
discipline of Theorem~\ref{thm:zcarry-not-hsame} keeps carry
classification separate from \(\HSame\), and the cannot-claim registry
of \autoref{ch:concrete-instances-axis-zeckendorf-cannot-claim} blocks
promotion to \(\RealUp\), a primitive dimension, or a replacement of
\(\NatUp\). The projection is therefore only a display interface over
finite ledger data.
\end{proof}
