\begin{principle}[Beta-sign selection for running deformations]
\label{prin:zeckendorf-beta-sign-selection}
For any candidate physical running deformation of the fixed infrared
denominator, write
$$
  D(t)=D_{6,\alpha}^{\star}+\delta D(t),
$$
where $t$ is any monotone activation coordinate for increasing
electromagnetic scale. If this deformation is to represent the physical
running direction of the electromagnetic coupling, then it must satisfy
$$
  \frac{d}{dt}D(t)\geq0
$$
on its activation interval. Since
\autoref{thm:zeckendorf-denominator-predictivity-requirement} gives
$dR/dD<0$, this condition is exactly the inverse-coupling sign condition
$$
  \frac{d}{dt}\alpha^{-1}(t)\leq0,
  \qquad
  \frac{d}{dt}\alpha(t)\geq0.
$$
Thus the beta-sign test is structural: it depends on the direction of
the denominator deformation, not on fitting the infrared value.
\end{principle}

\begin{theorem}[Walking branch is not the physical running direction]
\label{thm:zeckendorf-walking-branch-not-physical-running}
The walking-gate phase-reservoir denominator
$$
  D_{6,\mathrm{walk}}^{\mathrm{res}}
  =
  47+\varphi^{-7}-\frac12\varphi^{-17}+\frac89\varphi^{-29}
$$
cannot be the primary physical running deformation of the Window-six
electromagnetic readout. It is a cross-window refinement branch, not the
local beta seed for increasing electromagnetic scale.
\end{theorem}
\begin{proof}
The static low-energy phase-reservoir denominator is
$$
  D_{6,\alpha}^{\star}
  =
  47+\varphi^{-7}-\frac12\varphi^{-17}+\frac89\varphi^{-27}.
$$
Since $\varphi^{-29}<\varphi^{-27}$, one has
$$
  D_{6,\mathrm{walk}}^{\mathrm{res}}<D_{6,\alpha}^{\star}.
$$
By \autoref{thm:zeckendorf-denominator-predictivity-requirement}, the
readout $R(D)$ is strictly decreasing in $D$. Therefore
$$
  R(D_{6,\mathrm{walk}}^{\mathrm{res}})
  >
  R(D_{6,\alpha}^{\star}),
$$
so the walking deformation raises $\alpha^{-1}$ instead of lowering it.
It fails the beta-sign selection principle
\autoref{prin:zeckendorf-beta-sign-selection}. The same branch may still
record boundary-uplift refinement data, because the transition from
$\Delta^{\mathrm{bdry}}_6=\{0,F_9\}$ to the three-branch boundary
differences
$$
  \Delta^{\mathrm{bdry}}_7=\{0,F_{10},F_{11}\},
  \qquad
  \Delta^{\mathrm{bdry}}_8=\{0,F_{11},F_{12}\}
$$
exhibits the Fibonacci second-difference channel. That channel requires
protocol-gated window refinement before it can become a physical running
law.
\end{proof}

\begin{theorem}[Green-excess passes the beta-sign test]
\label{thm:zeckendorf-green-excess-passes-beta-sign-test}
The centered Green-excess deformation
$$
  D_{6,\alpha}(z)
  =
  D_{6,\alpha}^{\star}
  +
  \Delta_R(z)\varphi^{-37}
  +
  O(\varphi^{-47})
$$
has the correct physical running sign on the activation interval
$0<z\leq1$.
\end{theorem}
\begin{proof}
By \autoref{thm:zeckendorf-spectral-response-interpolation},
$\Delta_R(0)=0$, $\Delta_R(z)>0$ for $0<z\leq1$, and
$\Delta_R'(z)>0$ on $0\leq z\leq1$. Hence the Green-excess deformation
increases the denominator away from the static point. Applying
\autoref{prin:zeckendorf-beta-sign-selection} and
\autoref{thm:zeckendorf-denominator-predictivity-requirement} gives
$$
  \alpha_{Z6}^{-1}(z)<\alpha_{Z6}^{-1}(0)
  \qquad(0<z\leq1),
$$
so the coupling itself increases along the activation. The physically
admissible local seed is therefore the $571$-Green Green-excess branch,
while the walking branch remains a protocol-gated cross-window
refinement signal.
\end{proof}
