\chapter{Layer-Kernel Residual Induction}
\label{ch:visions-axiszeckendorf-layer-kernel-residual-induction}
\origin{ai}

\autoref{ch:visions-axiszeckendorf-window-six-inverse-limit-lifting}
separates the Window-six address carrier from the flux-operator tower.  The
present chapter records the induction form needed after that separation.  A
global RH counter residual is not attacked first at the completed
inverse-limit object.  It is inspected through the finite projection tower,
where the first nonzero residual must appear in a newly added kernel layer.

\section{Linearized residual tower}
\label{sec:axiszeckendorf-linearized-residual-tower}

\begin{definition}[Linearized address residual space]
\label{def:axiszeckendorf-linearized-address-residual-space}
For each $m\geq 6$, let
$$
V_m:=\mathbb Q[X_m]
$$
be the rational vector space freely generated by the no-adjacent address
window $X_m$ of
\autoref{def:axiszeckendorf-no-adjacent-address-windows}.  The prefix map
$\pi_{m+1,m}:X_{m+1}\to X_m$ induces the linear projection
$$
\Pi_{m+1,m}:V_{m+1}\to V_m.
$$
\end{definition}

\begin{definition}[Layer kernel]
\label{def:axiszeckendorf-layer-kernel}
The layer kernel added between levels $m$ and $m+1$ is
$$
K_{m+1}:=\ker(\Pi_{m+1,m}).
$$
It records residual information visible at level $m+1$ but invisible after
projection to level $m$.
\end{definition}

\begin{proposition}[Layer-kernel dimension]
\label{prop:axiszeckendorf-layer-kernel-dimension}
Assume the prefix projection is surjective on the no-adjacent address
windows.  Then
$$
\dim K_{m+1}=|X_{m+1}|-|X_m|=F_{m+1}.
$$
In particular,
$$
\dim K_7=F_7=13.
$$
\end{proposition}

\begin{proof}
The dimension of $V_m$ is $|X_m|$.  Since every word in $X_m$ extends by a
zero digit to a word in $X_{m+1}$, the prefix projection is surjective.
Therefore
$$
\dim K_{m+1}=\dim V_{m+1}-\dim V_m.
$$
Using $|X_m|=F_{m+2}$ gives
$$
\dim K_{m+1}=F_{m+3}-F_{m+2}=F_{m+1}.
$$
For $m=6$, this yields $\dim K_7=F_7=13$.
\end{proof}

\begin{definition}[Residual tower]
\label{def:axiszeckendorf-residual-tower}
A residual tower is a compatible family
$$
r_\infty=(r_m)_{m\geq 6},
\qquad
r_m\in V_m,
$$
such that
$$
\Pi_{m+1,m}(r_{m+1})=r_m
\qquad
\text{for every }m\geq 6.
$$
The tower is zero when every projection $r_m$ is zero.
\end{definition}

\begin{principle}[Projection separation]
\label{prin:axiszeckendorf-projection-separation}
Residual towers are separated by their finite projections:
$$
\left(\forall m\geq 6,\ r_m=0\right)
\Longrightarrow
r_\infty=0.
$$
Equivalently, a nonzero residual tower has a nonzero finite projection at
some level.
\end{principle}

\section{First nonzero layer}
\label{sec:axiszeckendorf-first-nonzero-layer}

\begin{proposition}[First nonzero residual lies in a layer kernel]
\label{prop:axiszeckendorf-first-nonzero-layer-kernel}
Let $r_\infty=(r_m)_{m\geq6}$ be a residual tower.  If $r_\infty\neq0$ and
$m_0$ is the least level with $r_{m_0}\neq0$, then either $m_0=6$ or
$$
r_{m_0}\in K_{m_0}.
$$
\end{proposition}

\begin{proof}
If $m_0=6$, there is nothing to prove.  If $m_0>6$, minimality gives
$r_{m_0-1}=0$.  Compatibility of the residual tower gives
$$
\Pi_{m_0,m_0-1}(r_{m_0})=r_{m_0-1}=0,
$$
so $r_{m_0}\in\ker(\Pi_{m_0,m_0-1})=K_{m_0}$.
\end{proof}

\begin{principle}[Induction target]
\label{prin:axiszeckendorf-layer-kernel-induction-target}
The projection-tower problem is therefore reduced to the following finite
layer question: can a newly added kernel $K_m$ contain a nonzero residual
that is simultaneously in the skew sector and fixed by the return operator?
If every such kernel rejects fixed skew residuals, no nonzero admitted
counter residual tower survives.
\end{principle}

\section{Return operators and spectral exclusion}
\label{sec:axiszeckendorf-return-operators-spectral-exclusion}

\begin{definition}[Compatible return operators]
\label{def:axiszeckendorf-compatible-return-operators}
A return-operator tower is a family of linear maps
$$
R_m:V_m\to V_m
\qquad
(m\geq 6)
$$
such that
$$
\Pi_{m+1,m}R_{m+1}=R_m\Pi_{m+1,m}
\qquad
\text{for every }m\geq 6.
$$
\end{definition}

\begin{definition}[Skew kernel sector]
\label{def:axiszeckendorf-skew-kernel-sector}
For each $m\geq 6$, a skew kernel sector is a displayed subspace
$$
K_m^{\mathrm{skew}}\subseteq K_m
$$
containing the kernel residuals admitted by the RH counter-residual bridge.
The sector is part of the bridge data; it is not determined by the address
carrier alone.
\end{definition}

\begin{definition}[Layer spectral exclusion]
\label{def:axiszeckendorf-layer-spectral-exclusion}
Layer spectral exclusion at level $m$ is the row
$$
1\notin\operatorname{spec}(R_m|_{K_m^{\mathrm{skew}}}).
$$
Equivalently, the linear map
$$
I-R_m:K_m^{\mathrm{skew}}\to K_m^{\mathrm{skew}}
$$
is invertible.
\end{definition}

\begin{proposition}[No fixed skew residual in an excluded layer]
\label{prop:axiszeckendorf-no-fixed-skew-residual-excluded-layer}
If $r\in K_m^{\mathrm{skew}}$, $R_mr=r$, and layer spectral exclusion holds
at $m$, then $r=0$.
\end{proposition}

\begin{proof}
The equation $R_mr=r$ is equivalent to $(I-R_m)r=0$.  Since $I-R_m$ is
invertible on $K_m^{\mathrm{skew}}$, its kernel is zero.  Hence $r=0$.
\end{proof}

\begin{remark}[Window-six anchor]
\label{rem:axiszeckendorf-window-six-anchor}
Window six supplies the anchor pattern for this exclusion: on the intended
skew sector, the return has the form
$$
R_6\simeq -\frac{1}{2}I+N_6
$$
with nilpotent tail $N_6$.  Then
$$
I-R_6\simeq\frac{3}{2}I-N_6,
$$
which is invertible over the rational residual module.  The induction
problem is to show that the same no-neutral-leak condition persists through
the added kernels.
\end{remark}

\section{Layer-kernel refutation}
\label{sec:axiszeckendorf-layer-kernel-refutation}

\begin{theorem}[Layer-kernel residual refutation]
\label{thm:axiszeckendorf-layer-kernel-residual-refutation}
Assume the following rows for an admitted RH counter residual tower
$r_\infty=(r_m)_{m\geq6}$:
$$
r_\infty\neq0,
$$
zero-event return invariance
$$
R_mr_m=r_m
\qquad
\text{for every }m\geq6,
$$
compatibility of the return operators, and, for every layer in which a
first nonzero residual can appear, skew-sector admission
$$
r_m\in K_m^{\mathrm{skew}}
$$
together with layer spectral exclusion
$$
1\notin\operatorname{spec}(R_m|_{K_m^{\mathrm{skew}}}).
$$
Then no such nonzero admitted RH counter residual tower exists.
\end{theorem}

\begin{proof}
By projection separation, choose the least $m_0\geq6$ with
$r_{m_0}\neq0$.  If $m_0=6$, the Window-six anchor places the residual in
the admitted skew sector, and spectral exclusion at level $6$ contradicts
$R_6r_6=r_6$ by
\autoref{prop:axiszeckendorf-no-fixed-skew-residual-excluded-layer}.

If $m_0>6$, then
\autoref{prop:axiszeckendorf-first-nonzero-layer-kernel} gives
$r_{m_0}\in K_{m_0}$.  The bridge-admission row places this residual in
$K_{m_0}^{\mathrm{skew}}$.  Zero-event return invariance gives
$R_{m_0}r_{m_0}=r_{m_0}$, while layer spectral exclusion at $m_0$ forces
$r_{m_0}=0$ by
\autoref{prop:axiszeckendorf-no-fixed-skew-residual-excluded-layer}.  This
contradicts the choice of $m_0$.
\end{proof}

\begin{corollary}[Projection induction form]
\label{cor:axiszeckendorf-projection-induction-form}
Equivalently, define
$$
P(m):\quad r_m=0.
$$
If the Window-six anchor proves $P(6)$ and every induction step proves that
$r_m=0$ forces $r_{m+1}\in K_{m+1}^{\mathrm{skew}}$ and hence
$r_{m+1}=0$ by spectral exclusion, then
$$
\forall m\geq6,\ r_m=0.
$$
Projection separation then gives $r_\infty=0$.
\end{corollary}

\begin{proof}
The induction step uses compatibility:
$$
\Pi_{m+1,m}(r_{m+1})=r_m=0,
$$
so $r_{m+1}\in K_{m+1}$.  The bridge-admission row places it in the skew
kernel sector, and spectral exclusion removes every fixed residual there.
Ordinary induction gives all finite projections zero, and projection
separation gives the zero tower.
\end{proof}

\section{Proof obligations}
\label{sec:axiszeckendorf-layer-kernel-proof-obligations}

\begin{definition}[Projection-tower proof obligations]
\label{def:axiszeckendorf-projection-tower-proof-obligations}
The layer-kernel induction route has four proof obligations:
$$
\mathcal B_{\zeta\to Z}:
\mathsf{CounterRH}(n,q)\to r_\infty\neq0,
$$
$$
\mathsf{ZeroEventReturnInvariant}_m(r_m):
R_mr_m=r_m,
$$
$$
\mathsf{NoNeutralLeak}_m:
r_m\in K_m^{\mathrm{skew}},
$$
and
$$
\mathsf{LayerSpectralExclusion}_m:
1\notin\operatorname{spec}(R_m|_{K_m^{\mathrm{skew}}}).
$$
\end{definition}

\begin{remark}[Independent work before RH bridge]
\label{rem:axiszeckendorf-independent-work-before-rh-bridge}
The address windows, kernel dimensions, compatible return operators, and
spectral exclusions can be studied before any RH bridge is supplied.  The
largest bridge is the first obligation, which must route a
$\mathsf{CounterRH}(n,q)$ event to a nonzero Zeckendorf residual tower.
Until that bridge is certified, the layer-kernel induction is a structured
target for residual elimination, not a proof of $\mathsf{RH}$.
\end{remark}
